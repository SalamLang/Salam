\chapter{Cheat Sheet}%
\label{app:cheatsheet}

A compact reference to Salam's syntax and the most-used standard-library calls.
Every snippet is a complete program that compiles and runs as it stands; refer
back to the named chapter for the full story.

\section*{Hello, world (Chapter 1)}

\begin{salamen}
func main:
    println "Hello, Salam!"
end
\end{salamen}

\noindent Run it: \code{salam run hello.salam} (or \code{salam exec hello.salam}
with the interpreter)\quad Build it: \code{salam build hello.salam --output=hello}

\section*{Variables and types (Chapters 2 and 3)}

\begin{salamen}
const MAX := 100                // compile-time constant

func main:
    name := "Sara"              // immutable, type inferred (str)
    mut count := 0              // mutable
    count += 1
    x := 3.14                   // a float literal is f64
    big := 250 as i64           // choose a type with 'as'
    b := 250 as int as u8       // explicit conversion
    println name, count, x, big, b, MAX
end
\end{salamen}

\noindent There are no typed declarations: write \code{name := value}, and use
\code{as} when the inferred type is not the one you want. Reassigning needs
\code{mut}; an unused variable, or a \code{mut} that is never reassigned, is a
compile error. \\
Types: \code{i8 i16 i32 i64}, \code{u8 u16 u32 u64}, \code{usize size},
\code{f32 f64}, \code{bool}, \code{char}, \code{uchar}, \code{str},
\code{void}; \code{int}, \code{uint} and \code{float} are the short spellings of
\code{i32}, \code{u32} and \code{f32}. \\
Operators: \code{+ - * / \%}, \code{**} (power, float), \code{== != < > <= >=}
(or the words \code{eq neq lt gt lte gte}), \code{and or not},
\code{cond ? a : b}, bitwise \code{\& | \textasciicircum{} \textasciitilde{} << >>} on integers,
\code{+= -= *= /= \%= **=}, \code{++ --}. Integer \code{/} truncates.

\section*{Control flow (Chapter 4)}

\begin{salamen}
func main:
    x := 7
    if x > 0:
        println "positive"
    else x == 0:                     // else-if
        println "zero"
    else:
        println "negative"
    end

    switch x:                        // arms fall through: end each with break
        1, 2: println "small" break end
        3 to 9: println "medium" break end
        else: println "large" end
    end

    size := match x:                 // match is an expression, no fallthrough
        1, 2: "small" end
        else: "bigger" end
    end
    println size
end
\end{salamen}

\section*{Loops (Chapter 5)}

\begin{salamen}
func main:
    mut n := 3
    until n > 0:                     // runs WHILE the condition holds
        n -= 1
    end
    repeat 3: print "hi " end        // 3 times
    repeat 4 in i: print i, "" end   // i = 0..3
    repeat 1 to 5 in i: print i, "" end        // 1..5 inclusive
    repeat 0 to 20 by 5 in i: print i, "" end  // step by 5
    xs := [10, 20, 30]
    each x in xs: print x, "" end              // walk a collection
    each i, x in xs: print i, x, "" end        // with the index
    println ""
    // break: exit the loop;  continue: next iteration
end
\end{salamen}

\section*{Functions (Chapter 6)}

\begin{salamen}
func add(a: int, b: int): int:
    ret a + b
end

func greet(name: str, greeting: str = "Hi"):   // no result type; a default
    println greeting + ",", name
end

func bump(n &: int):                 // reference parameter
    n += 1
end

func main:
    greet("Sara")
    mut total := add(2, 3)
    bump(total)
    println total
end
\end{salamen}

\noindent Parameters are passed by value unless marked \code{\&:}; functions can
be overloaded by parameter types; definition order does not matter.
\code{func main} returns the program's exit code (\code{ret 0} when omitted).

\section*{Arrays and vectors (Chapter 7)}

\begin{salamen}
func total(view: int[]): int:        // int[] takes any int array or slice
    mut s := 0
    each x in view: s += x end
    ret s
end

func main:
    a := [1, 2, 3]                   // fixed array int[3], indexed 0..2
    g := [[1, 2, 3], [4, 5, 6]]      // 2-D, int[2][3]
    println len(a), g[1][2], total(a)

    mut v := Vector {} as Vector<int>
    v.push(10)                       // append
    v.push(20)
    v.set(0, 99)                     // write
    println v.get(0), v.len()        // read, count
    v.pop()                          // remove the last element
    v.free()                         // release the memory
end
\end{salamen}

\section*{Strings (Chapter 8)}

\begin{salamen}
import str

func main:
    s := " Hello, Salam "
    t := str.Trim(s)
    println "a" + "b", len(t), str.Upper(t), str.Lower(t), str.Reverse(t)
    println str.Find(t, "Salam"), str.Contains(t, "lo"), str.Replace(t, "Hello", "Bye")
    println str.Substr(t, 0, 5), str.FromInt(42 as i64), str.ToInt("17") + 1
    parts := str.Split("a,b,c", ",")
    println str.Join(parts, " - ")
    parts.free()
end
\end{salamen}

\section*{Structs and methods (Chapter 9)}

\begin{salamen}
struct Point:
    pub x: int = 0                   // pub: usable outside the struct
    pub y: int = 0

    pub func move_by(dx: int, dy: int):
        this.x += dx
        this.y += dy
    end
end

func main:
    mut p := Point { x = 1, y = 2 }
    p.move_by(3, 4)
    println p.x, p.y
    println p                        // Point {x = 4, y = 6}
end
\end{salamen}

\section*{Enums (Chapter 10)}

\begin{salamen}
enum Day: Mon, Tue, Wed, Thu, Fri, Sat, Sun end

enum Shape:
    Circle(r: f64)
    Rect(w: f64, h: f64)
end

func area(s: Shape): f64:
    ret match s:                     // must cover every member
        Circle(r): 3.14 * r * r end
        Rect(w, h): w * h end
    end
end

func main:
    d := Day.Sat
    println d as int                 // 5
    println area(Shape.Rect(3.0, 4.0))
end
\end{salamen}

\section*{Interfaces and generics (Chapters 11 and 12)}

\begin{salamen}
interface Shape:
    func area(): f64
end

struct Square:                       // conforms by having the pub method
    pub side: f64 = 1.0
    pub func area(): f64: ret this.side * this.side end
end

struct Box<T>:                       // generic struct
    pub value: T
end

func describe<T: Shape>(s: T):       // static bound
    println s.area()
end

func draw(s: dyn Shape):             // dynamic dispatch
    println s.area()
end

func Max<T>(a: T, b: T): T:          // generic function
    if a > b: ret a end
    ret b
end

func main:
    sq := Square { side = 2.0 }
    describe(sq)
    draw(sq)
    b := Box { value = Max(3, 9) } as Box<int>
    println b.value
end
\end{salamen}

\section*{Closures (Chapter 13)}

\begin{salamen}
func apply(f: func (int) int, x: int): int:
    ret f(x)
end

func make_adder(n: int): func (int) int:
    ret (x: int): ret x + n end      // closure capturing n by value
end

func main:
    double := (x: int): ret x * 2 end
    println double(4)
    println apply((x: int): ret x + 1 end, 5)   // lambda as argument
    add10 := make_adder(10)
    println add10(1)
end
\end{salamen}

\section*{Errors and defer (Chapter 14)}

\begin{salamen}
import option

func safe_div(a: int, b: int): option.Option<int>:
    if b == 0: ret option.None() end
    ret option.Some(a / b)
end

func main:
    mut v := Vector {} as Vector<int>
    defer v.free()                   // runs at scope exit (last in, first out)
    v.push(1)
    r := safe_div(10, 0)
    println option.IsSome(r), option.UnwrapOr(r, -1)
end
\end{salamen}

\section*{Modules (Chapter 15)}

\begin{shell}
package main
import math                         // a standard library package, by name
import encoding.json                // a nested one; call json.Name(...)
include geo "geometry.salam"        // your own file; geo is an alias
include "@/shared/util.salam"       // path from the project root

pub func Area(r: f64): f64: ret math.PI * r * r end   // pub: exported
\end{shell}

\noindent File order: \code{package}, imports, includes, constants and globals,
types, then functions. Names are private to their package unless marked
\code{pub}.

\section*{Memory and FFI (Chapters 18 and 19)}

\begin{salamen}
import mem

extern:
    func printf(format: str, ...): int
    func sqrt(x: f64): f64
end

export:                              // a Salam function C can call by name
    func on_event(code: int): int:
        ret code * 2
    end
end

func main:
    p := mem.Allocate(8 as u64) as i64*
    p[0] = 42
    printf("%d %d\n", p[0] as int, on_event(21))
    println sqrt(2.0)
    mem.Free(p as void*)
end
\end{salamen}

\noindent Link an external C library with \code{link dynamic "name"} (or
\code{link static}, or \code{link framework} on macOS) at the top of the file.

\section*{Concurrency (Chapter 20)}

\begin{salamen}
import sync

mut counter := 0
mut lock := sync.NewMutex()

func work:
    sync.Lock(lock)                  // critical section
    counter += 1
    sync.Unlock(lock)
end

func main:
    t := spawn(work)                 // start a thread
    t2 := spawn(work)
    join(t)                          // wait for it
    join(t2)
    println counter
    sync.Destroy(lock)
end
\end{salamen}

\section*{The \texttt{salam} commands}

\begin{center}
\begin{tabularx}{\linewidth}{@{}lX@{}}
\toprule
\textbf{Command} & \textbf{Does} \\
\midrule
\code{salam run f.salam}    & build and run, keep nothing \\
\code{salam exec f.salam}   & run with the interpreter (no C toolchain) \\
\code{salam build f.salam --output=f} & build a native executable \\
\code{salam check f.salam}  & find errors without building \\
\code{salam format f.salam} & reformat source in place \\
\code{salam new name}       & scaffold a new project \\
\code{salam translate fa f.salam} & rewrite a program into Persian (or \code{en}) \\
\code{salam doc dir}        & generate an API page from doc comments \\
\code{salam js f.salam}     & compile to JavaScript for the browser \\
\code{salam debug f.salam}  & build with debug info, launch a debugger \\
\code{salam memcheck f.salam} & build with AddressSanitizer and run \\
\code{salam cli}            & start the interactive REPL \\
\code{salam help}           & list every command and option \\
\code{salam version}        & show the compiler version \\
\bottomrule
\end{tabularx}
\end{center}

\noindent The language of a source file (English or Persian) is detected from
its keywords; \code{--lang=fa} or \code{--lang=en} states it explicitly.
