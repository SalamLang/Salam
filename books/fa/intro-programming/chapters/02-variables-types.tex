\chapter{متغیرها و گونه‌های داده}%
\label{ch:variables}

در فصلِ پیش با متغیر آشنا شدیم و آن را به جعبه‌ای تشبیه کردیم که چیزی در آن
نگه می‌داریم. در این فصل ژرف‌تر می‌شویم: گونه‌های گوناگونِ داده را می‌شناسیم،
یاد می‌گیریم متغیر را چگونه بسازیم، و درمی‌یابیم چرا «گونهٔ» هر متغیر اهمیت
دارد.

\section{متغیر؛ جعبه‌ای با نام}

تصور کنید چند جعبه دارید و روی هر کدام برچسبی زده‌اید: «سن»، «قد»، «نام». در هر
جعبه چیزی گذاشته‌اید. هر وقت به محتوای جعبه نیاز داشتید، کافی است نامِ روی برچسب
را صدا بزنید. در برنامه‌نویسی، این جعبه‌ها همان \textbf{متغیر}ها هستند.

ساده‌ترین شکلِ ساختنِ یک متغیر در سلام چنین است:

\begin{salamfa}
تابع ریشه:
    سن := 30
    سرچاپ سن
پایان
\end{salamfa}

این خط را چنین بخوانید: «یک \textbf{متغیر} به نامِ \textbf{سن} بساز و مقدارِ
\textbf{۳۰} را در آن بگذار.» نشانهٔ \code{:=} یعنی «نامی تازه بساز و این مقدار را
در آن بگذار». لازم نیست گونهٔ متغیر را بنویسید: سلام از روی خودِ مقدار می‌فهمد که
\code{30} یک عددِ صحیح است، پس \code{سن} از گونهٔ \code{صحیح} می‌شود و از آن پس
همیشه صحیح می‌ماند.

ساختار همیشه یکسان است:

\begin{center}
\textit{نام} \code{:=} \textit{مقدار}
\end{center}

\begin{note}
هر متغیر از همان لحظهٔ ساخته شدن مقدار دارد. در سلام نمی‌توان نامی را حالا
ساخت و بعداً پُرش کرد؛ پس هیچ‌گاه متغیری را نمی‌خوانید که مقدارِ بی‌معنایی در
آن مانده باشد.
\end{note}

\section{چهار شیوهٔ ساختنِ متغیر}

سلام چهار شیوه برای ساختنِ متغیر دارد و هر کدام معنایی متفاوت دارد. به این
برنامهٔ کوچک نگاه کنید:

\begin{salamfa}
تابع ریشه:
    ناپایا سن := 30
    پایا سال‌تولد := 1994
    قد := 1.75
    جمعیت := 9000000 برگردان صحیح۶۴
    سن = سن + 1
    سرچاپ سن, سال‌تولد, قد, جمعیت
پایان
\end{salamfa}

خروجی:

\begin{salamen}[language={}]
31 1994 1.75 9000000
\end{salamen}

بیایید چهار شیوه را از هم تفکیک کنیم:

\subsection{ناپایا: جعبه‌ای که می‌توان عوضش کرد}

\begin{salamfa}
ناپایا سن := 30
\end{salamfa}

واژهٔ \code{ناپایا} (معادلِ \code{mut} در انگلیسی) جعبه‌ای می‌سازد که محتوایش را
بعداً می‌توان \emph{تغییر} داد. به همین دلیل در خطِ \code{سن = سن + 1}
توانستیم مقدارِ \code{سن} را یکی زیاد کنیم؛ حالا \code{سن} برابرِ ۳۱ است. برای
دادنِ مقدارِ تازه به متغیری که از پیش ساخته شده، تنها \code{=} می‌نویسیم، نه
\code{:=}.

\begin{warn}
سلام سخت‌گیر است: اگر متغیری را \code{ناپایا} بسازید اما هیچ‌گاه تغییرش ندهید،
یا متغیری بسازید و هیچ‌جا به کارش نبرید، برنامه کامپایل نمی‌شود. پس
\code{ناپایا} را تنها وقتی بنویسید که واقعاً قرار است مقدار عوض شود.
\end{warn}

\subsection{پایا: جعبه‌ای که قفل است}

\begin{salamfa}
پایا سال‌تولد := 1994
\end{salamfa}

واژهٔ \code{پایا} (معادلِ \code{const}) جعبه‌ای می‌سازد که مقدارش هنگامِ کامپایل
ثابت می‌شود و \emph{دیگر هرگز تغییر نمی‌کند}. سالِ تولدِ کسی هیچ‌گاه عوض
نمی‌شود، پس درست است که آن را پایا بگیریم. اگر بکوشید مقدارِ یک پایا را تغییر
دهید، کامپایلر خطا می‌گیرد.

\begin{note}
نامِ پایا باید \emph{یک‌واژه‌ای} باشد: \code{پایا سال تولد := 1994} خطای تجزیه
می‌دهد. برای همین در این نمونه دو واژه را با نیم‌فاصله به هم چسبانده‌ایم:
\code{سال‌تولد}. متغیرهای معمولی می‌توانند نامِ چندواژه‌ای داشته باشند.
\end{note}

\subsection{اعلانِ ساده: تغییرناپذیر و بی‌واژه}

\begin{salamfa}
قد := 1.75
\end{salamfa}

اگر هیچ واژه‌ای (\code{ناپایا} یا \code{پایا}) ننویسید و مستقیم با نام آغاز
کنید، یک متغیرِ \emph{تغییرناپذیر} ساخته می‌شود: نمی‌توان بعداً مقدارِ تازه‌ای
به \code{قد} داد. این شکلِ کوتاه و تمیز، شیوهٔ پیش‌فرضِ سلام است: \emph{مگر آنکه
واقعاً نیاز به تغییر داشته باشید، متغیر را تغییرناپذیر بگذارید.} چون
\code{1.75} نقطهٔ اعشار دارد، سلام خودش درمی‌یابد که \code{قد} از گونهٔ
\code{اعشار۶۴} است.

\subsection{برگردان: گونه را خودتان برگزینید}

\begin{salamfa}
جمعیت := 9000000 برگردان صحیح۶۴
\end{salamfa}

سلام همیشه از روی مقدار گونه را تشخیص می‌دهد: عددِ صحیح به‌طورِ پیش‌فرض
\code{صحیح} (۳۲ بیتی) است و عددِ اعشاری \code{اعشار۶۴}. اگر گونهٔ دیگری
می‌خواهید، مقدار را با واژهٔ \code{برگردان} (معادلِ \code{as}) به آن گونه ببرید.
اینجا جمعیت را از همان آغاز \code{صحیح۶۴} گرفته‌ایم تا بعداً هر قدر هم بزرگ شود،
در آن جا بگیرد.

\begin{tip}
در کدهای قدیمی‌ترِ سلام گونه پیش از مقدار نوشته می‌شد، مانندِ
\code{سن: صحیح = 30}، و واژه‌ای به نامِ \code{خودکار} هم برای تشخیصِ خودکارِ
گونه بود. این شکل‌ها حذف شده‌اند و اکنون خطا می‌دهند. امروز همیشه
\code{سن := 30} بنویسید و اگر گونه مهم است، \code{سن := 30 برگردان صحیح۶۴}.
شکلِ \code{نام: گونه} تنها برای پارامترهای تابع و فیلدهای ساختار باقی مانده است.
\end{tip}

\begin{note}
به شیوهٔ صدا زدنِ \code{سرچاپ} دقت کنید: آرگومان‌ها را \textbf{بدونِ پرانتز} و
جدا شده با ویرگول می‌نویسیم، مانندِ \code{سرچاپ "سلام،", نام} یا
\code{سرچاپ نام, سن}. \code{سرچاپ} یک دستور است، نه یک تابعِ معمولی.
پرانتز در دلِ خودِ آرگومان‌ها آزاد است و برای گروه‌بندیِ محاسبات به کار
می‌رود؛ مثلاً \code{سرچاپ 1, (3 * 4)} یا \code{سرچاپ (2 + 3) * 4} کاملاً درست‌اند.
\end{note}

\section{گونه‌های پایهٔ داده}

هر متغیر یک \textbf{گونه} دارد که می‌گوید چه جور داده‌ای را می‌تواند نگه دارد.
گونه، هم به سلام کمک می‌کند که برنامهٔ شما را درست بفهمد، و هم از بسیاری خطاها
جلوگیری می‌کند. مهم‌ترین گونه‌های پایه در سلام این‌ها هستند:

\subsection{اعدادِ درست (صحیح)}

عددهای بدونِ اعشار مانندِ ۰، ۷، ۱۵، و ۱۰۰- را اعدادِ \emph{درست} یا
\emph{صحیح} می‌نامیم. سلام چند گونهٔ عددِ درست دارد که در \emph{اندازه} با هم
فرق می‌کنند:

\begin{center}
\begin{tabular}{c c l}
\toprule
\textbf{نامِ فارسی} & \textbf{نامِ انگلیسی} & \textbf{توضیح} \\
\midrule
\code{صحیح} & \code{int} / \code{i32} & عددِ درستِ معمول (۳۲ بیتی) \\
\code{صحیح۳۲} & \code{i32} & عددِ درستِ ۳۲ بیتی \\
\code{صحیح۶۴} & \code{i64} & عددِ درستِ ۶۴ بیتی (برای عددهای خیلی بزرگ) \\
\bottomrule
\end{tabular}
\end{center}

برای کارهای روزمره \code{صحیح} کافی است. اگر با عددهای بسیار بزرگ سر و کار
دارید (مثلاً فاکتوریلِ اعداد یا شمارشِ چیزهای فراوان)، از \code{صحیح۶۴}
استفاده کنید.

\begin{note}
\code{صحیح} و \code{صحیح۳۲} دقیقاً \emph{یک} گونه‌اند: هر دو همان عددِ درستِ ۳۲
بیتی (\code{i32}) هستند و هر جا یکی را بنویسید، دیگری هم همان معنا را می‌دهد.
عددِ درست در سلام به‌طورِ پیش‌فرض ۳۲ بیتی است، پس نوشتنِ \code{۳۲} در پایانِ
نام ضروری نیست. در سراسرِ این کتاب همان شکلِ کوتاه، یعنی \code{صحیح}، را به‌کار
می‌بریم.
\end{note}

\begin{salamfa}
تابع ریشه:
    کوچک := 1000
    بزرگ := 9000000000
    سرچاپ کوچک, بزرگ
پایان
\end{salamfa}

عددِ \code{9000000000} در \code{صحیح} جا نمی‌شود، پس سلام خودبه‌خود آن را
\code{صحیح۶۴} می‌گیرد. برای عددی کوچک‌تر که بعداً بزرگ می‌شود، خودتان
\code{برگردان صحیح۶۴} بنویسید.

\subsection{اعدادِ اعشاری}

عددهایی که بخشِ اعشاری دارند مانندِ ۱٫۷۵، ۳٫۱۴ و ۰٫۵ را اعدادِ
\emph{اعشاری} می‌نامیم. در سلام هر عددِ اعشاری که می‌نویسید از گونهٔ
\code{اعشار۶۴} است (معادلِ \code{f64}):

\begin{salamfa}
تابع ریشه:
    قد := 1.75
    وزن := 68.5
    سرچاپ قد, وزن
پایان
\end{salamfa}

\begin{warn}
در عددهای اعشاری، نقطهٔ اعشار همیشه با نقطهٔ لاتین (\code{.}) نوشته می‌شود، نه
ممیزِ فارسی. پس بنویسید \code{1.75} یا \code{۱.۷۵}، نه «۱٫۷۵». همچنین برای آنکه
عددی اعشاری شمرده شود، باید نقطه داشته باشد: \code{2} یک عددِ درست است اما
\code{2.0} یک عددِ اعشاری.
\end{warn}

\begin{note}
گونهٔ \code{اعشار} (بدونِ رقم) همان \code{اعشار۳۲} است، یعنی عددِ اعشاریِ
کوچک‌تری که تنها حدودِ هفت رقم را درست نگه می‌دارد. برای کارهای معمولی همیشه
\code{اعشار۶۴} را به‌کار ببرید.
\end{note}

\subsection{متن (رشته)}

هر دنباله‌ای از نویسه‌ها، چه یک واژه، چه یک جمله، چه یک نام، یک
\emph{رشته} (string) است. رشته‌ها را همیشه میانِ دو نقل‌قول می‌نویسیم و گونه‌شان
\code{رشته} است (معادلِ \code{str}):

\begin{salamfa}
تابع ریشه:
    شهر := "اصفهان"
    پیام := "به سلام خوش آمدید"
    سرچاپ شهر, پیام
پایان
\end{salamfa}

\subsection{مقادیرِ منطقی (درست/نادرست)}

گاهی فقط دو پاسخ ممکن است: آری یا نه، روشن یا خاموش، درست یا نادرست. برای این
موارد گونهٔ \code{منطقی} (معادلِ \code{bool}) را داریم که تنها دو مقدار می‌پذیرد:
\code{درست} (true) و \code{نادرست} (false):

\begin{salamfa}
تابع ریشه:
    باران := درست
    آفتاب := نادرست
    سرچاپ باران, آفتاب
پایان
\end{salamfa}

خروجی:

\begin{salamen}[language={}]
true false
\end{salamen}

همان‌طور که می‌بینید، مقدارهای منطقی حتی در برنامهٔ فارسی به شکلِ
\code{true} و \code{false} چاپ می‌شوند؛ عددها هم همیشه با رقم‌های لاتین چاپ
می‌شوند. مقادیرِ منطقی هنگامِ تصمیم‌گیری بسیار به‌کار می‌آیند؛ در فصلِ شرط‌ها
(فصلِ~\ref{ch:conditionals}) نقشِ کلیدیِ آن‌ها را خواهیم دید.

\subsection{نویسه (تک‌نویسه)}

اگر تنها به \emph{یک} نویسه نیاز داشته باشید، مثلاً حرفِ \lr{A} یا حرفِ «س»،
از گونهٔ \code{نویسه} (معادلِ \code{char}) استفاده می‌کنید. نویسه‌ها را میانِ
\emph{تک‌نقل‌قول} می‌نویسیم. \code{'A'} تنها برای یک حرف یا رقمِ انگلیسی کار
می‌کند؛ هر حرفِ دیگری، از جمله همهٔ حرف‌های فارسی، پیشوندِ \code{u} می‌خواهد،
مانندِ \code{u'س'}:

\begin{salamfa}
تابع ریشه:
    حرف := 'A'
    حرف فارسی := u'س'
    سرچاپ حرف, حرف فارسی
پایان
\end{salamfa}

\begin{note}
تفاوتِ \code{'A'} و \code{"A"} چیست؟ نخستی یک \emph{نویسه} (تک‌نویسه) است و
دومی یک \emph{رشته} (دنباله‌ای از نویسه‌ها که اینجا فقط یک نویسه دارد). در
بسیاری از زبان‌ها این تمایز مهم است.
\end{note}

\section{جدولِ کاملِ گونه‌های پایه}

\begin{center}
\begin{tabular}{c c l}
\toprule
\textbf{فارسی} & \textbf{انگلیسی} & \textbf{چه چیزی نگه می‌دارد} \\
\midrule
\code{صحیح} / \code{صحیح۳۲} & \code{int} / \code{i32} & عددِ درستِ ۳۲ بیتی \\
\code{صحیح۶۴} & \code{i64} & عددِ درستِ ۶۴ بیتی \\
\code{اعشار۶۴} & \code{f64} & عددِ اعشاری \\
\code{اعشار} / \code{اعشار۳۲} & \code{f32} & عددِ اعشاریِ کوچک‌تر \\
\code{منطقی} & \code{bool} & درست یا نادرست \\
\code{نویسه} & \code{char} & یک نویسه \\
\code{رشته} & \code{str} & دنباله‌ای از نویسه‌ها \\
\bottomrule
\end{tabular}
\end{center}

\begin{tip}
هر پرونده در سلام به \emph{یک} زبان نوشته می‌شود و نام‌های گونه هم باید به همان
زبان باشند. در یک برنامهٔ فارسی باید بنویسید \code{ن := 5 برگردان صحیح۶۴}؛ اگر
نامِ انگلیسیِ همان گونه، یعنی \code{i64}، را در پروندهٔ فارسی به‌کار ببرید،
کامپایلر خطا می‌گیرد و می‌گوید نامِ گونه در یک فایلِ فارسی باید فارسی باشد.
نامِ انگلیسیِ گونه‌ها تنها در پرونده‌های انگلیسی معتبر است.
\end{tip}

\section{تبدیلِ گونه با «برگردان»}

گاهی می‌خواهید یک مقدار را از گونه‌ای به گونهٔ دیگر تبدیل کنید؛ مثلاً یک عددِ
درست را به اعشاری، یا برعکس. برای این کار هم از همان واژهٔ \code{برگردان} استفاده
می‌کنید:

\begin{salamfa}
تابع ریشه:
    شمار := 7
    دقیق := شمار برگردان اعشار۶۴
    سرچاپ دقیق / 2
پایان
\end{salamfa}

خروجی:

\begin{salamen}[language={}]
3.5
\end{salamen}

اینجا \code{شمار} (که یک عددِ درست است) به \code{اعشار۶۴} تبدیل شد و برای همین
تقسیم بر ۲ بخشِ اعشاری را نگه داشت. به این کار \textbf{تبدیلِ گونه} (cast)
می‌گویند و در فصل‌های بعد بارها به آن نیاز خواهیم داشت. تبدیلِ اعشاری به صحیح
بخشِ اعشاری را دور می‌ریزد و گرد نمی‌کند: \code{3.9 برگردان صحیح} می‌شود ۳.

\section{قاعده‌های نام‌گذاری}

نامِ متغیرها را خودتان برمی‌گزینید، اما چند قاعدهٔ ساده هست:

\begin{itemize}
  \item نام می‌تواند فارسی، انگلیسی یا ترکیبی باشد: \code{سن}، \code{age}،
        \code{شماره‌حساب}.
  \item نام می‌تواند چندواژه‌ای باشد؛ بینِ واژه‌ها فاصله بگذارید:
        \code{قیمت کتاب}، \code{شمارنده اصلی}. فاصله و نیم‌فاصله درونِ نام
        یکی‌اند، پس \code{قیمت کتاب} و \code{قیمت‌کتاب} یک متغیرند.
  \item نام نمی‌تواند با یک رقم آغاز شود.
  \item نام نباید یکی از واژگانِ کلیدیِ زبان (مانندِ \code{تابع}، \code{اگر}،
        \code{و} یا \code{در}) باشد، حتی به شکلِ یکی از واژه‌های یک نامِ
        چندواژه‌ای: \code{تعداد در انبار} پذیرفته نمی‌شود، اما
        \code{موجودی انبار} درست است.
\end{itemize}

\begin{tip}
نامِ خوب، نامی است که \emph{معنا}ی متغیر را برساند. \code{مجموع نمرات} بسیار
گویاتر از \code{م} است. کدی که شش ماه بعد بخوانید، باید همچنان قابلِ فهم باشد،
و نامِ خوب نیمی از این فهم را تأمین می‌کند.
\end{tip}

\begin{deep}[نگاهی به درون: گونه چگونه به ایمنی کمک می‌کند]
سلام یک زبانِ \emph{ایستا-گونه} (statically typed) است؛ یعنی گونهٔ هر متغیر پیش
از اجرا و در زمانِ کامپایل مشخص است، حتی وقتی آن را ننوشته‌اید و سلام خودش از
روی مقدار فهمیده است. این ویژگی به کامپایلر اجازه می‌دهد بسیاری از خطاها را
\emph{پیش} از اجرای برنامه بگیرد. برای نمونه، اگر بکوشید متنی را در متغیری عددی
بگذارید، یا مقدارِ اعشاری را بدونِ \code{برگردان} در جعبه‌ای از گونهٔ صحیح بگذارید،
کامپایلر همان لحظه به شما هشدار می‌دهد، به‌جای آنکه برنامه هنگامِ اجرا و
شاید در دستِ کاربر، ناگهان از کار بیفتد.
\end{deep}

\section{خلاصهٔ فصل}

\begin{itemize}
  \item متغیر، جعبه‌ای نام‌دار برای نگه‌داشتنِ داده است و با \code{:=} ساخته
        می‌شود؛ سلام گونه را از روی مقدار تشخیص می‌دهد.
  \item چهار شیوهٔ اعلان داریم: \code{ناپایا} برای تغییرپذیر، \code{پایا}
        برای مقدارِ ثابت، اعلانِ ساده که تغییرناپذیر است، و
        \code{برگردان} برای برگزیدنِ گونه‌ای غیر از گونهٔ پیش‌فرض.
  \item گونه‌های پایه: \code{صحیح}، \code{صحیح۶۴}، \code{اعشار۶۴}،
        \code{منطقی}، \code{نویسه} و \code{رشته}.
  \item با \code{برگردان} می‌توان گونهٔ یک مقدار را تبدیل کرد.
  \item در پروندهٔ فارسی نام‌های فارسیِ گونه‌ها و در پروندهٔ انگلیسی نام‌های
        انگلیسیِ آن‌ها به‌کار می‌رود.
\end{itemize}

\exercise عددِ درستِ \code{9} را در یک متغیر بگذارید، سپس آن را با

\code{ناپایا} کنید و پیش از چاپ یکی به آن بیفزایید.
