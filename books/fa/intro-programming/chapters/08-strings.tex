\chapter{رشته‌ها و کاراکترها}%
\label{ch:strings}

متن همه‌جا هست: نام‌ها، پیام‌ها، خروجیِ برنامه‌ها. در این فصل عمیق‌تر با
\textbf{رشته}ها کار می‌کنیم: آن‌ها را به هم می‌چسبانیم، تکه‌تکه می‌کنیم،
می‌شماریم و دگرگون می‌کنیم.

\section{یادآوری: رشته چیست؟}

رشته دنباله‌ای از نویسه‌هاست که میانِ دو نقل‌قول نوشته می‌شود:
\code{"سلام دنیا"}. هر رشته از کنارِ‌هم‌قرارگرفتنِ نویسه‌ها (کاراکترها) ساخته
می‌شود. سلام از یونی‌کد پشتیبانی می‌کند، پس می‌توانید در رشته‌ها از فارسی،
انگلیسی، نشانه‌ها و حتی شکلک‌ها استفاده کنید.

\section{چسباندنِ رشته‌ها}

ساده‌ترین کار با رشته‌ها، چسباندنِ آن‌هاست. دو راه دارید: عملگرِ \code{+} یا
متدِ \code{.پیوست()}. این مثالِ واقعی هر دو راه را در عمل نشان می‌دهد:

\begin{salamfa}
// وارونه‌سازی رشته
تابع ریشه:
    س := "Salam"
    ناپایا ر := ""
    ناپایا i := س.طول() - ۱
    تا i >= ۰:
        ر = ر.پیوست(س.زیررشته(i، ۱))
        i = i - ۱
    پایان
    سرچاپ "وارونه‌ی"، س، "می‌شود"، ر
پایان
\end{salamfa}

\begin{progoutfa}
وارونه‌ی Salam می‌شود malaS
\end{progoutfa}

این برنامه یک رشته را \emph{وارونه} می‌کند. بیایید بشکافیمش:
\begin{itemize}
  \item \code{س.طول()} طولِ رشته را می‌دهد؛ پس \code{س.طول() - 1} اندیسِ آخرین
        نویسه است. چون \code{س} ناپایا نیست، با \code{:=} یک بار مقدار
        می‌گیرد؛ \code{ر} و \code{i} عوض می‌شوند و برای همین \code{ناپایا}اند.
  \item حلقه از آخرین نویسه تا نخستین (\code{i >= 0}) عقب می‌رود.
  \item \code{س.زیررشته(i، ۱)} یک
        \emph{زیررشته} به طولِ ۱ از موقعیتِ \code{i} می‌گیرد یعنی تنها
        یک نویسه.
  \item \code{ر.پیوست(...)} آن نویسه را به انتهای
        \code{ر} می‌چسباند.
\end{itemize}
\begin{note}
متدهای رشته نیز مانندِ متدهای بردار در پرونده‌ی فارسی نامِ فارسی دارند:
\code{طول}، \code{زیررشته}، \code{پیوست}، \code{بیاب}، \code{بشکاف} و
\code{پیراست}. نامِ انگلیسیِ آن‌ها (مثلِ \code{len}) در پرونده‌ی فارسی
پذیرفته نیست و کامپایلر خطا می‌دهد.
\end{note}

\begin{warn}
رشته به شکلِ بایت‌های \lr{UTF-8} نگه داشته می‌شود. هر حرفِ انگلیسی یک بایت است،
اما هر حرفِ فارسی دو بایت. \code{طول()} و \code{زیررشته} \emph{بایت}
می‌شمارند، پس \code{"سلام".طول()} برابرِ ۸ است. برای متنِ فارسی از همتاهای
نویسه‌ایِ آن‌ها استفاده کنید: \code{شمارنویسه()} (که برای \code{"سلام"} عددِ ۴
را می‌دهد)، \code{زیررشته نویسه(آغاز، تعداد)} و \code{نویسه شماره(ش)}. برنامه‌ی
بالا تنها برای متنِ انگلیسی درست کار می‌کند؛ برای وارونه کردنِ متنِ فارسی
\code{زیررشته} را با \code{زیررشته نویسه} و \code{طول} را با \code{شمارنویسه}
جایگزین کنید.
\end{warn}

\section{پیمایشِ نویسه‌به‌نویسه}

بسیاری از کارهای متنی نیازمندِ آن‌اند که رشته را نویسه‌به‌نویسه بررسی کنیم. الگوی
آن همیشه یکسان است: یک حلقه از ۰ تا \code{طول - 1}، و در هر گام گرفتنِ یک
زیررشتهٔ تک‌نویسه‌ای. این مثالِ واقعی مصوت‌های یک متن را می‌شمارد:

\begin{salamfa}
تابع ریشه:
    س := "salam donya"
    ناپایا تعداد := ۰
    تکرار س.طول() در i:
        ک := س.زیررشته(i، ۱)
        اگر (ک == "a") یا (ک == "e") یا (ک == "i") یا (ک == "o") یا (ک == "u"):
            تعداد = تعداد + ۱
        پایان
    پایان
    سرچاپ "تعداد مصوت‌ها در"، س، "="، تعداد
پایان
\end{salamfa}

\begin{progoutfa}
تعداد مصوت‌ها در salam donya = 4
\end{progoutfa}

این الگو «برای هر نویسه، کاری انجام بده» را در شمارش، جست‌وجو، فیلتر
کردن و بسیاری کارهای دیگر به‌کار خواهید برد.

\begin{warn}
برای مقایسهٔ دو رشته از \code{==} استفاده کنید، نه از کارهای دیگر. در سلام
\code{"a" == "a"} درست است و دو رشته را نویسه‌به‌نویسه می‌سنجد. به یاد داشته
باشید که مقایسه به بزرگی و کوچکیِ حروف حساس است: \code{"A"} با \code{"a"}
برابر نیست.
\end{warn}

\section{ابزارهای آمادهٔ کتابخانهٔ رشته}

نوشتنِ همهٔ کارهای متنی از صفر خسته‌کننده است. خوشبختانه سلام یک کتابخانهٔ
آمادهٔ رشته دارد که با \code{واردسازی رشته} در دسترس قرار می‌گیرد. به این
مثالِ واقعی نگاه کنید:

\begin{salamfa}
واردسازی رشته

تابع ریشه:
    متن := "Salam Donya"
    سرچاپ "طول:"، رشته.طول(متن)
    سرچاپ "بزرگ:"، رشته.بزرگ(متن)
    سرچاپ "کوچک:"، رشته.کوچک(متن)
    سرچاپ "وارونه:"، رشته.برعکس("Salam")
    سرچاپ "پیوست:"، رشته.پیوست("سلام "، "دنیا")
پایان
\end{salamfa}

\begin{progoutfa}
طول: 11
بزرگ: SALAM DONYA
کوچک: salam donya
وارونه: malaS
پیوست: سلام دنیا
\end{progoutfa}

با \code{واردسازی رشته} بسته‌ی رشته را وارد می‌کنیم و سپس با همین نام،
\code{رشته}، به تابع‌هایش دسترسی پیدا می‌کنیم. در پرونده‌ی فارسی هم نامِ بسته و
هم نامِ تابع‌ها فارسی است؛ نامِ انگلیسی (\code{str} یا \code{Upper}) را
کامپایلر در پرونده‌ی فارسی نمی‌پذیرد و تنها برای پیدا کردنِ همان تابع در
مرجعِ انگلیسی به کار می‌آید. مهم‌ترین تابع‌ها:

\begin{center}
\begin{tabular}{l l l}
\toprule
\textbf{فارسی} & \textbf{انگلیسی} & \textbf{کار} \\
\midrule
\code{رشته.طول(م)} & \code{str.Len} & طولِ رشته \\
\code{رشته.بزرگ(م)} & \code{str.Upper} & تبدیل به حروفِ بزرگ \\
\code{رشته.کوچک(م)} & \code{str.Lower} & تبدیل به حروفِ کوچک \\
\code{رشته.برعکس(م)} & \code{str.Reverse} & وارونه‌سازی \\
\code{رشته.پیوست(آ، ب)} & \code{str.Concat} & چسباندنِ دو رشته \\
\bottomrule
\end{tabular}
\end{center}

\begin{note}
دو شیوهٔ کار با رشته را دیدیم: یکی به‌صورتِ \emph{متد} (مثلِ \code{متن.طول()})
و دیگری به‌صورتِ \emph{تابعِ ماژول} (مثلِ \code{رشته.طول(متن)}). هر دو درست‌اند؛
بسته‌ی \code{رشته} تابع‌های پیشرفته‌تری (مانندِ جداسازی، جست‌وجو و جایگزینی)
هم دارد که در فصلِ کتابخانهٔ استاندارد (فصلِ ۱۵) بیشتر می‌بینیم.
\end{note}

\section{دنباله‌های گریز}

گاهی می‌خواهیم نویسه‌هایی در رشته بگذاریم که نوشتنشان مستقیم دشوار است مثلاً
خودِ نقل‌قول، یا رفتن به خطِ تازه. برای این کار از \textbf{دنباله‌های گریز}
استفاده می‌کنیم که با یک \code{\textbackslash} (بک‌اسلش) آغاز می‌شوند:

\begin{center}
\begin{tabular}{c l}
\toprule
\textbf{دنباله} & \textbf{معنا} \\
\midrule
\lr{\texttt{\textbackslash n}} & خطِ تازه \\
\lr{\texttt{\textbackslash t}} & تب (فاصلهٔ افقی) \\
\lr{\texttt{\textbackslash"}} & نقل‌قولِ دوتایی \\
\lr{\texttt{\textbackslash\textbackslash}} & خودِ بک‌اسلش \\
\bottomrule
\end{tabular}
\end{center}

برای نمونه، \code{سرچاپ "خط یک\textbackslash nخط دو"} دو خط چاپ می‌کند، چون
\lr{\texttt{\textbackslash n}} میانِ آن‌ها یک شکستِ خط می‌گذارد.

\section{کاراکتر در برابرِ رشته}

در فصلِ ۲ دیدیم که کاراکتر (\code{'A'}) و رشته (\code{"A"}) دو چیزِ متفاوت‌اند.
کاراکتر تنها \emph{یک} نویسه است و میانِ تک‌نقل‌قول می‌آید، حال‌آنکه رشته
می‌تواند هیچ، یک، یا هزاران نویسه داشته باشد و میانِ نقل‌قولِ دوتایی می‌آید.
گونه‌ی کاراکتر در سلام \code{نویسه} است و تنها یک بایت را نگه می‌دارد؛ پس
\code{'A'} درست است، اما یک حرفِ فارسی در یک \code{نویسه} جا نمی‌شود. برای
نویسه‌های یونیکد گونه‌ی \code{یونیکد} هست که با \code{u'آ'} نوشته می‌شود، و
\code{نویسه شماره(ش)} نویسه‌ی
شماره‌ی \code{ش} از یک رشته را به شکلِ یک رشته‌ی کوتاه برمی‌گرداند.
در عمل، برای کار با متن بیشتر با رشته‌ها سروکار دارید؛ کاراکتر بیشتر در کارهای
سطحِ‌پایین به کار می‌آید.

\section{مثالِ کامل: شمارشِ واژه‌ها}

بیایید آموخته‌ها را در یک برنامهٔ کوچک به کار ببندیم که نویسه‌های یک رشته را
می‌شمارد و تعدادِ فاصله‌ها را نیز می‌یابد که تقریباً برابرِ تعدادِ واژه‌هاست:

\begin{salamfa}
تابع ریشه:
    متن := "سلام دنیا و سلام بر شما"
    ناپایا فاصله := ۰
    تکرار متن.شمارنویسه() در i:
        اگر متن.زیررشته نویسه(i، ۱) == " ":
            فاصله = فاصله + ۱
        پایان
    پایان
    سرچاپ "تعداد نویسه‌ها:"، متن.شمارنویسه()
    سرچاپ "تعداد واژه‌ها:"، فاصله + ۱
پایان
\end{salamfa}

\begin{progoutfa}
تعداد نویسه‌ها: 23
تعداد واژه‌ها: 6
\end{progoutfa}

چون متن فارسی است، به‌جای \code{طول()} و \code{زیررشته} از \code{شمارنویسه()}
و \code{زیررشته نویسه} استفاده کرده‌ایم تا نویسه‌ها شمرده شوند، نه بایت‌ها.

\section{خلاصهٔ فصل}

\begin{itemize}
  \item رشته دنباله‌ای از نویسه‌هاست و از یونی‌کد پشتیبانی می‌کند.
  \item با \code{+} یا \code{.پیوست()} رشته‌ها را به هم می‌چسبانیم.
  \item \code{.طول()} طول و \code{.زیررشته(i، n)} یک تکه از رشته را می‌دهد؛
        هر دو بایت می‌شمارند. برای متنِ فارسی \code{.شمارنویسه()} و
        \code{.زیررشته نویسه(i، n)} را به کار ببرید.
  \item الگوی پیمایشِ نویسه‌به‌نویسه برای شمارش و جست‌وجو بسیار کاربردی است.
  \item بسته‌ی \code{رشته} تابع‌های آماده‌ای مانندِ \code{بزرگ}، \code{کوچک} و
        \code{برعکس} دارد.
  \item دنباله‌های گریز مانندِ \lr{\texttt{\textbackslash n}} نویسه‌های ویژه
        را در رشته می‌گنجانند.
\end{itemize}

\section{تمرین‌ها}

\exercise برنامه‌ای بنویسید که یک رشته بگیرد و بگوید چند نویسه دارد.

\exercise با کمکِ بسته‌ی \code{رشته}، یک نام را هم با حروفِ بزرگ و هم با حروفِ
کوچک چاپ کنید.

\exercise برنامه‌ای بنویسید که بشمارد حرفِ «ا» چند بار در یک متنِ فارسی آمده
است. (راهنمایی: از \code{شمارنویسه()} و \code{زیررشته نویسه} استفاده کنید.)

\exercise یک رشته را وارونه کنید (یک بار با حلقهٔ دستی و یک بار با
\code{رشته.برعکس}) و نتیجه‌ها را مقایسه کنید.

\exercise برنامه‌ای بنویسید که یک رشته را بگیرد و بررسی کند آیا «متقارن»
(palindrome) است یعنی از چپ و راست یکسان خوانده می‌شود.
