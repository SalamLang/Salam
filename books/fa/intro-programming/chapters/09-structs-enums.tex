\chapter{ساختارها و شمارشی‌ها}%
\label{ch:structs}

تا اینجا با داده‌های ساده (عدد، رشته، منطقی) و مجموعه‌های هم‌گونه (آرایه، بردار)
کار کردیم. اما دنیای واقعی پیچیده‌تر است: یک «دانشجو» هم نام دارد، هم شماره، هم
معدل. چگونه این‌ها را با هم نگه داریم؟ پاسخ، \textbf{ساختار} است. در این فصل
ساختارها و خویشاوندشان، \textbf{شمارشی‌ها}، را می‌شناسیم.

\section{ساختار: ساختنِ گونهِ دلخواهِ خود}

ساختار به شما اجازه می‌دهد چند داده را زیرِ یک نام گرد آورید و یک \emph{گونهِ
تازه} بسازید. به این مثالِ واقعی نگاه کنید که یک «مستطیل» می‌سازد:

\begin{salamfa}
ساختار مستطیل:
    همگانی طول: اعشار۶۴ = ۰.۰
    همگانی عرض: اعشار۶۴ = ۰.۰
    همگانی تابع مساحت(): اعشار۶۴:
        برگشت این.طول * این.عرض
    پایان
    همگانی تابع محیط(): اعشار۶۴:
        برگشت ۲.۰ * (این.طول + این.عرض)
    پایان
پایان

تابع ریشه:
    ر := مستطیل { طول = ۴.۰، عرض = ۳.۰ }
    سرچاپ "مساحت:"، ر.مساحت()
    سرچاپ "محیط:"، ر.محیط()
پایان
\end{salamfa}

این مثال چند مفهومِ تازه دارد که یکی‌یکی می‌شکافیم.

\subsection{میدان‌ها (fields)}

\begin{salamfa}
ساختار مستطیل:
    همگانی طول: اعشار۶۴ = ۰.۰
    همگانی عرض: اعشار۶۴ = ۰.۰
پایان
\end{salamfa}

\code{طول} و \code{عرض} \textbf{میدان}های ساختار هستند یعنی داده‌هایی که هر
مستطیل نگه می‌دارد. هر میدان نامی، گونه‌ای و یک مقدارِ پیش‌فرضِ \emph{اختیاری} دارد
و به شکلِ \code{نام: گونه = پیش‌فرض} نوشته می‌شود. این تنها جایی (همراه با
پارامترهای تابع) است که نامِ گونه پس از دونقطه می‌آید؛ متغیرهای معمولی همیشه با
\code{:=} ساخته می‌شوند. مقدارِ پیش‌فرض (\code{= ۰.۰}) وقتی به کار می‌آید که
هنگامِ ساختنِ مستطیل، مقداری برای آن میدان ندهید.

واژه‌ی \code{همگانی} (\lr{pub}) پیش از میدان یعنی کدِ بیرون از ساختار هم
می‌تواند آن را بخواند و بنویسد. میدان‌ها و متدها به‌طورِ پیش‌فرض \emph{خصوصی}اند
و تنها متدهای خودِ ساختار به آن‌ها دسترسی دارند؛ اگر در \code{ریشه} بخواهید
میدانی خصوصی را بخوانید، کامپایلر خطای \code{E017} می‌دهد. چون در این مثال
\code{ریشه} میدان‌ها را مستقیم به کار نمی‌برد، می‌شد آن‌ها را خصوصی هم نگه داشت،
اما متدهای \code{مساحت} و \code{محیط} حتماً باید \code{همگانی} باشند.

مقدارِ پیش‌فرض اجباری نیست. میدانی که بدونِ \code{= مقدار} نوشته شود، یک میدانِ
\textbf{الزامی} است: هر نمونه باید آن را تعیین کند وگرنه کامپایلر خطا می‌دهد. به
این ترتیب داده‌ای را که ساختار بدونِ آن بی‌معناست الزامی می‌کنید و بقیه را به
پیش‌فرضِ معقول می‌سپارید:

\begin{salamfa}
ساختار کاربر:
    همگانی نام: رشته                 // الزامی، بدونِ پیش‌فرض
    همگانی نقش: رشته = "عضو"         // اختیاری، پیش‌فرض «عضو»
    همگانی فعال: منطقی = درست
پایان

تابع ریشه:
    کا := کاربر { نام = "سارا" }
    مدیر := کاربر { نام = "مکس"، نقش = "مدیر" }
    سرچاپ کا.نام، کا.نقش، کا.فعال
    سرچاپ مدیر.نام، مدیر.نقش، مدیر.فعال
پایان
\end{salamfa}

\begin{progoutfa}
سارا عضو true
مکس مدیر true
\end{progoutfa}

نوشتنِ \code{کاربر \{\}} اینجا خطای زمانِ کامپایل است، چون میدانِ الزامیِ
\code{نام} داده نشده:

\begin{progoutfa}
خطا[خطا۰۲۸]: مقدار ساختاری 'کاربر' به فیلد 'نام' نیاز دارد
\end{progoutfa}

میدان‌های دارای پیش‌فرض را همیشه می‌توان ننوشت.

\subsection{ساختنِ یک نمونه}

\begin{salamfa}
ر := مستطیل { طول = ۴.۰، عرض = ۳.۰ }
\end{salamfa}

این خط یک \textbf{نمونه} (instance) از مستطیل می‌سازد. نحوِ \code{مستطیل \{ ... \}}
یک مستطیلِ تازه با میدان‌های مشخص‌شده می‌سازد. حالا \code{ر} یک مستطیلِ واقعی
با طولِ ۴ و عرضِ ۳ است. میدان‌ها با ویرگول (\code{،} یا \code{,}) از هم جدا
می‌شوند.

\subsection{متدها: تابع‌های درونِ ساختار}

\begin{salamfa}
ساختار مستطیل:
    همگانی طول: اعشار۶۴ = ۰.۰
    همگانی عرض: اعشار۶۴ = ۰.۰
    همگانی تابع مساحت(): اعشار۶۴:
        برگشت این.طول * این.عرض
    پایان
پایان
\end{salamfa}

تابعی که \emph{درونِ} یک ساختار تعریف شود، \textbf{متد} نام دارد. متدها روی
داده‌های همان ساختار کار می‌کنند. واژهٔ کلیدیِ \code{این} (معادلِ \code{this})
به \emph{همان نمونه‌ای} اشاره می‌کند که متد رویش صدا زده شده. پس
\code{این.طول} یعنی «طولِ همین مستطیل».

متدها را با نقطه پس از نامِ نمونه صدا می‌زنیم: \code{ر.مساحت()}. این دقیقاً
همان شیوه‌ای است که در فصلِ بردارها برای \code{.بیفزا()} و \code{.طول()}
دیدیم و حالا می‌دانید که آن‌ها هم متد بودند.

\section{تغییرِ داده‌ها از درونِ متد}

متدها فقط نمی‌خوانند؛ می‌توانند داده‌های ساختار را \emph{تغییر} هم بدهند. این
مثالِ واقعی یک «شمارنده» می‌سازد:

\begin{salamfa}
ساختار شمارنده:
    مقدار: صحیح = ۰
    همگانی ناپایا تابع افزایش():
        این.مقدار = این.مقدار + ۱
    پایان
    همگانی تابع بگیر(): صحیح:
        برگشت این.مقدار
    پایان
پایان

تابع ریشه:
    ناپایا ش := شمارنده { مقدار = ۵ }
    ش.افزایش()
    ش.افزایش()
    سرچاپ ش.بگیر()
پایان
\end{salamfa}

متدِ \code{افزایش} مقدارِ درونیِ شمارنده را یکی زیاد می‌کند. پس از دو بار صدا
زدن، مقدار از ۵ به ۷ می‌رسد و خروجی \code{7} می‌شود.

به دو نکته دقت کنید. نخست، متدی که میدانی از \code{این} را تغییر می‌دهد با
\code{ناپایا تابع} تعریف می‌شود. وقتی ساختاری دست‌کم یک متدِ ناپایا داشته باشد،
کامپایلر همه‌ی متدهایش را می‌سنجد: متدِ معمولی‌ای که به میدانی از \code{این} مقدار
بدهد خطای \code{E105} می‌گیرد. دوم، متدِ ناپایا را تنها روی متغیری می‌توان صدا
زد که خودش \code{ناپایا} باشد (وگرنه خطای \code{E106})، پس \code{ش} با
\code{ناپایا ش :=} ساخته شده است.
با خواندنِ خطِ اولِ هر متد می‌فهمید که نمونه را تغییر می‌دهد یا نه. میدانِ
\code{مقدار} خصوصی مانده، چون بیرون از ساختار تنها از راهِ متدها به آن دسترسی
داریم. این الگو نگه‌داشتنِ
داده در ساختار و تغییرِ آن از راهِ متدها بنیادِ برنامه‌نویسیِ شیءگراست.

\section{مثالِ کامل: حسابِ بانکی}

این مثالِ واقعی همهٔ آموخته‌ها را گرد می‌آورد و یک حسابِ بانکیِ ساده می‌سازد که
واریز و برداشت دارد:

\begin{salamfa}
ساختار حساب:
    همگانی موجودی: صحیح = ۰
    همگانی ناپایا تابع واریز(مبلغ: صحیح): این.موجودی = این.موجودی + مبلغ پایان
    همگانی ناپایا تابع برداشت(مبلغ: صحیح):
        اگر مبلغ <= این.موجودی: این.موجودی = این.موجودی - مبلغ
        وگرنه:                   سرچاپ "موجودی کافی نیست"
        پایان
    پایان
پایان

تابع ریشه:
    ناپایا ح := حساب { موجودی = ۱۰۰ }
    ح.واریز(۵۰)
    ح.برداشت(۳۰)
    ح.برداشت(۲۰۰)
    سرچاپ "موجودی ="، ح.موجودی
پایان
\end{salamfa}

دقت کنید که \code{واریز} و \code{برداشت} هر دو \code{ناپایا تابع}اند، چون
موجودی را تغییر می‌دهند، و متدِ \code{برداشت} پیش از کم‌کردن، بررسی می‌کند که موجودی کافی
باشد. این یک قاعدهٔ مهم در طراحیِ ساختارهاست: \emph{متدها از درستیِ داده‌ها
پاسداری می‌کنند.} برداشتِ ۲۰۰ از حسابی که فقط ۱۲۰ دارد، رد می‌شود و خروجیِ
نهایی \code{موجودی = 120} است.

\begin{tip}
به دو شیوهٔ دسترسی به میدان دقت کنید: از \emph{بیرونِ} ساختار با نامِ نمونه
(\code{ح.موجودی})، و از \emph{درونِ} متد با \code{این} (\code{این.موجودی}).
هر دو به همان داده اشاره می‌کنند.
\end{tip}

\section{عملگرها برای ساختارها}

در فصلِ ۳ به‌اختصار دیدیم که می‌توان عملگرها را برای ساختارها تعریف کرد. اکنون
که ساختارها را می‌شناسید، این مثالِ کامل‌تر روشن‌تر است:

\begin{salamfa}
ساختار بردار:
    همگانی ایکس: صحیح = ۰
    همگانی وای: صحیح = ۰
    همگانی تابع +(دیگری: بردار): بردار:
        برگشت بردار { ایکس = این.ایکس + دیگری.ایکس، وای = این.وای + دیگری.وای }
    پایان
    همگانی تابع ==(دیگری: بردار): منطقی:
        برگشت (این.ایکس == دیگری.ایکس) و (این.وای == دیگری.وای)
    پایان
پایان

تابع ریشه:
    الف := بردار { ایکس = ۱، وای = ۲ }
    ب := بردار { ایکس = ۳، وای = ۴ }
    ج := الف + ب
    سرچاپ ج.ایکس، ج.وای
    سرچاپ الف == ب
    سرچاپ ج == بردار { ایکس = ۴، وای = ۶ }
پایان
\end{salamfa}

با تعریفِ \code{همگانی تابع +} و \code{همگانی تابع ==}، حالا \code{الف + ب} و
\code{الف == ب} روی بردارها معنا پیدا می‌کنند. خروجی به ترتیب: \code{4 6}،
\code{false} و \code{true}.

\begin{note}
سلام بسیاری عملگر را برای ساختارها می‌پذیرد: حسابی (\code{+ - * / \% **})،
مقایسه‌ای (\code{== != < > <= >=})، یکانی (\code{-} و \code{وارونه}) و حتی اندیس‌گذاری
(\code{[]}). اگر \code{==} را تعریف کنید اما \code{!=} را نه، سلام به‌طورِ
خودکار \code{!=} را از روی \code{==} می‌سازد. در واقع دو ساختار را بی هیچ کدِ
اضافه‌ای هم می‌شود با \code{==} میدان‌به‌میدان مقایسه کرد؛ تعریفِ \code{==}
تنها وقتی لازم است که برابری معنای دیگری داشته باشد.
\end{note}

\section{شمارشی: مجموعه‌ای از نام‌های ثابت}

گاهی متغیری داریم که فقط چند مقدارِ مشخص می‌تواند بگیرد: روزهای هفته، جهت‌های
اصلی، رنگ‌های چراغِ راهنما. برای این کار، \textbf{شمارشی} (enum) را داریم. به
این مثالِ واقعی نگاه کنید:

\begin{salamfa}
جداشمار رنگ:
    قرمز
    سبز = ۵
    آبی
پایان

تابع ریشه:
    سرچاپ رنگ.قرمز برگردان صحیح
    سرچاپ رنگ.سبز برگردان صحیح
    سرچاپ رنگ.آبی برگردان صحیح
پایان
\end{salamfa}

\code{جداشمار رنگ} یک گونهِ تازه می‌سازد که سه مقدارِ ممکن دارد: \code{قرمز}،
\code{سبز} و \code{آبی}. هر عضوِ شمارشی در پسِ پرده یک عددِ صحیح است که از صفر
آغاز می‌شود. اما می‌توانید مقدارِ دلخواه هم بدهید: اینجا \code{سبز = 5} گذاشته‌ایم،
پس \code{قرمز} برابرِ ۰، \code{سبز} برابرِ ۵ و \code{آبی} برابرِ ۶ (یکی پس از
سبز) می‌شود. با \code{برگردان صحیح} می‌توان عددِ پشتِ هر عضو را دید.

\subsection{شکلِ یک‌خطی و کاربردِ واقعی}

شمارشی‌ها را می‌توان فشرده‌تر هم نوشت و عضوها را با ویرگول از هم جدا کرد. این
مثالِ واقعیِ «روزهای هفته» شکلِ یک‌خطی را نشان می‌دهد و شمارشی را در یک تابع به
کار می‌برد:

\begin{salamfa}
جداشمار روز: دوشنبه، سه‌شنبه، چهارشنبه، پنجشنبه، جمعه، شنبه، یکشنبه پایان

تابع آخرهفته است(ر: روز): منطقی: برگشت (ر == روز.شنبه) یا (ر == روز.یکشنبه) پایان

تابع ریشه:
    ر := روز.شنبه
    سرچاپ "شماره‌ی روز ="، ر برگردان صحیح
    اگر آخرهفته است(ر): سرچاپ "آخر هفته است"
    وگرنه:               سرچاپ "روز کاری است"
    پایان
پایان
\end{salamfa}

اینجا تابعِ \code{آخرهفته است} یک \code{روز} می‌گیرد و با مقایسهٔ آن با
\code{روز.شنبه} و \code{روز.یکشنبه}، تصمیم می‌گیرد. شمارشی‌ها کد را
\emph{خواناتر} و \emph{ایمن‌تر} می‌کنند: به‌جای عددهای جادوییِ بی‌معنا (۵، ۶)،
نام‌های گویا (\code{شنبه}، \code{یکشنبه}) به کار می‌بریم.

\begin{tip}
هر جا متغیری دارید که فقط چند حالتِ مشخص می‌گیرد وضعیتِ سفارش (در انتظار،
ارسال‌شده، تحویل‌شده)، گونهِ کاربر (مهمان، عضو، مدیر) یک شمارشی بسازید. این
کار از خطاهای فراوان جلوگیری می‌کند، چون کامپایلر نمی‌گذارد مقداری بیرون از
فهرست به آن متغیر بدهید.
\end{tip}

\section{خلاصهٔ فصل}

\begin{itemize}
  \item ساختار چند داده را زیرِ یک نام گرد می‌آورد و گونه‌ای تازه می‌سازد.
  \item میدان‌ها داده‌های ساختارند و می‌توانند مقدارِ پیش‌فرضِ اختیاری داشته
        باشند؛ میدانِ بدونِ پیش‌فرض \emph{الزامی} است و باید در هر نمونه داده شود.
  \item متدها تابع‌های درونِ ساختارند و با \code{این} به نمونهٔ جاری دسترسی
        دارند. متدی که میدان‌ها را تغییر می‌دهد \code{ناپایا تابع} است.
  \item میدان‌ها و متدها به‌طورِ پیش‌فرض خصوصی‌اند؛ \code{همگانی} آن‌ها را
        بیرون از ساختار در دسترس می‌گذارد.
  \item نمونه را با نحوِ \code{نام \{ میدان = مقدار \}} می‌سازیم.
  \item عملگرها را می‌توان برای ساختارها تعریف کرد.
  \item شمارشی مجموعه‌ای از نام‌های ثابت است که کد را خوانا و ایمن می‌کند.
\end{itemize}

\section{تمرین‌ها}

\exercise ساختاری به نامِ \code{دایره} با میدانِ شعاع بسازید و متدهای
\code{مساحت} و \code{محیط} را برایش بنویسید (از $\pi \approx 3.1416$ استفاده
کنید).

\exercise ساختارِ \code{دانشجو} با میدان‌های نام، شماره و معدل بسازید و متدی
بنویسید که بگوید دانشجو مشروط است یا نه (معدلِ زیرِ ۱۲).
