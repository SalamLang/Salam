\chapter{میانجی‌ها و چندریختی}%
\label{ch:interfaces}

در فصلِ ۹ دیدیم که هر ساختار می‌تواند متدهای خود را داشته باشد. اما گاهی چند
ساختارِ متفاوت، \emph{رفتارِ} مشترکی دارند: هم مربع و هم دایره «مساحت» دارند، هر
چند فرمولشان فرق می‌کند. چگونه می‌توان با همهٔ آن‌ها به یک شیوهٔ یکسان کار کرد؟
پاسخ، \textbf{میانجی} است.

\section{مسئله: رفتارِ مشترک، پیاده‌سازیِ متفاوت}

فرض کنید چند شکلِ هندسی دارید و می‌خواهید تابعی بنویسید که مساحتِ \emph{هر}
شکلی را چاپ کند بی‌آنکه برای هر شکل، تابعی جدا بنویسید. مشکل اینجاست که مربع
و مستطیل دو گونهِ \emph{متفاوت}اند؛ یک تابع چگونه می‌تواند هر دو را بپذیرد؟

\textbf{میانجی} (interface) راهِ حل است: قراردادی که می‌گوید «هر گونه‌ای که این
متدها را داشته باشد، یک شکل به‌شمار می‌آید.»

\section{تعریف و پیاده‌سازیِ میانجی}

به این مثالِ واقعی نگاه کنید که قراردادِ «شکل» را تعریف و سپس دو گونه را با آن
هماهنگ می‌کند:

\begin{salamfa}
میانجی شکل:
    تابع مساحت(): صحیح
    تابع نام(): رشته
پایان

ساختار مربع:
    ضلع: صحیح = ۰
    همگانی تابع مساحت(): صحیح: برگشت این.ضلع * این.ضلع پایان
    همگانی تابع نام(): رشته: برگشت "مربع" پایان
پایان

ساختار مستطیل:
    طول: صحیح = ۰
    عرض: صحیح = ۰
    همگانی تابع مساحت(): صحیح: برگشت این.طول * این.عرض پایان
    همگانی تابع نام(): رشته: برگشت "مستطیل" پایان
پایان

تابع توصیف(ش: پویا شکل):
    سرچاپ ش.نام()، "->"، ش.مساحت()
پایان

تابع ریشه:
    توصیف(مربع { ضلع = ۵ })
    توصیف(مستطیل { طول = ۴، عرض = ۶ })
پایان
\end{salamfa}

این مثال چند بخشِ کلیدی دارد:

\subsection{اعلانِ میانجی}

\begin{salamfa}
میانجی شکل:
    تابع مساحت(): صحیح
    تابع نام(): رشته
پایان
\end{salamfa}

\code{میانجی شکل} می‌گوید: «هر چیزی که بخواهد یک \emph{شکل} باشد، باید دو متد
داشته باشد: \code{مساحت} که یک عدد می‌دهد، و \code{نام} که یک رشته می‌دهد.»
دقت کنید که در میانجی فقط \emph{امضا}ی متدها آمده، نه بدنه‌شان. میانجی می‌گوید
\emph{چه} رفتاری لازم است، نه \emph{چگونه}.

\subsection{پیاده‌سازیِ میانجی}

\code{مربع} و \code{مستطیل} هر دو متدهای \code{مساحت} و \code{نام} را دارند، پس
هر دو میانجیِ \code{شکل} را برآورده می‌کنند. در سلام لازم نیست صریحاً بنویسید
«مربع از شکل پیروی می‌کند»؛ همین که متدهای لازم را داشته باشد، کافی است.
تنها شرط این است که این متدها \code{همگانی} باشند، چون کدِ بیرون از ساختار
از راهِ میانجی صدایشان می‌زند. میدان‌ها (\code{ضلع}، \code{طول} و \code{عرض})
خصوصی مانده‌اند، چون تنها متدهای خودِ ساختار آن‌ها را می‌خوانند.

اگر گونه‌ای یکی از متدها را نداشته باشد، کامپایلر همان‌جا که مقدار به‌جای
میانجی به کار رفته خطا می‌دهد؛ برای نمونه:

\begin{progoutfa}
خطا[خطا۰۰۲]: 'مربع' میانجی 'شکل' را پیاده‌سازی نمی‌کند: متد 'نام' موجود نیست
\end{progoutfa}

\subsection{بدنه‌ی پیش‌فرض و کاربست}

متدِ میانجی می‌تواند بدنه‌ی پیش‌فرض هم داشته باشد. ساختاری که آن متد را ننویسد،
نسخه‌ی پیش‌فرض را می‌گیرد و ساختاری که بنویسد، نسخه‌ی خودش را نگه می‌دارد. با
\code{کاربست میانجی بر گونه} هم می‌توان یک میانجی را برای گونه‌ای که خودتان
ننوشته‌اید پیاده کرد، حتی گونه‌های پایه مانندِ \code{صحیح} و \code{رشته}:

\begin{salamfa}
میانجی رتبه‌دار:
    تابع رتبه(): صحیح
    تابع شرح(): رشته:
        برگشت "رتبه " + این.رتبه()
    پایان
پایان

کاربست رتبه‌دار بر رشته:
    تابع رتبه(): صحیح: برگشت طول(این) پایان
پایان

تابع ریشه:
    سرچاپ "salam".رتبه()
    سرچاپ "salam".شرح()
پایان
\end{salamfa}

\begin{progoutfa}
5
رتبه 5
\end{progoutfa}

درونِ متدهای کاربست، \code{این} خودِ مقدار است؛ اینجا همان رشته. \code{شرح}
را ننوشته‌ایم، پس رشته نسخه‌ی پیش‌فرضِ میانجی را می‌گیرد.

\subsection{استفادهٔ همگانی با «پویا»}

\begin{salamfa}
میانجی شکل:
    تابع مساحت(): صحیح
    تابع نام(): رشته
پایان

تابع توصیف(ش: پویا شکل):
    سرچاپ ش.نام()، "->"، ش.مساحت()
پایان
\end{salamfa}

اینجا جادو رخ می‌دهد. تابعِ \code{توصیف} یک پارامتر از گونهِ \code{پویا شکل}
می‌گیرد یعنی «هر چیزی که یک شکل باشد». حالا می‌توانیم هم مربع و هم مستطیل را
به آن بدهیم و هر کدام متدِ خودش را اجرا می‌کند. خروجی:

\begin{progoutfa}
مربع -> 25
مستطیل -> 24
\end{progoutfa}

به این توانایی اینکه یک تابع با گونه‌های گوناگون کار کند و هر گونه رفتارِ خود
را نشان دهد \textbf{چندریختی} (polymorphism) می‌گویند؛ واژه‌ای یونانی به
معنای «چند شکلی».

\begin{deep}[نگاهی به درون: «پویا» چگونه کار می‌کند]
وقتی یک مقدار را به‌صورتِ \code{پویا شکل} می‌فرستید، سلام در پسِ پرده یک
«اشاره‌گرِ چاق» (fat pointer) می‌سازد: یک اشاره به خودِ داده، و یک اشاره به
جدولی از متدهای آن گونه (که به آن \emph{جدولِ مجازی} یا vtable می‌گویند). وقتی
\code{ش.مساحت()} را صدا می‌زنید، سلام در زمانِ اجرا از این جدول، متدِ درستِ
همان گونه را پیدا و اجرا می‌کند. نامِ \code{پویا} از همین‌جاست: به این کار
«ارسالِ پویا» (dynamic dispatch) می‌گویند؛ تصمیمِ اینکه کدام متد اجرا شود،
در \emph{زمانِ اجرا} گرفته می‌شود، نه زمانِ کامپایل.
\end{deep}

\section{چرا میانجی‌ها ارزشمندند؟}

تصور کنید فردا بخواهید شکلِ تازه‌ای مثلاً «مثلث» بیفزایید. کافی است یک
ساختارِ \code{مثلث} با متدهای \code{مساحت} و \code{نام} بسازید. تابعِ
\code{توصیف} \emph{بدونِ هیچ تغییری} با مثلث هم کار می‌کند! این یعنی کدِ شما
\textbf{گسترش‌پذیر} است: می‌توانید گونه‌های تازه بیفزایید بی‌آنکه کدِ موجود را
دست بزنید.

\begin{tip}
میانجی‌ها به شما اجازه می‌دهند کد را بر پایهٔ \emph{رفتار} بنویسید، نه بر پایهٔ
گونهِ مشخص. این یکی از مهم‌ترین ابزارهای طراحیِ نرم‌افزارِ بزرگ و قابلِ نگه‌داری
است. هر جا دیدید که یک تابع را برای چند گونهِ مشابه تکرار می‌کنید، احتمالاً یک
میانجی می‌تواند آن تکرار را از میان بردارد.
\end{tip}

\section{میانجی در برابرِ کلی‌سازی}

سلام دو راه برای نوشتنِ کدی که با گونه‌های گوناگون کار کند دارد: \emph{میانجی}
(این فصل) و \emph{کلی‌سازی} (فصلِ بعد). تفاوتشان ظریف اما مهم است:

\begin{itemize}
  \item \textbf{میانجی با \code{پویا}:} انعطاف‌پذیر است؛ می‌توانید گونه‌های مختلف
        را در یک بردار (\code{وکتور<پویا شکل>}) بریزید و در زمانِ اجرا تصمیم
        بگیرید. اما اندکی هزینهٔ
        اجرایی دارد (به‌خاطرِ همان جدولِ مجازی).
  \item \textbf{کلی‌سازی:} در زمانِ کامپایل به کدِ مخصوصِ هر گونه تبدیل می‌شود،
        پس هیچ هزینهٔ اجرایی ندارد و سریع‌تر است؛ اما کمی انعطافِ کمتری دارد.
\end{itemize}

در فصلِ بعد کلی‌سازی را خواهیم دید و آن‌گاه می‌توانید آگاهانه میانِ این دو
انتخاب کنید.

\section{خلاصهٔ فصل}

\begin{itemize}
  \item میانجی قراردادی است که می‌گوید یک گونه چه متدهایی باید داشته باشد.
  \item هر ساختاری که متدهای لازم را به شکلِ \code{همگانی} داشته باشد، میانجی را
        برآورده می‌کند، بی‌نیاز به اعلانِ صریح.
  \item متدِ میانجی می‌تواند بدنه‌ی پیش‌فرض داشته باشد، و \code{کاربست ... بر}
        میانجی را به گونه‌ای موجود (حتی \code{صحیح} و \code{رشته}) می‌دهد.
  \item با \code{پویا میانجی} می‌توان تابعی نوشت که با هر گونهِ سازگار کار کند.
  \item چندریختی یعنی یک تابع با گونه‌های گوناگون کار کند و هر گونه رفتارِ خود را
        نشان دهد.
  \item میانجی‌ها کد را گسترش‌پذیر می‌کنند.
\end{itemize}

\section{تمرین‌ها}

\exercise میانجیِ \code{شکل} را گسترش دهید و یک ساختارِ \code{دایره} بیفزایید که
آن را برآورده کند، سپس آن را به \code{توصیف} بدهید.

\exercise میانجی‌ای به نامِ \code{جاندار} با متدِ \code{تابع صدا(): رشته} بسازید و
ساختارهای \code{گربه} و \code{سگ} را برایش پیاده کنید.

\exercise تابعی بنویسید که یک \code{پویا شکل} بگیرد و بگوید مساحتش از ۱۰۰ بیشتر
است یا نه.

\exercise میانجی‌ای به نامِ \code{قابل‌چاپ} با متدِ \code{تابع به‌رشته(): رشته} بسازید و
دو گونهِ متفاوت را برایش پیاده کنید.
