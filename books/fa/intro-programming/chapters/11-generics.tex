\chapter{کلی‌سازی}%
\label{ch:generics}

تصور کنید کارکردی نوشته‌اید که بزرگ‌ترِ دو عددِ صحیح را برمی‌گرداند. حالا همان را
برای اعدادِ اعشاری هم می‌خواهید. آیا باید کارکرد را دوباره با همان منطق و فقط
ریختِ متفاوت بنویسید؟ خوشبختانه نه. \textbf{کلی‌سازی} (generics) به ما اجازه
می‌دهد یک‌بار کدی بنویسیم که با \emph{هر} ریختی کار کند.

\section{مسئله: تکرارِ یک منطق برای ریخت‌های گوناگون}

بدونِ کلی‌سازی، ناچار بودیم چنین بنویسیم:

\begin{salam}
کارکرد بیشینه صحیح(الف : صحیح, ب : صحیح) : صحیح:
    اگر الف > ب: بازگشت الف پایان
    بازگشت ب
پایان

کارکرد بیشینه اعشاری(الف : اعشار۶۴, ب : اعشار۶۴) : اعشار۶۴:
    اگر الف > ب: بازگشت الف پایان
    بازگشت ب
پایان
\end{salam}

این دو کارکرد \emph{منطقِ یکسانی} دارند و تنها در ریخت فرق می‌کنند. این تکرار هم
خسته‌کننده است و هم خطاخیز: اگر روزی منطق را اصلاح کنید، باید در هر دو جا تغییر
دهید.

\section{راهِ حل: پارامترِ ریخت}

با کلی‌سازی، ریخت را هم مانندِ یک پارامتر می‌گیریم. این مثالِ واقعی را ببینید:

\begin{salam}
کارکرد بیشینه<ت>(الف : ت, ب : ت) : ت:
    اگر الف > ب:
        بازگشت الف
    پایان
    بازگشت ب
پایان
ساختار جعبه<ت>:
    مقدار : ت
پایان

کارکرد آغازین:
    چاپ بیشینه(3, 9)
    چاپ بیشینه(2.5, 1.5)
    گذرا ج : جعبه<صحیح> = جعبه { مقدار = 42 }
    چاپ ج.مقدار
پایان
\end{salam}

به \code{بیشینه<ت>} دقت کنید. آن \code{ت} درونِ \code{<\,>} یک \textbf{پارامترِ
ریخت} است یک ریختِ «جای‌خالی» که هنوز مشخص نیست. سپس در امضای کارکرد،
پارامترها و خروجی همگی از همان ریختِ \code{ت} هستند. معنایش این است: «هر دو
ورودی و خروجی، از یک ریختِ یکسان‌اند هر ریختی که باشد.»

حالا وقتی می‌نویسیم:
\begin{itemize}
  \item \code{بیشینه(3, 9)}: سلام می‌بیند که ورودی‌ها صحیح‌اند، پس \code{ت} را
        \code{صحیح} می‌گیرد.
  \item \code{بیشینه(2.5, 1.5)}: ورودی‌ها اعشاری‌اند، پس \code{ت} را
        \code{اعشاری} می‌گیرد.
\end{itemize}

یک کارکرد، دو کاربرد بی‌هیچ تکراری!

\begin{note}
دقت کنید که هنگامِ \emph{صدا زدنِ} \code{بیشینه}، لازم نبود ریخت را بنویسیم؛ سلام
آن را از روی ورودی‌ها \emph{حدس} زد. این را «استنتاجِ ریخت» می‌گویند و کد را
تمیزتر می‌کند.
\end{note}

\section{ساختارهای کلی}

نه‌تنها کارکردها، بلکه \emph{ساختارها} هم می‌توانند کلی باشند. در مثالِ بالا:

\begin{salam}
ساختار جعبه<ت>:
    مقدار : ت
پایان
\end{salam}

\code{جعبه<ت>} یک ساختارِ کلی است: جعبه‌ای که می‌تواند \emph{هر} ریختی را در خود
نگه دارد. وقتی می‌نویسیم \code{جعبه<صحیح>}، یک جعبهٔ مخصوصِ اعدادِ صحیح
می‌سازیم. می‌توانستیم \code{جعبه<رشته>} هم بسازیم تا متن نگه دارد.

\begin{tip}
بردار (\code{وکتور<T>}) و نگاشت (\code{نگاشت<K, V>}) که در فصل‌های پیش دیدیم،
خودشان ساختارهای کلی‌اند! به همین دلیل می‌توانستیم هم \code{وکتور<صحیح>} و هم
\code{وکتور<رشته>} بسازیم. اکنون می‌دانید که در پسِ این انعطاف، همین کلی‌سازی
نهفته است.
\end{tip}

\section{کلی‌سازی با چند پارامترِ ریخت}

یک ساختار یا کارکرد می‌تواند بیش از یک پارامترِ ریخت داشته باشد. برای نمونه، یک
«جفت» که دو مقدارِ احتمالاً متفاوت را نگه می‌دارد:

\begin{salam}
ساختار جفت<آ, ب>:
    اول : آ
    دوم : ب
پایان

کارکرد آغازین:
    گذرا ج : جفت<رشته, صحیح> = جفت { اول = "سن", دوم = 30 }
    چاپ ج.اول, ج.دوم
پایان
\end{salam}

اینجا \code{آ} و \code{ب} دو پارامترِ ریختِ مستقل‌اند؛ پس می‌توان جفتی ساخت که
عضوِ نخستش رشته و عضوِ دومش عدد باشد. همان‌گونه که \code{نگاشت<K, V>} از کلید و
مقدارِ متفاوت بهره می‌برد.

\section{چرا کلی‌سازی سریع است؟}

\begin{deep}[نگاهی به درون: تک‌ریخت‌سازی]
سلام کلی‌سازی را با روشی به نامِ \emph{تک‌ریخت‌سازی} (monomorphization) پیاده
می‌کند. وقتی شما \code{بیشینه(3, 9)} و \code{بیشینه(2.5, 1.5)} را می‌نویسید،
سلام در زمانِ کامپایل \emph{دو نسخهٔ جداگانه} از کارکرد می‌سازد: یکی مخصوصِ صحیح و
یکی مخصوصِ اعشاری. یعنی همان کدی که در آغازِ این فصل دستی نوشتیم، این بار به‌طورِ
خودکار ساخته می‌شود.

نتیجه این است که کلی‌سازی در سلام \emph{هیچ هزینهٔ اجرایی} ندارد: کدِ نهایی
دقیقاً به سرعتِ کدی است که برای هر ریخت جدا نوشته باشید. شما راحتیِ نوشتنِ یک
کارکرد را دارید و رایانه سرعتِ چند کارکردِ تخصصی را. این یکی از زیباترین معامله‌های
برنده-برنده در طراحیِ زبان است.
\end{deep}

\section{کلی‌سازی در برابرِ میانجی: کدام را برگزینیم؟}

اکنون که هر دو ابزار را دیدید، این راهنمای ساده کمکتان می‌کند:

\begin{center}
\begin{tabular}{l l}
\toprule
\textbf{کلی‌سازی را برگزینید اگر\ldots} & \textbf{میانجیِ \code{dyn} را برگزینید اگر\ldots} \\
\midrule
سرعت بسیار مهم است & به انعطافِ زمانِ اجرا نیاز دارید \\
ریخت‌ها در زمانِ کامپایل معلوم‌اند & می‌خواهید ریخت‌های مختلف را در یک بردار بریزید \\
منطقِ یکسان برای ریخت‌های گوناگون & رفتارِ مشترک با پیاده‌سازیِ متفاوت \\
\bottomrule
\end{tabular}
\end{center}

\begin{tip}
قاعدهٔ سرانگشتی: اگر شک دارید، کلی‌سازی را برگزینید. در بیشترِ موارد سریع‌تر و
ایمن‌تر است. تنها وقتی به \code{dyn} روی بیاورید که واقعاً به ذخیرهٔ ریخت‌های
متفاوت در کنارِ هم یا تصمیم‌گیریِ زمانِ اجرا نیاز داشته باشید.
\end{tip}

\section{خلاصهٔ فصل}

\begin{itemize}
  \item کلی‌سازی به ما اجازه می‌دهد یک‌بار کدی بنویسیم که با هر ریختی کار کند.
  \item پارامترِ ریخت (مثلِ \code{ت}) درونِ \code{<\,>} نوشته می‌شود.
  \item هم کارکردها و هم ساختارها می‌توانند کلی باشند.
  \item می‌توان چند پارامترِ ریخت داشت: \code{جفت<آ, ب>}.
  \item سلام ریخت را از روی ورودی‌ها استنتاج می‌کند.
  \item کلی‌سازی با تک‌ریخت‌سازی پیاده می‌شود و هیچ هزینهٔ اجرایی ندارد.
\end{itemize}

\section{تمرین‌ها}

\exercise کارکردِ کلیِ \code{کمینه<ت>} را بنویسید که کوچک‌ترِ دو مقدار را
برگرداند.

\exercise کارکردِ کلیِ \code{جابجا<ت>} را بنویسید که\ldots (راهنمایی: به مفهومِ
ارجاع در فصلِ ۱۳ نیاز دارید؛ فعلاً دو مقدار بگیرید و آن‌ها را در یک \code{جفت}ِ
جابه‌جا‌شده برگردانید).

\exercise ساختارِ کلیِ \code{پشته<ت>} را با متدهای \code{بیفزا} و
\code{دربیاور} بسازید (می‌توانید درونش از یک بردار استفاده کنید).

\exercise ساختارِ \code{جفت<آ, ب>} را گسترش دهید و متدی بیفزایید که جای دو عضو
را عوض کند و یک \code{جفت<ب, آ>} برگرداند.

\exercise توضیح دهید چرا برای ساختنِ یک «فهرستِ نام‌ها و سن‌ها»، \code{نگاشت}
بهتر از دو بردارِ جداگانه است.
