\chapter{کلی‌سازی}%
\label{ch:generics}

تصور کنید تابعی نوشته‌اید که بزرگ‌ترِ دو عددِ صحیح را برمی‌گرداند. حالا همان را
برای اعدادِ اعشاری هم می‌خواهید. آیا باید تابع را دوباره با همان منطق و فقط
گونهِ متفاوت بنویسید؟ خوشبختانه نه. \textbf{کلی‌سازی} (generics) به ما اجازه
می‌دهد یک‌بار کدی بنویسیم که با \emph{هر} گونه‌ای کار کند.

\section{مسئله: تکرارِ یک منطق برای گونه‌های گوناگون}

بدونِ کلی‌سازی، ناچار بودیم چنین بنویسیم:

\begin{salamfa}
تابع بیشینه صحیح(الف: صحیح، ب: صحیح): صحیح:
    اگر الف > ب: برگشت الف پایان
    برگشت ب
پایان

تابع بیشینه اعشاری(الف: اعشار۶۴، ب: اعشار۶۴): اعشار۶۴:
    اگر الف > ب: برگشت الف پایان
    برگشت ب
پایان

تابع ریشه:
    سرچاپ بیشینه صحیح(۳، ۹)، بیشینه اعشاری(۲.۵، ۱.۵)
پایان
\end{salamfa}

این دو تابع \emph{منطقِ یکسانی} دارند و تنها در گونه فرق می‌کنند. این تکرار هم
خسته‌کننده است و هم خطاخیز: اگر روزی منطق را اصلاح کنید، باید در هر دو جا تغییر
دهید.

\section{راهِ حل: پارامترِ گونه}

با کلی‌سازی، گونه را هم مانندِ یک پارامتر می‌گیریم. این مثالِ واقعی را ببینید:

\begin{salamfa}
ساختار جعبه<ت>:
    همگانی مقدار: ت
پایان

تابع بیشینه<ت>(الف: ت، ب: ت): ت:
    اگر الف > ب:
        برگشت الف
    پایان
    برگشت ب
پایان

تابع ریشه:
    سرچاپ بیشینه(۳، ۹)
    سرچاپ بیشینه(۲.۵، ۱.۵)
    ج := جعبه { مقدار = ۴۲ } برگردان جعبه<صحیح>
    سرچاپ ج.مقدار
پایان
\end{salamfa}

\begin{progoutfa}
9
2.5
42
\end{progoutfa}

به \code{بیشینه<ت>} دقت کنید. آن \code{ت} درونِ \code{<\,>} یک \textbf{پارامترِ
گونه} است یک گونهِ «جای‌خالی» که هنوز مشخص نیست. سپس در امضای تابع،
پارامترها و خروجی همگی از همان گونهِ \code{ت} هستند. معنایش این است: «هر دو
ورودی و خروجی، از یک گونهِ یکسان‌اند هر گونه‌ای که باشد.»

حالا وقتی می‌نویسیم:
\begin{itemize}
  \item \code{بیشینه(۳، ۹)}: سلام می‌بیند که ورودی‌ها صحیح‌اند، پس \code{ت} را
        \code{صحیح} می‌گیرد.
  \item \code{بیشینه(۲.۵، ۱.۵)}: ورودی‌ها اعشاری‌اند، پس \code{ت} را
        \code{اعشار۶۴} می‌گیرد.
\end{itemize}

یک تابع، دو کاربرد بی‌هیچ تکراری!

\begin{note}
دقت کنید که هنگامِ \emph{صدا زدنِ} \code{بیشینه}، لازم نبود گونه را بنویسیم؛ سلام
آن را از روی ورودی‌ها \emph{حدس} زد. این را «استنتاجِ گونه» می‌گویند و کد را
تمیزتر می‌کند. در واقع نوشتنِ گونه هنگامِ صدا زدن ممکن هم نیست:
\code{بیشینه<صحیح>(۳، ۹)} خطای تجزیه است. اگر می‌خواهید ورودی گونه‌ی دیگری
داشته باشد، آن را با \code{برگردان} تبدیل کنید.

بدنه‌ی تابعِ کلی باید برای هر گونه‌ای که به آن می‌دهید معنا داشته باشد.
\code{بیشینه} با \code{>} مقایسه می‌کند که برای عدد و رشته کار می‌کند؛ اگر دو
ساختار به آن بدهید، کامپایلر خطا می‌دهد، چون ساختارها را نمی‌توان با \code{>}
مقایسه کرد.
\end{note}

\section{ساختارهای کلی}

نه‌تنها تابع‌ها، بلکه \emph{ساختارها} هم می‌توانند کلی باشند. در مثالِ بالا:

\begin{salamfa}
ساختار جعبه<ت>:
    همگانی مقدار: ت
پایان
\end{salamfa}

\code{جعبه<ت>} یک ساختارِ کلی است: جعبه‌ای که می‌تواند \emph{هر} گونه‌ای را در خود
نگه دارد. ساختارِ کلی گونه‌اش را از تبدیل می‌گیرد: با
\code{جعبه \{ مقدار = ۴۲ \} برگردان جعبه<صحیح>} یک جعبهٔ مخصوصِ اعدادِ صحیح
می‌سازیم. می‌توانستیم \code{جعبه<رشته>} هم بسازیم تا متن نگه دارد. میدانِ
\code{مقدار} پیش‌فرض ندارد، پس هنگامِ ساختنِ جعبه باید داده شود؛ و چون
\code{ریشه} آن را می‌خواند، \code{همگانی} است.

\begin{tip}
بردار (\code{وکتور<T>}) و نگاشت (\code{نگاشت<K، V>}) که در فصل‌های پیش دیدیم،
خودشان ساختارهای کلی‌اند! به همین دلیل می‌توانستیم هم \code{وکتور<صحیح>} و هم
\code{وکتور<رشته>} بسازیم. اکنون می‌دانید که در پسِ این انعطاف، همین کلی‌سازی
نهفته است.
\end{tip}

\section{کلی‌سازی با چند پارامترِ گونه}

یک ساختار یا تابع می‌تواند بیش از یک پارامترِ گونه داشته باشد. برای نمونه، یک
«جفت» که دو مقدارِ احتمالاً متفاوت را نگه می‌دارد:

\begin{salamfa}
ساختار جفت<آ، ب>:
    همگانی اول: آ
    همگانی دوم: ب
پایان

تابع ریشه:
    ج := جفت { اول = "سن"، دوم = ۳۰ } برگردان جفت<رشته، صحیح>
    سرچاپ ج.اول، ج.دوم
پایان
\end{salamfa}

اینجا \code{آ} و \code{ب} دو پارامترِ گونهِ مستقل‌اند؛ پس می‌توان جفتی ساخت که
عضوِ نخستش رشته و عضوِ دومش عدد باشد، همان‌گونه که \code{نگاشت<K، V>} از کلید و
مقدارِ متفاوت بهره می‌برد. پارامترهای گونه با ویرگول از هم جدا می‌شوند.

\section{محدودیتِ گونه}

یک \code{<ت>} ساده هیچ چیزی درباره‌ی \code{ت} نمی‌داند، پس بدنه نمی‌تواند متدی
روی آن صدا بزند. با نوشتنِ نامِ یک میانجی پس از پارامترِ گونه، مثلِ
\code{<ت: شکل>}، قول می‌دهیم که هر \code{ت} متدهای آن میانجی را دارد، و بدنه آزاد
است آن‌ها را به کار ببرد:

\begin{salamfa}
میانجی شکل:
    تابع مساحت(): صحیح
پایان

ساختار مربع:
    ضلع: صحیح = ۰
    همگانی تابع مساحت(): صحیح: برگشت این.ضلع * این.ضلع پایان
پایان

ساختار مستطیل:
    طول: صحیح = ۰
    عرض: صحیح = ۰
    همگانی تابع مساحت(): صحیح: برگشت این.طول * این.عرض پایان
پایان

تابع بیشترین مساحت<ت: شکل>(الف: ت، ب: ت): صحیح:
    اگر الف.مساحت() > ب.مساحت():
        برگشت الف.مساحت()
    پایان
    برگشت ب.مساحت()
پایان

تابع ریشه:
    سرچاپ بیشترین مساحت(مربع { ضلع = ۳ }، مربع { ضلع = ۵ })
    سرچاپ بیشترین مساحت(مستطیل { طول = ۲، عرض = ۹ }، مستطیل { طول = ۴، عرض = ۴ })
پایان
\end{salamfa}

\begin{progoutfa}
25
18
\end{progoutfa}

دقت کنید که هر دو ورودی باید از \emph{یک} گونه باشند: \code{بیشترین مساحت} یک مربع
را با یک مستطیل مقایسه نمی‌کند. برای نگه‌داشتنِ گونه‌های گوناگون در کنارِ هم
همان \code{پویا}ی فصلِ پیش لازم است.

\section{چرا کلی‌سازی سریع است؟}

\begin{deep}[نگاهی به درون: تک‌گونه‌سازی]
سلام کلی‌سازی را با روشی به نامِ \emph{تک‌گونه‌سازی} (monomorphization) پیاده
می‌کند. وقتی شما \code{بیشینه(۳، ۹)} و \code{بیشینه(۲.۵، ۱.۵)} را می‌نویسید،
سلام در زمانِ کامپایل \emph{دو نسخهٔ جداگانه} از تابع می‌سازد: یکی مخصوصِ صحیح و
یکی مخصوصِ اعشاری. یعنی همان کدی که در آغازِ این فصل دستی نوشتیم، این بار به‌طورِ
خودکار ساخته می‌شود.

نتیجه این است که کلی‌سازی در سلام \emph{هیچ هزینهٔ اجرایی} ندارد: کدِ نهایی
دقیقاً به سرعتِ کدی است که برای هر گونه جدا نوشته باشید. شما راحتیِ نوشتنِ یک
تابع را دارید و رایانه سرعتِ چند تابعِ تخصصی را. این یکی از زیباترین معامله‌های
برنده-برنده در طراحیِ زبان است.
\end{deep}

\section{کلی‌سازی در برابرِ میانجی: کدام را برگزینیم؟}

اکنون که هر دو ابزار را دیدید، این راهنمای ساده کمکتان می‌کند:

\begin{center}
\begin{tabular}{@{}>{\centering\arraybackslash}p{0.44\linewidth}@{\hspace{2em}}>{\centering\arraybackslash}p{0.44\linewidth}@{}}
\toprule
\textbf{کلی‌سازی را برگزینید اگر\ldots} & \textbf{میانجیِ \code{پویا} را برگزینید اگر\ldots} \\
\midrule
سرعت بسیار مهم است & به انعطافِ زمانِ اجرا نیاز دارید \\
گونه‌ها در زمانِ کامپایل معلوم‌اند & می‌خواهید گونه‌های مختلف را در یک بردار بریزید \\
منطقِ یکسان برای گونه‌های گوناگون & رفتارِ مشترک با پیاده‌سازیِ متفاوت \\
\bottomrule
\end{tabular}
\end{center}

\begin{tip}
قاعدهٔ سرانگشتی: اگر شک دارید، کلی‌سازی را برگزینید. در بیشترِ موارد سریع‌تر و
ایمن‌تر است. تنها وقتی به \code{پویا} روی بیاورید که واقعاً به ذخیرهٔ گونه‌های
متفاوت در کنارِ هم یا تصمیم‌گیریِ زمانِ اجرا نیاز داشته باشید.
\end{tip}

\section{خلاصهٔ فصل}

\begin{itemize}
  \item کلی‌سازی به ما اجازه می‌دهد یک‌بار کدی بنویسیم که با هر گونه‌ای کار کند.
  \item پارامترِ گونه (مثلِ \code{ت}) درونِ \code{<\,>} نوشته می‌شود.
  \item هم تابع‌ها و هم ساختارها می‌توانند کلی باشند.
  \item می‌توان چند پارامترِ گونه داشت: \code{جفت<آ، ب>}.
  \item سلام گونه‌ی تابعِ کلی را از روی ورودی‌ها استنتاج می‌کند؛ ساختارِ کلی
        گونه‌اش را از \code{برگردان} می‌گیرد.
  \item محدودیتِ گونه (\code{<ت: شکل>}) به بدنه اجازه می‌دهد متدهای یک میانجی
        را صدا بزند.
  \item کلی‌سازی با تک‌گونه‌سازی پیاده می‌شود و هیچ هزینهٔ اجرایی ندارد.
\end{itemize}

\section{تمرین‌ها}

\exercise تابعِ کلیِ \code{کمینه<ت>} را بنویسید که کوچک‌ترِ دو مقدار را
برگرداند.

\exercise تابعِ کلیِ \code{جابجا<ت>} را بنویسید که دو مقدار بگیرد و آن‌ها را در
یک \code{جفت}ِ جابه‌جا‌شده برگرداند.

\exercise ساختارِ کلیِ \code{پشته<ت>} را با متدهای \code{بیفزا} و
\code{دربیاور} بسازید (می‌توانید درونش از یک \code{وکتور<ت>} استفاده کنید؛
متدهایی که آن را تغییر می‌دهند \code{ناپایا تابع}اند و فراموش نکنید متدی برای
آزاد کردنِ وکتور هم بگذارید).

\exercise ساختارِ \code{جفت<آ، ب>} را گسترش دهید و متدی بیفزایید که جای دو عضو
را عوض کند و یک \code{جفت<ب، آ>} برگرداند.

\exercise توضیح دهید چرا برای ساختنِ یک «فهرستِ نام‌ها و سن‌ها»، \code{نگاشت}
بهتر از دو بردارِ جداگانه است.
