\chapter{بستارها و عبارت‌های لامبدا}%
\label{ch:lambdas}

تا اینجا کارکردها را همیشه با نام می‌شناختیم. اما در سلام، کارکرد خودش یک \emph{مقدار}
است: می‌توان آن را در متغیر گذاشت، به کارکردِ دیگر داد، یا از یک کارکرد
بازگرداند. در این فصل با \textbf{کارکردهای بی‌نام} (لامبدا) و \textbf{بستارها}
آشنا می‌شویم ابزارهایی که برنامه‌نویسی را انعطاف‌پذیرتر می‌کنند.

\section{کارکرد به‌مثابهِ مقدار}

به این مثالِ واقعی نگاه کنید. اینجا کارکردها را در متغیر می‌گذاریم و حتی به کارکردِ
دیگری می‌دهیم:

\begin{salamfa}
کارکرد اعمال(ف : کارکرد (صحیح) صحیح, مقدار : صحیح) : صحیح:
    بازگشت ف(مقدار)
پایان

کارکرد آغازین:
    دوبرابر : کارکرد (صحیح) صحیح = (ن : صحیح) => ن * 2
    مربع : کارکرد (صحیح) صحیح = (ن : صحیح) => ن * ن
    چاپ اعمال(دوبرابر, 21)
    چاپ اعمال(مربع, 9)
    چاپ دوبرابر(مربع(3))
پایان
\end{salamfa}

چند چیزِ تازه اینجا هست:

\subsection{ریختِ کارکرد}

\begin{salamfa}
کارکرد (صحیح: ) : صحیح
\end{salamfa}

این یک \textbf{ریخت} است مانندِ \code{صحیح} یا \code{رشته}، اما ریختِ یک
\emph{کارکرد}. خوانده می‌شود: «کارکردی که یک عددِ صحیح می‌گیرد و یک عددِ صحیح
برمی‌گرداند.» پس \code{دوبرابر} متغیری است که نه عدد، بلکه یک \emph{کارکرد} در
آن نگه‌داشته می‌شود.

\subsection{عبارتِ لامبدا}

\begin{salamfa}
(ن صحیح) => ن * 2
\end{salamfa}

این یک \textbf{لامبدا} است: یک کارکردِ کوچکِ بی‌نام. ساختارش ساده است: درونِ
پرانتز پارامترها، سپس نشانهٔ \code{=>}، و سپس عبارتی که نتیجه را می‌دهد. این
لامبدا یک عدد می‌گیرد و دوبرابرش را برمی‌گرداند درست مانندِ یک کارکردِ معمولی،
اما بدونِ نام و در یک خط.

\subsection{کارکردی که کارکرد می‌گیرد}

کارکردِ \code{اعمال} در پارامترِ نخستش یک \emph{کارکرد} می‌گیرد (\code{ف}) و آن را
روی \code{مقدار} اجرا می‌کند. به همین دلیل توانستیم هم \code{دوبرابر} و هم
\code{مربع} را به آن بدهیم. کارکردهای که کارکردِ دیگر می‌گیرند یا برمی‌گردانند را
\textbf{کارکردهای مرتبهٔ بالاتر} می‌نامند و در پردازشِ داده بسیار کاربرد دارند.

\begin{note}
به آخرین خط دقت کنید: \code{دوبرابر(مربع(3))}. نخست \code{مربع(3)} برابرِ ۹، سپس
\code{دوبرابر(9)} برابرِ ۱۸ می‌شود. چون این‌ها متغیرهایی حاوی کارکرد‌اند، می‌توان
مستقیم صدایشان زد درست مانندِ کارکردهای معمولی.
\end{note}

\section{بستار: لامبدایی که محیطش را به خاطر می‌سپارد}

اینجا به یکی از قدرتمندترین ایده‌های برنامه‌نویسی می‌رسیم. یک لامبدا می‌تواند
متغیرهای \emph{بیرونِ} خودش را «به یاد بسپارد». به چنین لامبدایی \textbf{بستار}
(closure) می‌گویند. این مثالِ واقعی را ببینید:

\begin{salamfa}
کارکرد سازنده جمع(ن : صحیح) : کارکرد (صحیح) صحیح:
    بازگشت (x : صحیح) => x + ن
پایان

کارکرد آغازین:
    دوبرابر : کارکرد (صحیح) صحیح = (n : صحیح) => n * 2
    چاپ دوبرابر(21)
    جمع۱۰ : خودکار = سازنده جمع(10)
    چاپ جمع۱۰(5)
پایان
\end{salamfa}

کارکردِ \code{سازنده جمع} یک عدد (\code{ن}) می‌گیرد و یک \emph{لامبدا}
برمی‌گرداند. اما دقت کنید: آن لامبدا \code{(x صحیح) => x + ن} از
\code{ن} استفاده می‌کند که پارامترِ کارکردِ بیرونی است! وقتی
\code{سازنده جمع(10)} را صدا می‌زنیم، لامبدای بازگشتی مقدارِ \code{ن = 10} را
\emph{با خود نگه می‌دارد}. پس \code{جمع۱۰(5)} برابرِ $5 + 10 = 15$ می‌شود.

این یعنی \code{جمع۱۰} یک کارکردِ تازه است که «۱۰ را به یاد دارد». اگر
\code{سازنده جمع(100)} می‌ساختیم، کارکردی می‌داشتیم که ۱۰۰ را به یاد می‌آورد.

\section{بستار با چند دستور}

لامبداهایی که دیدیم یک عبارت داشتند. اما گاهی به چند خط منطق نیاز داریم. برای
این، شکلِ بلندترِ لامبدا را داریم که با \code{=> :} باز و با \code{پایان} بسته
می‌شود. این مثالِ واقعی یک \emph{شمارنده} می‌سازد:

\begin{salamfa}
کارکرد ساخت شمارنده() : کارکرد () صحیح:
    گذرا ن : صحیح = 0
    بازگشت () => :
        ن = ن + 1
        بازگشت ن
    پایان
پایان

کارکرد آغازین:
    بعدی : کارکرد () صحیح = ساخت شمارنده()
    چاپ بعدی(), بعدی(), بعدی(), بعدی()
پایان
\end{salamfa}

اینجا اتفاقِ جالبی می‌افتد. \code{ساخت شمارنده} متغیرِ \code{ن} را می‌سازد و
یک بستار برمی‌گرداند که آن را یکی زیاد و سپس بازمی‌گرداند. هر بار که
\code{بعدی()} را صدا می‌زنیم، همان \code{ن} که \emph{میانِ فراخوانی‌ها زنده
می‌ماند} یکی بیشتر می‌شود. پس خروجی \code{1 2 3 4} است.

\begin{deep}[نگاهی به درون: بستار حافظه را کجا نگه می‌دارد؟]
وقتی یک بستار متغیری از محیطِ بیرون را می‌گیرد، سلام آن متغیر را در \emph{پشته}
نمی‌گذارد (که با پایانِ کارکرد از میان می‌رود)، بلکه در \emph{پشتهٔ آزاد} (heap)
نگه می‌دارد و یک «محیطِ بسته‌شده» می‌سازد که همراهِ خودِ کارکرد زندگی می‌کند. به
همین دلیل است که \code{ن} حتی پس از پایان‌یافتنِ \code{ساخت شمارنده} همچنان
زنده می‌ماند. این مکانیزم گرفتنِ متغیرها و نگه‌داشتنشان روی پشتهٔ آزاد ---
قلبِ کارِ بستارهاست.
\end{deep}

\begin{tip}
بستارها به‌ویژه برای ساختنِ «کارخانه‌های کارکرد» (مانندِ \code{سازنده جمع}) و
نگه‌داشتنِ حالتِ خصوصی (مانندِ شمارنده) عالی‌اند. در فصلِ کتابخانهٔ استاندارد
خواهیم دید که بسیاری از کارکردهای پردازشِ داده، لامبدا می‌گیرند تا رفتارشان را
سفارشی کنید.
\end{tip}

\section{خلاصهٔ فصل}

\begin{itemize}
  \item در سلام، کارکرد یک مقدار است: می‌توان آن را در متغیر گذاشت، پاس داد و
        بازگرداند.
  \item ریختِ کارکرد چنین نوشته می‌شود: \code{کارکرد (ورودی) خروجی}.
  \item لامبدا کارکردی بی‌نام است: \code{(پارامتر) => عبارت}.
  \item برای چند دستور، شکلِ بلند را به‌کار می‌بریم: \code{() => :} \ldots
        \code{پایان}.
  \item بستار لامبدایی است که متغیرهای محیطِ خود را به یاد می‌سپارد.
  \item کارکردهای مرتبهٔ بالاتر، کارکردِ دیگر را می‌گیرند یا برمی‌گردانند.
\end{itemize}

\section{تمرین‌ها}

\exercise یک لامبدا بسازید که عددی بگیرد و مکعبش را برگرداند، و آن را در یک
متغیر بگذارید و صدا بزنید.

\exercise کارکردی به نامِ \code{سازنده ضرب} بنویسید که عددی بگیرد و یک بستار
برگرداند که هر ورودی را در آن عدد ضرب کند.

\exercise با کمکِ \code{ساخت شمارنده}، دو شمارندهٔ مستقل بسازید و نشان دهید که
هر کدام حالتِ خودش را دارد.

\exercise کارکردِ \code{اعمال دوبار} را بنویسید که یک کارکرد و یک مقدار بگیرد و
کارکرد را \emph{دو بار} پشتِ سرِ هم روی مقدار اجرا کند.

\exercise بستاری بسازید که مجموعِ همهٔ اعدادی را که تا کنون به آن داده‌اید نگه
دارد و در هر فراخوانی، مجموعِ تازه را برگرداند.
