\chapter{بستارها و عبارت‌های لامبدا}%
\label{ch:lambdas}

تا اینجا تابع‌ها را همیشه با نام می‌شناختیم. اما در سلام، تابع خودش یک \emph{مقدار}
است: می‌توان آن را در متغیر گذاشت، به تابعِ دیگر داد، یا از یک تابع
بازگرداند. در این فصل با \textbf{تابع‌های بی‌نام} (لامبدا) و \textbf{بستارها}
آشنا می‌شویم ابزارهایی که برنامه‌نویسی را انعطاف‌پذیرتر می‌کنند.

\section{تابع به‌مثابهِ مقدار}

به این مثالِ واقعی نگاه کنید. اینجا تابع‌ها را در متغیر می‌گذاریم و حتی به تابعِ
دیگری می‌دهیم:

\begin{salamfa}
تابع اعمال(ف: تابع (صحیح) صحیح، مقدار: صحیح): صحیح:
    برگشت ف(مقدار)
پایان

تابع ریشه:
    دوبرابر := (ن: صحیح): برگشت ن * ۲ پایان
    مربع := (ن: صحیح): برگشت ن * ن پایان
    سرچاپ اعمال(دوبرابر، ۲۱)
    سرچاپ اعمال(مربع، ۹)
    سرچاپ دوبرابر(مربع(۳))
پایان
\end{salamfa}

\begin{progoutfa}
42
81
18
\end{progoutfa}

چند چیزِ تازه اینجا هست:

\subsection{گونهِ تابع}

\begin{salamfa}
گونه عملگر = تابع (صحیح) صحیح
\end{salamfa}

\code{تابع (صحیح) صحیح} یک \textbf{گونه} است مانندِ \code{صحیح} یا \code{رشته}،
اما گونهِ یک \emph{تابع}. خوانده می‌شود: «تابعی که یک عددِ صحیح می‌گیرد و یک
عددِ صحیح برمی‌گرداند.» گونه‌ی ورودی‌ها درونِ پرانتز می‌آید و گونه‌ی خروجی پس
از آن؛ \code{تابع (صحیح)} تابعی است که چیزی برنمی‌گرداند. وقتی گونه‌ای بلند است
یا زیاد به کار می‌رود، می‌توان مانندِ خطِ بالا با \code{گونه} برایش نامی گذاشت.

پارامترِ \code{ف} در \code{اعمال} از همین گونه است. اما \code{دوبرابر} و
\code{مربع} در \code{ریشه} گونه‌ی صریح ندارند: مانندِ هر متغیرِ دیگری با
\code{:=} ساخته شده‌اند و سلام گونه‌شان را از خودِ لامبدا می‌فهمد. پس
\code{دوبرابر} متغیری است که نه عدد، بلکه یک \emph{تابع} در آن نگه‌داشته می‌شود.

\subsection{عبارتِ لامبدا}

\begin{salamfa}
دوبرابر := (ن: صحیح): برگشت ن * ۲ پایان
\end{salamfa}

این یک \textbf{لامبدا} است: یک تابعِ کوچکِ بی‌نام. ساختارش ساده است: درونِ
پرانتز پارامترها، سپس دونقطه، سپس بدنه، و در پایان \code{پایان}؛ نتیجه را با
\code{برگشت} می‌دهیم. این لامبدا یک عدد می‌گیرد و دوبرابرش را برمی‌گرداند،
درست مانندِ یک تابعِ معمولی، اما بدونِ نام و در یک خط.

برای لامبدا هیچ‌گاه گونه‌ی خروجی ننویسید: سلام آن را از \code{برگشت} می‌فهمد و
شکلی مانندِ \code{(ن: صحیح): صحیح: ...} پذیرفته نیست. نشانه‌ی \code{=>} هم که
در نگارش‌های قدیمی‌ترِ سلام به کار می‌رفت، از زبان برداشته شده است.

\subsection{تابعی که تابع می‌گیرد}

تابعِ \code{اعمال} در پارامترِ نخستش یک \emph{تابع} می‌گیرد (\code{ف}) و آن را
روی \code{مقدار} اجرا می‌کند. به همین دلیل توانستیم هم \code{دوبرابر} و هم
\code{مربع} را به آن بدهیم. تابع‌های که تابعِ دیگر می‌گیرند یا برمی‌گردانند را
\textbf{تابع‌های مرتبهٔ بالاتر} می‌نامند و در پردازشِ داده بسیار کاربرد دارند.

\begin{note}
به آخرین خط دقت کنید: \code{دوبرابر(مربع(3))}. نخست \code{مربع(3)} برابرِ ۹، سپس
\code{دوبرابر(9)} برابرِ ۱۸ می‌شود. چون این‌ها متغیرهایی حاوی تابع‌اند، می‌توان
مستقیم صدایشان زد درست مانندِ تابع‌های معمولی.
\end{note}

\section{بستار: لامبدایی که محیطش را به خاطر می‌سپارد}

اینجا به یکی از قدرتمندترین ایده‌های برنامه‌نویسی می‌رسیم. یک لامبدا می‌تواند
متغیرهای \emph{بیرونِ} خودش را «به یاد بسپارد». به چنین لامبدایی \textbf{بستار}
(closure) می‌گویند. این مثالِ واقعی را ببینید:

\begin{salamfa}
تابع سازنده جمع(ن: صحیح): تابع (صحیح) صحیح:
    برگشت (x: صحیح): برگشت x + ن پایان
پایان

تابع ریشه:
    دوبرابر := (n: صحیح): برگشت n * ۲ پایان
    سرچاپ دوبرابر(۲۱)
    جمع۱۰ := سازنده جمع(۱۰)
    سرچاپ جمع۱۰(۵)
پایان
\end{salamfa}

تابعِ \code{سازنده جمع} یک عدد (\code{ن}) می‌گیرد و یک \emph{لامبدا}
برمی‌گرداند. اما دقت کنید: آن لامبدا \code{(x: صحیح): برگشت x + ن پایان} از
\code{ن} استفاده می‌کند که پارامترِ تابعِ بیرونی است! وقتی
\code{سازنده جمع(10)} را صدا می‌زنیم، لامبدای بازگشتی مقدارِ \code{ن = 10} را
\emph{با خود نگه می‌دارد}. پس \code{جمع۱۰(۵)} برابرِ $5 + 10 = 15$ می‌شود و
خروجیِ برنامه \code{42} و سپس \code{15} است.

این یعنی \code{جمع۱۰} یک تابعِ تازه است که «۱۰ را به یاد دارد». اگر
\code{سازنده جمع(100)} می‌ساختیم، تابعی می‌داشتیم که ۱۰۰ را به یاد می‌آورد.

\section{بستار با چند دستور}

لامبداهایی که دیدیم یک عبارت داشتند. اما بدنهٔ لامبدا یک بلوک است و هر چند
دستور که لازم باشد می‌گیرد؛ چیزی در ساختار عوض نمی‌شود. این مثالِ واقعی یک
\emph{شمارنده} می‌سازد:

\begin{salamfa}
تابع ساخت شمارنده(): تابع () صحیح:
    ناپایا ن := ۰
    برگشت ():
        ن = ن + ۱
        برگشت ن
    پایان
پایان

تابع ریشه:
    بعدی := ساخت شمارنده()
    سرچاپ بعدی()، بعدی()، بعدی()، بعدی()
پایان
\end{salamfa}

اینجا اتفاقِ جالبی می‌افتد. \code{ساخت شمارنده} متغیرِ \code{ن} را می‌سازد و
یک بستار برمی‌گرداند که آن را یکی زیاد و سپس بازمی‌گرداند. هر بار که
\code{بعدی()} را صدا می‌زنیم، همان \code{ن} که \emph{میانِ فراخوانی‌ها زنده
می‌ماند} یکی بیشتر می‌شود. پس خروجی \code{1 2 3 4} است.

\begin{warn}
بستار از متغیرهای بیرونی‌ای که به کار می‌برد، در همان لحظه‌ی ساخته شدن یک
\emph{رونوشت} برمی‌دارد. اگر پس از آن متغیرِ بیرونی را تغییر دهید، بستار تغییر
را نمی‌بیند؛ و اگر بستار متغیرِ برداشته‌شده را تغییر دهد، تنها رونوشتِ خودش
عوض می‌شود:

\begin{salamfa}
تابع ریشه:
    ناپایا سقف := ۳
    نمایش := (): برگشت سقف پایان
    سقف = ۱۰
    سرچاپ سقف، نمایش()
پایان
\end{salamfa}

\begin{progoutfa}
10 3
\end{progoutfa}

در شمارنده‌ی بالا هم \code{ن}ی که بستار یکی‌یکی زیاد می‌کند رونوشتِ خودِ
بستار است و تا وقتی بستار هست می‌ماند. اگر لازم است بستار چیزی بیرون از خودش را
تغییر دهد، یک متغیرِ سراسریِ \code{ناپایا} به کار ببرید، یا بگذارید بستار مقدارِ
تازه را برگرداند.
\end{warn}

\begin{deep}[نگاهی به درون: بستار حافظه را کجا نگه می‌دارد؟]
وقتی یک بستار متغیری از محیطِ بیرون را می‌گیرد، سلام رونوشتِ آن متغیر را در
\emph{پشته} نمی‌گذارد (که با پایانِ تابع از میان می‌رود)، بلکه در \emph{پشتهٔ
آزاد} (heap) نگه می‌دارد و یک «محیطِ بسته‌شده» می‌سازد که همراهِ خودِ تابع
زندگی می‌کند. به همین دلیل است که \code{ن} حتی پس از پایان‌یافتنِ
\code{ساخت شمارنده} همچنان زنده می‌ماند. این حافظه هیچ‌گاه آزاد نمی‌شود؛ پس
بستاری را که متغیری از بیرون برمی‌دارد یک بار و بیرون از حلقه بسازید، نه در
هر دورِ حلقه.
\end{deep}

\begin{tip}
بستارها به‌ویژه برای ساختنِ «کارخانه‌های تابع» (مانندِ \code{سازنده جمع}) و
نگه‌داشتنِ حالتِ خصوصی (مانندِ شمارنده) عالی‌اند. در فصلِ کتابخانهٔ استاندارد
خواهیم دید که بسیاری از تابع‌های پردازشِ داده، لامبدا می‌گیرند تا رفتارشان را
سفارشی کنید.
\end{tip}

\section{خلاصهٔ فصل}

\begin{itemize}
  \item در سلام، تابع یک مقدار است: می‌توان آن را در متغیر گذاشت، پاس داد و
        بازگرداند.
  \item گونهِ تابع چنین نوشته می‌شود: \code{تابع (ورودی) خروجی}.
  \item لامبدا تابعی بی‌نام است: \code{(پارامتر): } \ldots \code{پایان}،
        و نتیجه را با \code{برگشت} می‌دهد. لامبدا گونه‌ی خروجیِ صریح ندارد و
        نشانهٔ \code{=>} در نگارش ۰٫۵٫۰ از زبان برداشته شد.
  \item بستار لامبدایی است که متغیرهای محیطِ خود را به یاد می‌سپارد؛ آن‌ها را
        هنگامِ ساخته شدن رونوشت می‌کند.
  \item تابع‌های مرتبهٔ بالاتر، تابعِ دیگر را می‌گیرند یا برمی‌گردانند.
\end{itemize}

\section{تمرین‌ها}

\exercise یک لامبدا بسازید که عددی بگیرد و مکعبش را برگرداند، و آن را در یک
متغیر بگذارید و صدا بزنید.

\exercise تابعی به نامِ \code{سازنده ضرب} بنویسید که عددی بگیرد و یک بستار
برگرداند که هر ورودی را در آن عدد ضرب کند.

\exercise با کمکِ \code{ساخت شمارنده}، دو شمارندهٔ مستقل بسازید و نشان دهید که
هر کدام حالتِ خودش را دارد.

\exercise تابعِ \code{اعمال دوبار} را بنویسید که یک تابع و یک مقدار بگیرد و
تابع را \emph{دو بار} پشتِ سرِ هم روی مقدار اجرا کند.

\exercise بستاری بسازید که مجموعِ همهٔ اعدادی را که تا کنون به آن داده‌اید نگه
دارد و در هر فراخوانی، مجموعِ تازه را برگرداند. (راهنمایی: مانندِ
\code{ساخت شمارنده}، مجموع را در متغیری بگذارید که بستار رونوشتش را نگه می‌دارد.)
