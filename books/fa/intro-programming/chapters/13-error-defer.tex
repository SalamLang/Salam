\chapter{مدیریتِ خطا و اجرای به‌تعویق‌افتاده}%
\label{ch:defer}

برنامه‌های واقعی همیشه در مسیرِ خوش پیش نمی‌روند: فایل ممکن است یافت نشود،
ورودیِ کاربر ممکن است نادرست باشد، یا منبعی که گرفته‌ایم باید آزاد شود. در این
فصل می‌آموزیم چگونه برنامه‌ای بنویسیم که با این موقعیت‌ها \emph{تمیز} و
\emph{مطمئن} برخورد کند.

\section{دیرکرد: کاری که حتماً باید انجام شود}

گاهی کاری هست که باید \emph{در پایانِ} یک تابع انجام شود هرچه پیش بیاید.
مثلاً اگر فایلی را باز کرده‌اید، باید آن را ببندید؛ اگر حافظه‌ای گرفته‌اید، باید
آزادش کنید. واژهٔ \code{دیرکن} (معادلِ \code{defer}) دقیقاً برای همین است. به
این مثالِ واقعیِ کوتاه نگاه کنید:

\begin{salamfa}
تابع ریشه:
    دیرکن سرچاپ "پاکسازی"
    سرچاپ "شروع"
    سرچاپ "پایان"
پایان
\end{salamfa}

شاید انتظار داشته باشید خروجی به ترتیبِ نوشتن باشد، اما چنین نیست:

\begin{progoutfa}
شروع
پایان
پاکسازی
\end{progoutfa}

دستوری که با \code{دیرکن} نشانه‌گذاری می‌شود، \emph{بی‌درنگ اجرا نمی‌شود}؛
بلکه کنار گذاشته می‌شود تا \emph{درست پیش از پایانِ بلوکی} که در آن نوشته شده
اجرا گردد؛ اینجا آن بلوک خودِ تابعِ \code{ریشه} است. پس هرچند «پاکسازی» را در
خطِ نخست نوشتیم، در پایان اجرا شد.

دقت کنید که \code{دیرکن} مالِ \emph{بلوکِ} خودش است، نه لزوماً کلِ تابع. اگر آن را
درونِ یک حلقه یا یک \code{اگر} بنویسید، در پایانِ همان بلوک اجرا می‌شود؛ در یک
حلقه یعنی در پایانِ هر دور:

\begin{salamfa}
تابع ریشه:
    تکرار ۳ در روز:
        دیرکن سرچاپ "  بستن روز"، روز
        سرچاپ "باز کردن روز"، روز
    پایان
    سرچاپ "تمام"
پایان
\end{salamfa}

\begin{progoutfa}
باز کردن روز 0
  بستن روز 0
باز کردن روز 1
  بستن روز 1
باز کردن روز 2
  بستن روز 2
تمام
\end{progoutfa}

یک نکته‌ی دیگر: دستورِ دیرکن‌شده متغیرهایش را هنگامِ \emph{اجرا} می‌خواند، نه
وقتی به خطِ \code{دیرکن} می‌رسیم. اگر متغیری در این فاصله عوض شود، دیرکن مقدارِ
تازه را می‌بیند.

\section{چرا دیرکرد این‌قدر مفید است؟}

ارزشِ واقعیِ \code{دیرکن} وقتی روشن می‌شود که تابع چند راهِ خروج دارد. این مثالِ
واقعی را ببینید:

\begin{salamfa}
تابع کار():
    دیرکن سرچاپ "۳. منبع آزاد شد"
    سرچاپ "۱. شروع کار"
    سرچاپ "۲. انجام کار"
پایان

تابع ریشه:
    کار()
    سرچاپ "۴. پایان برنامه"
پایان
\end{salamfa}

خروجی به ترتیبِ ۱، ۲، ۳، ۴ است. نکته اینجاست: از \emph{هر} راهی که از بلوک
بیرون برویم، چه به‌طورِ طبیعی، چه با \code{برگشت} زودهنگام، چه با \code{بشکن} از
یک حلقه، دستورِ \code{دیرکن} اجرا می‌شود. به همین دلیل، \code{دیرکن} بهترین جا برای «آزادسازیِ
منابع» است: کافی است \emph{بلافاصله} پس از گرفتنِ منبع، آزادسازی‌اش را
\code{دیرکن} کنید و دیگر هرگز نگرانِ فراموش‌کردنش نباشید.

\Needspace{24\baselineskip}
\begin{tip}
الگوی طلایی: هر بار منبعی گرفتید (فایل باز کردید، حافظه گرفتید، قفلی گرفتید)،
بی‌درنگ در خطِ بعد آزادسازی‌اش را \code{دیرکن} کنید. این‌گونه گرفتن و آزادکردن
کنارِ هم می‌مانند و خواندنِ کد آسان می‌شود. در این نمونه پرونده‌ای برای
نوشتن باز می‌شود و درست پس از آنکه مطمئن شدیم باز شده، بستنش را
\code{دیرکن} می‌کنیم:
\begin{salamfa}
واردسازی سیستم عامل

تابع ریشه:
    مسیر := سیستم عامل.پوشه موقت() + "/salam-note.txt"
    پ := سیستم عامل.بازکردن(مسیر، "w")
    اگر پ == پوچ:
        برگشت ۱
    پایان
    دیرکن پ.ببند()
    // ... هر کاری با پ، حتی با چند برگشتِ زودهنگام ...
    پ.نوشتن("سلام\n")
پایان
\end{salamfa}
\end{tip}

\section{چند دیرکرد و ترتیبِ اجرا}

اگر چند \code{دیرکن} داشته باشید، به ترتیبِ \emph{معکوس} اجرا می‌شوند یعنی
آخرین دیرکرد نخست اجرا می‌شود:

\begin{salamfa}
تابع ریشه:
    دیرکن سرچاپ "یک"
    دیرکن سرچاپ "دو"
    دیرکن سرچاپ "سه"
    سرچاپ "بدنه"
پایان
\end{salamfa}

خروجی:

\begin{progoutfa}
بدنه
سه
دو
یک
\end{progoutfa}

\begin{deep}[نگاهی به درون: چرا ترتیب معکوس است؟]
\code{دیرکن}ها روی یک \emph{پشته} انباشته می‌شوند. همان‌طور که در فصلِ بردارها
دیدیم، در پشته «آخرین چیزی که گذاشتی، نخستین چیزی است که برمی‌داری». این ترتیبِ
معکوس در عمل دقیقاً همان چیزی است که می‌خواهیم: اگر منبعِ «الف» را پیش از «ب»
گرفته‌اید، طبیعی است که «ب» را پیش از «الف» آزاد کنید درست مثلِ بازکردنِ
لایه‌های یک پیاز از بیرون به درون.
\end{deep}

\section{مدیریتِ خطا با مقدارِ بازگشتی}

سلام رویکردی \emph{صریح} به خطاها دارد: به‌جای آنکه خطاها به‌طورِ پنهان رخ
دهند، تابع‌ها معمولاً یک \emph{نشانه} برمی‌گردانند که می‌گوید کار موفق بود یا
نه. ساده‌ترین شکل، بازگرداندنِ یک مقدارِ منطقی یا یک کدِ وضعیت است:

\begin{salamfa}
تابع تقسیم امن(الف: صحیح، ب: صحیح، نتیجه &: صحیح): منطقی:
    اگر ب == ۰:
        برگشت نادرست
    پایان
    نتیجه = الف / ب
    برگشت درست
پایان

تابع ریشه:
    ناپایا خروجی := ۰
    اگر تقسیم امن(۱۰، ۲، خروجی):
        سرچاپ "نتیجه:"، خروجی
    وگرنه:
        سرچاپ "خطا: تقسیم بر صفر"
    پایان
    اگر تقسیم امن(۵، ۰، خروجی):
        سرچاپ "نتیجه:"، خروجی
    وگرنه:
        سرچاپ "خطا: تقسیم بر صفر"
    پایان
پایان
\end{salamfa}

\begin{progoutfa}
نتیجه: 5
خطا: تقسیم بر صفر
\end{progoutfa}

اینجا تابع \code{درست} یا \code{نادرست} برمی‌گرداند تا بگوید موفق شد یا نه، و
نتیجهٔ واقعی را از راهِ پارامترِ \emph{ارجاعی} \code{نتیجه \&: صحیح} بیرون
می‌دهد. (نشانهٔ \code{\&:} پارامتر را \emph{ارجاع} می‌کند و به تابع اجازه می‌دهد
متغیرِ بیرون را تغییر دهد؛ برای همین \code{خروجی} را \code{ناپایا} ساخته‌ایم.
آن را در فصلِ ۲۰ کامل‌تر می‌بینیم.) فراخواننده
\emph{موظف} است وضعیت را بررسی کند و همین صراحت، خطاها را آشکار و کنترل‌پذیر
می‌کند.

\begin{warn}
در زبان‌هایی که خطاها را پنهان مدیریت می‌کنند، آسان است که فراموش کنید خطایی
ممکن است رخ دهد. رویکردِ سلام شما را وادار می‌کند که \emph{آگاهانه} دربارهٔ
حالتِ خطا تصمیم بگیرید. این کمی بیشتر کد می‌خواهد، اما برنامه‌هایی می‌سازد که در
برابرِ شرایطِ ناخواسته مقاوم‌اند.
\end{warn}

\section{بررسیِ مرزها و اجرای امن}

یکی از رایج‌ترین خطاها، بیرون‌زدن از مرزِ آرایه است (که در فصلِ ۷ هشدارش را
دادیم). سلام این خطاها را به‌طورِ پیش‌فرض در زمانِ اجرا می‌گیرد: در ساختِ
معمولی (\code{salam run} یا \code{salam build}) هر دسترسیِ نامعتبر به آرایه یا
وکتور، به‌جای رفتارِ نامشخص، برنامه را با یک پیامِ روشن متوقف می‌کند:

\begin{salamfa}
تابع ریشه:
    آ := [۱، ۲، ۳]
    ناپایا i := ۲
    i += ۳
    سرچاپ آ[i]
پایان
\end{salamfa}

\begin{shell}
salam: array index 5 out of bounds [0,3)
\end{shell}

این برای یافتنِ خطاها در حینِ توسعه بسیار ارزشمند است. تنها وقتی برنامه را با
گزینهٔ \code{--release} بسازید، این بررسی‌ها برای سرعتِ بیشتر برداشته می‌شوند:

\begin{shell}
salam build app.salam --release --output=app
\end{shell}

پس نسخه‌ی \code{--release} را تنها برای برنامه‌ای بسازید که پیش‌تر به‌خوبی
آزموده‌اید.

\section{ترکیبِ دیرکرد و مدیریتِ خطا}

این دو ابزار با هم می‌درخشند. الگوی رایج چنین است: منبع را بگیر، آزادسازی‌اش را
دیرکرد کن، سپس کارها را انجام بده و در هر خطایی با خیالِ راحت \code{برگشت} کن،
چون می‌دانی \code{دیرکن} پاکسازی را تضمین کرده است:

\begin{salamfa}
تابع پردازش(داده: صحیح): منطقی:
    دیرکن سرچاپ "منبع آزاد شد"
    اگر داده < ۰:
        سرچاپ "ورودی نامعتبر"
        برگشت نادرست
    پایان
    سرچاپ "پردازش:"، داده
    برگشت درست
پایان

تابع ریشه:
    پردازش(۵)
    سرچاپ "---"
    پردازش(-۱)
پایان
\end{salamfa}

\begin{progoutfa}
پردازش: 5
منبع آزاد شد
---
ورودی نامعتبر
منبع آزاد شد
\end{progoutfa}

در هر دو فراخوانی چه موفق و چه ناموفق پیامِ «منبع آزاد شد» چاپ می‌شود،
چون با \code{دیرکن} نشانه‌گذاری شده بود.

\section{خلاصهٔ فصل}

\begin{itemize}
  \item \code{دیرکن} دستوری را تا پایانِ بلوکِ خودش (برای نمونه تابع یا یک دورِ
        حلقه) به تعویق می‌اندازد.
  \item هر مسیرِ خروج از بلوک، \code{دیرکن}ها را اجرا می‌کند؛ عالی برای
        آزادسازیِ منابع.
  \item چند \code{دیرکن} به ترتیبِ معکوس اجرا می‌شوند.
  \item سلام خطاها را \emph{صریح} مدیریت می‌کند: تابع‌ها نشانهٔ موفقیت
        برمی‌گردانند.
  \item ساختِ معمولی خطاهای مرزِ آرایه را در زمانِ اجرا می‌گیرد؛ تنها
        \code{--release} این بررسی‌ها را برمی‌دارد.
  \item ترکیبِ \code{دیرکن} و مدیریتِ صریحِ خطا، کدِ مقاوم می‌سازد.
\end{itemize}

\section{تمرین‌ها}

\exercise تابعی بنویسید که در آغازش با \code{دیرکن} پیامی چاپ کند و سپس دو خطِ
دیگر. ترتیبِ خروجی را پیش‌بینی و سپس آزمایش کنید.

\exercise تابعی به نامِ \code{ریشهٔ دوم امن} بنویسید که اگر ورودی منفی بود،
\code{نادرست} برگرداند و نتیجه را از راهِ پارامترِ ارجاعی بدهد.

\exercise تابعی بنویسید که سه \code{دیرکن} داشته باشد و با اجرای آن، ترتیبِ
معکوس را تأیید کنید.

\exercise برنامه‌ای بنویسید که آرایه‌ای بسازد و عمداً به اندیسی بیرون از مرز
دسترسی پیدا کند؛ آن را یک بار معمولی اجرا کنید و پیامِ خطا را ببینید. (اگر اندیس
یک عددِ ثابت باشد، کامپایلر همان هنگامِ ساخت خطا می‌دهد؛ پس اندیس را در یک
متغیرِ \code{ناپایا} حساب کنید.)

\exercise تابعی بنویسید که یک سن بگیرد و اگر بیرون از بازهٔ ۰ تا ۱۵۰ بود، با
\code{نادرست} خطا را اعلام کند.
