\chapter{ماژول‌ها و بسته‌ها}%
\label{ch:modules}

تا اینجا همهٔ برنامه‌ها را در یک فایل نوشتیم. اما برنامه‌های واقعی بزرگ‌اند و
نمی‌توان همه را در یک فایل گنجاند. \textbf{ماژول‌ها} و \textbf{بسته‌ها} به ما
اجازه می‌دهند برنامه را به فایل‌های کوچک‌تر و معنادار بشکنیم و کدها را میانِ
آن‌ها به اشتراک بگذاریم.

\section{چرا برنامه را بشکنیم؟}

تصور کنید یک برنامهٔ بزرگ دارید: بخشی برای محاسباتِ ریاضی، بخشی برای کار با
متن، بخشی برای رابطِ کاربری. اگر همه را در یک فایل بریزید، یافتنِ هر چیز دشوار
می‌شود. با تقسیمِ برنامه به ماژول‌ها:
\begin{itemize}
  \item هر فایل یک مسئولیتِ روشن دارد.
  \item کدها را می‌توان میانِ پروژه‌ها به اشتراک گذاشت و دوباره به‌کار برد.
  \item چند نفر می‌توانند هم‌زمان روی بخش‌های مختلف کار کنند.
\end{itemize}

\section{واردسازیِ کتابخانه‌های آماده}

ساده‌ترین شکلِ ماژول، کتابخانه‌های آمادهٔ خودِ سلام‌اند. با واژهٔ
\code{واردسازی} (معادلِ \code{import}) آن‌ها را به برنامه می‌آوریم. این مثالِ
واقعی ماژولِ ریاضی را وارد می‌کند:

\begin{salamfa}
واردسازی ریاضی

تابع ریشه:
    سرچاپ ریاضی.توان(۲.۰، ۱۰.۰)
    سرچاپ ریاضی.جذر(۱۴۴.۰)
پایان
\end{salamfa}

\begin{progoutfa}
1024
12
\end{progoutfa}

پس از \code{واردسازی ریاضی}، می‌توانیم با نامِ ماژول و یک نقطه، به تابع‌هایش
دسترسی پیدا کنیم: \code{ریاضی.توان(...)}، \code{ریاضی.جذر(...)}. این نقطه می‌گوید
«توانِ متعلق به ماژولِ ریاضی».

\begin{note}
نامِ ماژول‌های استاندارد هم مانندِ همه‌چیزِ دیگر در سلام دوزبانه است: هر بسته و
هر تابع یک نامِ انگلیسی و یک نامِ فارسی دارد. اما هر پرونده تنها زبانِ خودش را
می‌پذیرد. در پرونده‌ی فارسی باید \code{واردسازی ریاضی} و \code{ریاضی.جذر}
بنویسید؛ \code{واردسازی math} خطای «بسته یافت نشد» می‌دهد و
\code{ریاضی.Sqrt} خطای \code{E001} («شناسه در یک فایل فارسی باید فارسی
باشد»). در فصلِ ۱۵ کتابخانهٔ استاندارد را به‌تفصیل می‌بینیم.

واردسازی‌ای که به کار نرود هم خطاست (\code{E082})، پس فهرستِ واردسازی‌های بالای
هر پرونده همیشه دقیقاً می‌گوید آن پرونده به چه چیزهایی وابسته است.
\end{note}

\section{ساختنِ ماژولِ خودمان}

می‌توانیم کدهای خود را در یک فایلِ جدا بگذاریم و در فایلِ دیگر از آن‌ها استفاده
کنیم. نخست یک فایلِ کتابخانه می‌سازیم. این مثالِ واقعی، فایلی به نامِ
\code{falib.salam} است:

\begin{salamfa}
بسته ابزار

همگانی پایا پاسخ := ۴۲

تابع دوبرابر(ع: صحیح): صحیح:
    برگشت ع * ۲
پایان

همگانی تابع جمع(الف: صحیح، ب: صحیح): صحیح:
    برگشت الف + ب
پایان

همگانی تابع جمع دوبرابر(الف: صحیح، ب: صحیح): صحیح:
    برگشت دوبرابر(الف + ب)
پایان
\end{salamfa}

دو نکتهٔ کلیدی:
\begin{itemize}
  \item \code{بسته ابزار} (معادلِ \code{package}) به این فایل یک \emph{نام}
        می‌دهد و می‌گوید که این یک بستهٔ کتابخانه‌ای است، نه یک برنامهٔ اجرایی؛
        این فایل \code{ریشه} ندارد و به‌تنهایی اجرا نمی‌شود.
  \item \code{همگانی} (معادلِ \code{pub}) پیش از \code{تابع جمع}،
        \code{تابع جمع دوبرابر} و \code{پایا پاسخ} می‌گوید که این‌ها
        \emph{بیرون} از فایل هم در دسترس‌اند. \code{دوبرابر} همگانی نیست و تنها
        درونِ بسته به کار می‌رود.
\end{itemize}

ترتیبِ بخش‌های فایل هم مهم است و کامپایلر آن را بررسی می‌کند: نخست \code{بسته}،
سپس \code{واردسازی}ها و \code{فراخوانی}ها، سپس پایاها و متغیرهای سراسری، سپس
گونه‌ها (ساختار، جداشمار، میانجی، کاربست) و در پایان تابع‌ها، اول تابع‌های
خصوصی و بعد تابع‌های همگانی.

\subsection{همگانی در برابرِ خصوصی}

به‌طورِ پیش‌فرض، هر چیزی که در یک ماژول تعریف می‌کنید \emph{خصوصی} است؛ یعنی
فقط درونِ همان فایل دیده می‌شود. تنها چیزهایی که با \code{همگانی} نشانه‌گذاری
کنید، از بیرون قابلِ دسترس‌اند. اگر برنامه‌ی اصلی بخواهد \code{ابزار.دوبرابر}
را صدا بزند، کامپایلر می‌گوید:

\begin{progoutfa}
خطا[خطا۰۱۷]: 'دوبرابر' توسط بسته‌ی 'ابزار' صادر نشده است (آن را 'همگانی' کنید)
\end{progoutfa}

این یک اصلِ مهمِ طراحی است: \emph{فقط آنچه را
که دیگران واقعاً لازم دارند، همگانی کنید} و باقی را خصوصی نگه دارید.

\begin{tip}
چرا چیزها را خصوصی نگه داریم؟ چون هرچه کمتر همگانی باشد، آزادیِ شما برای تغییرِ
درونِ ماژول بیشتر است. اگر همه‌چیز همگانی باشد، هر تغییرِ کوچک ممکن است کدِ
دیگران را بشکند. مرزِ روشنِ «همگانی/خصوصی» به شما اجازه می‌دهد بی‌دغدغه درونِ
ماژول را بهبود دهید.
\end{tip}

\subsection{استفاده از ماژولِ خودمان}

حالا در فایلِ اصلی، ماژول را فراخوانی و از آن استفاده می‌کنیم. کتابخانه‌ها با
\code{واردسازی} و نامشان می‌آیند، اما فایل‌های خودمان با \code{فراخوانی} (معادلِ
\code{include}) و مسیرشان در گیومه. این مثالِ واقعی نشان می‌دهد چگونه با یک
\emph{نامِ مستعار} فایل را فراخوانی کنیم:

\begin{salamfa}
فراخوانی ابزار "falib.salam"

تابع ریشه:
    سرچاپ ابزار.جمع(۲، ۳)
    سرچاپ ابزار.جمع دوبرابر(۲، ۳)
    سرچاپ ابزار.پاسخ
پایان
\end{salamfa}

اینجا \code{فراخوانی ابزار "falib.salam"} فایل را می‌آورد و به آن نامِ
\code{ابزار} می‌دهد. سپس با \code{ابزار.جمع(۲، ۳)} و \code{ابزار.پاسخ} به
عضوهای همگانیِ آن دسترسی پیدا می‌کنیم. خروجی: \code{5}، \code{10} و \code{42}.

\begin{note}
مسیرِ \code{فراخوانی} به دو شکل نوشته می‌شود: نسبت به همین فایل، مانندِ
\code{"کمکی.salam"} یا \code{"../مشترک/ابزار.salam"}، یا از ریشه‌ی پروژه با
\code{@/} در آغاز، مانندِ \code{"@/مشترک/ابزار.salam"}. ریشه‌ی پروژه پوشه‌ی
فایلِ اصلیِ برنامه است. نوشتنِ \code{./} در آغازِ مسیر مجاز نیست. کامپایلر هنگامِ
ساخت، وجود و خوانا بودنِ هر فایلِ فراخوانی‌شده را بررسی می‌کند و فایلی که پیدا
نشود خطای \code{E008} می‌دهد.
\end{note}

\begin{note}
نامی که پس از \code{فراخوانی} می‌آید (اینجا \code{ابزار}) یک \emph{نامِ
مستعار} است و می‌تواند با نامِ فایل فرق کند. نوشتنش اختیاری است: بدونِ آن، فایل
با نامِ خودش در دسترس است. نامِ مستعار به‌ویژه وقتی مفید است که نامِ فایل طولانی
یا نامفهوم باشد یا دو ماژول نامِ یکسان داشته باشند. می‌توانید هر نامِ کوتاه و
گویایی برگزینید.
\end{note}

\section{دوزبانه‌سازیِ ماژول‌های خودمان}

یکی از زیباترین ویژگی‌های سلام این است که می‌توانید برای کدِ \emph{خودتان} هم
نام‌های فارسی و انگلیسی تعریف کنید. این کار با حاشیه‌نویسی‌های \code{@en} و
\code{@fa} انجام می‌شود. کتابخانه را یک بار به انگلیسی می‌نویسیم و کنارِ هر
نامِ همگانی، نامِ فارسی‌اش را با \code{@fa} می‌گذاریم. این مثالِ واقعی، فایلِ
\code{geo.salam}، یک ساختار و یک تابعِ کاملاً دوزبانه می‌سازد:

\begin{salamen}
package geo

@en "Point"
@fa "نقطه"
pub struct Point:
    @en "x"
    @fa "ایکس"
    pub x: int = 0
    @en "y"
    @fa "وای"
    pub y: int = 0
    @en "Sum"
    @fa "جمع"
    pub func Sum(): int: ret this.x + this.y end
end

@en "Twice"
@fa "دوبار"
pub func Twice(n: int): int:
    ret n * 2
end
\end{salamen}

و این برنامه‌ی فارسی آن را با نام‌های فارسی به کار می‌برد:

\begin{salamfa}
فراخوانی هندسه "geo.salam"

تابع ریشه:
    ن := هندسه.نقطه { ایکس = ۳، وای = ۴ }
    سرچاپ ن.ایکس
    سرچاپ ن.جمع()
    سرچاپ هندسه.دوبار(۲۱)
    سرچاپ زبان()
پایان
\end{salamfa}

\begin{progoutfa}
3
7
42
fa
\end{progoutfa}

با \code{@en "Point"} و \code{@fa "نقطه"} به ساختار دو نام دادیم. حالا در یک
برنامهٔ فارسی می‌توان ساختار، میدان‌ها و متدها را با نامِ فارسی صدا زد
(\code{هندسه.نقطه}، \code{ن.ایکس}، \code{ن.جمع()})، هرچند خودِ ساختار با
نام‌های انگلیسی تعریف شده است، و یک برنامه‌ی انگلیسی همان را با
\code{geo.Point} و \code{p.Sum()} به کار می‌برد. تابعِ ویژهٔ \code{زبان()}
(در پرونده‌ی انگلیسی \code{lang()}) نیز زبانِ پرونده‌ی جاری را برمی‌گرداند
(\code{"fa"} یا \code{"en"}).

\Needspace{9\baselineskip}
\begin{warn}
این ترجمه تنها از انگلیسی به فارسی کار می‌کند. اگر کتابخانه را به فارسی بنویسید
و با \code{@en} نامِ انگلیسی بدهید، یک برنامه‌ی انگلیسی نمی‌تواند آن نام‌ها را
پیدا کند. پس کتابخانه‌ای را که باید از هر دو زبان صدا زده شود، به انگلیسی
بنویسید و نام‌های فارسی را با \code{@fa} بیفزایید؛ کتابخانه‌ی استاندارد هم
همین‌گونه نوشته شده است.
\end{warn}

\begin{deep}[نگاهی به درون: چگونه یک کد، دو زبان دارد؟]
وقتی شما \code{@fa "نقطه"} می‌نویسید، کامپایلر یک \emph{نگاشت} میانِ نامِ فارسی
و نامِ انگلیسیِ اصلی می‌سازد. در زمانِ کامپایل، هر جا که \code{نقطه} ببیند، آن
را به همان \code{Point}ِ اصلی ترجمه می‌کند. پس در کدِ نهایی تنها یک گونه وجود
دارد؛ دو نام فقط دو \emph{درگاهِ ورودی} به همان چیزند. به همین دلیل است که
دوزبانه‌بودن هیچ هزینهٔ اجرایی ندارد: همه‌چیز در زمانِ کامپایل حل می‌شود.
\end{deep}

\begin{tip}
این ویژگی برای نوشتنِ کتابخانه‌هایی که هم فارسی‌زبانان و هم انگلیسی‌زبانان از
آن‌ها استفاده می‌کنند فوق‌العاده است. یک‌بار کتابخانه را با \code{@en} و
\code{@fa} می‌نویسید و هر کاربر با زبانِ خودش از آن بهره می‌برد، درست همان
کاری که خودِ کتابخانهٔ استانداردِ سلام انجام می‌دهد.
\end{tip}

\section{ساختارِ یک پروژهٔ چندفایلی}

فرمانِ \code{salam new my-project} پوشه‌ای با یک فایل به نامِ
\code{main.salam} می‌سازد که نقطه‌ی شروعِ برنامه است. یک پروژهٔ متوسط از آنجا
معمولاً چنین سامان می‌یابد:

\begin{shell}
my-project/
    main.salam
    geometry/geometry.salam
    text/text.salam
\end{shell}

\code{main.salam} برنامهٔ اصلی است و تابعِ \code{ریشه} در آن است. هر بخشِ
برنامه پوشه‌ای می‌شود که یک بسته است: پوشه هم‌نامِ بسته است و فایلی با همان نام
در آن هست، و با بزرگ شدنِ بسته، فایل‌های دیگری کنارش می‌آیند که همه جزوِ همان
بسته‌اند. \code{main.salam} با \code{فراخوانی} این بسته‌ها را می‌آورد (مثلاً
\code{فراخوانی هندسه "geometry/geometry.salam"}) و آن‌ها را به هم می‌پیوندد.
هنگامِ ساختِ برنامه، سلام به‌طورِ خودکار همهٔ فایل‌های فراخوانی‌شده را پیدا و
کامپایل می‌کند؛ کافی است درونِ پوشه‌ی پروژه بنویسید:

\begin{shell}
salam build
\end{shell}

بدونِ نامِ فایل، \code{salam build} همان \code{main.salam} را به برنامه‌ای
هم‌نامِ پوشه تبدیل می‌کند. می‌توانید فایلِ اصلی و نامِ خروجی را هم خودتان بدهید:

\begin{shell}
salam build main.salam --output=app
\end{shell}

\section{خلاصهٔ فصل}

\begin{itemize}
  \item \code{واردسازی} کتابخانه‌های آماده را با نامشان می‌آورد و
        \code{فراخوانی} فایل‌های خودمان را با مسیرشان.
  \item مسیرِ \code{فراخوانی} نسبت به همین فایل است، یا با \code{@/} از ریشه‌ی
        پروژه (پوشه‌ی فایلِ اصلی)؛ آغازِ مسیر با \code{./} خطاست.
  \item \code{بسته} به یک فایل نام می‌دهد و آن را کتابخانه می‌کند.
  \item هر چیز به‌طورِ پیش‌فرض خصوصی است؛ \code{همگانی} آن را بیرون‌دیدنی
        می‌کند.
  \item با نامِ مستعار (\code{فراخوانی ابزار "..."}) می‌توان ماژول را کوتاه صدا
        زد.
  \item با \code{@en} و \code{@fa} می‌توان به کدِ خود نام‌های دوزبانه داد؛
        کتابخانه‌ی دوزبانه به انگلیسی نوشته می‌شود و \code{@fa} نام‌های فارسی را
        می‌افزاید.
  \item \code{زبان()} زبانِ پرونده‌ی جاری را برمی‌گرداند.
\end{itemize}

\section{تمرین‌ها}

\exercise یک ماژول به نامِ \code{هندسه} بسازید که تابع‌های همگانیِ \code{مساحت دایره}
و \code{محیط دایره} داشته باشد، و در فایلِ اصلی از آن استفاده کنید.

\exercise در ماژولِ هندسه یک تابعِ کمکیِ \emph{خصوصی} بگذارید و نشان دهید که از
بیرون قابلِ دسترس نیست.

\exercise ماژولی به انگلیسی بسازید که با \code{@fa} برای تابع‌هایش نامِ فارسی هم
داشته باشد، و آن را یک بار از یک برنامهٔ فارسی و یک بار از یک برنامهٔ انگلیسی صدا
بزنید.

\exercise ماژولِ \code{ریاضی}ِ استاندارد را وارد کنید و با \code{ریاضی.جذر} و
\code{ریاضی.توان} مساحتِ یک دایره و قطرِ آن را حساب کنید.

\exercise یک پروژهٔ سه‌فایلی طراحی کنید (اصلی + دو ماژول) و توضیح دهید هر فایل
چه مسئولیتی دارد.
