\chapter{کتابخانهٔ استاندارد}%
\label{ch:stdlib}

هیچ برنامه‌نویسی همه‌چیز را از صفر نمی‌سازد. سلام با یک \textbf{کتابخانهٔ
استاندارد} غنی همراه است: مجموعه‌ای از ماژول‌های آماده برای ریاضی، متن،
ساختارهای داده، زمان، تصادف و بسیار چیزهای دیگر. در این فصل پرکاربردترین‌ها را
می‌بینیم.

\section{ریاضی}

ماژولِ \code{ریاضی} (به انگلیسی \code{math}) تابع‌های ریاضیِ رایج را در اختیار
می‌گذارد. در پرونده‌ی فارسی آن را با \code{واردسازی ریاضی} وارد می‌کنیم و
تابع‌هایش را با نامِ فارسی‌شان صدا می‌زنیم. این مثالِ واقعی چند تای آن‌ها را نشان
می‌دهد:

\begin{salamfa}
واردسازی ریاضی

تابع ریشه:
    سرچاپ "جذر ۱۴۴:"، ریاضی.جذر(۱۴۴.۰)
    سرچاپ "بیشینه:"، ریاضی.بیشینه(۳.۰، ۹.۰)
    سرچاپ "عدد پی:"، ریاضی.پی
پایان
\end{salamfa}

\begin{progoutfa}
جذر ۱۴۴: 12
بیشینه: 9
عدد پی: 3.141592653589793
\end{progoutfa}

مهم‌ترین تابع‌ها و پایاهای ماژولِ ریاضی (ستونِ انگلیسی نامی است که در پرونده‌ی
انگلیسی و در مرجعِ انگلیسیِ کتابخانه به کار می‌رود؛ در پرونده‌ی فارسی تنها نامِ
فارسی پذیرفته است):

\begin{center}
\begin{tabular}{l l l}
\toprule
\textbf{فارسی} & \textbf{انگلیسی} & \textbf{کار} \\
\midrule
\code{ریاضی.جذر(x)} & \code{Sqrt} & جذر \\
\code{ریاضی.توان(x، y)} & \code{Pow} & توان \\
\code{ریاضی.قدرمطلق(x)} & \code{Abs} & قدرِمطلق \\
\code{ریاضی.کف(x)} & \code{Floor} & گردکردن به پایین \\
\code{ریاضی.سقف(x)} & \code{Ceil} & گردکردن به بالا \\
\code{ریاضی.سینوس(x)} & \code{Sin} & سینوس \\
\code{ریاضی.کسینوس(x)} & \code{Cos} & کسینوس \\
\code{ریاضی.لگاریتم(x)} & \code{Log} & لگاریتم \\
\code{ریاضی.بیشینه(a، b)} & \code{Max} & بزرگ‌تر \\
\code{ریاضی.کمینه(a، b)} & \code{Min} & کوچک‌تر \\
\code{ریاضی.پی} & \code{PI} & عددِ $\pi$ \\
\bottomrule
\end{tabular}
\end{center}

\begin{warn}
\begin{sloppypar}
بیشترِ تابع‌های ریاضی روی اعدادِ \emph{اعشاری} کار می‌کنند. به همین دلیل
\code{ریاضی.جذر(۱۴۴.۰)} نوشتیم، نه \code{ریاضی.جذر(۱۴۴)}. اگر عددِ صحیح دارید،
نخست آن را با \code{برگردان اعشار۶۴} تبدیل کنید.
\end{sloppypar}
\end{warn}

\section{رشته}

ماژولِ \code{رشته} (به انگلیسی \code{str}) و متدهای پایه‌ی رشته را در فصلِ ۸
معرفی کردیم. خودِ رشته‌ها هم، بی‌نیاز به هیچ واردسازی، متدهای پیشرفته‌تری
دارند. این مثالِ واقعی چند تای آن‌ها را نشان می‌دهد:

\begin{salamfa}
تابع ریشه:
    متن := "salam,donya"
    سرچاپ "طول-متن"، متن.طول()
    سرچاپ "بخش"، متن.زیررشته(۰، ۵)
    سرچاپ "یافتن"، متن.بیاب("donya")
    سرچاپ "پیوست"، متن.پیوست("!")
    بخش‌ها := متن.بشکاف(",")
    دیرکن بخش‌ها.آزادکن()
    سرچاپ "تعداد-بخش"، بخش‌ها.طول()
    سرچاپ "عدد"، " ۴۲ ".پیراست()
    سرچاپ "به‌صحیح"، "42".به صحیح() + ۱
پایان
\end{salamfa}

\begin{progoutfa}
طول-متن 11
بخش salam
یافتن 6
پیوست salam,donya!
تعداد-بخش 2
عدد ۴۲
به‌صحیح 43
\end{progoutfa}

تابع‌های تازه‌ای که اینجا می‌بینیم:
\begin{itemize}
  \item \code{.بیاب("...")} (\code{find}): جای نخستین یافتنِ یک زیررشته را
        (به بایت) برمی‌گرداند، یا \code{-1} اگر پیدا نشود.
  \item \code{.بشکاف(",")} (\code{split}): رشته را بر اساسِ یک جداکننده به یک
        \code{وکتور<رشته>} از تکه‌ها می‌شکند. این وکتور باید آزاد شود؛ خطِ
        \code{دیرکن بخش‌ها.آزادکن()} برای همین است.
  \item \code{.پیراست()} (\code{trim}): فاصله‌های آغاز و پایان را می‌زداید.
  \item \code{.به صحیح()} (\code{to\_int}): یک رشتهٔ عددی را به عددِ صحیح تبدیل
        می‌کند. رقم‌های فارسی را هم می‌پذیرد، و اگر به نویسه‌ای برسد که جزوِ عدد
        نیست می‌ایستد؛ متنی که اصلاً عددی ندارد \code{0} می‌دهد.
\end{itemize}

بسته‌ی \code{رشته} هم برای کارهای بیشتر تابع‌هایی مانندِ
\code{رشته.تفکیک کردن}، \code{رشته.بپیوند}، \code{رشته.جایگزینی} و
\code{رشته.شامل است} دارد.

\begin{tip}
\code{.بشکاف()} یکی از پرکاربردترین تابع‌ها در پردازشِ داده است. هر وقت داده‌ای
دارید که با کاما، فاصله یا نشانهٔ دیگری جدا شده (مثلِ یک خطِ فایلِ CSV)، با
\code{بشکاف} آن را به اجزایش تقسیم کنید.
\end{tip}

\section{نگاشت: واژه‌نامهٔ کلید-مقدار}

\textbf{نگاشت} (HashMap) ساختاری است که هر \emph{کلید} را به یک \emph{مقدار}
وصل می‌کند درست مانندِ یک واژه‌نامه که هر واژه را به معنایش وصل می‌کند. این
مثالِ واقعی سن‌های چند نفر را نگه می‌دارد:

\begin{salamfa}
تابع ریشه:
    ناپایا سن‌ها := نگاشت {} برگردان نگاشت<رشته، صحیح>
    دیرکن سن‌ها.آزادکن()
    سن‌ها.درج("سارا"، ۳۰)
    سن‌ها.درج("علی"، ۲۵)
    سن‌ها.درج("رضا"، ۴۰)
    سرچاپ "سن سارا:"، سن‌ها.بگیر("سارا")
    سرچاپ "علی موجود است؟"، سن‌ها.دارد("علی")
    سرچاپ "حسن موجود است؟"، سن‌ها.دارد("حسن")
    سرچاپ "تعداد:"، سن‌ها.اندازه()
پایان
\end{salamfa}

\begin{progoutfa}
سن سارا: 30
علی موجود است؟ true
حسن موجود است؟ false
تعداد: 3
\end{progoutfa}

گونهِ \code{نگاشت<رشته، صحیح>} یعنی «نگاشتی که کلیدهایش رشته و مقدارهایش عددِ
صحیح‌اند». نگاشتِ تهی با \code{نگاشت \{\}} ساخته می‌شود و گونه‌اش را با
\code{برگردان} می‌گیرد. نگاشت حافظه می‌گیرد، پس درست پس از ساختنش آزاد کردنش را
با \code{دیرکن ... آزادکن()} به تعویق می‌اندازیم. متدهای اصلی:

\begin{center}
\begin{tabular}{l l l}
\toprule
\textbf{فارسی} & \textbf{انگلیسی} & \textbf{کار} \\
\midrule
\code{.درج(کلید، مقدار)} & \code{put} & افزودن یا به‌روزرسانی \\
\code{.بگیر(کلید)} & \code{get} & گرفتنِ مقدارِ یک کلید \\
\code{.دارد(کلید)} & \code{has} & آیا کلید هست؟ \\
\code{.حذف(کلید)} & \code{remove} & حذفِ یک کلید \\
\code{.اندازه()} & \code{size} & تعدادِ عناصر \\
\code{.پیمایش()} & \code{iter} & ساختنِ پیمایشگر \\
\bottomrule
\end{tabular}
\end{center}

\subsection{پیمایشِ یک نگاشت}

برای گذشتن از همهٔ عناصرِ یک نگاشت، ساده‌ترین راه حلقه‌ی \code{هر} است که در
هر دور یک کلید و مقدارش را می‌دهد. این مثالِ واقعی، مقدارهای یک نگاشت را جمع
می‌زند:

\begin{salamfa}
تابع ریشه:
    ناپایا امتیاز := نگاشت {} برگردان نگاشت<رشته، صحیح>
    دیرکن امتیاز.آزادکن()
    امتیاز.درج("الف"، ۳)
    امتیاز.درج("ب"، ۷)
    ناپایا مجموع := ۰
    هر _، عدد در امتیاز:
        مجموع = مجموع + عدد
    پایان
    سرچاپ "جمع"، مجموع
پایان
\end{salamfa}

\begin{progoutfa}
جمع 10
\end{progoutfa}

شکلِ کاملِ این حلقه \code{هر نام، عدد در امتیاز:} است که در هر دور کلید را در
\code{نام} و مقدار را در \code{عدد} می‌گذارد. اینجا کلید را لازم نداریم و متغیرِ
استفاده‌نشده در سلام خطاست، پس به‌جایش \code{\_} نوشته‌ایم. ترتیبِ عنصرها در این پیمایش ترتیبِ درج نیست و نباید
به آن تکیه کنید.

راهِ دیگر، ساختنِ یک \textbf{پیمایشگر} با \code{.پیمایش()} است. پیمایشگر سه متدِ
کلیدی دارد: \code{داردبعدی()} (آیا عنصرِ دیگری مانده؟)، \code{مقدار()} (و
\code{کلید()}) برای عنصرِ جاری، و \code{بعدی()} برای رفتن به عنصرِ بعد:

\begin{salamfa}
تابع ریشه:
    ناپایا امتیاز := نگاشت {} برگردان نگاشت<رشته، صحیح>
    دیرکن امتیاز.آزادکن()
    امتیاز.درج("الف"، ۳)
    امتیاز.درج("ب"، ۷)
    ناپایا مجموع := ۰
    ناپایا پیمایه := امتیاز.پیمایش()
    تا پیمایه.داردبعدی():
        مجموع = مجموع + پیمایه.مقدار()
        پیمایه.بعدی()
    پایان
    سرچاپ "جمع"، مجموع
پایان
\end{salamfa}

این الگو، یعنی «تا وقتی عنصری مانده، پردازش کن و جلو برو»، همان کاری است که
حلقه‌ی \code{هر} در پسِ پرده انجام می‌دهد.

\begin{tip}
نگاشت برای جست‌وجوی سریع عالی است. اگر می‌خواهید بدانید «سنِ سارا چند است؟»،
نگاشت پاسخ را تقریباً بی‌درنگ می‌دهد بی‌آنکه مجبور باشید مثلِ یک بردار،
همه‌چیز را یکی‌یکی بگردید. هر جا نیاز به نگاشتنِ «نام به داده» داشتید (شمارهٔ
دانشجو به نمره، نامِ کالا به قیمت)، نگاشت ابزارِ درست است.
\end{tip}

\section{دیگر ماژول‌های کاربردی}

کتابخانهٔ استانداردِ سلام بسیار گسترده است. در ادامه چند ماژولِ پرکاربردِ دیگر
را به‌اختصار معرفی می‌کنیم؛ شیوهٔ کارشان همان است: \code{واردسازی} با نامِ فارسی،
سپس دسترسی با نامِ ماژول و نقطه. نامِ انگلیسیِ هر ماژول را در پرانتز آورده‌ایم تا
در مرجعِ کتابخانه پیدایش کنید.

\begin{center}
\begin{tabular}{l p{0.5\linewidth}}
\toprule
\textbf{ماژول} & \textbf{کاربرد} \\
\midrule
\code{ورودی خروجی} (\code{io}) & ورودی و خروجی؛ خواندنِ ورودیِ کاربر، کار با فایل \\
\code{زمان} (\code{time}) & زمان و تاریخ؛ \code{اکنون}، \code{خواب} \\
\code{تصادفی} (\code{rand}) & عددهای تصادفی؛ \code{بذر}، \code{عدد تصادفی}، \code{عدد در بازه} \\
\code{قالب بندی} (\code{fmt}) & قالب‌بندیِ متن؛ ساختنِ رشته از عدد و مقدارهای دیگر \\
\code{سیستم عامل} (\code{os}) & سیستم‌عامل؛ آرگومان‌ها، متغیرهای محیطی، فایل‌ها \\
\code{کدگذاری.جیسون} (\code{encoding.json}) & خواندن و نوشتنِ داده‌های \lr{JSON} \\
\code{آزمایش} (\code{testing}) & نوشتنِ آزمون؛ \code{ادعا} و گزارشِ نتیجه \\
\code{همگام سازی} (\code{sync}) & هم‌روندی؛ قفل‌ها و هماهنگیِ نخ‌ها (فصلِ ۱۷) \\
\code{شبکه.اچ تی تی پی} (\code{net.http}) & درخواست‌های وب؛ \code{دریافت}، \code{ارسال} \\
\bottomrule
\end{tabular}
\end{center}

ماژول‌های \code{سیستم عامل}، \code{ورودی خروجی}، \code{زمان} و
\code{شبکه.اچ تی تی پی} را در فصل‌های آینده (۱۶ و ۱۷) بیشتر می‌بینیم؛
\code{آزمایش} را در فصلِ ۲۰. ماژولی که درونِ ماژولِ دیگری است با نقطه وارد
می‌شود (\code{واردسازی کدگذاری.جیسون}) و با بخشِ آخرِ نامش به کار می‌رود
(\code{جیسون.دریافت(...)}).

\subsection{نمونه: عددِ تصادفی}

\begin{salamfa}
واردسازی تصادفی

تابع ریشه:
    تصادفی.بذر(۷)
    سرچاپ "یک عدد تصادفی:"، تصادفی.عدد تصادفی()
    سرچاپ "یک تاس:"، تصادفی.عدد در بازه(۱، ۶)
پایان
\end{salamfa}

\begin{progoutfa}
یک عدد تصادفی: 8472296265413360938
یک تاس: 6
\end{progoutfa}

نخست با \code{تصادفی.بذر} «بذرِ» مولدِ تصادف را می‌کاریم (تا دنبالهٔ اعداد قابلِ
بازتولید باشد)، سپس با \code{تصادفی.عدد تصادفی} یک عددِ تصادفیِ بزرگ (از گونه‌ی
\code{صحیح۶۴}) می‌گیریم و با \code{تصادفی.عدد در بازه(۱، ۶)} عددی میانِ ۱ و ۶،
هر دو با خودشان. با همین بذرِ ۷، خروجی در هر اجرا همین است.

\begin{deep}[نگاهی به درون: تصادفِ واقعی یا ساختگی؟]
رایانه نمی‌تواند عددِ «واقعاً» تصادفی بسازد؛ آنچه می‌سازد دنباله‌ای است که
\emph{تصادفی به‌نظر می‌رسد} اما در واقع از یک فرمولِ معین و یک «بذرِ» ریشه
زاده می‌شود. به همین دلیل اگر بذرِ یکسان بدهید، همان دنباله را می‌گیرید که
برای آزمون و اشکال‌زدایی بسیار مفید است. اگر تصادفِ غیرقابلِ‌پیش‌بینی
می‌خواهید، \code{تصادفی.بذر خودکار()} را صدا بزنید که بذر را از زمانِ جاری
می‌گیرد؛ اگر هیچ بذری نکارید هم سلام خودش همین کار را می‌کند.
\end{deep}

\section{خلاصهٔ فصل}

\begin{itemize}
  \item ماژولِ \code{ریاضی} تابع‌های ریاضی و پایاهایی مانندِ $\pi$ دارد.
  \item رشته‌ها متدهای پیشرفتهٔ متن دارند: \code{بشکاف}، \code{بیاب}،
        \code{پیراست}، \code{به صحیح}؛ بسته‌ی \code{رشته} تابع‌های بیشتری دارد.
  \item \code{نگاشت} هر کلید را به یک مقدار وصل می‌کند و جست‌وجو را سریع
        می‌سازد.
  \item حلقه‌ی \code{هر کلید، مقدار در نگاشت} از همه‌ی عنصرها می‌گذرد؛
        پیمایشگر با \code{داردبعدی}، \code{مقدار} و \code{بعدی} همین کار را
        گام‌به‌گام می‌کند.
  \item در پرونده‌ی فارسی، نامِ ماژول‌ها و تابع‌هایشان فارسی است.
  \item ماژول‌های فراوانِ دیگری برای زمان، تصادف، فایل، شبکه و آزمون هست.
\end{itemize}

\section{تمرین‌ها}

\exercise با ماژولِ \code{ریاضی}، طولِ وترِ مثلثی با اضلاعِ ۳ و ۴ را حساب کنید
(قضیهٔ فیثاغورس).

\exercise یک نگاشت بسازید که نامِ چند میوه را به قیمتشان وصل کند، سپس قیمتِ یک
میوه را بپرسید.

\exercise با پیمایشِ یک نگاشتِ قیمت‌ها، گران‌ترین کالا را پیدا کنید.

\exercise رشتهٔ \code{"۱۰،۲۰،۳۰،۴۰"} را با \code{بشکاف("،")} به تکه‌ها بشکنید، هر
تکه را با \code{به صحیح()} به عدد تبدیل کنید و مجموعشان را بیابید.

\exercise با ماژولِ \code{تصادفی}، یک «تاسِ» شش‌وجهی بسازید که عددی میانِ ۱ تا ۶
بدهد؛ یک بار با \code{تصادفی.عدد در بازه} و یک بار با \code{تصادفی.عدد تصادفی} و
باقی‌ماندهٔ تقسیم بر ۶.
