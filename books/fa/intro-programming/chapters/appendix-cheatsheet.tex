\chapter{برگهٔ مرجعِ سریع}%
\label{app:cheatsheet}

این پیوست رایج‌ترین الگوهای زبانِ سلام را در یک نگاه گرد آورده است برای
وقتی که فقط می‌خواهید شکلِ درستِ چیزی را سریع به یاد بیاورید.

\section{اسکلتِ برنامه}

\begin{salamfa}
روال ریشه:
    سرچاپ "سلام، دنیا!"
پایان
\end{salamfa}

\section{متغیر و چاپ}

\begin{salamfa}
ناپایا شمار: صحیح = 0      // تغییرپذیر
نام: رشته = "سارا"          // تغییرناپذیر
سن: خودکار = 30             // استنتاج
سرچاپ نام, سن               // چاپ با خطِ تازه
چاپ "بدونِ خطِ تازه "
\end{salamfa}

\section{عملگرها}

\begin{salamfa}
+  -  *  /  %  ^^          // حسابی
+=  -=  *=  /=  %=  ^^=     // انتساب ترکیبی
==  !=  <  >  <=  >=        // مقایسه
&&  ||  !                   // منطقی
الف + " " + ب              // پیوندِ رشته
\end{salamfa}

\section{شرط}

\begin{salamfa}
اگر سن >= 18:
    سرچاپ "بزرگسال"
وگرنه سن >= 13:
    سرچاپ "نوجوان"
وگرنه:
    سرچاپ "کودک"
پایان
\end{salamfa}

\section{حلقه‌ها}

\begin{salamfa}
تا شرط:  ...  پایان
تکرار 10 از i:  ...  پایان        // i از 0 تا 9
تکرار 5:  ...  پایان
تکرار 1 تا 10:  ...  پایان
تکرار 1 تا 10 از i:  ...  پایان   // بازهٔ شامل، با شمارنده
تکرار 0 تا 100 هر 10:  ...  پایان
// بشکن  /  بگذر
\end{salamfa}

\section{روال}

\begin{salamfa}
روال جمع(الف: صحیح, ب: صحیح): صحیح:
    برگشت الف + ب
پایان
\end{salamfa}

\section{آرایه و بردار}

\begin{salamfa}
اعداد: صحیح[3] = [1, 2, 3]
سرچاپ اعداد[0], len(اعداد)

ناپایا ب: وکتور<صحیح> = وکتور {}
ب.بیفزا(10)        // افزودن
ب.بگیر(0)[0]       // خواندن
ب.طول()            // طول
ب.دربیاور()        // برداشتنِ آخر
\end{salamfa}

\section{رشته}

\begin{salamfa}
س.طول()                 // طول
س.زیررشته(0, 3)         // برش
س.پیوست("!")            // چسباندن
س.بشکاف(",")            // شکستن به بردار
"42".به‌صحیح()          // به عدد
\end{salamfa}

\section{نگاشت}

\begin{salamfa}
ناپایا م: نگاشت<رشته, صحیح> = نگاشت {}
م.درج("الف", 1)        // افزودن
م.بگیر("الف")          // خواندن
م.دارد("الف")          // وجود؟
م.اندازه()             // تعداد
\end{salamfa}

\section{ساختار و شمارشی}

\begin{salamfa}
ساختار نقطه:
    x: صحیح = 0
    y: صحیح = 0
    روال جمع(): صحیح: برگشت این.x + این.y پایان
پایان
ناپایا پ: نقطه = نقطه { x = 3, y = 4 }
پ.جمع()

جداشمار رنگ: قرمز, سبز, آبی پایان
رنگ.قرمز برگردان صحیح
\end{salamfa}

\section{میانجی و کلی‌سازی}

\begin{salamfa}
میانجی شکل:
    روال مساحت(): صحیح
پایان

روال توصیف(ش: dyn شکل): سرچاپ ش.مساحت() پایان

روال بیشینه<ت>(الف: ت, ب: ت): ت:
    اگر الف > ب: برگشت الف پایان
    برگشت ب
پایان
\end{salamfa}

\section{لامبدا و دیرکرد}

\begin{salamfa}
دوبرابر: func(صحیح) صحیح = (ن: صحیح) => ن * 2
دوبرابر(21)

دیرکن سرچاپ "پاکسازی"     // در پایانِ روال اجرا می‌شود
\end{salamfa}

\section{ماژول و بیرونی}

\begin{salamfa}
واردسازی "ریاضی"
ریاضی.جذر(144.0)

فراخوانه:
    روال printf(format: رشته, ...): صحیح
پایان
\end{salamfa}
