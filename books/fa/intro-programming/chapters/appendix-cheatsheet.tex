\chapter{برگهٔ مرجعِ سریع}%
\label{app:cheatsheet}

این پیوست رایج‌ترین الگوهای زبانِ سلام را در یک نگاه گرد آورده است، برای
وقتی که فقط می‌خواهید شکلِ درستِ چیزی را سریع به یاد بیاورید. هر تکه یک
برنامهٔ کامل است که همان‌طور که هست ساخته و اجرا می‌شود.

\section{اسکلتِ برنامه}

\begin{salamfa}
تابع ریشه:
    سرچاپ "سلام، دنیا!"
پایان
\end{salamfa}

\section{متغیر و چاپ}

\begin{salamfa}
تابع ریشه:
    ناپایا شمار := 0                // تغییرپذیر
    نام := "سارا"                  // تغییرناپذیر
    سن := 30                       // گونه از مقدار به دست می‌آید: صحیح
    بزرگ := 5 برگردان صحیح۶۴       // گزینشِ گونه با برگردان
    شمار += 1
    سرچاپ نام، سن، شمار، بزرگ       // چاپ با خطِ تازه
    چاپ "بدونِ خطِ تازه "
پایان
\end{salamfa}

\section{عملگرها}

\begin{salamfa}
تابع ریشه:
    الف := 7
    ب := 2
    سرچاپ الف + ب، الف - ب، الف * ب، الف / ب، الف % ب    // حسابی
    سرچاپ 2 ** 10                                    // توان (اعشار۶۴)
    ناپایا ج := الف
    ج += 1                                           // انتساب ترکیبی: += -= *= /= %=
    سرچاپ الف == ب، الف != ب، الف < ب، الف >= ب       // مقایسه
    سرچاپ الف بزرگتر ب، الف برابر ب                   // مقایسه با واژه
    سرچاپ (الف > 0 و ب > 0)، (الف < 0 یا ب < 0)، وارونه درست   // منطقی
    سرچاپ "سلام" + " " + "دنیا"، ج                    // پیوندِ رشته
پایان
\end{salamfa}

\section{شرط}

\begin{salamfa}
تابع ریشه:
    سن := 15
    اگر سن >= 18:
        سرچاپ "بزرگسال"
    وگرنه سن >= 13:
        سرچاپ "نوجوان"
    وگرنه:
        سرچاپ "کودک"
    پایان
پایان
\end{salamfa}

\section{حلقه‌ها}

\begin{salamfa}
تابع ریشه:
    ناپایا ک := 0
    تا ک < 3:                          // تا وقتی شرط برقرار است
        ک += 1
    پایان
    تکرار 10 در ای:  چاپ ای، ""  پایان          // ای از 0 تا 9
    تکرار 5:  چاپ "*"  پایان                    // پنج بار
    تکرار 1 تا 10 در ای:  چاپ ای، ""  پایان     // بازهٔ شامل، با شمارنده
    تکرار 0 تا 100 هر 10 در ای:  چاپ ای، ""  پایان   // با گام
    هر ع در [1، 2، 3]:  چاپ ع  پایان            // پیمایشِ عناصر
    هر ش، ع در [10، 20]:  چاپ ش، ع  پایان     // با اندیس
    // بشکن: خروج از حلقه   /   گذر: رفتن به دورِ بعد
    سرچاپ ""
پایان
\end{salamfa}

\section{تابع}

\begin{salamfa}
تابع جمع(الف: صحیح، ب: صحیح): صحیح:
    برگشت الف + ب
پایان

تابع ریشه:
    سرچاپ جمع(2، 3)
پایان
\end{salamfa}

\section{آرایه و بردار}

\begin{salamfa}
تابع ریشه:
    اعداد := [1، 2، 3]                    // آرایهٔ ثابت
    سرچاپ اعداد[0]، طول(اعداد)

    ناپایا ب := وکتور {} برگردان وکتور<صحیح>
    دیرکن ب.آزادکن()
    ب.بیفزا(10)                           // افزودن
    ب.بیفزا(20)
    سرچاپ ب.بگیر(0)، ب.طول()              // خواندن، طول
    سرچاپ ب.دربیاور()                     // برداشتنِ آخر
پایان
\end{salamfa}

\section{رشته}

\begin{salamfa}
تابع ریشه:
    س := "سلام، دنیا"
    سرچاپ س.طول()                  // طول به بایت
    سرچاپ س.شمارنویسه()            // شمارِ حرف‌ها
    سرچاپ س.زیررشته(0، 8)          // برش (بایت)
    سرچاپ س.پیوست("!")             // چسباندن
    تکه‌ها := س.بشکاف("، ")         // شکستن به بردار
    دیرکن تکه‌ها.آزادکن()
    سرچاپ تکه‌ها.طول()
    سرچاپ "42".به صحیح() + 1       // به عدد
پایان
\end{salamfa}

\section{نگاشت}

\begin{salamfa}
تابع ریشه:
    ناپایا م := نگاشت {} برگردان نگاشت<رشته، صحیح>
    دیرکن م.آزادکن()
    م.درج("الف"، 1)                // افزودن
    سرچاپ م.بگیر("الف")            // خواندن
    سرچاپ م.دارد("الف")            // وجود دارد؟
    سرچاپ م.اندازه()               // تعداد
    هر ک، م۲ در م:  سرچاپ ک، م۲  پایان
پایان
\end{salamfa}

\section{ساختار و شمارشی}

\begin{salamfa}
ساختار نقطه:
    همگانی ایکس: صحیح = 0
    همگانی ایگرگ: صحیح = 0
    همگانی تابع جمع(): صحیح: برگشت این.ایکس + این.ایگرگ پایان
پایان

جداشمار رنگ: قرمز، سبز، آبی پایان

تابع ریشه:
    پ := نقطه { ایکس = 3، ایگرگ = 4 }
    سرچاپ پ.جمع()
    ر := رنگ.سبز
    سرچاپ ر برگردان صحیح
    نام := همخوان ر:
        قرمز: "قرمز" پایان
        سبز، آبی: "سرد" پایان
    پایان
    سرچاپ نام
پایان
\end{salamfa}

\section{میانجی و کلی‌سازی}

\begin{salamfa}
میانجی شکل:
    تابع مساحت(): صحیح
پایان

ساختار مربع:
    همگانی ضلع: صحیح = 0
پایان

کاربست شکل بر مربع:
    تابع مساحت(): صحیح: برگشت این.ضلع * این.ضلع پایان
پایان

تابع توصیف(ش: پویا شکل): سرچاپ ش.مساحت() پایان

تابع بیشینه<ت>(الف: ت، ب: ت): ت:
    اگر الف > ب: برگشت الف پایان
    برگشت ب
پایان

تابع ریشه:
    توصیف(مربع { ضلع = 3 })
    سرچاپ بیشینه(4، 9)
پایان
\end{salamfa}
