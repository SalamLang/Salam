\chapter{دستورِ زبان (گرامر)}%
\label{app:grammar}

این پیوست ساختارهای نحویِ اصلیِ زبانِ سلام را به‌صورتِ فشرده گرد آورده است. هدف،
مرجعی سریع برای یادآوریِ شکلِ درستِ هر ساختار است، نه تعریفِ صوریِ کامل.

\section{ساختارِ کلیِ برنامه}

یک برنامهٔ اجرایی از چند تعریف (کارکرد، ساختار، شمارشی، \ldots) ساخته می‌شود و
حتماً یک کارکردِ \code{آغازین} دارد:

\begin{salam}
// تعریف‌های دلخواه (کارکردها، ساختارها، ...)
کارکرد آغازین:
    // بدنهٔ برنامه
پایان
\end{salam}

\section{اعلانِ متغیر}

\begin{salam}
گذرا نام : ریخت = مقدار      // تغییرپذیر
پایدار  نام : ریخت = مقدار      // تغییرناپذیر
نام : ریخت = مقدار            // تغییرناپذیر (شکلِ ساده)
نام خودکار = مقدار         // استنتاجِ ریخت
\end{salam}

\section{کارکرد}

\begin{salam}
کارکرد نام(پارامتر۱ : ریخت, پارامتر۲ : ریخت) : ریختِ‌خروجی:
    // بدنه
    بازگشت مقدار
پایان

کارکرد نام(پارامتر : ریخت):    // بدونِ مقدارِ بازگشتی
    // بدنه
پایان
\end{salam}

پارامترِ ارجاعی با \code{\&} مشخص می‌شود:

\begin{salam}
کارکرد نام(پارامتر &: ریخت):
    // تغییرِ پارامتر، متغیرِ بیرون را عوض می‌کند
پایان
\end{salam}

\section{شرط}

\begin{salam}
اگر شرط:
    // ...
وگرنه شرطِ‌دیگر:
    // ...
وگرنه:
    // ...
پایان
\end{salam}

\section{حلقه‌ها}

\begin{salam}
تاهنگام شرط:
    // ...
پایان

چرخه N:                    // N بار (بدونِ شمارنده)
چرخه N با i:                // N بار، با شمارندهٔ i از 0 تا N-1
چرخه آ تا ب:                // از آ تا بِ شامل
چرخه آ تا ب با i:            // با شمارندهٔ i
چرخه آ تا ب گام گ:           // با گامِ گ
چرخه آ تا ب گام گ با i:       // گام + شمارنده

هر عنصر در مجموعه:
    // ...
پایان

هر (i, عنصر) در مجموعه:       // با اندیس
    // ...
پایان
\end{salam}

کنترلِ حلقه: \code{بشکن} (خروج) و \code{بگذر} (پرش به تکرارِ بعد).

\section{ساختار و شمارشی}

\begin{salam}
ساختار نام:
    میدان : ریخت = پیش‌فرض
    کارکرد متد() : ریختِ‌خروجی:
        بازگشت این.میدان
    پایان
پایان

// ساختنِ نمونه:
نام { میدان = مقدار }

شمارش نام:
    عضو۱
    عضو۲ = 5
    عضو۳
پایان
// دسترسی: نام.عضو۱   و   نام.عضو۱ برگردان صحیح
\end{salam}

\section{میانجی و چندریختی}

\begin{salam}
میانجی نام:
    کارکرد متد() : ریختِ‌خروجی     // فقط امضا
پایان

کارکرد کاربر(پارامتر : dyn نام):
    پارامتر.متد()
پایان
\end{salam}

\section{کلی‌سازی}

\begin{salam}
کارکرد نام<ت>(پارامتر : ت) : ت:
    // ...
پایان

ساختار نام<آ, ب>:
    میدان۱ : آ
    میدان۲ : ب
پایان
// کاربرد: نام<صحیح>   یا   نام<رشته, صحیح>
\end{salam}

\section{لامبدا و بستار}

\begin{salam}
// ریختِ کارکرد:
func(ریختِ‌ورودی) ریختِ‌خروجی

// لامبدای یک‌عبارتی:
(پارامتر: ریخت) => عبارت

// لامبدای چنددستوری:
() => :
    // دستورها
    بازگشت مقدار
پایان
\end{salam}

\section{ماژول و دسترسی}

\begin{salam}
بسته نام                       // در فایلِ کتابخانه
همگانی کارکرد ...                  // عضوِ همگانی
همگانی پایدار ...

فراخوانی "نامِ‌ماژول"           // فراخوانیِ ساده
فراخوانی مستعار "فایل.salam"   // با نامِ مستعار
\end{salam}

\section{اجرای به‌تعویق‌افتاده و بیرونی}

\begin{salam}
دیرکرد دستور                    // در پایانِ کارکرد اجرا می‌شود

بیرونی:
    کارکرد نامِ‌بیرونی(پارامتر : ریخت) : ریختِ‌خروجی
پایان

پیوند پویا "نامِ‌کتابخانه"      // پیوند به کتابخانهٔ بیرونی
\end{salam}

\section{چیدمان (وب)}

\begin{salam}
چیدمان:
    ویژگی = مقدار
    عنصر:
        ویژگی = مقدار
        عنصرِ‌تودرتو: ویژگی = مقدار پایان
    پایان
پایان
\end{salam}

\section{قاعده‌های کلی}

\begin{itemize}
  \item بلوک‌ها با \code{:} باز و با \code{پایان} بسته می‌شوند.
  \item سازنده‌های مقدار (ساختار، آرایه) از \code{\{\,\}} و \code{[\,]} استفاده
        می‌کنند، نه \code{پایان}.
  \item توضیحاتِ تک‌خطی با \code{//} آغاز می‌شوند.
  \item رشته‌ها میانِ \code{"\,"} و کاراکترها میانِ \code{'\,'} می‌آیند.
  \item ارقامِ محاسباتی همیشه لاتین‌اند (\code{2}، نه ۲).
  \item اندیسِ آرایه‌ها و بردارها از صفر آغاز می‌شود.
\end{itemize}
