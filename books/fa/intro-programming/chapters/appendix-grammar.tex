\chapter{دستورِ زبان (گرامر)}%
\label{app:grammar}

این پیوست ساختارهای نحویِ اصلیِ زبانِ سلام را به‌صورتِ فشرده گرد آورده است. هدف،
مرجعی سریع برای یادآوریِ شکلِ درستِ هر ساختار است، نه تعریفِ صوریِ کامل.

\section{ساختارِ کلیِ برنامه}

یک برنامهٔ اجرایی از چند تعریف (تابع، ساختار، شمارشی، \ldots) ساخته می‌شود و
حتماً یک تابعِ \code{ریشه} دارد:

\begin{salamfa}
// ترتیبِ سطحِ بالا: بسته، واردسازی‌ها، فراخوانی‌ها، ثابت‌ها و متغیرهای سراسری،
// گونه‌ها (ساختار، جداشمار، میانجی، کاربست)، و سرانجام تابع‌ها
تابع ریشه:
    سرچاپ "بدنهٔ برنامه"
پایان
\end{salamfa}

\section{اعلانِ متغیر}

متغیر همیشه با \code{:=} و یک مقدارِ آغازین ساخته می‌شود و گونه‌اش از همان مقدار
به دست می‌آید؛ برای گونه‌ای دیگر از \code{برگردان} کمک بگیرید. شکلِ
\code{نام: گونه = مقدار} فقط برای میدان‌های ساختار و پارامترهای تابع است.

\begin{salamfa}
پایا بیشینه := 100                  // ثابت (نامِ یک‌واژه‌ای)

تابع ریشه:
    نام := "سارا"                   // تغییرناپذیر؛ گونه از مقدار
    ناپایا شمار := 0                 // تغییرپذیر
    کوچک := 5 برگردان صحیح۸          // با گزینشِ گونه
    شمار = شمار + بیشینه             // انتساب به متغیرِ موجود
    سرچاپ نام، شمار، کوچک
پایان
\end{salamfa}

\section{تابع}

\begin{salamfa}
تابع جمع(الف: صحیح، ب: صحیح): صحیح:  // نام(پارامتر: گونه، ...): گونهٔ خروجی
    برگشت الف + ب
پایان

تابع گزارش(پیام: رشته):              // بدونِ مقدارِ بازگشتی
    سرچاپ پیام
پایان

تابع ریشه:
    گزارش("حاصل:")
    سرچاپ جمع(2، 3)
پایان
\end{salamfa}

پارامترِ ارجاعی با \code{\&} مشخص می‌شود:

\begin{salamfa}
تابع افزایش(ن &: صحیح):          // پارامترِ ارجاعی
    ن += 1                       // متغیرِ بیرون را عوض می‌کند
پایان

تابع ریشه:
    ناپایا شمار := 1
    افزایش(شمار)
    سرچاپ شمار
پایان
\end{salamfa}

\section{شرط}

\begin{salamfa}
تابع ریشه:
    دما := 18
    اگر دما > 30:
        سرچاپ "گرم"
    وگرنه دما > 15:              // شرطِ دیگر
        سرچاپ "معتدل"
    وگرنه:
        سرچاپ "سرد"
    پایان
پایان
\end{salamfa}

\section{همخوان و ترابرد}

\begin{salamfa}
جداشمار رنگ: قرمز، سبز، آبی پایان

تابع ریشه:
    ر := رنگ.سبز
    نام := همخوان ر:                     // همخوان یک مقدار برمی‌گرداند
        قرمز: "گرم" پایان
        سبز، آبی: "سرد" پایان            // چند الگو در یک حالت
    پایان
    سرچاپ نام
    ک := 2
    ترابرد ک:
        1:
            سرچاپ "یک"
            بشکن                         // بدونِ بشکن به حالتِ بعد می‌ریزد
        پایان
        2، 3:
            سرچاپ "دو یا سه"
            بشکن
        پایان
        وگرنه:
            سرچاپ "دیگر"
        پایان
    پایان
پایان
\end{salamfa}

\section{حلقه‌ها}

\begin{salamfa}
تابع ریشه:
    ناپایا ک := 0
    تا ک < 3:                      // تا وقتی شرط برقرار است
        ک += 1
    پایان
    تکرار 3:  چاپ "*"  پایان               // N بار، بدونِ شمارنده
    تکرار 3 در ای:  چاپ ای  پایان          // شمارندهٔ ای از 0 تا N-1
    تکرار 1 تا 3:  چاپ "+"  پایان          // از آ تا ب، شامل
    تکرار 1 تا 3 در ای:  چاپ ای  پایان     // با شمارنده
    تکرار 0 تا 10 هر 5:  چاپ "-"  پایان    // با گام
    تکرار 0 تا 10 هر 5 در ای:  چاپ ای  پایان   // گام و شمارنده
    مجموعه := [7، 8، 9]
    هر عنصر در مجموعه:
        چاپ عنصر
    پایان
    هر اندیس، عنصر در مجموعه:       // با اندیس
        چاپ اندیس، عنصر
    پایان
    سرچاپ ""
پایان
\end{salamfa}

کنترلِ حلقه: \code{بشکن} (خروج) و \code{گذر} (پرش به تکرارِ بعد).

\section{ساختار و شمارشی}

\begin{salamfa}
ساختار نقطه:
    همگانی ایکس: صحیح = 0                // میدان: گونه = پیش‌فرض
    همگانی ایگرگ: صحیح = 0
    همگانی تابع جمع(): صحیح:            // متد
        برگشت این.ایکس + این.ایگرگ
    پایان
پایان

جداشمار وضعیت:
    آماده
    درکار
    تمام
پایان

تابع ریشه:
    پ := نقطه { ایکس = 1، ایگرگ = 2 }   // ساختنِ نمونه
    سرچاپ پ.جمع()
    و۱ := وضعیت.تمام                     // دسترسی به عضو
    سرچاپ و۱ برگردان صحیح
پایان
\end{salamfa}

\section{میانجی و چندریختی}

\begin{salamfa}
میانجی نامدار:
    تابع نام(): رشته                     // فقط امضا
پایان

ساختار گربه:
    همگانی لقب: رشته = ""
پایان

کاربست نامدار بر گربه:
    تابع نام(): رشته: برگشت این.لقب پایان
پایان

تابع معرفی(چیز: پویا نامدار):          // هر گونه‌ای که میانجی را دارد
    سرچاپ چیز.نام()
پایان

تابع ریشه:
    معرفی(گربه { لقب = "پشمک" })
پایان
\end{salamfa}

\section{کلی‌سازی}

\begin{salamfa}
ساختار جفت<آ، ب>:
    همگانی یکم: آ
    همگانی دوم: ب
پایان

تابع همان<ت>(مقدار: ت): ت:
    برگشت مقدار
پایان

تابع ریشه:
    سرچاپ همان(5)، همان("سلام")
    ج := جفت { یکم = "سن"، دوم = 30 } برگردان جفت<رشته، صحیح>
    سرچاپ ج.یکم، ج.دوم
پایان
\end{salamfa}

\section{لامبدا و بستار}

\begin{salamfa}
تابع اعمال(ت: تابع (صحیح) صحیح، ع: صحیح): صحیح:   // گونهٔ تابع
    برگشت ت(ع)
پایان

تابع ریشه:
    سه‌برابر := (ن: صحیح): برگشت ن * 3 پایان   // لامبدای یک‌خطی
    خوشامد := ():                                // لامبدای چنددستوری
        سرچاپ "درود"
        برگشت 1
    پایان
    سرچاپ اعمال(سه‌برابر، 4)، خوشامد()
پایان
\end{salamfa}

\section{ماژول و دسترسی}

\begin{salamfa}
// پرونده‌ی ابزار.salam
بسته ابزار

همگانی پایا نسخه := 2

همگانی تابع دوبرابر(ع: صحیح): صحیح:
    برگشت ع * 2
پایان
\end{salamfa}

و برنامه‌ای که آن را به کار می‌برد:

\begin{salamfa}
// پرونده‌ی main.salam
واردسازی ریاضی

فراخوانی ابزار "ابزار.salam"         // فایلِ خودتان، با نامِ ابزار
// فراخوانی "ابزار.salam"            نام از «بسته»ی همان فایل
// فراخوانی ابزار "@/lib/ابزار.salam"   مسیر از ریشه‌ی پروژه

تابع ریشه:
    سرچاپ ابزار.دوبرابر(ابزار.نسخه)
    سرچاپ ریاضی.جذر(16.0)
پایان
\end{salamfa}

\section{اجرای به‌تعویق‌افتاده و بیرونی}

\begin{salamfa}
اگر سلام سیستم ویندوز:
    پیوند پویا "winsqlite3"
وگرنه:
    پیوند پویا "sqlite3"              // پیوند به کتابخانهٔ بیرونی
پایان

درون‌داد:
    تابع sqlite3_libversion(): رشته    // نامِ بیرونی(پارامتر: گونه): گونهٔ خروجی
پایان

برون‌داد:                              // تابعِ سلام با نامِ ثابتِ C
    تابع on_event(کد: صحیح): صحیح:
        برگشت کد * 2
    پایان
پایان

تابع ریشه:
    دیرکن سرچاپ "پایان"                // در پایانِ تابع اجرا می‌شود
    سرچاپ sqlite3_libversion() != ""
پایان
\end{salamfa}

\section{صفحه (وب)}

\begin{salamfa}
صفحه:
    عنوان = "نامِ صفحه"                  // ویژگیِ صفحه = مقدار
    جعبه:                                // عنصر
        رنگ = "آبی"                      // ویژگیِ عنصر = مقدار
        پاراگراف: محتوا = "متن" پایان     // عنصرِ تودرتو در یک خط
    پایان
پایان
\end{salamfa}

\section{قاعده‌های کلی}

\begin{itemize}
  \item بلوک‌ها با \code{:} باز و با \code{پایان} بسته می‌شوند.
  \item سازنده‌های مقدار (ساختار، آرایه) از \code{\{\,\}} و \code{[\,]} استفاده
        می‌کنند، نه \code{پایان}.
  \item توضیحاتِ تک‌خطی با \code{//} آغاز می‌شوند.
  \item رشته‌ها میانِ \code{"\,"} و نویسه‌ها میانِ \code{'\,'} می‌آیند.
  \item عددها را می‌توان با رقم‌های فارسی یا لاتین نوشت (\code{۱۲} همان
        \code{12} است)، اما ممیزِ اعشار همیشه نقطه است (\code{3.5})؛ عددها
        همیشه با رقم‌های لاتین چاپ می‌شوند.
  \item ویرگولِ فارسی (\code{،}) و ویرگولِ لاتین (\code{,}) هر دو جداکننده‌اند.
  \item بلوک‌های \code{ترابرد} به حالتِ بعد می‌ریزند مگر با \code{بشکن}؛
        \code{همخوان} نمی‌ریزد و باید همهٔ حالت‌ها را پوشش دهد.
  \item متغیر، پارامتر، واردسازی یا تابعِ بی‌استفاده، و \code{ناپایا}یی که هرگز
        دوباره مقدار نگیرد، خطای کامپایل است.
  \item اندیسِ آرایه‌ها و بردارها از صفر آغاز می‌شود.
\end{itemize}
