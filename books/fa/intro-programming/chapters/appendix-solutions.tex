\chapter{پاسخِ تمرین‌ها}%
\label{app:solutions}

در این پیوست پاسخِ تمرین‌های منتخبِ هر فصل آمده است. به یاد داشته باشید که در
برنامه‌نویسی اغلب \emph{چند} راهِ درست وجود دارد؛ اگر پاسخِ شما با اینجا فرق
دارد اما درست کار می‌کند، آن هم پاسخی معتبر است. پیش از دیدنِ پاسخ، خودتان
بکوشید.

\textbf{تمرینِ ۳.۵} (مکعب با \code{**=}):

\begin{salamfa}
تابع ریشه:

    ضلع **= 3
    سرچاپ ضلع
پایان
\end{salamfa}

\begin{progoutfa}

\end{progoutfa}

\code{**} همیشه \code{اعشار۶۴} می‌دهد و یک \code{اعشار۶۴} بدونِ تبدیلِ صریح در متغیرِ \code{صحیح} جا نمی‌گیرد؛ پس \code{ضلع **= 3} به همان دلیلی رد می‌شود که \code{ضلع = ضلع ** 3} رد می‌شد.

\textbf{تمرینِ ۳.۶} (نامِ کامل و طولش):
\begin{salamfa}

    خانوادگی := "احمدی"
    کامل := نام + " " + خانوادگی
    سرچاپ کامل
    سرچاپ "طول به بایت:"، کامل.طول()
    سرچاپ "شمارِ نویسه‌ها:"، کامل.شمارنویسه()
پایان
\end{salamfa}

\begin{progoutfa}

شمارِ نویسه‌ها: 10
\end{progoutfa}

\code{طول()} بایت‌ها را می‌شمارد و هر حرفِ فارسی دو بایت است؛ شمارِ حرف‌ها را \code{شمارنویسه()} می‌دهد.
