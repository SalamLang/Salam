\chapter{متغیرها و گونه‌های داده}%
\label{ch:variables}

در فصلِ~\ref{ch:algorithms}، هنگامِ طراحیِ الگوریتمِ «بزرگ‌ترین عدد»، به
جایی برای نگهداریِ «بزرگ‌ترینِ فعلی» نیاز پیدا کردیم. برنامه‌ها تقریباً
همیشه باید چیزهایی را به خاطر بسپارند: امتیازِ بازیکن، نامِ کاربر، مجموعِ
نمره‌ها. ابزارِ این «به خاطر سپردن»، \textbf{متغیر} است.

\section{متغیر؛ جعبه‌ای با نام و برچسب}

متغیر را مانندِ جعبه‌ای تصور کنید که رویش نامی نوشته‌اید و درونش یک مقدار
گذاشته‌اید. هر وقت آن نام را در برنامه بیاورید، یعنی «محتوای این جعبه».
ساختنِ متغیر در سلام چنین است:

\begin{salamfa}
تابع ریشه:
    امتیاز := 100
    سرچاپ امتیاز
پایان
\end{salamfa}

\begin{progout}
100
\end{progout}

خطِ دوم را چنین بخوانید: «جعبه‌ای به نامِ \textbf{امتیاز} بساز و مقدارِ
\textbf{۱۰۰} را در آن بگذار.» نشانهٔ \code{:=} یعنی «بساز و مقدار بده».
ساختار همیشه یکسان است:

\begin{center}
\textit{نام} \code{:=} \textit{مقدار}
\end{center}

لازم نیست بگویید درونِ جعبه چه جور داده‌ای است؛ سلام از خودِ مقدار
می‌فهمد که \code{100} یک عددِ صحیح است. دربارهٔ این «جور داده» یا
\emph{گونه} کمی جلوتر بیشتر می‌گوییم.

دقت کنید که در \code{سرچاپ امتیاز} خبری از نقل‌قول نیست: می‌خواهیم
\emph{محتوای} جعبه چاپ شود (۱۰۰)، نه واژهٔ «امتیاز».

\section{تغییر دادنِ مقدار}

اسمش رویش است: متغیر می‌تواند تغییر کند. اما سلام محتاط است و جعبه‌ای را که
با \code{:=} ساخته‌اید، به‌طورِ پیش‌فرض \emph{قفل} می‌کند. اگر می‌خواهید
بعدها محتوایش را عوض کنید، هنگامِ ساختن واژهٔ \code{ناپایا} (یعنی «تغییرپذیر»)
را جلویش بنویسید. از آن به بعد با نوشتنِ \code{نام = مقدارِ تازه} محتوای جعبه
عوض می‌شود (این بار دیگر \code{ناپایا} و \code{:=} را نمی‌نویسیم، چون جعبه از
قبل ساخته شده است):

\begin{salamfa}
تابع ریشه:
    ناپایا امتیاز := 100
    سرچاپ "امتیاز اول:", امتیاز
    امتیاز = 250
    سرچاپ "امتیاز بعدی:", امتیاز
    امتیاز = امتیاز + 50
    سرچاپ "با جایزه:", امتیاز
پایان
\end{salamfa}

\begin{progoutfa}
امتیاز اول: 100
امتیاز بعدی: 250
با جایزه: 300
\end{progoutfa}

خطِ \code{امتیاز = امتیاز + 50} برای تازه‌کارها عجیب است: مگر می‌شود چیزی
با خودش به‌علاوهٔ ۵۰ برابر باشد؟! راز اینجاست که علامتِ \code{=} در
برنامه‌نویسی «مساوی بودن» نیست؛ \textbf{انتساب} است، یعنی: «اول سمتِ راست
را حساب کن، بعد نتیجه را در جعبهٔ سمتِ چپ بگذار.» پس سمتِ راست با مقدارِ
فعلی (۲۵۰) حساب می‌شود و می‌شود ۳۰۰، بعد ۳۰۰ در جعبه گذاشته می‌شود. این
همان نمادِ $\leftarrow$ شبه‌کدهای فصلِ~\ref{ch:algorithms} است.

\begin{note}
ترتیبِ خط‌ها اهمیت دارد. برنامه از بالا به پایین اجرا می‌شود و هر
\code{سرچاپ} مقدارِ \emph{همان لحظهٔ} متغیر را نشان می‌دهد؛ به همین دلیل سه
خروجیِ متفاوت دیدیم.
\end{note}

\section{جعبه‌های قفل و ثابت‌ها}

پس جعبه‌ای که بی‌\code{ناپایا} ساخته شود، قفل است. اگر جایی از برنامه سهواً
به آن مقدارِ تازه بدهید، کامپایلر خطا می‌گیرد و جلوی اشتباه را همان اول
می‌گیرد. برعکسش هم درست است: اگر جعبه‌ای را \code{ناپایا} بسازید و هیچ‌وقت
عوضش نکنید، کامپایلر باز هم خطا می‌گیرد و می‌گوید \code{ناپایا} را بردارید.
قاعده این است: \emph{مگر واقعاً به تغییر نیاز دارید، قفل‌شده بسازید}؛
برنامه‌ای که جعبه‌های کمتری در آن عوض می‌شوند، کم‌اشتباه‌تر است.

بعضی مقدارها هم از همان لحظهٔ نوشتنِ برنامه معلوم‌اند و هرگز عوض نمی‌شوند:
تعدادِ روزهای هفته، تعدادِ ماه‌های سال. به چنین مقدارهایی \textbf{ثابت}
می‌گویند و با واژهٔ \code{پایا} ساخته می‌شوند:

\begin{salamfa}
تابع ریشه:
    پایا روزهای‌هفته := 7
    سرچاپ روزهای‌هفته
پایان
\end{salamfa}

\begin{progout}
7
\end{progout}

اگر بنویسید \code{روزهای‌هفته = 8}، کامپایلر خطا می‌گیرد. نامِ ثابت باید یک
واژه باشد؛ برای همین «روزهای» و «هفته» را با نیم‌فاصله به هم چسبانده‌ایم.

\section{گونه‌های داده}

تا اینجا فقط با عددِ صحیح کار کردیم، اما داده‌ها جورواجورند. مهم‌ترین
گونه‌های سلام این‌هاست:

\begin{center}
\begin{tabular}{lll}
\toprule
گونه & چه چیزی نگه می‌دارد & نمونه \\
\midrule
\code{صحیح}  & عددِ درست (بی‌اعشار)         & \code{42} یا \code{-7} \\
\code{اعشار۶۴} & عددِ اعشاری                & \code{3.14} یا \code{-0.5} \\
\code{رشته}  & متن                          & \code{"سلام"} \\
\code{منطقی} & فقط \code{درست} یا \code{نادرست} & \code{درست} \\
\code{نویسه} & یک نویسهٔ لاتینِ تنها        & \code{'a'} \\
\bottomrule
\end{tabular}
\end{center}

\begin{salamfa}
تابع ریشه:
    نام := "سارا"
    سن := 12
    قد := 1.48
    عضو باشگاه := درست
    سرچاپ نام, سن, قد, عضو باشگاه
پایان
\end{salamfa}

\begin{progoutfa}
سارا 12 1.48 true
\end{progoutfa}

هر چهار جعبه گونهٔ خودشان را از مقدارشان گرفته‌اند: \code{نام} رشته است،
\code{سن} صحیح، \code{قد} اعشار۶۴ و \code{عضو باشگاه} منطقی. دقت کنید که
مقدارِ منطقی در خروجی به انگلیسی چاپ می‌شود: \code{درست} به صورتِ
\lr{true} و \code{نادرست} به صورتِ \lr{false}. عددها هم همیشه با رقم‌های
انگلیسی چاپ می‌شوند، حتی اگر در برنامه با رقمِ فارسی نوشته باشید.

گونهٔ \code{منطقی} شاید غریب به نظر برسد اما ستارهٔ فصلِ~\ref{ch:conditionals} است: هر جا برنامه بخواهد «تصمیم» بگیرد، پای یک مقدارِ
منطقی در میان است.

\subsection{چرا گونه مهم است؟}

گونه به کامپایلر می‌گوید با محتوای جعبه چه کارهایی مجاز است. دو عددِ صحیح را
می‌شود ضرب کرد، اما ضربِ دو رشته بی‌معنی است. اگر بنویسید
\code{"سلام" - 3}، کامپایلر همان‌جا خطا می‌گیرد، چون کم کردنِ عدد از متن
معنایی ندارد. این سخت‌گیری دوستِ شماست:
بسیاری از اشتباه‌ها پیش از اجرا، همان لحظهٔ ساختنِ برنامه، پیدا می‌شوند.

\subsection{انتخابِ گونه با \texorpdfstring{\code{برگردان}}{برگردان}}

دیدیم که سلام گونهٔ جعبه را از مقدارش حدس می‌زند: \code{12} صحیح است و
\code{1.48} اعشار۶۴. گاهی اما گونهٔ دیگری می‌خواهیم؛ مثلاً عددی آن‌قدر بزرگ
که در \code{صحیح} جا نمی‌شود. آن وقت با واژهٔ \code{برگردان} گونه را خودمان
انتخاب می‌کنیم:

\begin{salamfa}
تابع ریشه:
    جمعیت جهان := 8100000000 برگردان صحیح۶۴
    سن := 12
    سرچاپ جمعیت جهان, سن
پایان
\end{salamfa}

\begin{progout}
8100000000 12
\end{progout}

\code{صحیح۶۴} همان عددِ صحیح است، با جای بیشتر برای عددهای بسیار بزرگ.
هر جعبه پس از ساخته شدن گونهٔ دقیقِ خودش را دارد و تا آخر همان می‌ماند؛ در
جعبهٔ \code{سن} نمی‌توان بعدها متن گذاشت.

\section{نام‌گذاریِ خوب}

نامِ متغیر را شما انتخاب می‌کنید و این انتخاب مهم‌تر از آن است که به نظر
می‌رسد. برنامه‌ای با نام‌های \code{الف} و \code{ب} و \code{ج} را یک هفته
بعد خودتان هم نمی‌فهمید. چند قاعدهٔ سرانگشتی:

\begin{itemize}
  \item نام باید بگوید جعبه \emph{چه چیزی} نگه می‌دارد: \code{مجموع نمرات}
        بهتر از \code{م}، و \code{تعداد دانش آموزان} بهتر از \code{ت} است.
  \item نام با عدد شروع نمی‌شود و علامت‌هایی مانندِ \code{+} و \code{-} در
        آن راه ندارند.
  \item دو متغیرِ هم‌نام نمی‌توان ساخت؛ هر جعبه برچسبِ یکتای خود را دارد.
\end{itemize}

\begin{deep}[نگاهی به درون: جعبه‌ها کجا هستند؟]
جعبه‌ها در \emph{حافظهٔ اصلیِ} رایانه (\lr{RAM}) ساخته می‌شوند. حافظه را
مانندِ دیواری از میلیون‌ها خانهٔ کوچکِ شماره‌دار تصور کنید. وقتی می‌نویسید
\code{امتیاز := 100}، سلام چند خانهٔ کنارِ هم را برمی‌دارد،
عددِ ۱۰۰ را در آن‌ها می‌گذارد و نامِ «امتیاز» را به نشانیِ آن خانه‌ها گره
می‌زند. گونهٔ متغیر تعیین می‌کند چند خانه لازم است و الگوی صفر و یک‌ها چگونه
تفسیر شود.
\end{deep}

\section{تمرین‌ها}

\exercise سه متغیر بسازید: نامِ کتابِ محبوبتان (رشته)، تعدادِ صفحه‌هایش
(صحیح) و امتیازی که به آن می‌دهید (عددی اعشاری از ۵، مثلاً \code{4.5}). هر سه را در یک خط چاپ
کنید.

\exercise متغیری به نامِ \code{موجودی} با مقدارِ ۵۰۰ بسازید. با سه خطِ
جداگانه: ۲۰۰ به آن اضافه کنید، بعد ۱۵۰ از آن کم کنید و در پایان چاپش کنید.
پیش از اجرا حدس بزنید چه عددی چاپ می‌شود.

\exercise این برنامه را بدونِ اجرا دنبال کنید و بگویید چه چاپ می‌کند:
\begin{salamfa}
تابع ریشه:
    ناپایا آ := 4
    ناپایا ب := 9
    آ = ب
    ب = آ
    سرچاپ آ, ب
پایان
\end{salamfa}
اگر هدف «جابه‌جا کردنِ» مقدارِ دو جعبه بود، این برنامه موفق شده است؟ چرا؟

\exercise با یک متغیرِ \emph{سوم} برنامهٔ تمرینِ قبل را طوری بازنویسی کنید
که مقدارها واقعاً جابه‌جا شوند (راهنمایی: قبل از خراب کردنِ \code{آ}،
مقدارش را جایی امن نگه دارید).

\exercise کدام‌یک برای «تعدادِ خواهر و برادرها» گونهٔ مناسب‌تری است: صحیح یا
اعشار؟ برای «وزن» چطور؟ برای «آیا چراغ روشن است؟» چطور؟
