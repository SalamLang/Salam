\chapter{کارکردها: شکستنِ برنامه به قطعه‌ها}%
\label{ch:functions}

هر سه آجرِ الگوریتم (توالی، انتخاب، تکرار) را داریم و از این پس هیچ
الگوریتمی از دستمان درنمی‌رود. اما با بزرگ‌تر شدنِ برنامه، مشکلِ تازه‌ای سر
برمی‌آورد: شلوغی. راهِ‌حلِ برنامه‌نویس‌ها از روزِ اول یکی بوده است: برنامهٔ
بزرگ را به قطعه‌های کوچکِ نام‌دار بشکن. به این قطعه‌ها \textbf{کارکرد}
می‌گویند.

\section{کارکرد چیست؟}

کارکرد بسته‌ای از دستورهاست که نامی دارد و کارِ مشخصی می‌کند. یک بار
می‌نویسیدش و هر بار که لازم شد، فقط \emph{صدایش می‌زنید}. با یکی از
آشناترین‌ها شروع کنیم؛ همان «بزرگ‌ترین از میان دو عدد»:

\begin{salamfa}
کارکرد بیشینه(الف : صحیح, ب : صحیح) : صحیح:
    اگر الف > ب:
        بازگشت الف
    پایان
    بازگشت ب
پایان

کارکرد آغازین:
    چاپ بیشینه(7, 12)
    چاپ بیشینه(40, 3)
    چاپ بیشینه(5, 5)
پایان
\end{salamfa}

\begin{progout}
12
40
5
\end{progout}

ساختارِ تعریفِ کارکرد:

\begin{center}
\code{کارکرد} \;\textit{نام}\code{(}\textit{پارامترها}\code{)}\;:\;\textit{ریختِ خروجی}\;\code{:}
\quad\ldots\;بدنه\;\ldots\quad \code{پایان}
\end{center}

تکه‌تکه‌اش کنیم:

\begin{itemize}
  \item \code{بیشینه} نامِ کارکرد است؛ همان قاعده‌های نام‌گذاریِ متغیرها
        اینجا هم برقرار است: نام باید بگوید کارکرد \emph{چه می‌کند}.
  \item \code{(الف صحیح, ب صحیح)} دو \textbf{پارامتر} است: ورودی‌های
        کارکرد، هر یک با نام و ریخت.
  \item \code{صحیح}ِ پس از پرانتز، ریختِ \textbf{خروجی} است: این کارکرد در
        پایان یک عددِ صحیح تحویل می‌دهد.
  \item \code{بازگشت} خروجی را تحویل می‌دهد و کارِ کارکرد را \emph{همان
        لحظه} تمام می‌کند. به همین دلیل در بدنهٔ بالا به \code{وگرنه}
        نیازی نبود: اگر \code{الف > ب} باشد، با \code{بازگشت الف} کارِ
        کارکرد تمام است و هرگز به خطِ \code{بازگشت ب} نمی‌رسیم.
\end{itemize}

وقتی در کارکردِ آغازین می‌نویسیم \code{بیشینه(7, 12)}، می‌گوییم: «کارکردِ بیشینه
را با الفِ ۷ و بِ ۱۲ اجرا کن و هر چه برگرداند، همین‌جا بگذار.» به این کار
\textbf{فراخوانی} می‌گویند. ترتیب مهم است: مقدارِ اول در پارامترِ اول
می‌نشیند و به همین ترتیب.

\section{چرا کارکرد؟}

سه دلیلِ بزرگ:

\begin{itemize}
  \item \textbf{پرهیز از تکرار:} اگر جایی سه بار یک منطق را کپی کرده‌اید،
        وقتِ کارکرد ساختن است. فردا که اشکالی پیدا شد، یک جا را درست
        می‌کنید نه سه جا را.
  \item \textbf{خوانایی:} \code{چاپ بیشینه(الف, ب)} را هر کسی می‌فهمد؛
        نسخهٔ باز شده‌اش را باید خط‌به‌خط خواند.
  \item \textbf{آزمون‌پذیری:} هر کارکرد را می‌توان جدا آزمود. سه فراخوانیِ
        مثالِ بالا در واقع سه \emph{آزمایش} بودند؛ حالتِ سوم (دو مقدارِ
        برابر) همان حالتِ مرزی‌ای است که در تمرینِ فصلِ
        \ref{ch:algorithms} بررسی کردید.
\end{itemize}

\begin{tip}
قاعدهٔ طلایی: \emph{هر کارکرد فقط یک کار بکند و آن را خوب بکند.} کارکردی که هم
حساب می‌کند، هم چاپ می‌کند و هم تصمیم می‌گیرد، نه خوانا است، نه
دوباره‌استفاده‌شدنی و نه آزمودنی. کوتاه نگهش دارید؛ اگر از حدودِ بیست خط
گذشت، احتمالاً دو کار در آن قاطی شده و باید دو کارکرد شود.
\end{tip}

\section{کارکردِ بدونِ خروجی}

هر کارکردی چیزی برنمی‌گرداند؛ بعضی فقط \emph{کاری می‌کنند} (مثلاً چاپ). در
این حالت ریختِ خروجی را نمی‌نویسیم:

\begin{salamfa}
کارکرد خط بکش:
    چرخه 30:
        بنویس "-"
    پایان
    چاپ ""
پایان

کارکرد آغازین:
    خط بکش()
    چاپ "گزارش فروش"
    خط بکش()
پایان
\end{salamfa}

\begin{progoutfa}
------------------------------
گزارش فروش
------------------------------
\end{progoutfa}

دقت کنید که کارکردِ بی‌پارامتر را با پرانتزِ خالی صدا می‌زنیم:
\code{خط بکش()}.

\section{متغیرهای محلی}

متغیری که درونِ کارکرد ساخته می‌شود، فقط درونِ همان کارکرد وجود دارد و بیرون
از آن دیده نمی‌شود؛ به آن \textbf{متغیرِ محلی} می‌گویند. پارامترها هم
محلی‌اند. این جداسازی نعمت است: هر کارکرد آشپزخانهٔ خودش را دارد و کارکردهای
دیگر نمی‌توانند ندانسته ظرف‌هایش را جابه‌جا کنند. اگر کارکردی به داده‌ای
نیاز دارد، آن را از راهِ پارامتر بدهید و نتیجه را از راهِ \code{بازگشت}
پس بگیرید؛ این تنها درِ ورودی و خروجیِ تمیزِ یک کارکرد است.

\section{ترکیبِ کارکردها}

خروجیِ یک کارکرد می‌تواند ورودیِ کارکردی دیگر باشد؛ کارکردها مثلِ قطعه‌های لگو در
هم چفت می‌شوند. بزرگ‌ترین از میانِ \emph{سه} عدد، بدونِ نوشتنِ حتی یک
مقایسهٔ تازه:

\begin{salamfa}
کارکرد بیشینه(الف : صحیح, ب : صحیح) : صحیح:
    اگر الف > ب:
        بازگشت الف
    پایان
    بازگشت ب
پایان

کارکرد بیشینه سه تایی(الف : صحیح, ب : صحیح, ج : صحیح) : صحیح:
    بازگشت بیشینه(بیشینه(الف, ب), ج)
پایان

کارکرد آغازین:
    چاپ بیشینه سه تایی(7, 12, 5)
پایان
\end{salamfa}

\begin{progout}
12
\end{progout}

خطِ \code{بیشینه(بیشینه(الف, ب), ج)} از درون به بیرون خوانده می‌شود: اول
بزرگ‌ترینِ الف و ب پیدا می‌شود، بعد حاصلش با ج سنجیده می‌شود. این «ساختن از
قطعه‌های کوچکِ مطمئن» جانِ کلامِ مهندسیِ نرم‌افزار است.

یک نمونهٔ کامل‌تر که شرط و حلقه و کارکرد را با هم دارد؛ آزمونِ اول بودنِ یک
عدد (عددی که فقط بر ۱ و خودش بخش‌پذیر است):

\begin{salamfa}
کارکرد اول است(عدد : صحیح) : منطقی:
    اگر عدد < 2:
        بازگشت نادرست
    پایان
    گذرا مقسوم علیه : صحیح = 2
    تاهنگام (مقسوم علیه * مقسوم علیه) <= عدد:
        اگر (عدد % مقسوم علیه) == 0:
            بازگشت نادرست
        پایان
        مقسوم علیه = مقسوم علیه + 1
    پایان
    بازگشت درست
پایان

کارکرد آغازین:
    چرخه 2 تا 20 با n:
        اگر اول است(n):
            بنویس n
            بنویس " "
        پایان
    پایان
    چاپ ""
پایان
\end{salamfa}

\begin{progout}
2 3 5 7 11 13 17 19
\end{progout}

دو ریزه‌کاریِ این کارکرد خواندنی است: نگهبانِ اولِ کار حالت‌های مرزی (اعدادِ
کمتر از ۲) را یک‌جا کنار می‌گذارد؛ و شرطِ
\code{(مقسوم علیه * مقسوم علیه) <= عدد} یعنی فقط تا جذرِ عدد می‌گردیم، چون
اگر عددی مقسوم‌علیهی بزرگ‌تر از جذرش داشته باشد، حتماً یکی کوچک‌تر از جذر
هم دارد. با همین یک شرط، آزمودنِ اعدادِ بزرگ به‌مراتب سریع‌تر می‌شود؛
نمونه‌ای کوچک از اینکه \emph{فکرِ الگوریتمی} چگونه به برنامهٔ بهتر تبدیل
می‌شود.

\section{تمرین‌ها}

\exercise کارکردِ \code{کمینه(الف صحیح, ب صحیح) صحیح} را بنویسید که
کوچک‌ترین را برگرداند، و با سه فراخوانی بیازمایید (فراموش نکنید: یکی از
آزمایش‌ها دو مقدارِ برابر باشد).

\exercise کارکردِ \code{قدر مطلق(عدد صحیح) صحیح} را بنویسید: اگر عدد منفی
بود قرینه‌اش را برگرداند، وگرنه خودش را.

\exercise کارکردِ \code{زوج است(عدد صحیح) منطقی} را بنویسید و سپس با کمکِ
آن (نه با نوشتنِ دوبارهٔ \code{\%})، اعدادِ زوجِ ۱ تا ۳۰ را چاپ کنید.

\exercise کارکردِ \code{توان(پایه صحیح, نما صحیح) صحیح} را با الگوی انباشتنِ
ضربی بنویسید؛ \code{توان(2, 10)} باید \code{1024} شود.

\exercise کارکردِ \code{شمارش مقسوم علیه ها(عدد صحیح) صحیح} را بنویسید که
تعدادِ مقسوم‌علیه‌های عدد را برگرداند. سپس در کارکردِ آغازین، با آن عددهای ۱
تا ۳۰ را بررسی کنید و هر عددی که دقیقاً دو مقسوم‌علیه دارد را چاپ کنید.
خروجی باید همان اعدادِ اول باشد. چرا؟
