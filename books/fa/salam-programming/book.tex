\def\salamfa@mergeloop#1{%
  \ifx#1\relax\else
    \ifnum`#1>127
      \ifnum`#1<1776 \salamfa@mergeablefalse
      \else\ifnum`#1>1785 \salamfa@mergeablefalse
      \else \salamfa@hasfadigittrue\salamfa@hasanydigittrue
      \fi\fi
    \else
      \ifnum`#1>47 \ifnum`#1<58 \salamfa@hasanydigittrue\fi\fi
      \ifnum`#1>64 \ifnum`#1<91 \salamfa@haslettertrue\fi\fi
      \ifnum`#1>96 \ifnum`#1<123 \salamfa@haslettertrue\fi\fi
    \fi
    \expandafter\salamfa@mergeloop
  \fi}

\newcommand{\salamfa@laddexp}[1]{\g@addto@macro\salamfa@lineacc{#1}}
\newcommand{\salamfa@tokC}[2]{%
  \salamfa@mergeabletrue\salamfa@hasfadigitfalse\salamfa@hasanydigitfalse\salamfa@hasletterfalse
  \salamfa@mergeloop#2\relax
  \ifnum\salamfa@brkdepth=\z@\else\salamfa@mergeablefalse\fi
  \ifsalamfa@mergeable
    \ifsalamfa@lineopen
      \salamfa@standalonefalse
    \else
      \ifsalamfa@hasanydigit\salamfa@standalonefalse\else
        \ifsalamfa@hasletter\salamfa@standalonefalse\else\salamfa@standalonetrue\fi
      \fi
    \fi
    \ifsalamfa@standalone
      \ifsalamfa@spacepending
        \salamfa@gadd{\space}%
        \salamfa@spacependingfalse
      \fi
      \salamfa@opswap{#2}%
      \edef\salamfa@tokCtmp{\noexpand#1{{\noexpand\latincodefont\salamfa@opswapped}}}%
      \expandafter\salamfa@gadd\expandafter{\salamfa@tokCtmp}%
    \else
    \salamfa@brkloop#2\relax
    \ifsalamfa@commapending
      \g@addto@macro\salamfa@lineacc{,}%
      \global\salamfa@commapendingfalse
    \fi
    \ifsalamfa@spacepending
      \g@addto@macro\salamfa@lineacc{\space}%
      \salamfa@spacependingfalse
    \fi
    \global\salamfa@lineopentrue
    \IfEndWith{#2}{,}%
      {\StrGobbleRight{#2}{1}[\salamfa@tokbody]\global\salamfa@commapendingtrue}%
      {\def\salamfa@tokbody{#2}}%
    \ifsalamfa@hasfadigit
      \edef\salamfa@tokCtmp{\noexpand#1{{\noexpand\salamcodefont\salamfa@tokbody}}}%
    \else
      \edef\salamfa@tokCtmp{\noexpand#1{\salamfa@tokbody}}%
    \fi
    \expandafter\salamfa@laddexp\expandafter{\salamfa@tokCtmp}%
    \fi
  \else
    \salamfa@lineflush
    \ifsalamfa@spacepending
      \salamfa@gadd{\space}%
      \salamfa@spacependingfalse
    \fi
    \salamfa@tokbrk\z@
    \salamfa@tokbrkloop#2\relax
    \global\advance\salamfa@brkdepth\salamfa@tokbrk
    \edef\salamfa@tokCtmp{\noexpand#1{#2}}%
    \expandafter\salamfa@gadd\expandafter{\salamfa@tokCtmp}%
  \fi}

\renewcommand{\salamfa@classify}[1]{%
  \ifsalamfa@linecmt
    \salamfa@tokC{\cmtword}{#1}%
  \else\ifsalamfa@incmt
    \salamfa@tokC{\cmtword}{#1}%
    \IfBeginWith{#1}{/*}{\global\advance\salamfa@cmtdepth\@ne}{}%
    \IfEndWith{#1}{*/}{\salamfa@closecmt}{}%
  \else\ifsalamfa@instr
    \salamfa@tokC{\stword}{#1}%
    \IfSubStr{#1}{"}{\salamfa@instrfalse}{}%
  \else
    \IfBeginWith{#1}{//}%
      {\salamfa@lineflush \salamfa@linecmttrue \salamfa@tokC{\cmtword}{#1}}%
      {\IfBeginWith{#1}{/*}{\salamfa@lineflush\salamfa@opencmt{#1}}{\salamfa@word{#1}}}%
  \fi\fi\fi}
\renewcommand{\salamfa@opencmt}[1]{%
  \salamfa@tokC{\cmtword}{#1}%
  \global\salamfa@cmtdepth\@ne \global\salamfa@incmttrue
  \IfEndWith{#1}{*/}{\salamfa@closecmt}{}}
\renewcommand{\salamfa@word}[1]{%
  \IfBeginWith{#1}{"}%
    {\StrGobbleLeft{#1}{1}[\salamfa@rest]%
     \expandafter\IfSubStr\expandafter{\salamfa@rest}{"}%
       {\salamfa@tokC{\stword}{#1}}%
       {\salamfa@tokC{\stword}{#1}\salamfa@instrtrue}}%
    {\ifcsname salamkw@#1\endcsname
       \salamfa@tokC{\kw}{#1}%
     \else\ifcsname salamty@#1\endcsname
       \salamfa@tokC{\tyword}{#1}%
     \else
       \StrLeft{#1}{1}[\salamfa@fc]%
       \ifcsname salamfa@dig@\salamfa@fc\endcsname
         \salamfa@tokC{\salamfaNM}{#1}%
       \else
         \salamfa@tokC{\salamfa@id}{#1}%
       \fi
     \fi\fi}}
\renewcommand{\salamfa@procline}[1]{%
  \global\advance\salamfa@nlines\@ne
  \salamfa@lead=0
  \salamfa@linecmtfalse \salamfa@instrfalse  %
  \global\salamfa@lineopenfalse \global\let\salamfa@lineacc\@empty
  \global\salamfa@spacependingfalse
  \global\salamfa@commapendingfalse
  \global\salamfa@brkdepth\z@
  \ifnum\salamfa@nlines>\@ne \salamfa@gadd{\\}\fi
  \salamfa@gadd{\strut}%

  \salamfa@inleadtrue
  \expandafter\salamfa@loop\salamfa@line
}
\def\salamfa@loop#1 {%
  \def\salamfa@tmp{#1}%
  \ifx\salamfa@tmp\salamfa@endmac
    \salamfa@lineflush
  \else
    \ifx\salamfa@tmp\@empty
      \ifsalamfa@inlead \advance\salamfa@lead\@ne \else \salamfa@feedspace \fi
    \else
      \ifsalamfa@inlead
        \salamfa@inleadfalse
        \ifnum\salamfa@lead>\z@
          \edef\salamfa@tmp{\noexpand\hspace*{\the\dimexpr\salamfa@lead\salamfa@unit\relax}}%
          \expandafter\salamfa@gadd\expandafter{\salamfa@tmp}%
        \fi
      \else
        \salamfa@feedspace
      \fi
      \salamfa@classify{#1}%
    \fi
    \expandafter\salamfa@loop
  \fi}
