% اصلاح چهار مشکل ماکروهای مشترک کارت کد برای متن راست‌به‌چپ:
% ۱) توکن تمام‌لاتین (عدد، عملگر، رشته‌ی انگلیسی) باید چپ‌به‌راست و بیرون
%    از حالت فارسی حروف‌چینی شود، وگرنه xepersian ترتیب رقم‌ها و
%    علامت‌ها را وارونه می‌کند («3.5,» به «,5.3»).
% ۲) رشته‌ای مثل «"abc",» که بعد از نقل قول پایانی علامت دارد، در ماکروی
%    اصلی «رشته‌ی باز» تلقی می‌شد و توکن‌های بعدی را قرمز و وارونه می‌کرد.
% ۳) از همه مهم‌تر: چند توکنِ تمام‌اسکی پشت سر هم (مثلاً سه توکن «(12,»
%    و «25),» و «19)») که هرکدام را جدا در \lr بگذاریم، به‌عنوان چند
%    قلم هم‌سطحِ جدا از نگاه bidi دیده می‌شوند و قاعده‌ی L2 آن‌ها را
%    نسبت به هم وارونه می‌کند؛ محتوای هر \lr درست است اما ترتیب‌شان
%    نسبت به هم به‌هم می‌ریزد («12, 25), 19)» می‌شود «19) 25), (12,»).
% راه‌حل برای ۳: توکن‌های تمام‌اسکیِ پشت‌سرهم (با فاصله‌ی معمولی میانشان)
% بلافاصله در \lr جداگانه گذاشته نمی‌شوند؛ در یک انباره‌ی سطحِ خط جمع
% می‌شوند و تازه وقتی به توکنِ غیرتمام‌اسکی یا پایان خط برسیم، کل انباره
% یک‌جا در یک \lr واحد گذاشته می‌شود.
% نکته‌ی مهم: توکنی که فارسی و لاتین را بی‌فاصله به هم چسبانده (مثل
% فراخوانی تودرتوی «بزرگتر(بزرگتر(12,» یا اندیس «نمرات[3]») را نباید به
% زیررشته‌های اسکی/فارسی شکست: یک پرانتزِ تنها که میان دو کلمه‌ی فارسی
% در \lr جدا بیفتد، حتی تنها، وارونه (آینه‌ای) چاپ می‌شود، چون در آن
% embedding چیز دیگری برای تطبیق نیست؛ اما همان پرانتز، دست‌نخورده و
% کنار همسایه‌های فارسی‌اش، درست چاپ می‌شود (قاعده‌ی N0 خودِ bidi برای
% جفت پرانتز). پس چنین توکنی دست‌نخورده و یک‌جا صادر می‌شود - دقیقاً مثل
% کد اصلیِ فایل، فقط با پرچمِ رنگ.
% ---------------------------------------------------------------------

\newif\ifsalamfa@lineopen       % آیا ناحیه‌ی اسکیِ رها (هنوز چاپ‌نشده) باز است؟
\newcommand{\salamfa@lineacc}{}% محتوای رنگ‌شده‌ی همان ناحیه، هنوز بدون \lr بیرونی

% \salamfa@lineacc از پیش پر است از فراخوانی‌های آماده‌ی رنگ‌کننده‌ها
% (مثل «\cmtword{//}»)؛ این‌ها هنوز اجرا نشده‌اند (با \noexpand محافظت
% شده بودند) و اجرای دستورهایی مثل \textcolor تنها زمانی درست است که واقعاً
% حروف‌چینی شوند. اگر این‌جا \salamfa@lineacc را مستقیم در یک \edef بگذاریم،
% \edef همه‌چیز را تا آخر باز می‌کند و همان \textcolor را همین الان - در دل
% edef - اجرا می‌کند که نتیجه‌اش دنباله‌ای درهم و شمارش نامتوازنِ if/fi است.
% به همین دلیل با یک ثبات توکن (\toks@) محتوا را فقط یک لایه باز می‌کنیم و
% خودِ محتوا دست‌نخورده می‌ماند؛ \the\toks@ درون edef هرگز دوباره باز
% نمی‌شود - همان ترفندی که \g@addto@macro در هسته‌ی لاتک به کار می‌برد.
% نکته‌ی ظریفِ آخر: \lr (بستهٔ bidi) یک embedding چپ‌به‌راست می‌سازد، نه
% یک override؛ درونِ یک embedding قاعده‌های حلِ نوعِ ضعیفِ یونیکد هنوز
% اجرا می‌شوند، و برای الگوی «عدد، یک عملگرِ تنها، عددِ دیگر» (مثلاً
% «۱ + ۲» با فاصله دو طرفِ «+») bidi آن سه‌تکه را نسبت به بافتِ بیرونی
% وارونه می‌نشاند. راه‌حل، نویسه‌های override یونیکد است: LRO
% (U+202D) پیش و PDF (U+202C) پس از محتوا. تنها با override به‌تنهایی
% حروف‌چینیِ پیوستنِ حروفِ فارسیِ مجاور در ادامه‌ی خط خراب می‌شد - چون
% \lr خودش محدوده‌ی فونتِ \latincodefont را در یک گروه می‌بست و اینجا،
% وقتی \lr را برداشتیم، آن گروه هم رفت و \latincodefont بدونِ مرز به
% بقیه‌ی خط نشت می‌کرد. برای همین کل محتوا - override و تعویضِ فونت و
% همه - داخل یک {...} صریح گذاشته شده تا هم override درست کار کند و هم
% فونت به بیرون درز نکند.
% اما خودِ override یک باگِ تازه باز می‌کند: توکنی مثل «0.3» (رقم، نقطه،
% رقم، بدونِ هیچ پرانتزی) زیرِ override دو تکه‌اش («0» و «3») جا به جا
% می‌شوند («0.30000000000000004» می‌شود «30000000000000004.0»؛
% «۷.۰ / ۲.۰» می‌شود «۰.۷ / ۰.۲»). \lr (که خودش یک embedding معمولی است،
% نه override) این مورد را درست چاپ می‌کند. اما \lr به نوبه‌ی خودش
% فراخوانیِ تودرتو («بزرگتر(...), ۱۹)») را خراب می‌کند: وقتی نیمه‌ی بازِ
% یک پرانتز در توکنِ خامِ فارسیِ قبل از ادغام نشسته و نیمه‌ی بسته‌اش
% («), ۱۹)») تویِ ناحیه‌ی ادغام‌شده است، یعنی «عمقِ پرانتزِ» این ناحیه
% صفر نیست (بسته‌ها بیشتر از بازها هستند) - دقیقاً همان حالتی که override
% لازم دارد. برعکس، آرایه‌ای مثلِ «[17, 15, 19, 12, 20]» که کلِ پرانتزش
% (این‌جا کروشه) خودش تویِ همان یک ناحیه جفت می‌شود، عمقش صفر است و مثل
% حالتِ بی‌پرانتز با \lr درست چاپ می‌شود؛ زیرِ override برعکس خراب
% می‌شد. پس معیار «عمقِ خالصِ پرانتز/کروشه» است، نه صرفِ وجودِ آن‌ها:
% اگر بازها و بسته‌های ناحیه با هم برابر نباشند (عمق ≠ صفر) از override
% استفاده کن؛ وگرنه (چه بدونِ هیچ پرانتزی، چه با پرانتزِ کاملاً جفت‌شده
% در همان ناحیه) از \lr ساده.
\newcommand{\salamfa@lineflush}{%
  \ifsalamfa@lineopen
    \toks@\expandafter{\salamfa@lineacc}%
    \ifnum\salamfa@brkdepth=\z@
      \edef\salamfa@linetmp{\noexpand\lr{\noexpand\latincodefont \the\toks@}}%
    \else
      \edef\salamfa@linetmp{{‭\noexpand\latincodefont \the\toks@‬}}%
    \fi
    \expandafter\salamfa@gadd\expandafter{\salamfa@linetmp}%
    \global\let\salamfa@lineacc\@empty
    \global\salamfa@lineopenfalse
    \global\salamfa@brkdepth\z@
  \fi
  \ifsalamfa@commapending
    \salamfa@gadd{,}%
    \global\salamfa@commapendingfalse
  \fi}

% فاصله‌ی میان‌واژه: اگر وسط یک ناحیه‌ی اسکیِ باز باشیم به همان انباره
% می‌رود (تا ناحیه پاره نشود)؛ وگرنه مثل قبل مستقیم روی خروجی می‌رود.
% نکته‌ی مهم: در یک embedding چپ‌به‌راستِ تویِ متنِ راست‌به‌چپ، لبه‌ی
% «آخرِ» محتوا (همان‌جا که این فاصله بعد از عددی مثل «18» می‌نشیند) بصریاً
% کنارِ چیزی می‌افتد که در ترتیبِ منطقی قبل از آن embedding آمده، نه بعدِ
% آن؛ همان‌طور که در یک جمله‌ی فارسیِ عادی، خودِ رقم‌های یک عدد
% (بدونِ \lr هم) چپ‌به‌راست خوانده می‌شوند اما «عدد» به‌عنوانِ یک واحد در
% جایگاهِ منطقی‌اش می‌نشیند. پس فاصله‌ای که قرار است بینِ «18» و کلمه‌ی
% فارسیِ بعدی («سال») دیده شود را نباید تویِ همان \lr، به‌عنوانِ آخرین
% چیز، جا داد - آن‌جا کنارِ چیزی می‌نشیند که قبل از عدد بوده، و «سال»
% بی‌فاصله به «18» می‌چسبد. راه‌حل: چنین فاصله‌ای را «معلق» نگه می‌داریم؛
% اگر توکنِ اسکیِ دیگری در ادامه آمد، همان‌جا (وسطِ ناحیه) اضافه‌اش
% می‌کنیم؛ اگر توکنِ غیرقابل‌ادغام (فارسی) آمد، ناحیه را می‌بندیم و
% فاصله را بیرون از \lr، به‌شکلِ معمولی، پیش از آن توکن می‌گذاریم.
\newif\ifsalamfa@spacepending
\newcommand{\salamfa@feedspace}{%
  \ifsalamfa@lineopen
    \salamfa@spacependingtrue
  \else
    \salamfa@gadd{\space}%
  \fi}

% فرق دارد: ویرگولِ تنها و بی‌همسایه‌ی عددیِ بعد از خودش، تویِ

% (تویِ ناحیه) برش می‌گردانیم، وگرنه هنگامِ بستنِ ناحیه بیرون از
% \lr/override صادرش می‌کنیم.

% \salamfa@tokC{<رنگ‌کننده>}{<توکن>} - اگر توکن تمام‌اسکی باشد به
% انباره‌ی سطحِ خط می‌رود (تا با توکن‌های اسکیِ همسایه در یک \lr ادغام
% شود)؛ وگرنه (فارسیِ خالص یا مخلوط) دست‌نخورده و مستقیم صادر می‌شود،
% بعد از آن‌که هر ناحیه‌ی اسکیِ رهاشده را ببندد.
% علامت‌ها و عملگرهای اسکی همیشه باید ادغام شوند، اما رقم‌های فارسی هم
% (که این کتاب حالا در کد به کار می‌برد) باید همراهشان ادغام شوند - وگرنه
% توکنی مثل «۲۵),» چون دیگر «تمام‌اسکی» نیست، از مسیر ادغام بیرون می‌افتد و
% همان مشکلِ ترتیب دوباره برمی‌گردد. پس معیار «قابل‌ادغام» می‌شود: هیچ

% فارسی، آن تکه باید با قلمِ کدِ فارسی نوشته شود، نه لاتین.
\newif\ifsalamfa@mergeable
\newif\ifsalamfa@hasfadigit
\def\salamfa@mergeloop#1{%
  \ifx#1\relax\else
    \ifnum`#1>127
      \ifnum`#1<1776 \salamfa@mergeablefalse
      \else\ifnum`#1>1785 \salamfa@mergeablefalse
      \else \salamfa@hasfadigittrue
      \fi\fi
    \fi
    \expandafter\salamfa@mergeloop
  \fi}
% \salamfa@brkdepth: عمقِ خالصِ پرانتز/کروشه‌ی ناحیه‌ی ادغام‌شده‌ی جاری
% (بازها منهای بسته‌ها)؛ برای انتخاب override در برابر \lr در
% \salamfa@lineflush - نگاه کنید به توضیح آن‌جا.
\newcount\salamfa@brkdepth
\def\salamfa@brkloop#1{%
  \ifx#1\relax\else
    \ifnum`#1=40 \advance\salamfa@brkdepth\@ne\fi
    \ifnum`#1=91 \advance\salamfa@brkdepth\@ne\fi
    \ifnum`#1=41 \advance\salamfa@brkdepth\m@ne\fi
    \ifnum`#1=93 \advance\salamfa@brkdepth\m@ne\fi
    \expandafter\salamfa@brkloop
  \fi}
% توکنِ مخلوطِ فارسی+اسکی که پرانتزِ جفت‌نشده دارد (مثل «بزرگتر(۱۲,»)
% دست‌نخورده و خام صادر می‌شود؛ اما همین پرانتزِ تنها، چون جفتش («)») در
% یک ناحیه‌ی جداگانه (ادغام‌شده) نشسته، به‌جای «(» آینه‌ای («)») چاپ
% می‌شود. راه‌حل: پیش از صدور، «(» و «)» را در متنِ خام جابه‌جا می‌کنیم؛
% وقتی خودِ bidi دوباره آینه‌اش می‌کند، شکلِ درست («(») از آب درمی‌آید.
\newcommand{\salamfa@laddexp}[1]{\g@addto@macro\salamfa@lineacc{#1}}
\newcommand{\salamfa@tokC}[2]{%
  \salamfa@mergeabletrue\salamfa@hasfadigitfalse\salamfa@mergeloop#2\relax
  \ifsalamfa@mergeable
    \salamfa@brkloop#2\relax
    \ifsalamfa@commapending
      \g@addto@macro\salamfa@lineacc{,}%
      \global\salamfa@commapendingfalse
    \fi
    \ifsalamfa@spacepending
      \g@addto@macro\salamfa@lineacc{\space}%
      \salamfa@spacependingfalse
    \fi
    \global\salamfa@lineopentrue
    \IfEndWith{#2}{,}% chktex 18
      {\StrGobbleRight{#2}{1}[\salamfa@tokbody]\global\salamfa@commapendingtrue}%
      {\def\salamfa@tokbody{#2}}%
    \ifsalamfa@hasfadigit
      \edef\salamfa@tokCtmp{\noexpand#1{{\noexpand\salamcodefont\salamfa@tokbody}}}%
    \else
      \edef\salamfa@tokCtmp{\noexpand#1{\salamfa@tokbody}}%
    \fi
    \expandafter\salamfa@laddexp\expandafter{\salamfa@tokCtmp}%
  \else

      \salamfa@gadd{\space}%

    \fi

    \StrSubstitute{\salamfa@pmB}{ZzLPARzZ}{)}[\salamfa@pmC]%

% حلقه‌ی خط را دوباره تعریف می‌کنیم تا فاصله‌های میان‌واژه هم از
% \salamfa@feedspace بگذرند (نه مستقیم \salamfa@gadd{\space})، و پایان
% خط هم هر ناحیه‌ی اسکیِ نیمه‌کاره را ببندد.
\renewcommand{\salamfa@procline}[1]{%
  \global\advance\salamfa@nlines\@ne
  \salamfa@lead=0
  \salamfa@linecmtfalse \salamfa@instrfalse  % these never span lines
  \global\salamfa@lineopenfalse \global\let\salamfa@lineacc\@empty
  \global\salamfa@spacependingfalse
  \global\salamfa@commapendingfalse
  \global\salamfa@brkdepth\z@
  \ifnum\salamfa@nlines>\@ne \salamfa@gadd{\\}\fi
  \salamfa@gadd{\strut}% keep blank lines non-empty (avoids consecutive \\)
  \edef\salamfa@line{\detokenize{#1}\space\salamfa@end\space}%
  \salamfa@inleadtrue
  \expandafter\salamfa@loop\salamfa@line
}
\def\salamfa@loop#1 {%
  \def\salamfa@tmp{#1}%
  \ifx\salamfa@tmp\salamfa@endmac
    \salamfa@lineflush
  \else
    \ifx\salamfa@tmp\@empty
      \ifsalamfa@inlead \advance\salamfa@lead\@ne \else \salamfa@feedspace \fi
    \else
      \ifsalamfa@inlead
        \salamfa@inleadfalse
        \ifnum\salamfa@lead>\z@
          \edef\salamfa@tmp{\noexpand\hspace*{\the\dimexpr\salamfa@lead\salamfa@unit\relax}}%
          \expandafter\salamfa@gadd\expandafter{\salamfa@tmp}%
        \fi
      \else
        \salamfa@feedspace
      \fi
      \salamfa@classify{#1}%
    \fi
    \expandafter\salamfa@loop
  \fi}
\makeatother
