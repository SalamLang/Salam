\section{یک نام برای یک کار}

از فصل اول با یک تابع کار می‌کنیم: \fcode{تابع اصلی}. حالا وقت آن است که
ببینیم «تابع» واقعاً چیست. \textbf{تابع} بلوکی از دستورهاست که یک نام
دارد. هر جا آن نام را بنویسیم (به این کار \textbf{فراخوانی} می‌گویند)،
دستورهای درون تابع اجرا می‌شوند. تابع اصلی هم همین است، با این تفاوت که
سلام آن را خودش، در آغاز برنامه، فراخوانی می‌کند.

فرض کنید در برنامه‌ای چند بار باید یک خط جداکننده چاپ کنیم. می‌توانیم هر
بار \fcode{چاپ "--------------------"} بنویسیم، یا این کار را یک بار در
تابعی تعریف کنیم و هر جا لازم بود، نامش را صدا بزنیم:

\program{تعریف و فراخوانی تابع}
\begin{salam}
تابع خط جداکننده():
    چاپ "--------------------"
تمام

تابع اصلی:
    خط جداکننده()
    چاپ "فهرست خرید"
    خط جداکننده()
    چاپ "نان، پنیر، شیر"
    خط جداکننده()
تمام
\end{salam}

\begin{outputfa}
--------------------
فهرست خرید
--------------------
نان، پنیر، شیر
--------------------
\end{outputfa}

تعریف تابع با واژه‌ی \kwd{تابع} شروع می‌شود، بعد نام تابع می‌آید، بعد یک
جفت پرانتز و دونقطه، و در پایان \fcode{تمام}. نام تابع هم مثل نام متغیر
می‌تواند چند واژه‌ای باشد. فراخوانی، نوشتن نام تابع همراه با پرانتز است:
\fcode{خط جداکننده()}. پرانتزها لازم‌اند؛ بدون آن‌ها سلام فکر می‌کند از یک
متغیر حرف می‌زنید.

به دو نکته دقت کنید. اول این که تابع را \emph{بیرون} تابع اصلی تعریف
کردیم، در بالای برنامه؛ تابع‌ها همیشه در سطح بالای برنامه و کنار هم
تعریف می‌شوند، نه درون یکدیگر. دوم این که ترتیب تعریف مهم نیست؛ می‌توانستیم
\fcode{خط جداکننده} را زیر \fcode{تابع اصلی} هم بنویسیم. سلام اول همه‌ی
تعریف‌ها را می‌بیند و بعد از \fcode{تابع اصلی} شروع می‌کند.

\begin{note}[چرا تابع؟]
سه دلیل اصلی دارد. \textbf{تکرار نکردن}: کاری که در چند جا لازم است، یک بار
نوشته می‌شود و اگر روزی باید عوض شود، یک جا عوض می‌شود. \textbf{نام
گذاشتن}: خواندن \fcode{خط جداکننده()} بسیار آسان‌تر از خواندن
\fcode{چاپ "--------------------"} است؛ نام، مقصود را می‌گوید.
\textbf{تقسیم کار}: یک مسئله‌ی بزرگ را به چند تابع کوچک می‌شکنیم و هر
کدام را جداگانه می‌نویسیم و آزمایش می‌کنیم.
\end{note}

\section{پارامتر: فرستادن مقدار به تابع}

تابع \fcode{خط جداکننده} همیشه یک کار ثابت می‌کند. اما بیشتر تابع‌ها به
اطلاعاتی نیاز دارند تا کارشان را انجام دهند: نام کسی که باید به او سلام
کرد، عددی که باید مربعش را حساب کرد. این اطلاعات را از راه
\textbf{پارامتر} به تابع می‌فرستیم. پارامترها را درون پرانتز تعریف تابع
می‌نویسیم و برای هر یک باید \emph{نوعش} را هم بگوییم:

\program{تابع با پارامتر}
\begin{salam}
تابع سلام کن(نام: رشته):
    چاپ "سلام", نام + "! خوش آمدی."
تمام

تابع اصلی:
    سلام کن("سارا")
    سلام کن("رضا")
    دوست := "مینا"
    سلام کن(دوست)
تمام
\end{salam}

\begin{outputfa}
سلام سارا! خوش آمدی.
سلام رضا! خوش آمدی.
سلام مینا! خوش آمدی.
\end{outputfa}

\fcode{نام: رشته} می‌گوید این تابع یک پارامتر به نام \fcode{نام} از نوع
رشته می‌خواهد. در هر فراخوانی، مقداری که درون پرانتز می‌نویسیم (به آن
\textbf{آرگومان} می‌گویند) در جعبه‌ی \fcode{نام} گذاشته می‌شود و تابع با
آن کار می‌کند. آرگومان می‌تواند یک مقدار مستقیم باشد، یک متغیر، یا حتی
یک محاسبه.

تابع می‌تواند چند پارامتر داشته باشد که با ویرگول جدا می‌شوند. هنگام
فراخوانی، آرگومان‌ها به همان ترتیب به پارامترها می‌رسند:

\program{چند پارامتر}
\begin{salam}
تابع ردیف علامت(تعداد: صحیح, علامت: رشته):
    متغیر خط := ""
    تکرار تعداد:
        خط += علامت
    تمام
    چاپ خط
تمام

تابع اصلی:
    ردیف علامت(5, "*")
    ردیف علامت(3, "#")
    ردیف علامت(8, "=")
تمام
\end{salam}

\begin{outputfa}
*****
###
========
\end{outputfa}

\begin{warn}
نوع آرگومان باید با نوع پارامتر بخواند. فراخوانی \fcode{ردیف علامت("5", "*")}
خطاست، چون \code{"5"} رشته است و پارامتر اول عدد صحیح می‌خواهد. تعداد
آرگومان‌ها هم باید درست باشد؛ نه کمتر و نه بیشتر.
\end{warn}

\section{بازگشت: گرفتن نتیجه از تابع}

تابع‌های بالا کاری می‌کردند (چاپ)، اما چیزی به فراخواننده پس نمی‌دادند.
بسیاری از تابع‌ها باید نتیجه‌ای \emph{بدهند}: مساحت، قیمت نهایی، جواب یک
سؤال. برای این کار دو چیز لازم است: نوع نتیجه را بعد از پرانتز اعلام
کنیم، و درون تابع با \kwd{بازگشت} نتیجه را پس بدهیم:

\program{تابع با مقدار بازگشتی}
\begin{salam}
تابع مساحت مستطیل(طول: صحیح, عرض: صحیح): صحیح:
    بازگشت طول * عرض
تمام

تابع اصلی:
    چاپ "مساحت اتاق:", مساحت مستطیل(4, 5)
    چاپ "مساحت حیاط:", مساحت مستطیل(12, 8)
    مجموع := مساحت مستطیل(4, 5) + مساحت مستطیل(12, 8)
    چاپ "مجموع:", مجموع
تمام
\end{salam}

\begin{outputfa}
مساحت اتاق: 20
مساحت حیاط: 96
مجموع: 116
\end{outputfa}

به سرآغاز تابع نگاه کنید: \fcode{تابع مساحت مستطیل(طول: صحیح, عرض: صحیح):
صحیح:}. بعد از پرانتز، یک دونقطه و نوع نتیجه (\fcode{صحیح}) آمده و بعد
دونقطه‌ی همیشگی که بلوک را باز می‌کند. دستور \fcode{بازگشت طول * عرض}
مقدار را حساب می‌کند، آن را به فراخواننده می‌دهد و تابع همان‌جا تمام
می‌شود.

مهم‌ترین نکته این است که فراخوانی چنین تابعی، خودش یک \emph{مقدار} است و
می‌توان آن را هر جا که یک مقدار می‌رود گذاشت: در \fcode{چاپ}، در یک
محاسبه، در تعریف یک متغیر، یا حتی به عنوان آرگومان تابعی دیگر.

\subsection{تابعی که بله یا نه می‌گوید}

تابعی که نتیجه‌ی منطقی می‌دهد، در شرط‌ها بسیار خوش‌دست است:

\program{تابع منطقی}
\begin{salam}
تابع زوج است(عدد: صحیح): منطقی:
    بازگشت عدد % 2 == 0
تمام

تابع اصلی:
    تکرار 1 تا 6 با عدد:
        اگر زوج است(عدد):
            چاپ عدد, "زوج"
        وگرنه:
            چاپ عدد, "فرد"
        تمام
    تمام
تمام
\end{salam}

\begin{outputfa}
1 فرد
2 زوج
3 فرد
4 زوج
5 فرد
6 زوج
\end{outputfa}

خط \fcode{اگر زوج است(عدد):} تقریباً مثل فارسی خوانده می‌شود. این یکی از
زیبایی‌های نام‌گذاری خوب تابع‌هاست. رسم خوبی است که نام تابع‌های منطقی را
به شکل پرسش بگذاریم: \fcode{زوج است}، \fcode{معتبر است}، \fcode{خالی است}.

\subsection{چند بازگشت در یک تابع}

تابع می‌تواند در چند جا \fcode{بازگشت} داشته باشد. به محض اجرای اولین
\fcode{بازگشت}، تابع تمام می‌شود و بقیه‌ی خط‌ها اجرا نمی‌شوند. این ویژگی
زنجیره‌های شرطی را کوتاه می‌کند:

\program{چند بازگشت}
\begin{salam}
تابع برچسب نمره(نمره: صحیح): رشته:
    اگر نمره >= 18: بازگشت "عالی" تمام
    اگر نمره >= 15: بازگشت "خوب" تمام
    اگر نمره >= 10: بازگشت "قابل قبول" تمام
    بازگشت "مردود"
تمام

تابع اصلی:
    چاپ برچسب نمره(19)
    چاپ برچسب نمره(16)
    چاپ برچسب نمره(12)
    چاپ برچسب نمره(7)
تمام
\end{salam}

\begin{outputfa}
عالی
خوب
قابل قبول
مردود
\end{outputfa}

با نمره‌ی ۱۶، شرط اول نادرست است، شرط دوم درست است و تابع با «خوب»
برمی‌گردد؛ خط‌های بعدی هرگز دیده نمی‌شوند. مقایسه کنید با همان کار در فصل
\ref{ch:conditions} که با \fcode{وگرنه} نوشته بودیم.

\begin{warn}
سلام با تابع‌ها هم مثل متغیرها سخت‌گیر است: تابعی که تعریف شده اما
هیچ‌جا فراخوانی نشده، خطاست. همچنین تابعی که نوع بازگشتی اعلام کرده،
باید در \emph{همه‌ی} مسیرها چیزی برگرداند. اگر \fcode{بازگشت "مردود"} آخر را حذف کنید، سلام خطا می‌دهد که
تابع ممکن است بدون بازگشت تمام شود. برعکس، تابعی که نوع بازگشتی ندارد
(مثل \fcode{سلام کن}) نباید مقداری برگرداند.
\end{warn}

\subsection{تابع با عدد اعشاری}

\program{تبدیل دما به شکل تابع}
\begin{salam}
تابع به فارنهایت(سلسیوس: اعشار۶۴): اعشار۶۴:
    بازگشت سلسیوس * 9.0 / 5.0 + 32.0
تمام

تابع اصلی:
    چاپ "آب یخ می‌زند:", به فارنهایت(0.0)
    چاپ "دمای بدن:", به فارنهایت(37.0)
    چاپ "آب می‌جوشد:", به فارنهایت(100.0)
تمام
\end{salam}

\begin{outputfa}
آب یخ می‌زند: 32
دمای بدن: 98.6
آب می‌جوشد: 212
\end{outputfa}

فرمولی که در فصل \ref{ch:loops} درون حلقه نوشته بودیم، حالا یک نام دارد
و هر جا لازم باشد قابل استفاده است. دقت کنید که آرگومان‌ها را با نقطه‌ی
اعشار نوشتیم (\code{0.0} نه \code{0})، چون پارامتر اعشاری است.

\section{مقدار پیش‌فرض برای پارامتر}

گاهی یک پارامتر بیشتر وقت‌ها مقدار مشخصی دارد و فقط گاهی عوض می‌شود.
می‌توان در تعریف تابع برایش \textbf{مقدار پیش‌فرض} گذاشت؛ آن وقت
فراخواننده می‌تواند آن آرگومان را ننویسد:

\program{پارامتر با مقدار پیش‌فرض}
\begin{salam}
تابع سلام کن(نام: رشته, پیشوند: رشته = "سلام"):
    چاپ پیشوند, نام
تمام

تابع اصلی:
    سلام کن("سارا")
    سلام کن("دکتر اسماعیلی", "درود بر")
تمام
\end{salam}

\begin{outputfa}
سلام سارا
درود بر دکتر اسماعیلی
\end{outputfa}

پارامترهای پیش‌فرض‌دار باید بعد از پارامترهای معمولی بیایند.

\section{محدوده‌ی متغیرها}

متغیری که درون یک تابع تعریف می‌شود، فقط درون همان تابع وجود دارد. به
این ویژگی \textbf{محدوده} می‌گویند. تابع اصلی نمی‌تواند متغیرهای درون
\fcode{سلام کن} را ببیند و برعکس. پارامترها هم همین‌طورند: جعبه‌هایی که
با شروع تابع ساخته می‌شوند و با پایان آن از بین می‌روند.

نکته‌ی مهم‌تر این است که آرگومان‌ها به شکل \emph{کپی} به تابع می‌رسند. تابع
هر کاری با نسخه‌ی خودش بکند، متغیر فراخواننده دست‌نخورده می‌ماند:

\program{تابع کپی می‌گیرد}
\begin{salam}
تابع دو برابر کن(عدد: صحیح): صحیح:
    متغیر نتیجه := عدد
    نتیجه *= 2
    بازگشت نتیجه
تمام

تابع اصلی:
    مقدار := 21
    چاپ "نتیجه‌ی تابع:", دو برابر کن(مقدار)
    چاپ "مقدار اصلی:", مقدار
تمام
\end{salam}

\begin{outputfa}
نتیجه‌ی تابع: 42
مقدار اصلی: 21
\end{outputfa}

این همان چیزی است که در فصل \ref{ch:variables} درباره‌ی کپی شدن مقدارها
دیدیم. اگر تابع بخواهد نتیجه‌ای به بیرون بدهد، راه درستش \fcode{بازگشت}
است، نه دست‌کاری آرگومان.

\subsection{ثابت‌های سراسری}

گاهی مقداری در چند تابع لازم است. \fcode{ثابت}‌هایی که در بالای برنامه و
بیرون همه‌ی تابع‌ها تعریف شوند، در همه جا در دسترس‌اند:

\program{ثابت سراسری}
\begin{salam}
ثابت مالیات := 9

تابع قیمت نهایی(قیمت: صحیح): صحیح:
    بازگشت قیمت + قیمت * مالیات / 100
تمام

تابع اصلی:
    چاپ قیمت نهایی(100000)
    چاپ قیمت نهایی(250000)
    چاپ "نرخ:", مالیات, "درصد"
تمام
\end{salam}

\begin{outputfa}
109000
272500
نرخ: 9 درصد
\end{outputfa}

بیرون از تابع‌ها فقط \fcode{ثابت} تعریف کنید، نه متغیر تغییرپذیر. متغیر
سراسری‌ای که هر تابعی بتواند عوضش کند، پیدا کردن اشکال‌ها را بسیار سخت
می‌کند؛ بهتر است هر چه یک تابع لازم دارد را از راه پارامتر بگیرد و هر چه
تولید می‌کند را از راه بازگشت پس بدهد.

\begin{deep}[تابعی که خودش را صدا می‌زند]
تابع می‌تواند درون خودش، خودش را فراخوانی کند. به این کار
\textbf{بازگشتی} می‌گویند و کارهای جالبی با آن می‌شود کرد. برنامه‌ی زیر
با همین ترفند شمارش معکوس می‌کند: هر فراخوانی، عددی یکی کمتر را
می‌فرستد، تا وقتی که به صفر برسد و زنجیره قطع شود. \fcode{بازگشت} بدون
مقدار، تابع بی‌نتیجه را همان‌جا تمام می‌کند. در این کتاب به این موضوع
بیشتر نمی‌پردازیم، اما بدانید که وجود دارد.
\end{deep}

\begin{salam}
تابع پس شماری(عدد: صحیح):
    اگر عدد == 0:
        چاپ "پایان"
        بازگشت
    تمام
    چاپ عدد
    پس شماری(عدد - 1)
تمام

تابع اصلی:
    پس شماری(3)
تمام
\end{salam}
\begin{outputfa}
3
2
1
پایان
\end{outputfa}

\section{یک برنامه‌ی کامل‌تر}

صورت‌حساب یک رستوران، که هر بخش کارش را به یک تابع سپرده است:

\program{صورت‌حساب رستوران}
\begin{salam}
ثابت خدمات := 10

تابع مبلغ سفارش(قیمت واحد: صحیح, تعداد: صحیح): صحیح:
    بازگشت قیمت واحد * تعداد
تمام

تابع اعمال تخفیف(مبلغ: صحیح, درصد: صحیح): صحیح:
    بازگشت مبلغ - مبلغ * درصد / 100
تمام

تابع نمایش سطر(عنوان: رشته, مبلغ: صحیح):
    چاپ عنوان + ":", مبلغ, "تومان"
تمام

تابع اصلی:
    چلوکباب := مبلغ سفارش(180000, 2)
    نوشابه := مبلغ سفارش(15000, 3)
    جمع := چلوکباب + نوشابه
    حق خدمات := جمع * خدمات / 100

    نمایش سطر("چلوکباب", چلوکباب)
    نمایش سطر("نوشابه", نوشابه)
    نمایش سطر("جمع", جمع)

تابع اصلی این برنامه را بخوانید: تقریباً مثل یک متن فارسی است، چون هر
جزئیات محاسبه در تابعی با نام گویا پنهان شده است. به آخرین خط دقت کنید:
نتیجه‌ی \fcode{اعمال تخفیف} مستقیم به عنوان آرگومان به \fcode{نمایش سطر}
داده شده است. تابع‌ها این‌گونه در هم می‌نشینند.

\begin{summary}
\begin{itemize}
  \item تابع بلوکی نام‌دار است: \fcode{تابع نام(پارامترها): نوع بازگشتی:}
        ... \fcode{تمام}. فراخوانی با نام و پرانتز انجام می‌شود.
  \item تابع‌ها در سطح بالای برنامه تعریف می‌شوند؛ ترتیبشان مهم نیست.
  \item پارامترها با نوعشان نوشته می‌شوند: \fcode{نام: رشته}. آرگومان‌ها
        به ترتیب و به شکل کپی به آن‌ها می‌رسند.
  \item \fcode{بازگشت مقدار} نتیجه را پس می‌دهد و تابع را تمام می‌کند.
        فراخوانی چنین تابعی خودش یک مقدار است.
  \item پارامتر می‌تواند مقدار پیش‌فرض داشته باشد.
  \item متغیرهای درون تابع فقط در همان تابع وجود دارند. برای مقدارهای
        مشترک، \fcode{ثابت} سراسری تعریف کنید.
\end{itemize}
\end{summary}

\section*{تمرین‌ها}

\exercise تابعی به نام \fcode{بزرگتر} بنویسید که دو عدد صحیح بگیرد و
بزرگ‌ترشان را برگرداند. سپس با کمک آن، بزرگ‌ترین سه عدد را پیدا کنید.

\exercise تابعی بنویسید که شعاع دایره (اعشاری) را بگیرد و مساحتش را
برگرداند (عدد پی را ۳٫۱۴۱۶ بگیرید). با یک حلقه، مساحت دایره‌هایی با شعاع
۱ تا ۵ را چاپ کنید.

\exercise تابعی به نام \fcode{سال کبیسه است} بنویسید که یک سال شمسی
بگیرد و بگوید آیا کبیسه است یا نه. قاعده‌ی ساده‌شده: سالی کبیسه است که
باقی‌مانده‌ی تقسیمش بر ۳۳ یکی از ۱، ۵، ۹، ۱۳، ۱۷، ۲۲، ۲۶ یا ۳۰ باشد. سپس
برای سال‌های ۱۴۰۳ تا ۱۴۰۸ نتیجه را چاپ کنید.

\exercise تابعی بنویسید که تعداد ثانیه بگیرد و آن را به شکل رشته‌ای مثل
\code{"1:02:05"} (ساعت:دقیقه:ثانیه) برگرداند. راهنمایی: از محاسبه‌ی فصل
\ref{ch:operators} و چسباندن رشته‌ها استفاده کنید.

\exercise برنامه‌ی جدول ضرب فصل \ref{ch:loops} را با تابعی به نام
\fcode{سطر جدول} بازنویسی کنید که شماره‌ی سطر و تعداد ستون‌ها را بگیرد و
یک سطر جدول را چاپ کند. تعداد ستون‌ها مقدار پیش‌فرض ۱۰ داشته باشد.
