\chapter{پاسخ تمرین‌های منتخب}\label{app:solutions}

در این پیوست پاسخ شماری از تمرین‌های کتاب آمده است. پاسخ‌ها یک راه حل‌اند،
نه تنها راه حل؛ اگر برنامه‌ی شما خروجی درست می‌دهد و خواناست، درست است،
حتی اگر با این‌جا فرق داشته باشد. خواهش می‌کنیم پیش از نگاه کردن به هر
پاسخ، خودتان تلاش کنید.

\section*{فصل~\ref{ch:first}}

\solutionhead{تمرین ۱.۱}
\begin{salamfa}
تابع ریشه:
    سرچاپ "سارا"
    سرچاپ "احمدی"
    سرچاپ "کاشان"
پایان
\end{salamfa}

\solutionhead{تمرین ۱.۳}
\begin{salamfa}
تابع ریشه:
    سرچاپ "حاصل جمع ۱۲۵ و ۳۷۵:", ۱۲۵ + ۳۷۵
    سرچاپ "حاصل ضرب ۲۴ در ۷:", ۲۴ * ۷
    سرچاپ "ساعت‌های یک هفته:", ۷ * ۲۴
پایان
\end{salamfa}
\begin{outputfa}
حاصل جمع ۱۲۵ و ۳۷۵: 500
حاصل ضرب ۲۴ در ۷: 168
ساعت‌های یک هفته: 168
\end{outputfa}

\subsection*{تمرین ۱.۴}
سه اشتباه: دونقطه‌ی بعد از \fcode{تابع ریشه} فراموش شده، در خط دوم
\fcode{print} انگلیسی به جای \fcode{سرچاپ} نوشته شده، و رشته‌ی خط سوم نقل قول
پایانی ندارد.

\section*{فصل~\ref{ch:variables}}

\solutionhead{تمرین ۲.۱}
\begin{salamfa}
تابع ریشه:
    طول := ۵
    عرض := ۴
    ارتفاع := ۳
    سرچاپ "مساحت کف:", طول * عرض, "متر مربع"
    سرچاپ "حجم اتاق:", طول * عرض * ارتفاع, "متر مکعب"
پایان
\end{salamfa}
\begin{outputfa}
مساحت کف: 20 متر مربع
حجم اتاق: 60 متر مکعب
\end{outputfa}

\subsection*{تمرین ۲.۳}
خروجی \code{7 1} است. دنبال کنید: \fcode{الف} برابر ۳ و \fcode{ب} برابر
۶ می‌شود. سپس \fcode{الف} برابر ۷ می‌شود. در آخر \fcode{ب} برابر
\scode{۷ - ۶} یعنی ۱ می‌شود.

\subsection*{تمرین ۲.۴}
\fcode{امتیاز} بدون \fcode{ناپایا} تعریف شده اما در خط بعد عوض می‌شود؛ باید
\scode{ناپایا امتیاز := ۱۰} باشد. \fcode{نام بازیکن} تعریف شده اما
استفاده نشده؛ یا در \fcode{سرچاپ} به کارش ببرید یا خطش را حذف کنید.

\section*{فصل~\ref{ch:operators}}

\solutionhead{تمرین ۴.۲}
\begin{salamfa}
تابع ریشه:
    عدد := ۴۷۲۹
    یکان := عدد % ۱۰
    دهگان := (عدد / ۱۰) % ۱۰
    صدگان := (عدد / ۱۰۰) % ۱۰
    هزارگان := عدد / ۱۰۰۰
    سرچاپ "مجموع رقم‌ها:", یکان + دهگان + صدگان + هزارگان
پایان
\end{salamfa}
\begin{outputfa}

\end{outputfa}

\section*{فصل~\ref{ch:conditions}}

\solutionhead{تمرین ۵.۳}
\begin{salamfa}
تابع ریشه:
    ساعت := ۱۵
    اگر ساعت < ۱۲:
        سرچاپ "صبح بخیر"
    وگرنه ساعت < ۱۷:
        سرچاپ "ظهر بخیر"
    وگرنه ساعت < ۲۰:
        سرچاپ "عصر بخیر"
    وگرنه:
        سرچاپ "شب بخیر"
    پایان
پایان
\end{salamfa}
\begin{outputfa}
ظهر بخیر
\end{outputfa}

\subsection*{تمرین ۵.۶}
سه خط چاپ می‌شود: «بزرگ‌تر از ده»، «مضرب پنج» و «مضرب سه». زنجیره‌ی اول
با نخستین شرط درست تمام می‌شود و «بزرگ‌تر از پنج» چاپ نمی‌شود؛ اما دو
\fcode{اگر} بعدی جداگانه‌اند و هر دو بررسی می‌شوند.

\section*{فصل~\ref{ch:loops}}

\solutionhead{تمرین ۶.۱}
\begin{salamfa}
تابع ریشه:
    ناپایا خط := ""
    تکرار ۱ تا ۵۰ در عدد:
        اگر (عدد % ۷) == ۰:
            خط = خط + عدد + " "
        پایان
    پایان
    سرچاپ خط
پایان
\end{salamfa}
\begin{outputfa}
7 14 21 28 35 42 49
\end{outputfa}

\solutionhead{تمرین ۶.۲}

    تکرار ۱ تا ۱۰ در عدد:

    پایان
    سرچاپ حاصل
پایان
\end{salamfa}

\subsection*{تمرین ۶.۶}
شش خط. حلقه‌ی بیرونی سه دور دارد و در هر دور، حلقه‌ی درونی فقط برای
\fcode{ب} برابر ۰ و ۱ چاپ می‌کند و در ۲ می‌شکند.

\section*{فصل~\ref{ch:functions}}

\solutionhead{تمرین ۷.۱}
\begin{salamfa}
تابع بیشینه(الف: صحیح, ب: صحیح): صحیح:
    اگر الف > ب: برگشت الف پایان
    برگشت ب
پایان

تابع ریشه:
    دوتای_اول := بیشینه(۱۲, ۲۵)
    سرچاپ بیشینه(دوتای_اول, ۱۹)
پایان
\end{salamfa}
\begin{outputfa}
25
\end{outputfa}
بزرگ‌ترین سه عدد، بزرگ‌ترِ «بزرگ‌ترِ دوتای اول» و سومی است.

\subsection*{تمرین ۷.۶}
وقتی \fcode{عدد} صفر است، هیچ‌کدام از دو شرط برقرار نیست و تابع بدون
\fcode{برگشت} تمام می‌شود؛ سلام برای این هشدار می‌دهد (پیوست~\ref{app:errors})
و اگر تابع را با صفر صدا بزنید، به جای عدد \code{null} برمی‌گردد. راه حل: شرط دوم را
به \scode{وگرنه:} بی‌شرط تبدیل کنید، یا در پایان تابع
\fcode{برگشت عدد} بگذارید.

\section*{فصل~\ref{ch:arrays}}

\solutionhead{تمرین ۸.۲}
\begin{salamfa}
تابع میانگین(اعداد: صحیح[:]): اعشار۶۴:
    ناپایا جمع := ۰
    هر عدد در اعداد:
        جمع += عدد
    پایان
    تعداد := اعداد.طول()
    برگشت (جمع برگردان اعشار۶۴) / (تعداد برگردان اعشار۶۴)
پایان

تابع بیشترین(اعداد: صحیح[:]): صحیح:
    ناپایا بزرگترین := اعداد[۰]
    هر عدد در اعداد:
        اگر عدد > بزرگترین: بزرگترین = عدد پایان
    پایان
    برگشت بزرگترین
پایان

تابع ریشه:
    نمرات := [۱۷, ۱۵, ۱۹, ۱۲, ۲۰]
    سرچاپ "میانگین:", میانگین(نمرات)
    سرچاپ "بیشترین:", بیشترین(نمرات)
پایان
\end{salamfa}
\begin{outputfa}
میانگین: 16.6
بیشترین: 20
\end{outputfa}

\subsection*{تمرین ۸.۶}
خروجی \code{[10, 2, 30, 4, 50]} است: عضوهایی که اندیس زوج دارند (۰، ۲
و ۴) ده برابر می‌شوند.

\section*{فصل~\ref{ch:strings}}

\solutionhead{تمرین ۹.۴}
\begin{salamfa}
تابع ریشه:
    متن := "17,15,19,12"
    تکه ها := متن.بشکاف(",")
    ناپایا مجموع := ۰
    تکرار تکه ها.طول() در ک:
        مجموع += تکه ها.بگیر(ک).به صحیح()
    پایان
    تعداد := تکه ها.طول()
    سرچاپ "مجموع:", مجموع
    میانگین := (مجموع برگردان اعشار۶۴) / (تعداد برگردان اعشار۶۴)
    سرچاپ "میانگین:", میانگین
پایان
\end{salamfa}
\begin{outputfa}
مجموع: 63
میانگین: 15.75
\end{outputfa}
