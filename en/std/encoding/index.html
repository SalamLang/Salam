Standard library · Encoding and data

encoding

Turning text and data into Base64, hexadecimal and URL encoding.

import encoding
On this page
  1. Functions

This package holds the common ways of writing bytes as plain text: Base64 for fitting binary data into JSON or email, hexadecimal for showing bytes, and URL encoding for putting text or special characters into a web address.

The larger formats, such as JSON, YAML and XML, have sub-packages of their own, listed in this same group.

Functions

Base64Encode

func Base64Encode(data: str): str
Turns text into standard Base64, padded with =.
encoding-Base64.salam
import encoding
import mem

func main:
    e := encoding.Base64Encode("hello world")
    println e
    println encoding.Base64Decode(e)
    mut ok := false
    println "[" + encoding.Base64DecodeStrict("***", ok) + "]", ok
    println encoding.Base64DecodeStrict("YWJj", ok), ok
    b := mem.Allocate(3 as u64)
    mem.Set(b, 255, 3 as u64)
    println encoding.Base64EncodeBytes(b, 3)
    out := mem.Allocate(8 as u64)
    println encoding.Base64DecodeBytes("AAEC", out, 8, ok), ok
    mem.Free(b)
    mem.Free(out)
end
OutputaGVsbG8gd29ybGQ= hello world [] false abc true //// 3 true

HexEncode

func HexEncode(data: str): str
Writes each byte of the text as two lower case hexadecimal digits.
encoding-Hex.salam
import encoding

func main:
    h := encoding.HexEncode("Hi!")
    println h, encoding.HexDecode(h)
    println encoding.HexVal("f"), encoding.HexVal("A"), encoding.HexVal("z")
    u := encoding.URLEncode("name=Ali & age=30")
    println u
    println encoding.URLDecode(u)
end
Output486921 Hi! 15 10 -1 name%3DAli%20%26%20age%3D30 name=Ali & age=30
