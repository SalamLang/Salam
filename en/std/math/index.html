math

The basic mathematics: powers and roots, logarithms, trigonometry, rounding, minimum and maximum, and the greatest common divisor.

import math
On this page
  1. Constants and variables
  2. Functions

The math package holds what you would expect from a scientific calculator. Most of its functions work on f64; the ones whose names end in I or I64 are the whole-number versions.

The angles in the trigonometric functions are in radians, not degrees. Use Radians and Degrees to convert.

Constants and variables

PI

const PI: f64 = 3.14159
Pi (π), the ratio of a circle's circumference to its diameter.
math-Sqrt.salam
import math

func main:
    println math.Sqrt(2.0), math.Cbrt(27.0), math.Pow(2.0, 10.0), math.Pow10(3)
    println math.Hypot(3.0, 4.0), math.Exp(1.0)
    println math.Log(math.E), math.Log10(1000.0), math.Log2(1024.0)
    println math.PI, math.TAU
end
Output1.4142135623730951 3 1024 1000 5 2.718281828459045 1 3 10 3.141592653589793 6.283185307179586

Sin

func Sin(x: f64): f64
The sine of an angle given in radians.
math-Sin.salam
import math
import fmt

func main:
    angle := math.Radians(30.0)
    println fmt.FloatPrec(math.Sin(angle), 4), fmt.FloatPrec(math.Cos(angle), 4), fmt.FloatPrec(math.Tan(angle), 4)
    println fmt.FloatPrec(math.Degrees(math.Asin(0.5)), 4), fmt.FloatPrec(math.Degrees(math.Acos(0.0)), 4), fmt.FloatPrec(math.Degrees(math.Atan(1.0)), 4)
    println fmt.FloatPrec(math.Degrees(math.Atan2(1.0, -1.0)), 4)
    println fmt.FloatPrec(math.Sinh(1.0), 4), fmt.FloatPrec(math.Cosh(1.0), 4), fmt.FloatPrec(math.Tanh(1.0), 4)
end
Output0.5000 0.8660 0.5774 30.0000 90.0000 45.0000 135.0000 1.1752 1.5431 0.7616

Floor

func Floor(x: f64): f64
The largest whole number not greater than the value; for negatives it goes down (−2.5 gives −3).
math-Floor.salam
import math

func main:
    each x in [2.5, -2.5, 2.49]:
        println x, math.Floor(x), math.Ceil(x), math.Round(x), math.Trunc(x)
    end
    println math.Abs(-3.5), math.Sign(-3.5), math.Sign(0.0), math.Mod(7.5, 2.0)
end
Output2.5 2 3 3 2 -2.5 -3 -2 -2 -2 2.49 2 3 2 2 3.5 -1 0 1.5

Min

func Min(a: f64, b: f64): f64
The smaller of two decimal numbers.
math-Min.salam
import math

func main:
    println math.Min(2.5, 1.5), math.Max(2.5, 1.5), math.ClampF(1.7, 0.0, 1.0)
    println math.MinI(3, 8), math.MaxI(3, 8), math.AbsI(-9), math.ClampI(120, 0, 100)
    big := 9000000000 as i64
    println math.MinI64(big, 5 as i64), math.MaxI64(big, 5 as i64), math.AbsI64(-big)
    println math.Lerp(10.0, 20.0, 0.25)
end
Output1.5 2.5 1 3 8 9 100 5 9000000000 9000000000 12.5

IsNaN

func IsNaN(x: f64): bool
True when the number is "not a number". NaN is not equal to itself, so == cannot detect it.
math-IsNaN.salam
import math

func main:
    inf := math.Inf()
    nan := math.NaN()
    println inf, -inf, math.IsInf(inf), math.IsInf(1.0)
    println math.IsNaN(nan), math.IsNaN(1.0), nan == nan
end
Outputinf -inf true false true false false
