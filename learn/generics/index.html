Generics and interfaces

Generic functions work for any type, and interfaces describe behaviour that different types share.

In this lesson
  1. Generic functions and structs
  2. Interfaces
  3. impl and type constraints

Generic functions and structs

Write <T> after the name and the function or struct is written once for any type T. Salam generates a separate, fully specialised version for every type you use it with.

generics.salam
struct Box<T>:
    pub inside: T
end

func larger<T>(a: T, b: T): T:
    if a > b:
        ret a
    end
    ret b
end

func main:
    println larger(3, 9), larger("a", "b"), larger(2.5, 1.5)
    b := Box { inside = "a gift" } as Box<str>
    println b.inside
end
Output9 b 2.5 a gift

Interfaces

An interface is a list of methods. Any struct that has all of them implements the interface automatically. A parameter of type dyn Shape accepts any value that implements Shape.

interface.salam
interface Shape:
    func area(): int
    func name(): str
end

struct Square:
    side: int = 0
    pub func area(): int: ret this.side * this.side end
    pub func name(): str: ret "square" end
end

struct Rectangle:
    width: int = 0
    height: int = 0
    pub func area(): int: ret this.width * this.height end
    pub func name(): str: ret "rectangle" end
end

func describe(s: dyn Shape):
    println s.name(), s.area()
end

func main:
    describe(Square { side = 5 })
    describe(Rectangle { width = 4, height = 6 })
end
Outputsquare 25 rectangle 24

An interface method may carry a default body. A struct that does not write that method gets the default, and a struct that does write it gets its own:

interface-default.salam
interface Shape:
    func area(): f64
    func describe(): str:
        ret "area " + this.area()
    end
end

struct Circle:
    pub radius: f64 = 1.0
    pub func area(): f64:
        ret 3.14 * this.radius * this.radius
    end
end

struct Square:
    pub side: f64 = 1.0
    pub func area(): f64:
        ret this.side * this.side
    end
    pub func describe(): str:
        ret "a square with side " + this.side
    end
end

func main:
    println Circle { radius = 2.0 }.describe()
    println Square { side = 3.0 }.describe()
end
Outputarea 12.56 a square with side 3

impl and type constraints

impl Interface on Type adds an interface's methods to any type at all, including int and str. <T: Ranked> means the function accepts only types that implement that interface.

impl.salam
interface Ranked:
    func rank(): int
end

impl Ranked on int:
    func rank(): int: ret this end
end

impl Ranked on str:
    func rank(): int: ret len(this) end
end

func higher<T: Ranked>(a: T, b: T): int:
    if a.rank() > b.rank():
        ret a.rank()
    end
    ret b.rank()
end

func main:
    println higher(3, 9)
    println higher("hi", "salam")
end
Output9 5
