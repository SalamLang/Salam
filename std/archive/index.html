archive

Making and opening zip and tar archives, reading RAR, and gzipping a file.

import archive
On this page
  1. Types
  2. Functions

An archive gathers several files into one. Zip compresses each file on its own; tar only puts the files one after another and is usually gzipped afterwards (.tar.gz).

The functions work both in memory and on files. They take the archive data as an address and a length, such as b.data and b.len of a byte buffer.

archive-Tar.salam
import archive
import compress

func main:
    b := compress.BufFromStr("first line\nsecond line\n")
    mut files := Vector {} as Vector<archive.TarFileInput>
    files.push(archive.MakeTarFileInput("notes", null, 0, true))
    files.push(archive.MakeTarFileInput("notes/a.txt", b.data, b.len, false))
    t := archive.TarCreate(files)
    println t.len % 512
    mut entries := archive.TarList(t.data, t.len)
    each e in entries:
        println e.name, e.size, e.is_dir
    end
    archive.FreeEntries(entries)
    mut ok := false
    a := archive.TarExtractByName(t.data, t.len, "notes/a.txt", ok)
    println ok, compress.BufToStrUnsafe(a)
    println archive.TarCreateFile(files, "out.tar"), archive.TarExtractAll(t.data, t.len, "untarred")
    println archive.GzipCompressFile("out.tar", "out.tar.gz"), archive.GzipDecompressFile("out.tar.gz", "back.tar")
    back := archive.ReadBinaryFile("back.tar", ok)
    println ok, back.len == t.len
    archive.EnsureDir("a/b/c")
    println archive.WriteBinaryFile("a/b/c/x.bin", b.data, 3), archive.WriteBinaryFileBuf("a/b/c/y.bin", b)
    mut b2 := compress.NewByteBuf()
    archive.LEPut16(b2, 513)
    archive.LEPut32(b2, 70000)
    println archive.LEGet16(b2.data, 0), archive.LEGet32(b2.data, 2)
end
Output0 notes 0 true notes/a.txt 23 false true first line second line true true true true true true true true 513 70000

Entry

struct Entry
One archive entry.
NameTypeDescription
namestr = ""The name and path inside the archive.
sizei64 = 0The original size in bytes.
compressed_sizei64 = 0The compressed size.
is_dirbool = falseThe entry is a directory.
methodint = 0The compression method: zero for plain storage, 8 for DEFLATE.
crc32u32 = 0The checksum of the contents.
mtimei64 = 0The modification time in seconds since the Unix epoch.
archive-Zip.salam
import archive
import compress

func main:
    text := "hello archive, hello archive, hello archive"
    b := compress.BufFromStr(text)
    mut files := Vector {} as Vector<archive.ZipFileInput>
    files.push(archive.MakeZipFileInput("docs/", null, 0, 0))
    files.push(archive.MakeZipFileInput("docs/a.txt", b.data, b.len, 8))
    files.push(archive.MakeZipFileInput("b.txt", b.data, 5, 0))
    z := archive.ZipCreate(files)
    mut entries := archive.ZipList(z.data, z.len)
    each e in entries:
        println e.name, e.size, e.compressed_size < e.size, e.is_dir, e.method
    end
    archive.FreeEntries(entries)
    mut ok := false
    a := archive.ZipExtractByName(z.data, z.len, "docs/a.txt", ok)
    println ok, compress.BufToStrUnsafe(a) == text
    _ := archive.ZipExtractByName(z.data, z.len, "missing", ok)
    println ok
    println archive.ZipCreateFile(files, "out.zip"), archive.ZipExtractAll(z.data, z.len, "unzipped")
    read := archive.ReadBinaryFile("unzipped/docs/a.txt", ok)
    println ok, read.len == b.len
    println archive.IsRar(z.data, z.len), archive.RarList(z.data, z.len).len()
    _ := archive.RarExtractByName(z.data, z.len, "x", ok)
    println ok, archive.RarExtractAll(z.data, z.len, "rar-out")
end
