audio/midi

Reading, writing and playing standard MIDI files.

import audio.midi
On this page
  1. Types
  2. Constants and variables
  3. Functions

This is an official package that does not ship with Salam. Add it to a project first: salam add audio.midi

A MIDI file holds notes, not sound: which note, when, how hard and on what instrument. That makes it tiny, and it makes it editable in a way a recording is not.

This package reads and writes standard MIDI files, and Render turns one into sound through audio/synth, so a file of notes becomes a clip you can save as WAV or FLAC.

Things to know

  • Everything is in ticks, and division says how many ticks a quarter note is. TicksToSeconds needs the tempo as well, which is why it takes the whole song.
  • A note is two events, on and off. AddNote writes both, and Notes pairs them back up and gives you lengths in seconds.
  • A note on with velocity 0 means a note off, and the reader turns it into one so you never have to check.
  • Release a song with FreeSong: it owns a vector per track plus the vector of tracks.
  • Render uses a single plain envelope for every note. For anything more, take Notes and render it yourself.

Types

Event

struct Event
One MIDI event, with the tick it happens at.
NameTypeDescription
tickint = 0When it happens, in ticks from the start.
statusint = 0Which kind of event it is.
channelint = 0Which of the sixteen channels it is on.
data1int = 0The first byte, a note number or a controller.
data2int = 0The second byte, a velocity or a value.
metaint = 0Which meta event it is, when the status says meta.
valuei64 = 0The number a meta event carries, such as a tempo.
midi-Write.salam
import audio.midi
import compress

func main:
    mut f := midi.NewSong(480)
    println f.division, f.format, f.tempo as int
    println midi.DefaultDivision

    mut t := midi.NewTrack()
    midi.AddTempo(t, 0, 120.0)
    midi.AddProgram(t, 0, 0, 0)
    midi.AddControl(t, 0, 7, 100, 0)
    mut s := 0
    until s < 4:
        midi.AddNote(t, s * 480, 240, 60 + s * 2, 100, 0)
        s += 1
    end
    println t.events.len()
    midi.AddTrack(f, t)
    println f.tracks.len(), midi.TotalTicks(f)

    one := t.events.get(1)
    println one.tick, one.status == midi.NoteOn, one.data1, one.data2, one.channel

    println midi.NoteOn, midi.NoteOff, midi.ControlChange, midi.ProgramChange, midi.MetaEvent

    enc := midi.EncodeMIDI(f)
    println compress.BufLen(enc) > 22, midi.IsMIDI(enc.data, enc.len)
    mut empty := midi.NewSong(480)
    println compress.BufLen(midi.EncodeMIDI(empty))
    midi.FreeSong(empty)

    compress.BufFree(enc)
    midi.FreeSong(f)
end

IsMIDI

func IsMIDI(p: void*, n: int): bool
True when these bytes start with the MThd header.
midi-Read.salam
import audio
import audio.midi
import compress
import filepath
import math
import mem
import os

func main:
    mut f := midi.NewSong(480)
    mut t := midi.NewTrack()
    midi.AddTempo(t, 0, 120.0)
    mut s := 0
    until s < 4:
        midi.AddNote(t, s * 480, 240, 60 + s * 2, 100, 0)
        s += 1
    end
    midi.AddTrack(f, t)
    enc := midi.EncodeMIDI(f)

    println midi.IsMIDI(enc.data, enc.len), midi.IsMIDI(enc.data, 8)
    println audio.FormatName(audio.DetectFormat(enc.data, enc.len))
    mut ok := false
    mut back := midi.DecodeMIDI(enc.data, enc.len, ok)
    println ok, back.division, back.tracks.len()
    println math.Abs(midi.Bpm(back) - 120.0) < 0.01
    println midi.TotalTicks(back)
    println midi.TicksToSeconds(back, 480), midi.SecondsToTicks(back, 0.5)

    mut notes := midi.Notes(back)
    println notes.len(), notes.get(0).note, notes.get(3).note
    println notes.get(1).start, notes.get(0).length
    notes.free()

    rendered := midi.Render(back, 22050, 1)
    println rendered.frames > 0, audio.Peak(rendered) > 0.05
    println midi.ProgramName(0), midi.ProgramName(200) == ""

    path := filepath.Join(os.TempDir(), "salam_midi_example.mid")
    defer mem.Free(path as void*)
    println midi.SaveMIDI(f, path)
    mut dok := false
    mut disk := midi.OpenMIDI(path, dok)
    println dok, disk.tracks.len()
    mut gone := false
    missing := filepath.Join(os.TempDir(), "salam_midi_none.mid")
    defer mem.Free(missing as void*)
    midi.OpenMIDI(missing, gone)
    println gone

    mut bad := false
    midi.DecodeMIDI(enc.data, 8, bad)
    println bad

    compress.BufFree(enc)
    mut r := rendered
    audio.Free(r)
    midi.FreeSong(f)
    midi.FreeSong(back)
    midi.FreeSong(disk)
end
