bigint

Whole numbers of unlimited size: large factorials, cryptography, and any arithmetic that will not fit in 64 bits.

import bigint
On this page
  1. Types
  2. Functions

An i64 holds about nine quintillion. For anything larger, such as the factorial of 30 or a cryptographic key, use bigint.Int, whose size is limited only by memory.

Most functions return a new number and leave their arguments alone. The ones whose names say To, From or By, such as AddTo, change the number in place and are faster in a loop because they take no new memory. These numbers allocate; release the ones you no longer need with Free.

bigint-Int.salam
import bigint

func main:
    mut a := bigint.FromInt(123456789)
    mut b := bigint.Pow(bigint.FromInt(2), 100)
    defer a.free()
    defer b.free()
    println b.to_str()
    println a.is_zero(), a.sign(), a.is_odd(), a.is_even()
    println b.bit_len(), b.limb_count()
    c := a.clone()
    println c.to_str()
end

bigint-AddTo.salam
import bigint

func main:
    mut acc := bigint.Zero()
    defer bigint.Free(acc)
    bigint.SetInt(acc, 10)
    bigint.AddTo(acc, bigint.FromInt(5))
    bigint.SubFrom(acc, bigint.FromInt(3))
    bigint.MulBy(acc, bigint.FromInt(4))
    println bigint.ToStr(acc)
    bigint.AddIntTo(acc, 2)
    bigint.MulIntBy(acc, 10)
    println bigint.ToStr(acc)
    bigint.NegateIn(acc)
    bigint.ShlBy(acc, 1)
    println bigint.ToStr(acc)
    bigint.ShrBy(acc, 2)
    println bigint.ToStr(acc)
    bigint.Set(acc, bigint.FromInt(7))
    println bigint.ToStr(acc)
end
Output48 500 -1000 -250 7

bigint-Parse.salam
import bigint

func main:
    mut ok := false
    n := bigint.Parse("123456789012345678901234567890", ok)
    println ok, bigint.ToStr(n)
    h := bigint.ParseBase("ff", 16, ok)
    println ok, bigint.ToStr(h), bigint.ToStrBase(h, 2), bigint.ToHex(n)
    println bigint.ToStr(bigint.ParseOr("xyz", 10))
    mut fits := false
    v := bigint.ToInt(h, fits)
    println v, fits
    println bigint.ToIntClamped(n), bigint.ToFloat(n)
end

ModExp

func ModExp(base: Int, e: Int, m: Int): Int
Computes base to the power e modulo m quickly, without building the huge intermediate number; the basis of RSA.
bigint-ModExp.salam
import bigint

func main:
    println bigint.ToStr(bigint.Factorial(30))
    println bigint.ToStr(bigint.ModExp(bigint.FromInt(4), bigint.FromInt(13), bigint.FromInt(497)))
    println bigint.ToStr(bigint.Gcd(bigint.FromInt(48), bigint.FromInt(18))), bigint.ToStr(bigint.Lcm(bigint.FromInt(4), bigint.FromInt(6)))
    mut x := bigint.Zero()
    mut y := bigint.Zero()
    g := bigint.ExtGcd(bigint.FromInt(240), bigint.FromInt(46), x, y)
    println bigint.ToStr(g), bigint.ToStr(x), bigint.ToStr(y)
    mut ok := false
    inv := bigint.ModInverse(bigint.FromInt(3), bigint.FromInt(11), ok)
    println ok, bigint.ToStr(inv)
    println bigint.ToStr(bigint.Sqrt(bigint.Pow10(20)))
end

DivMod

func DivMod(a: Int, b: Int, q &: Int, r &: Int): bool
Puts the quotient and the remainder into q and r together; the quotient is rounded towards zero. It returns false when the divisor is zero.
bigint-Div.salam
import bigint

func main:
    a := bigint.FromInt(-17)
    b := bigint.FromInt(5)
    println bigint.ToStr(bigint.Div(a, b)), bigint.ToStr(bigint.Rem(a, b))
    println bigint.ToStr(bigint.FloorDiv(a, b)), bigint.ToStr(bigint.FloorMod(a, b)), bigint.ToStr(bigint.Mod(a, b))
    mut q := bigint.Zero()
    mut r := bigint.Zero()
    println bigint.DivMod(a, b, q, r), bigint.ToStr(q), bigint.ToStr(r)
    println bigint.FloorDivMod(a, b, q, r), bigint.ToStr(q), bigint.ToStr(r)
    println bigint.DivMod(a, bigint.Zero(), q, r)
    mut rem := 0 as i64
    println bigint.ToStr(bigint.DivModInt(bigint.FromInt(100), 7, rem)), rem, bigint.ModInt(bigint.FromInt(100), 7)
end
