calendar

The Jalali, Hijri and Gregorian calendars: conversion, day arithmetic, month names and time zones.

import calendar
On this page
  1. Types
  2. Functions

This package works with three calendars: Jalali (Solar Hijri, the official calendar of Iran and Afghanistan), Hijri (lunar) and Gregorian. A date is a plain struct of year, month and day, and each calendar has its own conversion, validation, day arithmetic and month name functions.

The Hijri calendar here is arithmetic: months alternate between 30 and 29 days and are not adjusted by sighting the moon, so a date can be one day off from the official calendar of a given country.

There are three ways to turn an instant into a date: EpochToJalali and its siblings use the machine local time, WallClockJalali takes a fixed time zone such as Tehran, and LoadZone covers zones that observe daylight saving.

struct Zone
A full time zone from the zone database (tzdata), with its history of offsets and daylight saving.
NameTypeDescription
namestr = "UTC"The zone name.
okbool = falseWhether the zone loaded.
type_offsetVectorThe offset of each time type, in seconds.
type_isdstVectorWhether each type is daylight saving.
type_abbrevVectorThe abbreviation of each type, such as EST.
timesVectorThe instants at which the offset changes.
type_atVectorThe time type in force after each change.

calendar-Zone.salam
import calendar

func main:
    mut z := calendar.LoadZone("America/New_York")
    println z.name, z.ok
    winter := 1704067200
    summer := 1719792000
    println calendar.ZoneOffsetMinutes(z, winter), calendar.ZoneOffsetMinutes(z, summer), calendar.ZoneOffsetSeconds(z, summer)
    println calendar.ZoneIsDST(z, winter), calendar.ZoneIsDST(z, summer)
    println calendar.ZoneAbbrevAt(z, winter), calendar.ZoneAbbrevAt(z, summer)
    tz := calendar.TimeZoneAt(z, summer)
    println tz.name, tz.offset_minutes
    println calendar.ZoneAt("Asia/Tehran", winter).offset_minutes
    e := calendar.ZonedToEpoch(calendar.GregorianDate { year = 2024, month = 6, day = 30 }, calendar.ClockTime { hour = 20 }, z)
    println e == summer
    calendar.FreeZone(z)
    println calendar.LoadZone("Mars/Olympus").ok
    mut l := calendar.LocalZone()
    println l.ok or not l.ok
    calendar.FreeZone(l)
end

import calendar
import time

func main:
    t := 1710943509
    zone := calendar.TZLocal(t)
    g := calendar.EpochToGregorian(t)
    println calendar.CompareGregorian(g, calendar.WallClockGregorian(t, zone)) == 0
    println calendar.CompareJalali(calendar.EpochToJalali(t), calendar.WallClockJalali(t, zone)) == 0
    println calendar.CompareHijri(calendar.EpochToHijri(t), calendar.WallClockHijri(t, zone)) == 0
    println calendar.FormatClockTime(calendar.EpochToClockTime(t)) == calendar.FormatClockTime(calendar.WallClockTime(t, zone))
    j := calendar.JalaliDate { year = 1403, month = 1, day = 1 }
    println calendar.JalaliToUnixUTC(j, 14, 5, 9) == t
    println calendar.GregorianToUnixUTC(calendar.GregorianDate { year = 2024, month = 3, day = 20 }, 14, 5, 9)
    println calendar.HijriToUnixUTC(calendar.HijriDate { year = 1445, month = 9, day = 10 }, 0, 0, 0)
    println calendar.CompareGregorian(calendar.TodayGregorian(), calendar.EpochToGregorian(time.Now())) >= 0, calendar.TodayGregorian().year >= 2024, calendar.TodayJalali().year >= 1403, calendar.TodayHijri().year >= 1445
    c := calendar.NowClockTime()
    println c.hour < 24, c.minute < 60, c.second < 61
end

calendar-TimeZone.salam
import calendar

func main:
    t := 1710943509
    tehran := calendar.TZTehran()
    println tehran.name, tehran.offset_minutes
    println calendar.FormatJalali(calendar.WallClockJalali(t, tehran)), calendar.FormatClockTime(calendar.WallClockTime(t, tehran))
    println calendar.FormatGregorian(calendar.WallClockGregorian(t, calendar.TZTokyo())), calendar.FormatClockTime(calendar.WallClockTime(t, calendar.TZTokyo()))
    println calendar.FormatHijri(calendar.WallClockHijri(t, calendar.TZMecca()))
    each zone in [calendar.TZUTC(), calendar.TZKabul(), calendar.TZRiyadh(), calendar.TZDubai(), calendar.TZIstanbul(), calendar.TZLondon(), calendar.TZNewYork()]:
        println zone.name, zone.offset_minutes, calendar.FormatClockTime(calendar.WallClockTime(t, zone))
    end
    g := calendar.GregorianDate { year = 2024, month = 3, day = 20 }
    c := calendar.ClockTime { hour = 23, minute = 0, second = 0 }
    println calendar.ZonedGregorianToEpoch(g, c, tehran)
    println calendar.FormatGregorian(calendar.ConvertZonedGregorian(g, c, tehran, calendar.TZTokyo())), calendar.FormatClockTime(calendar.ConvertZonedTime(g, c, tehran, calendar.TZTokyo()))
    fixed := calendar.FixedOffsetZone("UTC+05:30", 330)
    println fixed.name, calendar.FormatClockTime(calendar.WallClockTime(t, fixed))
    local := calendar.TZLocal(t)
    println local.name, local.offset_minutes == calendar.LocalOffsetMinutes(t)
end

import calendar

func main:
    println calendar.IsLeapJalali(1403), calendar.IsLeapJalali(1404), calendar.DaysInJalaliYear(1403)
    println calendar.DaysInJalaliMonth(1403, 1), calendar.DaysInJalaliMonth(1403, 7), calendar.DaysInJalaliMonth(1403, 12), calendar.DaysInJalaliMonth(1404, 12)
    println calendar.IsValidJalaliDate(1404, 12, 30), calendar.IsValidJalaliDate(1403, 12, 30)
    println calendar.JalaliMonthNameFa(1), calendar.JalaliMonthNameEn(1), calendar.JalaliMonthNameFa(12)
    j := calendar.JalaliDate { year = 1403, month = 6, day = 31 }

    println calendar.JalaliWeekday(d), calendar.JalaliWeekdayNameFa(calendar.JalaliWeekday(d))

    nowruz := calendar.JalaliDate { year = 1404, month = 1, day = 1 }

IsLeapHijri

func IsLeapHijri(year: int): bool
True when the Hijri year is a leap year, meaning Dhu al-Hijjah has 30 days.
calendar-Hijri.salam
import calendar

func main:
    h := calendar.HijriDate { year = 1445, month = 9, day = 29 }
    println calendar.IsLeapHijri(1445), calendar.IsLeapHijri(1446)
    println calendar.DaysInHijriMonth(1445, 9), calendar.DaysInHijriMonth(1445, 12), calendar.DaysInHijriMonth(1446, 12)
    println calendar.IsValidHijriDate(1445, 9, 30), calendar.IsValidHijriDate(1445, 10, 30)
    println calendar.HijriMonthNameFa(9), calendar.HijriMonthNameAr(9), calendar.HijriMonthNameEn(9)
    println calendar.FormatHijri(calendar.AddDaysHijri(h, 1)), calendar.FormatHijri(calendar.AddWeeksHijri(h, 1))
    println calendar.FormatHijri(calendar.AddMonthsHijri(h, 3)), calendar.FormatHijri(calendar.AddYearsHijri(h, 1))
    println calendar.FormatHijri(calendar.StartOfMonthHijri(h)), calendar.FormatHijri(calendar.EndOfMonthHijri(h))
    muharram := calendar.HijriDate { year = 1446, month = 1, day = 1 }
    println calendar.DaysBetweenHijri(h, muharram), calendar.CompareHijri(h, muharram)
    println calendar.WeekdayNameFa(calendar.WeekdayHijri(h))
end
