compress

Byte buffers, compression methods from RLE through DEFLATE, gzip and checksums.

import compress
On this page
  1. Types
  2. Constants and variables
  3. Functions

This package is the base of all binary data work in Salam: ByteBuf and ByteReader are what packages such as encoding/binary, archive and encoding/msgpack pass around.

For compression, reach for gzip when the data has to be read by other programs. The other methods are mostly there to learn from and experiment with, and their output format belongs to this package alone. The examples show that for small data the overhead of Huffman and DEFLATE comes to more than the data itself.

Types

Method

enum Method
The compression method, as an enum.
NameTypeDescription
NoneNo compression.
RLERun length encoding.
HuffmanHuffman.
LZ77LZ77.
LZWLZW.
DeflateLZ77 and Huffman.
BWTBurrows Wheeler.
compress-Methods.salam
import compress

func main:
    text := "ababababab hello hello hello hello hello hello hello hello"
    b := compress.BufFromStr(text)
    each method in [compress.MethodNone, compress.MethodRLE, compress.MethodHuffman, compress.MethodLZ77, compress.MethodLZW, compress.MethodDeflate, compress.MethodBWT]:
        mut c := compress.CompressBuf(method, b)
        mut back := compress.DecompressBuf(method, c)
        println method, compress.BufLen(c), compress.BufEquals(b, back)
        compress.BufFree(c)
        compress.BufFree(back)
    end
    mut best := 0
    c := compress.CompressBest(b.data, b.len, best)
    println best, compress.Ratio(c.len, b.len) < 1.0
    s := compress.CompressStr(compress.MethodDeflate, "aaaaaaaaaa")
    b3 := compress.Decompress(compress.MethodDeflate, s.data, s.len)
    println compress.BufToStrUnsafe(b3)
    b4 := compress.Compress(compress.MethodRLE, b.data, b.len)
    println compress.BufLen(b4) > 0, compress.Method.Huffman
    println compress.BufLen(compress.RLEDecompress(compress.RLECompress(b.data, b.len).data, compress.RLECompress(b.data, b.len).len)) == b.len
    h := compress.HuffmanCompress(b.data, b.len)
    println compress.BufEquals(compress.HuffmanDecompress(h.data, h.len), b)
    l := compress.LZ77Compress(b.data, b.len)
    println compress.BufEquals(compress.LZ77Decompress(l.data, l.len), b)
    w := compress.LZWCompress(b.data, b.len)
    println compress.BufEquals(compress.LZWDecompress(w.data, w.len), b)
    bw := compress.BWTCompress(b.data, b.len)
    println compress.BufEquals(compress.BWTDecompress(bw.data, bw.len), b)
    mut primary := 0
    t := compress.BWTTransform(b.data, 3, primary)
    println primary, compress.BufToStrUnsafe(compress.BWTInverse(t.data, t.len, primary)) == "aba"
    m := compress.MTFEncode(b)
    println compress.BufEquals(compress.MTFDecode(m), b)
end
Output0 62 true 1 63 true 2 281 true 3 22 true 4 35 true 5 545 true 6 282 true 3 true aaaaaaaaaa true 2 true true true true true 1 true true

ByteBuf

struct ByteBuf
A byte buffer: an array of bytes that grows by itself. Most packages in this group take and give binary data in one.
NameTypeDescription
datavoid*The memory address of the bytes.
lenint = 0The number of bytes.
capint = 0The capacity.
compress-Buffer.salam
import compress

func main:
    mut b := compress.NewByteBufCap(16)
    compress.BufPush(b, 72)
    compress.BufPush(b, 105)
    mut other := compress.BufFromStr("!!")
    compress.BufPushBuf(b, other)
    compress.BufPushBytes(b, other.data, 1)
    compress.BufPushU32(b, 10)
    println compress.BufLen(b), compress.BufGet(b, 0), compress.BufToBase64(b)
    compress.BufSet(b, 0, 104)
    c := compress.BufClone(b)
    println compress.BufEquals(b, c), compress.BufToStrUnsafe(compress.BufFromStr("hello"))
    mut r := compress.ReaderFromBuf(b)
    println compress.RdLeft(r), compress.RdByte(r), compress.RdByte(r)
    three := compress.RdBytes(r, 3)
    println compress.BufLen(three), compress.RdU32(r), compress.RdLeft(r), r.bad
    _ := compress.RdByte(r)
    println r.bad
    mut r2 := compress.NewByteReader(c.data, 2)
    println compress.RdByte(r2)
    compress.BufReset(b)
    println compress.BufLen(b)
    compress.BufFree(b)
    compress.BufFree(c)
    compress.BufFree(other)
end
Output9 72 SGkhISEAAAAK true hello 9 104 105 3 10 0 false true 104 0

BitWriter

struct BitWriter
A writer of bits; it fills each byte from the least significant bit.
NameTypeDescription
bufByteBufThe bytes written.
accint = 0The bits waiting.
nbitsint = 0The number of bits waiting.
compress-Bits.salam
import compress

func main:
    mut w := compress.NewBitWriter()
    compress.BitWriteBit(w, 1)
    compress.BitWriteBits(w, 5, 3)
    compress.BitWriteBits(w, 255, 8)
    compress.BitFlush(w)
    println compress.BitWriterLen(w)
    mut r := compress.BitReaderFromBuf(w.buf)
    println compress.BitReadBit(r), compress.BitReadBits(r, 3), compress.BitReadBits(r, 8)
    mut r2 := compress.NewBitReader(w.buf.data, 1)
    println compress.BitReadBits(r2, 4)
    compress.BitWriterFree(w)
end
Output2 1 5 255 11

GzipCompress

func GzipCompress(p: void*, n: int): ByteBuf
Compresses the data into the gzip format, the one in .gz files.
compress-Gzip.salam
import compress

func main:
    text := "hello gzip! hello gzip! hello gzip!"
    g := compress.GzipCompressStr(text)
    println compress.BufGet(g, 0), compress.BufGet(g, 1)
    mut ok := false
    b := compress.GzipDecompressBuf(g, ok)
    println ok, compress.BufToStrUnsafe(b) == text
    b2 := compress.BufFromStr(text)
    g2 := compress.GzipCompressBuf(b2)
    g3 := compress.GzipCompress(b2.data, b2.len)
    println compress.BufEquals(g2, g3)
    _ := compress.GzipDecompress(b2.data, b2.len, ok)
    println ok
    d := compress.DeflateCompress(b2.data, b2.len)
    println compress.BufEquals(compress.DeflateDecompress(d.data, d.len), b2)
    raw := compress.RawDeflate(b2.data, b2.len)
    back := compress.RawInflate(raw.data, raw.len, b2.len, ok)
    println ok, compress.BufEquals(back, b2)
end
Output31 139 true true true false true true true

CRC32

func CRC32(p: void*, n: int): u32
The CRC-32 checksum, the one in ZIP and gzip, for checking that data is intact.
compress-Checksum.salam
import compress

func main:
    println compress.CRC32Str("123456789"), compress.Adler32Str("Wikipedia")
    b := compress.BufFromStr("123456789")
    println compress.CRC32(b.data, b.len) == compress.CRC32Buf(b)
    println compress.Adler32(b.data, b.len) == compress.Adler32Buf(b)
    println compress.CRC32Str("hello")
end
