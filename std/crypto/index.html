Standard library · Security and cryptography

crypto

SHA and MD5 digests, HMAC signatures, secure random numbers and password hashing.

import crypto
On this page
  1. Types
  2. Functions

This package gives the basic cryptography tools, all of them written in Salam. For the digest of a piece of text the ...Hex functions are the simplest way; the ...Bytes functions write the digest into memory you pass in yourself.

Never store a user password raw or under a plain digest; use HashPassword and VerifyPassword. To compare tokens, use SecretEquals rather than ==. To encrypt data, see the tls package.

Types

Sha256

struct Sha256
The SHA-256 state for data that arrives in pieces, such as a large file.
NameTypeDescription
hu32[]The internal state.
blku8[]The partial block.
blenint = 0The length of the partial block.
totalu64 = 0The total number of bytes.
crypto-Hash.salam
import crypto
import encoding
import mem

func main:
    println crypto.Sha256Hex("abc")
    println crypto.Sha512Hex("abc").len()
    println crypto.Sha1Hex("abc")
    println crypto.Sha1Base64("abc")
    println crypto.Md5Hex("abc")

    text := "abc"
    out := mem.Allocate(64 as u64)
    crypto.Sha256Bytes(text as void*, 3, out)
    println encoding.Base64EncodeBytes(out, 32)
    crypto.Sha384Bytes(text as void*, 3, out)
    println encoding.Base64EncodeBytes(out, 48)
    crypto.Sha512Bytes(text as void*, 3, out)
    println encoding.Base64EncodeBytes(out, 16).len()
    crypto.Sha1Bytes(text as void*, 3, out)
    println encoding.Base64EncodeBytes(out, 20)
    crypto.Md5Bytes(text as void*, 3, out)
    println encoding.Base64EncodeBytes(out, 16)

    mut h := crypto.Sha256New()
    crypto.Sha256Update(h, "a" as void*, 1)
    crypto.Sha256Update(h, "bc" as void*, 2)
    crypto.Sha256Final(h, out)
    println encoding.Base64EncodeBytes(out, 32)
    mut b := crypto.Sha512New()
    crypto.Sha512Update(b, text as void*, 3)
    crypto.Sha512Final(b, out)
    println b.outlen
    mut s := crypto.Sha384New()
    crypto.Sha512Update(s, text as void*, 3)
    crypto.Sha512Final(s, out)
    println s.outlen, encoding.Base64EncodeBytes(out, 48)
    mem.Free(out)

    println crypto.HashStr("hello") == crypto.HashStr("hello"), crypto.HashStr("a") != crypto.HashStr("b")
    println crypto.HashBytes(text as void*, 3 as u64) == crypto.HashStr("abc")
    println crypto.HashInt(42 as u64) != crypto.HashInt(43 as u64)
end

crypto-Password.salam
import crypto
import str
import mem

func main:
    stored := crypto.HashPassword("s3cret-password")
    println str.StartsWith(stored, "pbkdf2-sha256$")
    println crypto.VerifyPassword("s3cret-password", stored), crypto.VerifyPassword("wrong", stored)
    println stored != crypto.HashPassword("s3cret-password")

    println crypto.Pbkdf2HmacSha256Hex("password", "salt", 1, 32)
    println crypto.SecretEquals("abc", "abc"), crypto.SecretEquals("abc", "xyz")

    println crypto.RandomHex(16).len(), crypto.RandomToken(32).len() > 0
    b := mem.Allocate(8 as u64)
    crypto.RandomBytes(b, 8)
    mem.Free(b)
end

HmacSha256Hex

func HmacSha256Hex(key: str, msg: str): str
The HMAC-SHA256 signature of a message under a key, as hex; for checking that the message is untouched and comes from whoever holds the key.
crypto-Hmac.salam
import crypto
import encoding
import mem

func main:
    key := "key"
    message := "The quick brown fox jumps over the lazy dog"
    println crypto.HmacSha256Hex(key, message)
    println crypto.HmacSha1Hex(key, message)
    println crypto.HmacSha1Base64(key, message)
    println crypto.HmacSha384Hex(key, message).len(), crypto.HmacSha512Hex(key, message).len()

    out := mem.Allocate(64 as u64)
    crypto.HmacSha256Bytes(key, message, out)
    a := encoding.Base64EncodeBytes(out, 32)
    crypto.HmacSha256Raw(key as void*, 3, message as void*, message.len(), out)
    println a == encoding.Base64EncodeBytes(out, 32)
    crypto.HmacSha1Bytes(key, message, out)
    b := encoding.Base64EncodeBytes(out, 20)
    crypto.HmacSha1Raw(key as void*, 3, message as void*, message.len(), out)
    println b == encoding.Base64EncodeBytes(out, 20), b == crypto.HmacSha1Base64(key, message)
    crypto.HmacSha384Bytes(key, message, out)
    c := encoding.Base64EncodeBytes(out, 48)
    crypto.HmacSha384Raw(key as void*, 3, message as void*, message.len(), out)
    println c == encoding.Base64EncodeBytes(out, 48)
    crypto.HmacSha512Bytes(key, message, out)
    d := encoding.Base64EncodeBytes(out, 64)
    crypto.HmacSha512Raw(key as void*, 3, message as void*, message.len(), out)
    println d == encoding.Base64EncodeBytes(out, 64)
    mem.Free(out)
end
