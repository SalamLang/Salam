data

Reading data for machine learning: CSV tables, batching samples, MNIST and images, and synthetic data.

import data
On this page
  1. Types
  2. Constants and variables
  3. Functions

This package readies data for training a model. LoadCsv builds a table that turns text columns into category numbers by itself, and split_xy divides it into a feature matrix and a label.

The Make... functions build synthetic data from a fixed seed, for trying algorithms out without needing a file. All of it works with the matrix and tensor packages.

Types

Table

struct Table
A data table with named columns, for machine learning.
NameTypeDescription
namesVectorThe column names.
kindsVectorThe kind of each column; one of the COL_ constants.
catsVector>The levels of each categorical column.
rawVector>The original text of the cells.
cellsmatrix.MatrixThe numeric value of the cells; a categorical column holds the level number.
data-Table.salam
import data
import matrix

func main:
    text := "age,city,score,ok\n30,tehran,1.5,1\n25,shiraz,2.5,0\n41,tehran,3.0,1\n"
    mut t := data.LoadCsv(text, true)
    println t.rows(), t.cols(), t.names, t.col_index("score"), t.col_index("x")
    println t.kinds[0] == data.COL_INT, t.kinds[1] == data.COL_CAT, t.kinds[2] == data.COL_FLOAT, data.COL_STR
    println t.cat_levels("city")
    mut m := t.to_matrix()
    println m.rows, m.cols, m.at(2, 1)
    mut x := matrix.Matrix {}
    mut y := matrix.Matrix {}
    println t.split_xy("ok", x, y), x.cols, y.at(1, 0)
    mut good := true
    _ := data.LoadCsvFile("missing.csv", true, good)
    println good
    matrix.Free(m)
    matrix.Free(x)
    matrix.Free(y)
    t.free()
end
Output3 4 ["age", "city", "score", "ok"] 2 -1 true true true 3 ["tehran", "shiraz"] 3 4 0 true 3 0 false

data-Batch.salam
import data
import matrix
import tensor

func main:
    mut y := matrix.Matrix {}
    x := data.MakeBlobs(10, 2, 3, 0.2, 1, y)
    mut set := data.MatrixDataset { x = x, y = y }
    println set.len()
    mut it := set.batches(4, false, 0)
    println it.batches_per_epoch()
    mut batch_x := matrix.Matrix {}
    mut batch_y := matrix.Matrix {}
    until it.next(batch_x, batch_y):
        println batch_x.rows, batch_x.cols, batch_y.rows
    end
    it.reset(1)
    mut t := tensor.Tensor {}
    mut labels := Vector {} as Vector<int>
    println it.next_tensor(t, labels), labels.len()
    println data.YLabels(y).len(), data.XTensor(x).shape
    mut good := true
    _ := data.ParseIdx(null, 0, good)
    println good
    _ := data.LoadIdx("missing.idx", good)
    println good, data.LoadMnist("a", "b", batch_x, batch_y)
    mut classes := Vector {} as Vector<str>
    println data.LoadImageFolder("missing-folder", 8, 8, true, t, labels, classes)
    it.free()
    set.free()
end

data-Synth.salam
import data
import matrix

func count_ones(y: matrix.Matrix): int:
    mut n := 0
    repeat y.rows in i:
        if y.at(i, 0) == 1.0:
            n += 1
        end
    end
    ret n
end

func main:
    mut y := matrix.Matrix {}
    blobs := data.MakeBlobs(90, 3, 2, 0.5, 42, y)
    println blobs.rows, blobs.cols, y.rows
    moons := data.MakeMoons(100, 0.1, 7, y)
    println moons.rows, moons.cols, count_ones(y)
    circles := data.MakeCircles(60, 0.05, 0.5, 7, y)
    println circles.rows, count_ones(y)
    mut coef := matrix.Matrix {}
    reg := data.MakeRegression(50, 3, 0.1, 1, y, coef)
    println reg.rows, reg.cols, coef.rows
    cls := data.MakeClassification(40, 4, 2, 1.0, 3, y)
    println cls.rows, cls.cols, count_ones(y) > 0
end
Output90 2 90 100 2 50 60 30 50 3 3 40 4 true
