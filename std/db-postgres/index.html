db/postgres

Connecting to PostgreSQL.

import db.postgres
On this page
  1. Types
  2. Functions

This package implements the Postgres protocol directly and needs no outside library; the example needs a running Postgres server, so it is checked and not run. Every row of a result is fetched from the server at once, so RowCount is known before you read any of them.

db-postgres-Client.salam
import db
import db.postgres

func main:
    if not postgres.Available():
        println "the Postgres client is not available"
        ret
    end
    println postgres.ClientVersion()
    d := postgres.Open("host=127.0.0.1 port=5432 dbname=shop user=app password=s3cret")
    if not postgres.Ok(d):
        println postgres.Error(d)
        ret
    end
    println postgres.ServerVersion(d), postgres.DatabaseName(d), postgres.UserName(d), postgres.HostName(d), postgres.Port(d)
    println postgres.Ping(d)
    _ := postgres.Exec(d, "CREATE TABLE IF NOT EXISTS items (id SERIAL PRIMARY KEY, name TEXT, price NUMERIC)")
    _ := postgres.Exec(d, "INSERT INTO items (name, price) VALUES (" + postgres.Quote(d, "tea") + ", 12.5)")
    println postgres.ExecCount(d, "UPDATE items SET price = price * 1.1")

    rows := postgres.Query(d, "SELECT id, name, price FROM items ORDER BY id")
    println rows.Ok(), rows.ColumnCount(), rows.RowCount(), rows.ColumnName(1)
    until rows.Next():
        println rows.Int(0), rows.Text(1), rows.Float(2), rows.Int64(0), rows.IsNull(2), rows.ColumnType(1)
    end
    _ := rows.Close()
    println postgres.QueryInt(d, "SELECT COUNT(*) FROM items"), postgres.QueryText(d, "SELECT version()")

    c := postgres.Conn(d)
    tx := c.Begin()
    _ := tx.Exec("DELETE FROM items")
    _ := tx.Rollback()
    println db.QueryInt(c, "SELECT COUNT(*) FROM items"), d.Kind()
    postgres.Reset(d)
    _ := postgres.Close(d)
    _ := postgres.OpenFull("db.example.com", 5432, "app", "s3cret", "shop")
end
