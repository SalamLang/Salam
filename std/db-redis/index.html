db/redis

A Redis client: keys and values, hashes, lists, sets and publishing.

import db.redis
On this page
  1. Types
  2. Constants and variables
  3. Functions

Redis is an in memory database for caches, sessions, counters and queues. This package implements the Redis protocol directly and with no outside library; the example needs a running Redis server, so it is only checked. For a command with no function of its own, call Command with the command name and its arguments.

db-redis-Client.salam
import db.redis

func main:
    c := redis.Connect("127.0.0.1", 6379)
    if not redis.Ok(c):
        println redis.Error(c)
        ret
    end
    _ := redis.Auth(c, "s3cret")
    _ := redis.Select(c, 1)
    println redis.Ping(c)

    _ := redis.Set(c, "greeting", "hello")
    _ := redis.SetEx(c, "session:42", "token", 3600)
    println redis.Get(c, "greeting"), redis.Exists(c, "greeting"), redis.TTL(c, "session:42")
    println redis.Incr(c, "visits"), redis.IncrBy(c, "visits", 10), redis.Decr(c, "visits")
    println redis.Append(c, "greeting", "!"), redis.StrLen(c, "greeting"), redis.Type(c, "greeting")
    _ := redis.Expire(c, "greeting", 60)
    println redis.Keys(c, "session:*").len()
    _ := redis.Del(c, "greeting")

    _ := redis.HSet(c, "user:1", "name", "Maryam")
    _ := redis.HSet(c, "user:1", "city", "Tehran")
    println redis.HGet(c, "user:1", "name"), redis.HExists(c, "user:1", "age"), redis.HLen(c, "user:1")
    all := redis.HGetAll(c, "user:1")
    println all.size()
    _ := redis.HDel(c, "user:1", "city")

    _ := redis.RPush(c, "jobs", "a")
    _ := redis.LPush(c, "jobs", "b")
    println redis.LLen(c, "jobs"), redis.LRange(c, "jobs", 0, -1).len()
    println redis.LPop(c, "jobs"), redis.RPop(c, "jobs")

    _ := redis.SAdd(c, "tags", "go")
    _ := redis.SAdd(c, "tags", "salam")
    println redis.SIsMember(c, "tags", "salam"), redis.SCard(c, "tags"), redis.SMembers(c, "tags").len()
    _ := redis.SRem(c, "tags", "go")
    println redis.Publish(c, "news", "fresh news")

    mut args := Vector {} as Vector<str>
    args.push("GET")
    args.push("visits")
    r := redis.Command(c, args)
    if redis.ReplyOk(r) and not redis.ReplyIsError(r) and not redis.ReplyIsNil(r):
        println redis.ReplyType(r) == redis.REPLY_STRING, redis.ReplyStr(r)
    end
    redis.FreeReply(r)
    mut args2 := Vector {} as Vector<str>
    args2.push("LRANGE")
    args2.push("jobs")
    args2.push("0")
    args2.push("-1")
    r2 := redis.Command(c, args2)
    if redis.ReplyType(r2) == redis.REPLY_ARRAY:
        repeat redis.ReplyArrayLen(r2) in i:
            println redis.ReplyStr(redis.ReplyArrayGet(r2, i))
        end
    end
    println redis.ReplyInt(r2), redis.REPLY_INTEGER, redis.REPLY_NIL, redis.REPLY_STATUS, redis.REPLY_ERROR, redis.REPLY_DOUBLE
    redis.FreeReply(r2)
    _ := redis.FlushDB(c)
    redis.Close(c)
end
