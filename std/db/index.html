db

One interface shared by the database engines, schema migrations and a connection pool.

import db

Every database engine has its own package: db/sqlite, db/mysql, db/postgres, for key and value db/redis, and for search db/typesense, db/meilisearch, db/elasticsearch and, for search and analytics, db/opensearch; the last two also answer SQL queries through Connection. This package gives the Connection interface that the three SQL engines implement; write your code against it and changing engine is one line. To check the SQL itself before it ever reaches a server, db/sql parses it for each engine's dialect and resolves it against your schema.

Migrations move the database schema along with the code: every change has a number and MigrateUp applies the ones numbered above the current schema version. Never paste a value a user typed straight into SQL; send it through QuoteLiteral or as a prepared parameter.

Things that catch people out:

  • QuoteLiteral only doubles single quotes, which is the SQL standard that SQLite and PostgreSQL follow. MySQL also treats a backslash as an escape by default, so on MySQL quote with the connection's own c.Quote, which escapes backslashes as well.
  • MigrateUp compares version numbers, not a list of applied names. A migration numbered below the highest applied one, for example one merged from an older branch, is skipped without a word. Always give a new migration the next free number.
  • MySQL commits every CREATE, ALTER and DROP on its own, so on MySQL a migration that fails halfway can leave the first statements applied. SQLite and PostgreSQL roll the whole migration back.
  • Migration SQL is split into statements at each ; outside single quotes. A PostgreSQL function body in $$ ... $$ or a trigger with several statements is cut apart; run those with c.Exec instead.

Writing code against Connection

Give your data functions a dyn db.Connection parameter instead of a SQLite or PostgreSQL database. They then work with every SQL engine, and tests can pass an in-memory SQLite database while production passes PostgreSQL. db.Run runs one statement and returns true on success; c.Error() says what went wrong; c.LastInsertId() gives the id of the row just inserted.

db-guide-store.salam
import db
import db.sqlite

func setup(c: dyn db.Connection): bool:
    ret db.Run(c, "CREATE TABLE notes (id INTEGER PRIMARY KEY, author TEXT NOT NULL, body TEXT NOT NULL)")
end

func add_note(c: dyn db.Connection, author: str, body: str): i64:
    sql := "INSERT INTO notes (author, body) VALUES (" + db.QuoteLiteral(author) + ", " + db.QuoteLiteral(body) + ")"
    if not db.Run(c, sql):
        println "insert failed:", c.Error()
        ret 0
    end
    ret c.LastInsertId()
end

func print_notes(c: dyn db.Connection, author: str):
    rows := c.Query("SELECT id, body FROM notes WHERE author = " + db.QuoteLiteral(author) + " ORDER BY id")
    until rows.Next():
        println "#" + rows.Int(0), rows.Text(1)
    end
    _ := rows.Close()
end

func main:
    c := sqlite.Conn(sqlite.Open(":memory:"))
    if not setup(c):
        println "setup failed:", c.Error()
        ret 1
    end
    add_note(c, "Sara", "buy milk")
    add_note(c, "Omid", "call the bank")
    add_note(c, "Sara", "it's Reza's birthday")
    add_note(c, "x' OR '1'='1", "trying to break in")
    print_notes(c, "Sara")
    println "notes by Omid:", db.QueryInt(c, "SELECT COUNT(*) FROM notes WHERE author = 'Omid'")
    println "all notes:", db.QueryInt(c, "SELECT COUNT(*) FROM notes")
    _ := c.Close()
    ret 0
end
Output#1 buy milk #3 it's Reza's birthday notes by Omid: 1 all notes: 4

Every value that came from a user goes through QuoteLiteral, which wraps it in quotes and doubles any quote inside. That is why the apostrophe in Reza's is stored correctly, and why the author name x' OR '1'='1 is stored as an odd name instead of changing the meaning of the query. Never build SQL by gluing raw input into it.

A query returns dyn db.Rows. Next moves to the next row and returns false at the end, so until rows.Next(): visits every row; read columns by position with Int, Text, Float and IsNull, and always Close the rows. For a single number or string, QueryInt and QueryText do all of that in one call.

Transactions: all or nothing

When several statements must succeed or fail together, run them inside a transaction. c.Begin() returns a dyn db.Tx with its own Exec and Query; Commit makes the changes permanent and Rollback throws them all away. The transfer below takes money from one account, gives it to the other, then checks the result and rolls back if the first account went below zero.

db-guide-transfer.salam
import db
import db.sqlite

func transfer(c: dyn db.Connection, from_id: int, to_id: int, amount: int): str:
    tx := c.Begin()
    if not tx.Ok():
        ret "could not start: " + c.Error()
    end
    _ := tx.Exec("UPDATE accounts SET balance = balance - " + amount + " WHERE id = " + from_id)
    _ := tx.Exec("UPDATE accounts SET balance = balance + " + amount + " WHERE id = " + to_id)
    rows := tx.Query("SELECT balance FROM accounts WHERE id = " + from_id)
    mut left := 0
    if rows.Next():
        left = rows.Int(0)
    end
    _ := rows.Close()
    if left < 0:
        _ := tx.Rollback()
        ret "refused, only " + (left + amount) + " left"
    end
    _ := tx.Commit()
    ret "moved " + amount
end

func balances(c: dyn db.Connection): str:
    ret db.QueryText(c, "SELECT group_concat(name || '=' || balance, ' ') FROM accounts")
end

func main:
    c := sqlite.Conn(sqlite.Open(":memory:"))
    mut setup := Vector {} as Vector<str>
    defer setup.free()
    setup.push("CREATE TABLE accounts (id INTEGER PRIMARY KEY, name TEXT, balance INTEGER)")
    setup.push("INSERT INTO accounts VALUES (1, 'ali', 100), (2, 'sara', 50)")
    println db.InTx(c, setup)
    println transfer(c, 1, 2, 30), "|", balances(c)
    println transfer(c, 1, 2, 500), "|", balances(c)
    println transfer(c, 2, 1, 80), "|", balances(c)
    _ := c.Close()
end
Outputtrue moved 30 | ali=70 sara=80 refused, only 70 left | ali=70 sara=80 moved 80 | ali=150 sara=0

Inside the transaction the queries already see the uncommitted changes, which is what makes the check possible. Other connections do not see them until Commit. When you have a fixed list of statements and no checks in between, db.InTx runs them in one transaction and returns false, with nothing applied, if any of them fails.

Migrations at start up

Call MigrateUp every time the program starts. It records each applied version in a table of its own, so the first run creates the schema, later runs do nothing, and a release that adds migration 3 applies only that one. Existing data stays where it is.

db-guide-startup.salam
import db
import db.sqlite

func migrations(): db.Migrator:
    mut m := db.NewMigrator()
    db.AddMigration(m, 1, "create users", "CREATE TABLE users (id INTEGER PRIMARY KEY, email TEXT NOT NULL UNIQUE)", "DROP TABLE users")
    db.AddMigration(m, 2, "add names", "ALTER TABLE users ADD COLUMN name TEXT DEFAULT ''", "ALTER TABLE users DROP COLUMN name")
    ret m
end

func start(c: dyn db.Connection):
    mut m := migrations()
    before := db.SchemaVersion(m, c)
    applied := db.MigrateUp(m, c)
    println "schema", before, "->", db.SchemaVersion(m, c), "(" + applied + " applied)"
    db.FreeMigrator(m)
end

func main:
    c := sqlite.Conn(sqlite.Open(":memory:"))
    start(c)
    println db.Run(c, "INSERT INTO users (email, name) VALUES ('a@x.ir', 'Ali')")
    start(c)
    mut m := migrations()
    db.AddMigration(m, 3, "add city", "ALTER TABLE users ADD COLUMN city TEXT", "ALTER TABLE users DROP COLUMN city")
    println "pending:", db.PendingCount(m, c)
    println "applied:", db.MigrateUp(m, c), "now at", db.SchemaVersion(m, c)
    println db.QueryText(c, "SELECT name FROM users WHERE email = 'a@x.ir'")
    db.FreeMigrator(m)
    _ := c.Close()
end
Outputschema 0 -> 2 (2 applied) true schema 2 -> 2 (0 applied) pending: 1 applied: 1 now at 3 Ali

Never change a migration that has already run somewhere: add a new one instead. Each migration runs in its own transaction, so on SQLite and PostgreSQL a failing one leaves the schema at the previous version. Rollback undoes the last migration with its down SQL and MigrateTo moves to any version, which is useful in development but rarely in production.

Choosing the engine at run time

Because the rest of the program only sees dyn db.Connection, the engine can be picked from a setting. Here the APP_DB environment variable chooses PostgreSQL, MySQL or a SQLite file, and everything after connect is the same for all three.

db-guide-engines.salam
import db
import db.sqlite
import db.postgres
import db.mysql
import env

func connect(): dyn db.Connection:
    engine := env.GetOr("APP_DB", "sqlite")
    if engine == "postgres":
        ret postgres.Conn(postgres.Open("host=127.0.0.1 port=5432 user=app password=secret dbname=app"))
    else engine == "mysql":
        ret mysql.Conn(mysql.Open("127.0.0.1", "app", "secret", "app", 3306))
    end
    ret sqlite.Conn(sqlite.Open("app.db"))
end

func main:
    c := connect()
    if not c.Ok():
        println "cannot connect to", c.Kind() + ":", c.Error()
        ret 1
    end
    println "connected to", c.Kind()
    _ := db.Run(c, "CREATE TABLE IF NOT EXISTS visits (page VARCHAR(100))")
    _ := db.Run(c, "INSERT INTO visits (page) VALUES ('home')")
    println "visits so far:", db.QueryInt(c, "SELECT COUNT(*) FROM visits")
    _ := c.Close()
    ret 0
end

Without APP_DB it uses SQLite and needs nothing else; each run adds a visit, so the count grows by one each time. To try PostgreSQL, start a server with Docker and run the program with APP_DB=postgres:

docker run -d -p 5432:5432 -e POSTGRES_USER=app -e POSTGRES_PASSWORD=secret -e POSTGRES_DB=app postgres:17
APP_DB=postgres salam run db-guide-engines.salam

Stick to SQL that all your engines understand, or keep the engine specific parts in one function. Placeholders and quoting differ between engines, which is one more reason to let QuoteLiteral or c.Quote do the quoting.

db-Pool.salam
import db
import db.sqlite

func main:
    mut pool := db.NewPool(2)
    repeat 3:
        println db.Add(pool, sqlite.Conn(sqlite.Open(":memory:")))
    end
    println db.Size(pool), db.Idle(pool), db.InUse(pool)

    one := db.Acquire(pool)
    two := db.Acquire(pool)
    three := db.Acquire(pool)
    println db.Leased(one), db.Leased(three), three == db.NO_LEASE
    println db.Idle(pool), db.InUse(pool)

    c := db.Conn(pool, one)
    println db.QueryInt(c, "SELECT 6 * 7")
    db.Release(pool, one)
    db.Release(pool, two)
    println db.Idle(pool), db.DropDead(pool)
    db.CloseAll(pool)
    db.Free(pool)
end
Outputtrue true false 2 2 0 true false true 0 2 42 2 0

Rows

interface Rows
The interface of the rows of a query; its methods are next, text, int, close and the like.
db-Connection.salam
import db
import db.sqlite

func report(c: dyn db.Connection):
    println c.Kind(), c.Ok(), c.Ping()
    _ := c.Exec("CREATE TABLE users (id INTEGER PRIMARY KEY, name TEXT, score REAL)")
    _ := c.Exec("INSERT INTO users (name, score) VALUES (" + c.Quote("O'Brien") + ", 7.5)")
    _ := c.Exec("INSERT INTO users (name, score) VALUES ('Maryam', 9)")
    println c.LastInsertId(), c.Changes()

    rows := c.Query("SELECT id, name, score FROM users ORDER BY id")
    println rows.Ok(), rows.ColumnCount(), rows.ColumnName(1)
    until rows.Next():
        println rows.Int(0), rows.Text(1), rows.Float(2), rows.Int64(0), rows.IsNull(2), rows.ColumnType(1)
    end
    _ := rows.Close()

    tx := c.Begin()
    println tx.Ok()
    _ := tx.Exec("UPDATE users SET score = 0")
    txrows := tx.Query("SELECT SUM(score) FROM users")
    _ := txrows.Next()
    println txrows.Float(0)
    _ := txrows.Close()
    _ := tx.Rollback()
    tx2 := c.Begin()
    _ := tx2.Exec("DELETE FROM users WHERE id = 1")
    _ := tx2.Commit()

    println db.QueryInt(c, "SELECT COUNT(*) FROM users"), db.QueryInt64(c, "SELECT MAX(id) FROM users")
    println db.QueryFloat(c, "SELECT score FROM users"), db.QueryText(c, "SELECT name FROM users")
    println db.CountRows(c, "SELECT * FROM users")
    println db.Run(c, "INSERT INTO users (name) VALUES ('Ali')"), db.Run(c, "INSERT INTO nowhere VALUES (1)")
    println c.Error()
    mut stmts := Vector {} as Vector<str>
    stmts.push("INSERT INTO users (name) VALUES ('Sara')")
    stmts.push("INSERT INTO users (name) VALUES ('Reza')")
    println db.RunAll(c, stmts)
    stmts.push("INSERT INTO nowhere VALUES (1)")
    println db.InTx(c, stmts), db.QueryInt(c, "SELECT COUNT(*) FROM users")
    println db.QuoteLiteral("it's")
    _ := c.Close()
end

func main:
    report(sqlite.Conn(sqlite.Open(":memory:")))
    println db.SQLITE, db.MYSQL, db.POSTGRES
    println db.TYPE_NULL, db.TYPE_INT, db.TYPE_FLOAT, db.TYPE_TEXT, db.TYPE_BLOB
end

db-Migrate.salam
import db
import db.sqlite

func main:
    c := sqlite.Conn(sqlite.Open(":memory:"))
    mut m := db.NewMigrator()
    db.AddMigration(m, 1, "users", "CREATE TABLE users (id INTEGER PRIMARY KEY, email TEXT NOT NULL)", "DROP TABLE users")
    db.AddMigration(m, 2, "posts", "CREATE TABLE posts (id INTEGER PRIMARY KEY, title TEXT); INSERT INTO posts (title) VALUES ('hello')", "DROP TABLE posts")
    db.AddMigration(m, 3, "tags", "ALTER TABLE posts ADD COLUMN tag TEXT", "ALTER TABLE posts DROP COLUMN tag")
    println db.MigrationCount(m), db.SchemaVersion(m, c), db.PendingCount(m, c)

    println db.MigrateTo(m, c, 2)
    println db.SchemaVersion(m, c), db.IsApplied(m, c, 2), db.IsApplied(m, c, 3)
    println db.MigrateUp(m, c), db.SchemaVersion(m, c)
    println db.MigrateUp(m, c)
    println db.Rollback(m, c), db.SchemaVersion(m, c)
    println db.MigrateTo(m, c, 0), db.SchemaVersion(m, c)
    println db.QueryInt(c, "SELECT COUNT(*) FROM " + db.MIGRATION_TABLE)
    db.FreeMigrator(m)

    mut m2 := db.NewMigratorIn("app_versions")
    db.AddMigration(m2, 1, "t", "CREATE TABLE t (x INTEGER)", "DROP TABLE t")
    println db.EnsureTable(m2, c), db.MigrateUp(m2, c), m2.table
    db.FreeMigrator(m2)

    parts := db.SplitStatements("INSERT INTO t VALUES (1); SELECT ';' ; ")
    println parts.len(), parts.get(1)
end
Output3 0 3 2 2 true false 1 3 0 true 2 2 0 0 0 1 app_versions 2 SELECT ';'
