decimal

Exact decimal numbers for money and accounting, with none of the rounding error of floating point.

import decimal
On this page
  1. Types
  2. Functions

In f64, 0.1 + 0.2 is not exactly 0.3. For science and graphics that tiny error does not matter; for money it does, because a lost cent in the accounts is not acceptable. The decimal package keeps a number as a large whole number and a "scale", the count of digits after the point, so 10.05 is 1005 with scale 2 and no error can creep in.

Addition, subtraction and multiplication are always exact. Division may not terminate, as one third does not, so you say how many decimal places you want and how to round. Rounding offers seven methods; accounting usually wants HalfEven.

Types

Rounding

enum Rounding
How to round when digits have to be dropped. The example below shows every method applied to −2.5.
NameTypeDescription
DownTowards zero: the extra digits are thrown away.
UpAway from zero.
FloorDownwards, towards negative infinity.
CeilUpwards, towards positive infinity.
HalfUpTo the nearest, with a half going away from zero. This is the rounding taught at school.
HalfDownTo the nearest, with a half going towards zero.
HalfEvenTo the nearest, with a half going to the even number, known as banker's rounding. Over many additions it keeps the rounding error at zero on average.
decimal-Round.salam
import decimal

func main:
    x := decimal.FromScaled(-25, 1)
    println decimal.ToStr(decimal.Round(x, 0)), decimal.ToStr(decimal.RoundHalfUp(x, 0)), decimal.ToStr(decimal.Truncate(x, 0))
    each mode in [decimal.Rounding.Down, decimal.Rounding.Up, decimal.Rounding.Floor, decimal.Rounding.Ceil, decimal.Rounding.HalfUp, decimal.Rounding.HalfDown, decimal.Rounding.HalfEven]:
        print decimal.ToStr(decimal.Rescale(x, 0, mode)), ""
    end
    println ""
    println decimal.ToStr(decimal.Trim(decimal.FromScaled(12500, 4)))
end
Output-2 -3 -2 -2 -3 -3 -2 -3 -2 -2 1.25

Dec

struct Dec
An exact decimal number.
NameTypeDescription
unscaledbigint.IntThe whole number without the point; for 10.05 it is 1005.
scaleint = 0The number of digits after the point.
decimal-Dec.salam
import decimal
import bigint

func main:
    mut a := decimal.FromScaled(1005, 2)
    defer a.free()
    println a.to_str(), a.is_zero(), a.sign()
    b := a.clone()
    println decimal.ToStr(b), decimal.Scale(b), bigint.ToStr(decimal.Unscaled(b))
    println decimal.ToStr(decimal.Zero()), decimal.ToStr(decimal.FromInt(42))
    println decimal.ToStr(decimal.New(bigint.FromInt(31415), 4))
    println decimal.ToStr(decimal.FromFloat(0.1, 3))
    println decimal.IsZero(decimal.Zero()), decimal.IsNegative(decimal.FromInt(-1)), decimal.Sign(a)
    mut c := decimal.Clone(a)
    decimal.Free(c)
end
Output10.05 false 1 10.05 2 1005 0 42 3.1415 0.100 true true 1

Split

func Split(total: Dec, n: int): Vector<Dec>
Divides an amount into n equal shares whose total is exactly the amount; the odd remainder goes to the first shares.
decimal-Split.salam
import decimal

func main:
    total := decimal.FromScaled(10000, 2)
    mut shares := decimal.Split(total, 3)
    defer decimal.FreeAll(shares)
    each s in shares:
        print decimal.ToStr(s), ""
    end
    println ""
    println decimal.ToStr(decimal.Sum(shares))
    mut ratios := Vector {} as Vector<i64>
    defer ratios.free()
    ratios.push(1 as i64)
    ratios.push(2 as i64)
    mut parts := decimal.Allocate(total, ratios)
    defer decimal.FreeAll(parts)
    each p in parts:
        print decimal.ToStr(p), ""
    end
    println ""
end
Output33.34 33.33 33.33 100.00 33.34 66.66

decimal-Add.salam
import decimal

func main:
    a := decimal.FromScaled(10, 1)
    b := decimal.FromScaled(20, 1)
    println decimal.ToStr(decimal.Add(a, b)), decimal.ToStr(decimal.Sub(a, b)), decimal.ToStr(decimal.Mul(a, b))
    println decimal.ToStr(decimal.Div(a, decimal.FromInt(3), 4, decimal.Rounding.HalfEven))
    println decimal.ToStr(decimal.DivRound(a, decimal.FromInt(3), 2))
    println decimal.ToStr(decimal.Pow(b, 3)), decimal.ToStr(decimal.Neg(a)), decimal.ToStr(decimal.Abs(decimal.Neg(a)))
    println decimal.ToStr(decimal.MulInt(a, 3)), decimal.ToStr(decimal.AddInt(a, 1)), decimal.ToStr(decimal.SubInt(a, 1))
    println decimal.ToStr(decimal.DivInt(decimal.FromInt(10), 4, 2, decimal.Rounding.Floor))
    println decimal.ToStr(decimal.Percent(decimal.FromInt(250000), decimal.FromInt(9), 0))
end
Output3.0 -1.0 2.00 0.3333 0.33 8.000 -1.0 1.0 3.0 2.0 0.0 2.50 22500

ToStr

func ToStr(d: Dec): str
The number as text with all its decimal places.
decimal-Parse.salam
import decimal

func main:
    mut ok := false
    d := decimal.Parse("1234567.891", ok)
    println ok, decimal.ToStr(d)
    println decimal.ToStrFixed(d, 1), decimal.FormatGrouped(d, 2, ",", ".")
    println decimal.ToStr(decimal.ParseOr("not a number"))
    mut fits := false
    println decimal.ToInt(decimal.FromInt(42), fits), fits, decimal.ToFloat(d)
end
Outputtrue 1234567.891 1234567.9 1,234,567.89 0 42 true 1234567.891

Compares two numbers whatever their scales; 1.50 and 1.5 are equal.
decimal-Cmp.salam
import decimal

func main:
    a := decimal.FromScaled(150, 2)
    b := decimal.FromScaled(15, 1)
    println decimal.Cmp(a, b), decimal.Equals(a, b)
    println decimal.Less(a, decimal.FromInt(2)), decimal.Greater(a, decimal.FromInt(2))
    println decimal.ToStr(decimal.Min(a, decimal.FromInt(1))), decimal.ToStr(decimal.Max(a, decimal.FromInt(1)))
end
