dom

Working with a web page in the browser: elements, events, timers and JavaScript objects.

import dom
On this page
  1. Functions

A program that uses this package runs in the browser. Translate it to JavaScript with salam js app.salam --output=app.js and put it in a page with a <script src="app.js"> tag; main runs once the page is ready, and println writes to the browser console. That is why the examples on this page are not run, only checked.

Elements and JavaScript objects are held as one i64 number. Events and timers take an anonymous function, such as (): ... end; keep a function you will want to drop later with OffEvent in a variable.

Functions

Query

func Query(sel: str): i64
The first element that matches the CSS selector; zero when there is none.
dom-Elements.salam
import dom

func main:
    list := dom.Create("ul")
    dom.SetAttr(list, "id", "books")
    dom.Append(dom.Body(), list)
    repeat 3 in i:
        item := dom.Create("li")
        dom.SetText(item, "book " + (i + 1))
        dom.AddClass(item, "book")
        dom.Append(list, item)
    end
    println dom.Exists("books"), dom.Exists("nothing")

    all := dom.QueryAll("li.book")
    println dom.CountOf(all)
    second := dom.ItemAt(all, 1)
    println dom.GetText(second)
    dom.ToggleClass(second, "selected")
    println dom.GetAttr(second, "class")
    dom.RemoveClass(second, "book")
    println dom.GetAttr(second, "class")

    dom.SetStyle(second, "color", "red")
    println dom.GetStyle(second, "color")
    first := dom.Query("li")
    dom.Remove(first)
    println dom.CountOf(dom.QueryAll("li"))

    box := dom.ById("books")
    dom.SetHtml(box, "<li>new</li>")
end

Document

func Document(): i64
The document object.
dom-Objects.salam
import dom

func main:
    d := dom.Document()
    println dom.GetStr(d, "title")
    dom.SetStr(d, "title", "my page")
    println dom.GetStr(d, "title")
    w := dom.Window()
    println dom.GetNum(w, "innerWidth") == dom.WinWidth()
    println dom.DocWidth() > 0, dom.DocHeight() > 0

    box := dom.Create("input")
    dom.SetBool(box, "disabled", true)
    println dom.GetBool(box, "disabled")
    dom.SetNum(box, "tabIndex", 3)
    println dom.GetNum(box, "tabIndex")
    dom.SetProp(box, "parentRef", dom.Body())
    println dom.GetProp(box, "parentRef") == dom.Body()

    math := dom.Global("Math")
    println dom.Call1(math, "abs", -5), dom.Call2(math, "max", 2, 9)
    println dom.Call3(math, "max", 1, 2, 3), dom.Call4(math, "min", 4, 3, 2, 1)
    _ := dom.Call0(box, "focus")
    _ := dom.CallS1(box, "removeAttribute", "placeholder")
    _ := dom.CallS2(box, "setAttribute", "name", "q")
    println dom.GetAttr(box, "name")

    canvas := dom.Create("canvas")
    ctx := dom.GetContext(canvas, "2d")
    println ctx != 0

    dom.StoreSetInt("best", 42)
    dom.StoreSetStr("name", "Maryam")
    println dom.StoreGetInt("best"), dom.StoreGetStr("name")

    r := dom.Random()
    println r >= 0.0 and r < 1.0, dom.RandomInt(6) < 6
    println dom.Now() >= 0.0

    button := dom.Create("button")
    greet := ():
        println "hello"
    end
    dom.OnEvent(button, "click", greet)
    dom.OffEvent(button, "click", greet)

    image := dom.LoadImage("logo.png")
    println dom.ImageReady(image)
    dom.Tone(440, 200)
    if dom.Confirm("shall we go on?"):
        name := dom.Prompt("what is your name?")
        dom.Alert("hello " + name)
    end
end

Window

func Window(): i64
The window object.
dom-Events.salam
import dom

mut count := 0

func main:
    button := dom.Create("button")
    dom.SetText(button, "click me")
    dom.SetAttr(button, "data-autofire", "click")
    dom.Append(dom.Body(), button)
    dom.OnEvent(button, "click", ():
        count += 1
        println "click", count
    end)

    box := dom.Create("input")
    dom.SetAttr(box, "data-autofire", "keydown")
    dom.Append(dom.Body(), box)
    dom.SetValue(box, "hello")
    println dom.GetValue(box)
    dom.OnEventE(box, "keydown", (e: i64):
        if dom.EventKey(e) == "Enter":
            dom.PreventDefault(e)
        end
        dom.StopPropagation(e)
        println "target:", dom.GetValue(dom.EventTarget(e))
    end)

    dom.OnWindowE("mousemove", (e: i64):
        println dom.EventX(e), dom.EventY(e)
    end)
    dom.OnWindowE("touchstart", (e: i64):
        println dom.TouchX(e), dom.TouchY(e)
    end)
    dom.OnWindow("scroll", ():
        println dom.ScrollX(), dom.ScrollY()
    end)
    dom.OnReady(():
        println "the page is ready"
    end)
    dom.OnResize(():
        println dom.WinWidth(), dom.WinHeight()
    end)
end

RequestFrame

func RequestFrame(handler: func(f64))
Calls the function once before the next frame is drawn; the function takes the time in milliseconds.
dom-Timers.salam
import dom

mut tick := 0
mut every := 0 as i64

func main:
    dom.SetTitle("timers")
    start := dom.Now()
    _ := dom.SetTimeout(():
        println "one second has passed"
    end, 1000)
    cancelled := dom.SetTimeout(():
        println "this never runs"
    end, 5000)
    dom.ClearTimeout(cancelled)
    every = dom.SetInterval(():
        tick += 1
        if tick == 3:
            dom.ClearInterval(every)
        end
    end, 200)
    dom.RequestFrame((now: f64):
        println "frame after", now - start
    end)
    dom.EveryFrame((now: f64):
        _ := now
    end)
end
