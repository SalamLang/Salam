encoding/json

Building and reading JSON: from a plain string through automatic struct conversion to streaming.

import encoding.json
On this page
  1. Types
  2. Functions

There are three ways to work with JSON here. For simple jobs, functions such as Member, Object and Get work directly on JSON text. For real programs, Marshal and Unmarshal convert a Salam struct to JSON and back. For data whose shape you do not know in advance, Decode returns a dynamic Value.

Set the key name of a field with @json "name" above it; @json "-" leaves the field out. Without it, the field's own name becomes the key.

JSONOptions

struct JSONOptions
The options for Encode.
NameTypeDescription
indentstr = ""The indent unit; empty means one line.
sort_keysbool = falseWrite the keys in alphabetical order. Without it the key order is not fixed.
ignore_nullbool = falseLeave out the keys whose value is null.
encoding-json-Value.salam
import encoding.json
import encoding.value
import encoding.encerr
import fs.file

func main:
    mut e := encerr.NoError()
    v := json.Decode(`{"city": "Tehran", "zones": [1, 2, 22]}`, e)
    println encerr.IsError(e), value.AsString(value.ObjectGet(v, "city"))
    println json.EncodeDefault(v)
    opts := json.JSONOptions { indent = "  ", sort_keys = true }
    println json.Encode(v, opts)
    _ := json.Decode("[1, 2", e)
    println encerr.IsError(e), e.position
    mut enc := json.NewBufEncoder()
    enc.Encode(value.ValueInt(1))
    enc.Encode(value.ValueString("two"))
    println enc.String()
    enc.Close()
    mut pretty := json.NewBufEncoderIndent("  ")
    pretty.Encode(v)
    println pretty.String()
    pretty.Close()
    out := file.Open("lines.ndjson", "wb")
    mut lines := json.NewEncoder(out)
    lines.Encode(value.ValueInt(10))
    lines.Encode(value.ValueInt(20))
    lines.Close()
    mut tabbed := json.NewEncoderIndent(file.Open("pretty.json", "wb"), "\t")
    tabbed.Encode(v)
    tabbed.Close()
    mut reader := json.NewDecoder(file.Open("lines.ndjson", "rb"))
    mut total := 0 as i64
    until reader.More():
        total += value.AsInt(reader.Decode())
    end
    println total
    mut two := json.NewBufDecoder("{\"a\": 1}\n{\"a\": 2}\n")
    until two.More():
        println json.EncodeDefault(two.Decode())
    end
end
Outputfalse Tehran {"city": "Tehran", "zones": [1, 2, 22]} { "city": "Tehran", "zones": [ 1, 2, 22 ] } true 5 1 "two" { "city": "Tehran", "zones": [ 1, 2, 22 ] } 30 {"a": 1} {"a": 2}

Valid

func Valid(text: str): bool
True when the text is valid JSON.
encoding-json-Build.salam
import encoding.json

func main:
    mut tags := Vector {} as Vector<str>
    tags.push(json.Str("hello"))
    tags.push(json.Str(`a "quoted" line`))
    mut members := Vector {} as Vector<str>
    members.push(json.Member("name", "Ali"))
    members.push(json.MemberInt("age", 30))
    members.push(json.MemberBool("admin", false))
    members.push(json.MemberFloat("score", 18.5))
    members.push(json.MemberRaw("tags", json.Array(tags)))
    text := json.Object(members)
    println text
    println json.Valid(text), json.Valid("{broken")
    println json.Get(text, "name"), json.GetInt(text, "age"), json.GetFloat(text, "score")
    println json.Has(text, "admin"), json.Has(text, "email"), json.Keys(text)
    println json.Escape("one\ttwo")
    pretty := json.Indent(text)
    println pretty
    println json.Minify(pretty) == json.Minify(text)
    println json.Indent4(`{"a":1}`)
    println json.IndentTabs(`{"a":1}`) == "{\n\t\"a\": 1\n}", json.IndentWith(`[1]`, "--")
end
Output{"name": "Ali", "age": 30, "admin": false, "score": 18.5, "tags": ["hello", "a \"quoted\" line"]} true false Ali 30 18.5 true false ["name", "age", "admin", "score", "tags"] one\ttwo { "name": "Ali", "age": 30, "admin": false, "score": 18.5, "tags": [ "hello", "a \"quoted\" line" ] } true { "a": 1 } true [ --1 ]

SchemaEnvelope

func SchemaEnvelope<T>(v: T): str
Builds the JSON Schema of a value's type, along with the definitions of the nested types, inside an envelope to be combined later with SchemaMerge.
encoding-json-Marshal.salam
import encoding.json
import encoding.encerr

struct User:
    @json "name"
    pub name: str = ""
    @json "age"
    pub age: int = 0
    @json "tags"
    pub tags: Vector<str> = Vector {}
    @json "-"
    pub password: str = ""
end

func main:
    mut u := User { name = "Maryam", age = 28, password = "secret" }
    u.tags.push("admin")
    println json.Marshal(u)
    println json.MarshalIndent(u, "  ")
    mut loaded := User {}
    mut e := encerr.NoError()
    println json.Unmarshal(`{"name": "Reza", "age": 41, "tags": ["a", "b"]}`, loaded, e)
    println loaded.name, loaded.age, loaded.tags.len()
    println json.Unmarshal(`{"name": "Reza", "age": "forty"}`, loaded, e), e.message
    mut lenient := User {}
    println json.UnmarshalLenient(`{"name": "Sara", "extra": 1}`, lenient, e), lenient.name
    println json.Schema(u)
    envelope := json.SchemaEnvelope(u)
    println json.SchemaMerge(envelope) == json.Schema(u)
end
Output{"name":"Maryam","age":28,"tags":["admin"]} { "name": "Maryam", "age": 28, "tags": [ "admin" ] } true Reza 41 2 false value does not match the expected type true Sara {"$schema":"https://json-schema.org/draft/2020-12/schema","$defs":{"User":{"title":"User","type":"object","properties":{"name":{"type":"string"},"age":{"type":"integer","format":"int32"},"tags":{"type":"array","items":{"type":"string"}}},"required":["name","age","tags"],"additionalProperties":false}},"$ref":"#/$defs/User"} true
