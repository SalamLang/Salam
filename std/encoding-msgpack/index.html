encoding/msgpack

The compact binary MessagePack format for dynamic values.

import encoding.msgpack
On this page
  1. Types
  2. Functions

MessagePack holds the same data as JSON in a shorter binary form that is quicker to read; it suits sending data between services or putting it in a cache. This package turns a value into a byte buffer and back.

encoding-msgpack-Marshal.salam
import encoding.msgpack
import encoding.value
import encoding.encerr
import encoding
import compress

func main:
    mut v := value.NewValueObject()
    value.ObjectSet(v, "id", value.ValueInt(7))
    value.ObjectSet(v, "ok", value.ValueBool(true))
    b := msgpack.Marshal(v)
    println compress.BufLen(b), compress.BufToBase64(b)
    mut err := encerr.NoError()
    back := msgpack.Unmarshal(b, err)
    println encerr.IsError(err), value.Equal(v, back), value.AsInt(value.ObjectGet(back, "id"))
    back2 := msgpack.UnmarshalBytes(b.data, b.len, err)
    println value.Equal(v, back2)
    o := msgpack.MsgpackOptions {}
    println compress.BufLen(msgpack.MarshalWith(v, o))
    println encoding.HexEncode(compress.BufToStrUnsafe(msgpack.Marshal(value.ValueInt(5))))
end
Output9 gqJpZAeib2vD false true 7 true 9 05
