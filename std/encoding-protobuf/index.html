encoding/protobuf

Reading .proto files, the Protocol Buffers wire encoding, and generating Salam code from them.

import encoding.protobuf
On this page
  1. Types
  2. Constants and variables
  3. Functions

Protocol Buffers is Google's binary format for structured messages. The shape of the messages is defined in a .proto file, and this package reads it, generates Salam code for the messages and services, and also carries the low level tools for writing and reading fields.

In the bytes, every field comes as a tag, the field number and the wire type, followed by its value. Numbers are usually written as a varint: each byte holds 7 bits of the number and its top bit says whether another byte follows.

encoding-protobuf-Parse.salam
import encoding.protobuf

func main:
    src := "syntax = \"proto3\";\npackage shop;\n\nenum Status { OPEN = 0; DONE = 1; }\n\nmessage Order {\n  int64 id = 1;\n  repeated string items = 2;\n  map<string, int32> counts = 3;\n  Status status = 4;\n  oneof pay { string card = 5; bool cash = 6; }\n}\n\nservice Orders {\n  rpc Get (Order) returns (Order);\n  rpc Watch (Order) returns (stream Order);\n}\n"
    mut ok := false
    p := protobuf.ParseFile(src, ok)
    println ok, p.packageName, p.messages.len(), p.enums.len(), p.services.len(), p.imports.len()
    msg := p.messages[0]
    println msg.name, msg.fields.len(), msg.oneofs[0].name, msg.oneofs[0].fieldIndices
    each f in msg.fields:
        println f.number, f.name, protobuf.FieldKindName(f.kind), f.typeName, f.repeated, f.oneofIndex
    end
    m := msg.fields[2]
    println m.kind == protobuf.FK_MAP, protobuf.FieldKindName(m.keyKind), protobuf.FieldKindName(m.mapValueKind)
    e := p.enums[0]
    println e.name, e.values[1].name, e.values[1].number
    s := p.services[0]
    println s.name, s.methods[1].name, s.methods[1].inputType, s.methods[1].serverStream, s.methods[1].clientStream
    println protobuf.ScalarKindFromName("sint64") == protobuf.FK_SINT64, protobuf.ScalarKindFromName("Order")
    mut toks := protobuf.Tokenize("message A { int32 x = 1; }", ok)
    println ok, toks.toks.len(), toks.toks[0].lexeme, toks.toks[0].kind == protobuf.PTK_IDENT, toks.toks[2].kind == protobuf.PTK_LBRACE
    println toks.toks[6].ival, toks.toks[6].line
    _ := protobuf.ParseFile("message {", ok)
    println ok
end

encoding-protobuf-Wire.salam
import encoding.protobuf
import compress

func main:
    mut b := compress.NewByteBuf()
    protobuf.WriteTag(b, 1, protobuf.WT_VARINT)
    protobuf.WriteVarint(b, 150 as u64)
    println compress.BufToBase64(b), protobuf.MakeTag(1, 0), protobuf.MakeTag(2, protobuf.WT_LEN)
    protobuf.WriteTag(b, 2, protobuf.WT_LEN)
    protobuf.WriteLenPrefixedStr(b, "hello")
    protobuf.WriteTag(b, 3, protobuf.WT_FIXED32)
    protobuf.WriteFixed32(b, 7 as u32)
    protobuf.WriteTag(b, 4, protobuf.WT_FIXED64)
    protobuf.WriteFixed64(b, 9 as u64)
    protobuf.WriteFloat(b, 0.5)
    protobuf.WriteDouble(b, 2.25)
    protobuf.WriteLenPrefixedBytes(b, b.data, 2)
    mut r := compress.ReaderFromBuf(b)
    mut good := true
    mut number := 0
    mut wire := 0
    protobuf.ReadTag(r, number, wire, good)
    println number, wire, protobuf.ReadVarint(r, good)
    protobuf.ReadTag(r, number, wire, good)
    println number, protobuf.ReadLenPrefixedStr(r, good)
    protobuf.ReadTag(r, number, wire, good)
    protobuf.SkipField(r, wire, good)
    protobuf.ReadTag(r, number, wire, good)
    println number, protobuf.ReadFixed64(r, good)
    println protobuf.ReadFloat(r, good), protobuf.ReadDouble(r, good)
    b2 := protobuf.ReadLenPrefixedBuf(r, good)
    println compress.BufLen(b2), good
    _ := protobuf.ReadVarint(r, good)
    println good
    println protobuf.ZigZagEncode32(-1), protobuf.ZigZagDecode32(3 as u32), protobuf.ZigZagEncode64(-2), protobuf.ZigZagDecode64(4 as u64)
    mut b3 := compress.NewByteBuf()
    protobuf.WriteFixed32(b3, 42 as u32)
    mut r3 := compress.ReaderFromBuf(b3)
    good = true
    println protobuf.ReadFixed32(r3, good)
    compress.BufFree(b)
    compress.BufFree(b3)
end
OutputCJYB 8 18 1 0 150 2 hello 4 9 0.5 2.25 2 true false 1 -2 3 2 42

GenerateSingle

func GenerateSingle(protoSource: str, ok &: bool): str
Generates Salam code from .proto text: one struct per message, with its serialising functions.
encoding-protobuf-Generate.salam
import encoding.protobuf
import str

func main:
    src := "syntax = \"proto3\";\nmessage Ping { string text = 1; }\nservice Echo {\n  rpc Say (Ping) returns (Ping);\n  rpc Stream (stream Ping) returns (stream Ping);\n}\n"
    mut ok := false
    code := protobuf.GenerateSingle(src, ok)
    println ok, str.Contains(code, "struct Ping")
    p := protobuf.ParseFile(src, ok)
    println protobuf.ServiceCount(p), protobuf.ServiceNameAt(p, 0), protobuf.ServiceMethodCount(p, 0)
    println protobuf.ServiceMethodSignature(p, 0, 0)
    println protobuf.ServiceMethodSignature(p, 0, 1)
    println str.Contains(protobuf.GenerateServices(p), "Echo")
    all := protobuf.GenerateWithServices(src, ok)
    println ok, str.Len(all) > str.Len(code)
    println protobuf.ServiceSummary(src, ok)
end
Outputtrue true 1 Echo 2 Say(Ping) returns (Ping) Stream(stream Ping) returns (stream Ping) true true true service Echo (2 methods) rpc Say(Ping) returns (Ping) rpc Stream(stream Ping) returns (stream Ping)
