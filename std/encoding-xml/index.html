encoding/xml

Reading, building and writing XML, with XPath, DTD validation, namespaces and a light simple interface.

import encoding.xml
On this page
  1. Types
  2. Constants and variables
  3. Functions

This package has two interfaces. The document interface is the complete one: Parse reads the text into a document, and every node has a numeric id you use to get its name, attributes, children and text. XPath, DTD validation, namespaces, entities and the canonical form all work on that document.

The simple node interface (Decode, Child, Attr) is for small jobs, such as reading a small configuration file, and keeps the tree directly in plain structs. The XMLTokenizer reads the text without building a tree at all.

encoding-xml-Build.salam
import encoding.xml

func main:
    mut d := xml.NewDocument()
    root := xml.CreateElement(d, "note")
    xml.Append(d, xml.DocumentNode(d), root)
    xml.SetAttribute(d, root, "id", "7")
    xml.SetAttribute(d, root, "draft", "yes")
    recipient := xml.CreateElement(d, "to")
    xml.Append(d, root, recipient)
    xml.Append(d, recipient, xml.CreateText(d, "Maryam & Reza"))
    body := xml.CreateElement(d, "body")
    xml.Append(d, root, body)
    xml.Append(d, body, xml.CreateCData(d, "<b>bold</b>"))
    xml.InsertBefore(d, root, xml.CreateComment(d, "recipient"), recipient)
    xml.Append(d, xml.DocumentNode(d), xml.CreateProcessingInstruction(d, "app", "v=1"))
    println xml.Write(d)
    println xml.RemoveAttribute(d, root, "draft"), xml.RemoveChild(d, root, body)
    xml.SetText(d, recipient, "everyone")
    println xml.WriteIndent(d, "  ")
    println xml.WriteNode(d, recipient), xml.WriteNodeIndent(d, recipient, "\t")
    mut o := xml.DefaultWriteOptions()
    o.declaration = true
    o.empty_as_pair = true
    println xml.WriteWith(d, o)
    println xml.WriteDocument(d)
    mut err := xml.NoError()
    mut d2 := xml.Parse("<r b=\"2\"   a=\"1\"><x/></r>", err)
    println xml.Canonical(d2), xml.CanonicalNode(d2, xml.ChildAt(d2, xml.Root(d2), 0))
    xml.Free(d2)
    xml.Free(d)
end

encoding-xml-DTD.salam
import encoding.xml

func main:
    dtd := xml.ParseDTD("<!ELEMENT note (to, body)>\n<!ELEMENT to (#PCDATA)>\n<!ELEMENT body (#PCDATA)>\n<!ATTLIST note id ID #REQUIRED kind (a|b) \"a\">")
    println dtd.ok, xml.DeclaredElementCount(dtd), xml.DeclaredAttributeCount(dtd)
    el := dtd.elements[0]
    println el.name, el.content == xml.ContentChildren, el.model
    at := dtd.attributes[1]
    println at.name, at.kind == xml.AttrEnumeration, at.mode == xml.DefaultValue, at.value, at.values
    mut err := xml.NoError()
    mut good := xml.Parse("<note id=\"n1\"><to>me</to><body>hello</body></note>", err)
    mut report := xml.Validate(good, dtd)
    println report.ok
    xml.FreeReport(report)
    println xml.ApplyDefaults(good, dtd), xml.AttributeValue(good, xml.Root(good), "kind")
    mut bad := xml.Parse("<note><body>no recipient</body></note>", err)
    mut r2 := xml.Validate(bad, dtd)
    println r2.ok, r2.items.len()
    println xml.ReportText(r2)
    xml.FreeReport(r2)
    mut inner := xml.Parse("<!DOCTYPE a [<!ELEMENT a EMPTY>]><a>text</a>", err)
    mut r3 := xml.ValidateInternalSubset(inner)
    println r3.ok, inner.doctype
    xml.FreeReport(r3)
    mut ns := xml.Parse("<a xmlns:x=\"urn:x\"><x:b/></a>", err)
    mut r4 := xml.CheckNamespaces(ns)
    println r4.ok
    xml.FreeReport(r4)
    mut dtd2 := dtd
    xml.FreeDTD(dtd2)
    xml.Free(good)
    xml.Free(bad)
    xml.Free(inner)
    xml.Free(ns)
end

encoding-xml-Parse.salam
import encoding.xml

func main:
    src := "<?xml version=\"1.0\"?>\n<library xmlns:b=\"urn:books\">\n  <!-- the list -->\n  <b:book id=\"1\" lang=\"en\"><title>Dune</title><year>1965</year></b:book>\n  <b:book id=\"2\"><title>Solaris</title></b:book>\n</library>"
    mut err := xml.NoError()
    mut d := xml.Parse(src, err)
    println err.ok, d.version, xml.NodeCount(d) > 0
    r := xml.Root(d)
    println xml.Name(d, r), xml.Kind(d, r) == xml.KindElement, xml.Parent(d, r) == xml.DocumentNode(d)
    mut books := xml.Elements(d, r)
    println books.items.len()
    b := books.items[0]
    println xml.Name(d, b), xml.LocalName(d, b), xml.Prefix(d, b), xml.NamespaceURI(d, b)
    println xml.Line(d, b), xml.Column(d, b), xml.Depth(d, b), xml.Path(d, b)
    println xml.AttributeCount(d, b), xml.HasAttribute(d, b, "lang"), xml.AttributeValue(d, b, "id")
    a := xml.AttributeAt(d, b, 1)
    println a.name, a.value
    t := xml.FirstElementNamed(d, b, "title")
    println xml.Text(d, t), xml.Text(d, b), xml.IsElement(d, t)
    println xml.ChildCount(d, b), xml.Name(d, xml.ChildAt(d, b, 1))
    println xml.Name(d, xml.NextSibling(d, t)), xml.PreviousSibling(d, t)
    mut all := xml.Descendants(d, r)
    mut named := xml.ElementsNamed(d, r, "b:book")
    mut up := xml.Ancestors(d, t)
    mut kids := xml.ChildNodes(d, r)
    println all.items.len(), named.items.len(), up.items.len(), kids.items.len()
    println xml.Exists(d, 9999), xml.Value(d, xml.ChildAt(d, t, 0))
    xml.FreeList(all)
    xml.FreeList(named)
    xml.FreeList(up)
    xml.FreeList(kids)
    xml.FreeList(books)
    _ := xml.Parse("<a><b></a>", err)
    println err.ok, err.line, xml.ErrorText(err)
    println xml.WellFormed("<x/>", err)
    xml.Free(d)
    mut o := xml.DefaultOptions()
    o.keep_comments = false
    mut d2 := xml.ParseWith("<!DOCTYPE p [<!ENTITY site \"hello\">]><p xmlns:x=\"urn:x\" x:id=\"7\">&site; <b>b</b> end</p>", o, err)
    p := xml.Root(d2)
    println err.ok, xml.EntityCount(d2), xml.EntityValue(d2, "site")
    println "[" + xml.DirectText(d2, p) + "]", xml.AttributeValueNS(d2, p, "urn:x", "id")
    println xml.EntityDepthLimit, xml.EntityExpansionLimit, o.max_depth
    xml.Free(d2)
end

encoding-xml-Simple.salam
import encoding.xml
import encoding.value

func main:
    n := xml.Decode("<config env=\"prod\"><db host=\"localhost\" port=\"5432\"/><feature>search</feature><feature>chat</feature></config>")
    println n.tag, xml.HasAttr(n, "env"), xml.Attr(n, "env")
    db := xml.Child(n, "db")
    println xml.Found(db), xml.Attr(db, "port"), xml.Found(xml.Child(n, "cache"))
    feats := xml.Children(n, "feature")
    println feats.items.len(), feats.items[1].text
    mut item := xml.NewXMLNode("item")
    xml.SetAttr(item, "id", "3")
    mut name := xml.NewXMLNode("name")
    name.text = "book"
    xml.AppendChild(item, name)
    println xml.Encode(item)
    println xml.EncodeIndent(item, "  ")
    println xml.EncodeDeclaration(item)
    println xml.Valid("<a><b/></a>"), xml.Valid("<a><b></a>")
    mut ok := false
    _ := xml.DecodeStrict("<a>", ok)
    println ok
    v := xml.ToValue(n)
    println value.ToDebugString(v)
    println xml.Encode(xml.FromValue(v, "root"))
    println xml.EscapeText("a < b & c"), xml.EscapeAttr(`"x"`), xml.UnescapeEntities("&lt;p&gt; &amp; &#115;")
    mut tz := xml.NewXMLTokenizer("<a x=\"1\">hello<!--c--></a>")
    until true:
        t := xml.NextToken(tz)
        if t.kind == xml.XMLTokEOF or t.kind == xml.XMLTokError:
            break
        end
        println t.kind, t.tag, t.text
    end
    println xml.XMLTokStartElement, xml.XMLTokEndElement, xml.XMLTokText, xml.XMLTokComment, xml.XMLTokCData
end

encoding-xml-XPath.salam
import encoding.xml

func main:
    mut err := xml.NoError()
    mut d := xml.Parse("<shop><item price=\"120\">tea</item><item price=\"80\">sugar</item><item price=\"45\">bread</item></shop>", err)
    r := xml.DocumentNode(d)
    mut cheap := xml.Select(d, r, "//item[@price < 100]", err)
    println cheap.items.len()
    each n in cheap.items:
        println xml.Text(d, n)
    end
    xml.FreeList(cheap)
    println xml.SelectString(d, r, "/shop/item[1]", err), xml.SelectNumber(d, r, "sum(//item/@price)", err), xml.SelectBoolean(d, r, "count(//item) = 3", err)
    last := xml.SelectFirst(d, r, "//item[last()]", err)
    println xml.NodeName(d, last), xml.StringValue(d, last)
    mut v := xml.Evaluate(d, r, "//item/@price", err)
    println v.kind == xml.ValueNodeSet, v.nodes.items.len()
    first := v.nodes.items[0]
    println xml.IsAttributeRef(first), xml.ReferencedAttribute(d, first).value, xml.IsAttributeRef(xml.AttributeRef(0))
    xml.FreeValue(v)
    num := xml.Evaluate(d, r, "string-length('hello')", err)
    println num.kind == xml.ValueNumber, num.number, xml.ValueString, xml.ValueBoolean
    _ := xml.Select(d, r, "//item[@", err)
    println err.ok
    xml.Free(d)
end
