encoding/yaml

Reading YAML by path, building and writing it, and converting it to JSON.

import encoding.yaml
On this page
  1. Functions

YAML is the usual format for configuration files. This package converts YAML to JSON internally, which is what makes its reading functions simple and what lets it work alongside the json package.

The reading functions take a path: keys separated by dots, with a list index in square brackets, as in app.hosts[1]. A path that is not there gives an empty string or zero; to tell that apart from a real value, use Has.

Functions

Decode

func Decode(text: str, err &: encerr.EncodingError): value.Value
Turns YAML into a dynamic Value.
encoding-yaml-Build.salam
import encoding.yaml
import encoding.value
import encoding.encerr
import str

func main:
    mut members := Vector {} as Vector<str>
    members.push(yaml.Member("name", "salam"))
    members.push(yaml.MemberInt("port", 8080))
    members.push(yaml.MemberBool("debug", false))
    members.push(yaml.MemberFloat("ratio", 0.25))
    mut hosts := Vector {} as Vector<str>
    hosts.push(yaml.Str("a: b"))
    members.push(yaml.MemberRaw("hosts", yaml.Array(hosts)))
    object := yaml.Object(members)
    println object
    println yaml.Dump(object)
    println str.Contains(yaml.DumpTabs(`{"a": {"b": 1}}`), "\tb: 1"), yaml.DumpWith(`{"a": [1]}`, "    ")
    println yaml.Escape("one: two")
    println yaml.Parse("x: 1")
    mut e := encerr.NoError()
    v := yaml.Decode("city: Tehran\npopulation: 9000000\n", e)
    println value.AsInt(value.ObjectGet(v, "population"))
    println yaml.Encode(v)
end
Output{"name": "salam", "port": 8080, "debug": false, "ratio": 0.25, "hosts": ["a: b"]} name: salam port: 8080 debug: false ratio: 0.25 hosts: - "a: b" true a: - 1 one: two {"x": 1} 9000000 city: Tehran population: 9000000

Get

func Get(text: str, path: str): str
The value at a path as text; strings come back without their quotes.
encoding-yaml-Query.salam
import encoding.yaml

func main:
    text := "app:\n  name: salam\n  port: 8080\n  debug: true\n  ratio: 0.5\n  hosts:\n    - a.example\n    - b.example\n"
    println yaml.Valid(text), yaml.Has(text, "app.name"), yaml.Has(text, "app.user")
    println yaml.Get(text, "app.name"), yaml.GetInt(text, "app.port"), yaml.GetBool(text, "app.debug"), yaml.GetFloat(text, "app.ratio")
    println yaml.TypeOf(text, "app.hosts"), yaml.Len(text, "app.hosts"), yaml.Get(text, "app.hosts[1]")
    println yaml.Keys(text, "app")
    mut required := Vector {} as Vector<str>
    required.push("app")
    println yaml.HasKeys(text, required)
    required.push("db")
    println yaml.HasKeys(text, required)
    println yaml.ToJSON(text)
    println yaml.SplitDocuments("a: 1\n---\nb: 2\n").len()
end
Outputtrue true false salam 8080 true 0.5 list 2 b.example ["name", "port", "debug", "ratio", "hosts"] true false {"app": {"name": "salam", "port": 8080, "debug": true, "ratio": 0.5, "hosts": ["a.example", "b.example"]}} 2
