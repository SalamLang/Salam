error

Structured errors with a kind, a message, a chain of causes, metadata and a retry policy.

import error
On this page
  1. Types
  2. Constants and variables
  3. Functions

A function that may fail can return an error.Error: either "nil", meaning everything went well, or an error with a kind, such as "not found", and a message. The kind is for the code to decide on; the message is for a person to read.

When an error comes up from a lower layer, Wrap it so the current layer's explanation is added. The final message shows the whole chain, as in "could not save the report: the disk is full", and RootMessage gives the original cause.

For transient errors, such as a dropped network connection, RetryPolicy says when and how often to try again and how long to wait between attempts; the delay doubles each time.

An error.
NameTypeDescription

error-New.salam
import error

func find_user(id: int): error.Error:
    if id <= 0:
        ret error.New(error.KindInvalidArgument, "the id has to be positive")
    end
    if id > 100:
        ret error.NewAt(error.KindNotFound, "no such user", "find_user")
    end
    ret error.Nil()
end

func main:
    each id in [7, -1, 500]:
        mut e := find_user(id)
        if error.Ok(e):
            println id, "found", error.IsNil(e)
        else:
            println id, error.Kind(e), error.Message(e)
            println error.Is(e, error.KindNotFound), error.IsNotFound(e), error.IsInvalidArgument(e)
        end
        error.Free(e)
    end
    mut args := Vector {} as Vector<str>
    defer args.free()
    args.push("config.ini")
    e2 := error.Newf(error.KindIO, "cannot read {}", args)
    println error.Message(e2)
end
Output7 found true -1 invalid_argument the id has to be positive false false true 500 not_found no such user true true false cannot read config.ini

RetryPolicy

struct RetryPolicy
A retry policy.
NameTypeDescription
max_attemptsint = 3The greatest number of attempts; three by default.
base_delay_msint = 100The delay after the first failed attempt, in milliseconds; 100 by default.
max_delay_msint = 5000The ceiling on the delay, in milliseconds; 5000 by default.
error-RetryPolicy.salam
import error

func main:
    mut policy := error.NewRetryPolicy(4, 100, 1000)
    println policy.max_attempts, policy.base_delay_ms, policy.max_delay_ms
    e := error.New(error.KindTimeout, "the server did not answer")
    mut attempt := 1
    until error.ShouldRetry(policy, e, attempt):
        println "attempt", attempt, "after", error.DelayForAttempt(policy, attempt), "milliseconds"
        attempt += 1
    end
    println error.ShouldRetry(policy, error.New(error.KindInvalidArgument, "bad input"), 1)
end
Output4 100 1000 attempt 1 after 100 milliseconds attempt 2 after 200 milliseconds attempt 3 after 400 milliseconds false

error-Wrap.salam
import error

func main:
    root := error.New(error.KindIO, "the disk is full")
    a := error.Wrap(root, "could not save the report")
    b := error.WrapAt(a, "the nightly run stopped", "nightly job")
    c := error.WrapKind(b, error.KindInternal, "internal error")
    d := error.WithMeta(c, "disk", "/dev/sda1")
    println error.Message(d)
    println error.RootMessage(d)
    println error.Kind(d), error.IsInternal(d), error.Meta(d, "disk")
    println error.Trace(d)
    println error.Format(d)
end
