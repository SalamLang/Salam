flag

Reading command line options such as --port 8080 and -v with the right type, and generating the help text.

import flag
On this page
  1. Types
  2. Constants and variables
  3. Functions

The flag package reads command line options. Each flag has a long name (--port), a short name (-p), a default value and a line of help. The value can be given either way: --port=8080 or --port 8080. Any argument that is not a flag is "positional", and after -- everything counts as positional.

--help and -h are always recognised and the Usage text is generated for you. Pass Parse the arguments without the program's own name.

Types

FlagSet

struct FlagSet
A program's set of flags and the result of parsing.
NameTypeDescription
progstr = ""The program's name, for the help text.
descstr = ""The program's description, for the help text.
namesVectorThe long names of the flags.
shortsVectorThe short names of the flags.
kindsVectorThe kind of each flag.
valsVectorThe current value of each flag, as text.
defsVectorThe default value of each flag.
helpsVectorThe help line of each flag.
was_setVectorWhether each flag appeared on the command line.
posVectorThe positional arguments.
errstr = ""The message of the last parse error.

flag-Parse.salam
import flag

func main:
    mut f := flag.New("copy", "copies files")
    defer flag.Free(f)
    flag.AddBool(f, "verbose", "v", false, "report everything")
    flag.AddStr(f, "out", "o", "output", "the destination directory")
    flag.AddInt(f, "jobs", "j", 1, "how many jobs at once")
    flag.AddFloat(f, "ratio", "r", 1.0, "the compression ratio")
    mut args := Vector {} as Vector<str>
    defer args.free()
    each a in ["-v", "--out=backup", "-j", "4", "a.txt", "b.txt"]:
        args.push(a)
    end
    println flag.Parse(f, args), "[" + flag.Error(f) + "]"
    println flag.GetBool(f, "verbose"), flag.GetStr(f, "out"), flag.GetInt(f, "jobs"), flag.GetFloat(f, "ratio")
    println flag.WasSet(f, "jobs"), flag.WasSet(f, "ratio")
    println flag.Positional(f), flag.HelpWanted(f)
    println flag.FlagKindBool, flag.FlagKindStr, flag.FlagKindInt, flag.FlagKindFloat
end
Outputtrue [] true backup 4 1 true false ["a.txt", "b.txt"] false 0 1 2 3

flag-Usage.salam
import flag

func main:
    mut f := flag.New("tool", "an example tool")
    defer flag.Free(f)
    flag.AddBool(f, "quiet", "q", false, "say nothing")
    flag.AddInt(f, "port", "p", 8080, "the port")
    mut args := Vector {} as Vector<str>
    defer args.free()
    args.push("--help")
    flag.Parse(f, args)
    if flag.HelpWanted(f):
        print flag.Usage(f)
    end
    mut bad := Vector {} as Vector<str>
    defer bad.free()
    bad.push("--port=abc")
    mut f2 := flag.New("tool", "")
    defer flag.Free(f2)
    flag.AddInt(f2, "port", "p", 8080, "the port")
    println flag.Parse(f2, bad), flag.Error(f2)
end
OutputUsage: tool [options] [args] an example tool Options: -q, --quiet say nothing -p, --port <int> the port (default: 8080) -h, --help show this help false invalid int value for --port: abc
