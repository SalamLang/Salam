ftp

An FTP and FTPS client: listing a directory, uploading and downloading files.

import ftp
On this page
  1. Types
  2. Functions

This package connects to FTP servers, for uploading a site to shared hosting or pulling files off old equipment. Plain FTP sends the password and the data unencrypted; wherever the server supports it, use DialTLS, which encrypts both the commands and the data with AUTH TLS.

Transfers go in passive mode (PASV or EPSV), which works from behind a firewall. The parsing functions, such as ParseListLine, are available for writing tools of your own too.

The information about one file from the server listing.
NameTypeDescription
namestr = ""The name.
sizei64 = 0The size.
is_dirbool = falseIt is a directory.
is_linkbool = falseIt is a link.
modestr = ""The permissions, such as rw-r--r--.
raw_datestr = ""The date exactly as the server wrote it.
ftp-Parse.salam
import ftp

func main:
    println ftp.ReplyCode("230 Login successful."), ftp.ReplyCode("broken")
    println ftp.IsFinalLine("211-Features:"), ftp.IsFinalLine("211 End")
    println ftp.ParsePasv("227 Entering Passive Mode (192,168,1,10,19,137)")
    println ftp.ParseEpsv("229 Entering Extended Passive Mode (|||50000|)")
    file := ftp.ParseListLine("-rw-r--r--    1 ftp      ftp          1024 Sep 30 12:00 notes.txt")
    println file.name, file.size, file.is_dir, file.is_link, file.mode, file.raw_date
    folder := ftp.ParseListLine("drwxr-xr-x    2 ftp      ftp          4096 Jan 01  2026 backup")
    println folder.name, folder.is_dir
    link := ftp.ParseListLine("lrwxrwxrwx    1 ftp      ftp             7 Mar 03 09:15 current -> backup")
    println link.name, link.is_link
end
Output230 0 false true 192.168.1.10:5001 50000 notes.txt 1024 false false rw-r--r-- Sep 30 12:00 backup true current true

ftp-Client.salam
import ftp
import tls

func main:
    mut c := ftp.DialTLS("ftp.example.com", 21)
    if not ftp.Ok(c) or not ftp.Login(c, "ali", "s3cret"):
        println ftp.LastCode(c), ftp.LastReply(c)
        ret
    end
    println ftp.Secure(c), ftp.SecureData(c)
    println ftp.System(c)
    println ftp.Features(c)
    _ := ftp.Binary(c)
    println ftp.Pwd(c)
    _ := ftp.MkdirAll(c, "/site/assets")
    _ := ftp.Cwd(c, "/site")
    _ := ftp.Write(c, "index.html", "<h1>hello</h1>")
    _ := ftp.Append(c, "log.txt", "one more line\n")
    println ftp.Read(c, "index.html")
    println ftp.Upload(c, "logo.png", "assets/logo.png")
    println ftp.Download(c, "log.txt", "log-copy.txt")
    each entry in ftp.List(c, "."):
        println entry.name, entry.size, entry.is_dir
    end
    each name in ftp.NameList(c, "assets"):
        println name
    end
    println ftp.ListRaw(c, ".").len() > 0
    println ftp.Size(c, "index.html"), ftp.ModTime(c, "index.html")
    println ftp.IsDir(c, "assets"), ftp.IsLink(c, "current"), ftp.Exists(c, "old.html")
    _ := ftp.Rename(c, "log.txt", "log-1.txt")
    _ := ftp.Move(c, "log-1.txt", "archive/log-1.txt")
    _ := ftp.Delete(c, "archive/log-1.txt")
    _ := ftp.CdUp(c)
    _ := ftp.Mkdir(c, "tmp")
    _ := ftp.Rmdir(c, "tmp")
    _ := ftp.RemoveAll(c, "/site/old")
    _ := ftp.UploadDir(c, "public", "/site")
    _ := ftp.DownloadDir(c, "/site", "backup")
    println ftp.Command(c, "SITE CHMOD 644 index.html")
    println ftp.Noop(c)
    _ := ftp.Ascii(c)
    ftp.Quit(c)
    ftp.Close(c)

    mut plain := ftp.Dial("ftp.example.com", 21)
    println ftp.AuthTLS(plain)
    mut cfg := tls.NewConfig("ftp.example.com")
    cfg.ca_file = "company-ca.pem"
    mut configured := ftp.DialTLSWith("ftp.example.com", 21, cfg)
    mut upgraded := ftp.Dial("ftp.example.com", 21)
    println ftp.AuthTLSWith(upgraded, cfg)
    ftp.Close(plain)
    ftp.Close(configured)
    ftp.Close(upgraded)
end
