graphql

A GraphQL server: schemas, resolvers, query execution and HTTP handling.

import graphql
On this page
  1. Types
  2. Constants and variables
  3. Functions

GraphQL is a language for asking an API for data: the client asks for exactly the fields it needs and the server sends back just those. This package is a complete server side implementation of the specification. Write the schema as SDL text and pass it to ParseSchema, or build it with the Add... functions, register resolvers for the fields and run a query with ExecuteSource.

A resolver is a function that takes a slot number; it reads the arguments and the parent object with ArgStr, ParentField and their siblings, and answers with one of the Resolve... functions or with Fail. A field without a resolver takes the value of the same name from the parent object. For a web server, call HandleHttpSimple inside a net/http handler.

graphql-Http.salam
import graphql
import encoding.value
import net.http
import net.router

mut schema := graphql.NewSchema()

func me(slot: i64):
    ctx := graphql.Context(slot)
    mut name := "guest"
    if value.ObjectHas(ctx, "user"):
        name = value.AsString(value.ObjectGet(ctx, "user"))
    end
    graphql.ResolveStr(slot, name + " (" + graphql.ParentTypeName(slot) + "." + graphql.FieldName(slot) + ")")
end

func add(slot: i64):
    graphql.ResolveInt(slot, graphql.ArgInt(slot, "a") + graphql.ArgInt(slot, "b"))
end

func endpoint(ctx: i64):
    res := graphql.HandleHttpSimple(schema, http.Ctx_method(ctx), http.Ctx_header(ctx, "Content-Type"), http.Ctx_body(ctx), http.Ctx_raw_query(ctx))
    http.Ctx_status(ctx, res.status)
    http.Ctx_set_header(ctx, "Content-Type", res.contentType)
    http.Ctx_text(ctx, res.body)
end

func main:
    mut err := graphql.NewError(graphql.ERR_SYNTAX, "")
    graphql.ParseSchemaInto(schema, "type Query { me: String! add(a: Int!, b: Int!): Int! }", err)
    _ := graphql.SetResolver(schema, "Query", "me", me)
    _ := graphql.SetResolver(schema, "Query", "add", add)

    mut r := router.NewRouter()
    router.Post(r, "/graphql", endpoint)
    router.Get(r, "/graphql", endpoint)
    mut srv := http.NewServer(19640)
    http.Use(srv, r)
    _ := http.RunBackground(srv)

    mut vars := value.NewValueObject()
    value.ObjectSet(vars, "x", value.ValueInt(40))
    body := graphql.BuildRequestBody("query Sum($x: Int!) { add(a: $x, b: 2) }", "Sum", vars)
    println body
    mut headers := http.NewHeaders()
    headers.insert("Content-Type", "application/json")
    resp := http.PostWith("http://127.0.0.1:19640/graphql", body, headers)
    println http.Status(resp), http.Body(resp)
    mut cr := graphql.ParseResponse(http.Body(resp))
    println graphql.ResponseOk(cr), cr.hasData, value.AsInt(graphql.ResponseField(cr, "add"))
    graphql.FreeClientResponse(cr)

    qs := graphql.BuildRequestQueryString("{ me }", "", value.ValueNull())
    resp2 := http.Get("http://127.0.0.1:19640/graphql?" + qs)
    println http.Body(resp2)
    resp3 := http.PostWith("http://127.0.0.1:19640/graphql", graphql.BuildQueryBody("{ nope }"), headers)
    mut cr3 := graphql.ParseResponse(http.Body(resp3))
    println http.Status(resp3), graphql.ResponseOk(cr3), graphql.ResponseErrorMessage(cr3)
    graphql.FreeClientResponse(cr3)

    mut req := graphql.ParseHttpRequestJSON(`{"query":"{ me }"}`)
    mut context := value.NewValueObject()
    value.ObjectSet(context, "user", value.ValueString("Maryam"))
    mut res := graphql.HandleRequest(schema, req, value.ValueNull(), context)
    println graphql.ResultToJSON(res)
    graphql.FreeResult(res)
    graphql.FreeHttpRequest(req)
    bad := graphql.ParseHttpRequest("PUT", "application/json", "{}", "")
    println bad.ok, bad.error
    resp4 := graphql.HandleHttp(schema, "POST", "application/json", "not json", "", value.ValueNull(), value.ValueNull())
    println resp4.status
end

graphql-Execute.salam
import graphql
import encoding.value

func book(id: str, title: str, year: i64): value.Value:
    mut b := value.NewValueObject()
    value.ObjectSet(b, "id", value.ValueString(id))
    value.ObjectSet(b, "title", value.ValueString(title))
    value.ObjectSet(b, "year", value.ValueInt(year))
    ret b
end

func hello(slot: i64):
    graphql.ResolveStr(slot, "Hello " + graphql.ArgStr(slot, "name"))
end

func books(slot: i64):
    mut list := value.NewValueArray()
    value.ArrayPush(list, book("1", "The Blind Owl", 1936))
    value.ArrayPush(list, book("2", "Savushun", 1969))
    graphql.Resolve(slot, list)
end

func one_book(slot: i64):
    if graphql.ArgInt(slot, "id") == 2:
        graphql.Resolve(slot, book("2", "Savushun", 1969))
    else:
        graphql.ResolveNull(slot)
    end
end

func age(slot: i64):
    year := value.AsInt(graphql.ParentField(slot, "year"))
    graphql.ResolveInt(slot, 2026 - year)
end

func score(slot: i64):
    graphql.ResolveFloat(slot, graphql.ArgFloat(slot, "base") * 2.0)
end

func in_stock(slot: i64):
    graphql.ResolveBool(slot, graphql.ArgBool(slot, "only") or graphql.HasArg(slot, "x"))
end

func broken(slot: i64):
    graphql.Fail(slot, "service unavailable")
end

func main:
    mut err := graphql.NewError(graphql.ERR_SYNTAX, "")
    mut s := graphql.ParseSchema(`
type Query {
  hello(name: String = "world"): String!
  books: [Book!]!
  book(id: Int!): Book
  score(base: Float!): Float!
  inStock(only: Boolean!): Boolean!
  broken: String
}
type Book {
  id: ID!
  title: String!
  year: Int!
  age: Int!
}`, err)
    _ := graphql.SetResolver(s, "Query", "hello", hello)
    _ := graphql.SetResolver(s, "Query", "books", books)
    _ := graphql.SetResolver(s, "Query", "book", one_book)
    _ := graphql.SetResolver(s, "Query", "score", score)
    _ := graphql.SetResolver(s, "Query", "inStock", in_stock)
    _ := graphql.SetResolver(s, "Query", "broken", broken)
    _ := graphql.SetResolver(s, "Book", "age", age)
    println graphql.HasResolver(s, "Book", "age"), graphql.HasResolver(s, "Book", "title")

    mut r := graphql.Run(s, `{ hello books { title year age } }`)
    println graphql.ResultOk(r)
    println graphql.ResultToJSONStable(r)
    graphql.FreeResult(r)

    mut vars := value.NewValueObject()
    value.ObjectSet(vars, "id", value.ValueInt(2))
    mut r2 := graphql.ExecuteSource(s, `query One($id: Int!) { book(id: $id) { title } missing: book(id: 9) { title } score(base: 1.5) inStock(only: true) hello(name: "Maryam") }`, "One", vars, value.ValueNull(), value.ValueNull())
    println graphql.ResultToJSONStable(r2)
    graphql.FreeResult(r2)

    mut r3 := graphql.Run(s, `{ broken hello }`)
    println graphql.ResultOk(r3), graphql.ResultErrorMessage(r3)
    println graphql.ResultToJSONStable(r3)
    graphql.FreeResult(r3)

    mut r4 := graphql.Run(s, `{ nope }`)
    println graphql.ResultErrorMessage(r4)
    graphql.FreeResult(r4)
    graphql.ResetSlots()
    graphql.FreeSchema(s)
end
