gui.reactive

Reactive properties for user interfaces: lazy bindings with automatic dependencies, trackers, change handlers, two-way links, CSS transitions and list models.

import gui.reactive
On this page
  1. A counter component
  2. A CSS transition on hover
  3. Types
  4. Functions

gui.reactive is the state runtime of Salam's native user interface, modelled on the property system of Slint. Every value that can change is a property. A property either holds a value or has a binding, a function that computes it from other properties. Reading a property inside a binding records a dependency by itself, so there is nothing to declare.

The work is split in two halves. A Set only compares, stores and marks the dependents dirty, stopping at nodes that are already dirty. Nothing is computed then: a dirty binding runs the next time somebody reads it, and it forgets its old dependencies and records new ones, so a binding with an if depends only on the branch it took.

Things to know:

  • Handles are small values. IntProp, StrProp and the others hold an index into the runtime, so a lambda that captures one by value still sees the live property. A constant made with ConstInt and friends holds its value in the handle and costs nothing in the runtime.
  • Setting a bound property removes the binding, as an assignment does in Slint. Use Link for a value that two places edit.
  • Text has ownership rules. StrProp.Get returns a string owned by the property, valid until the value changes. Bind takes ownership of the new string its function returns; BindCopy copies it, for literals and other properties' text.
  • Change handlers never run inside a set. OnChange queues them, the event loop calls RunPendingHandlers, and many changes give one call.
  • Binding loops are reported with the property names, for example binding loop: 'min width' -> 'preferred width' -> 'min width', instead of overflowing the stack. Name properties with Name to get readable reports.
  • The runtime is for one thread, the user interface thread. A capturing lambda keeps its environment until the program ends, so bindings made per row of a long list should read package-level state rather than capture.

A counter component

This is what a component with state count turns into. The label and the wide flag are bindings on count. A tracker stands for the screen: it runs the paint function, and its callback only notes that a repaint is needed. The loop plays the event loop: it handles a click, runs the change handlers, and repaints when asked. Clicking three times with one jump to 10 gives four paints and three saves.

gui-reactive-guide-counter.salam
import gui.reactive

mut count := reactive.IntProp {}
mut label := reactive.StrProp {}
mut wide := reactive.BoolProp {}
mut needs_paint := false

func paint():
    println "  paint:", label.Get(), wide.Get() ? "(wide)" : ""
end

func click():
    count.Set(count.Peek() + 1)
end

func main:
    count = reactive.NewInt(0).Name("count")
    label = reactive.NewStr("").Name("label")
    label.Bind((): ret "Count: " + count.Get() end)
    wide = reactive.NewBool(false)
    wide.Bind((): ret count.Get() >= 10 end)
    screen := reactive.Track((): paint() end, (): needs_paint = true end)
    saved := reactive.OnChange(count, (): println "  save", count.Get() end)

    repeat 3 in frame:
        click()
        if frame == 2:
            count.Set(10)
        end
        _n := reactive.RunPendingHandlers()
        if needs_paint:
            needs_paint = false
            _r := screen.Refresh()
        end
    end
    saved.Free()
    screen.Free()
end
Output paint: Count: 0 save 1 paint: Count: 1 save 2 paint: Count: 2 save 10 paint: Count: 10 (wide)

A CSS transition on hover

An animated property moves to each new value over time. Here the background is a binding on hovered, animated with ease-in-out over 300 ms and interpolated per colour channel. Each frame sets the clock with SetTime; the animated property depends on the clock only while it moves, and AnimationsRunning tells the loop when it can stop drawing.

gui-reactive-guide-transition.salam
import conv
import gui.reactive

mut hovered := reactive.BoolProp {}
mut background := reactive.IntProp {}
mut dirty := false

func show(ms: i64):
    c := ((background.Get() as i64) & 0xFFFFFFFF) as u64
    println "  t", ms, "background", conv.FormatHex(c)
end

func main:
    reactive.SetTime(0 as i64)
    hovered = reactive.NewBool(false)
    background = reactive.NewInt(0)
    background.Bind((): ret hovered.Get() ? 0xFF3366CC as int : 0xFFFFFFFF as int end)
    background.Animate(reactive.Animation { duration = 300, easing = reactive.ParseEasing("ease-in-out", reactive.Ease()), color = true })
    screen := reactive.Track((): _c := background.Get() end, (): dirty = true end)

    println "pointer enters at 0 ms"
    hovered.Set(true)
    mut now := 0 as i64
    until reactive.AnimationsRunning() or dirty:
        if dirty:
            dirty = false
            _r := screen.Refresh()
            show(now)
        end
        now += 100
        reactive.SetTime(now)
    end
    println "idle at", now, "ms, running", reactive.AnimationsRunning()
    screen.Free()
end
Outputpointer enters at 0 ms t 0 background ffffffff t 100 background ffd0dcf3 t 200 background ff6289d8 t 300 background ff3366cc idle at 400 ms, running false

gui-reactive-Easing.salam
import gui.reactive
import math

func r3(v: f64): f64:
    ret math.Round(v * 1000.0) / 1000.0
end

func main:
    each e in [reactive.Linear(), reactive.Ease(), reactive.EaseIn(), reactive.EaseOut(), reactive.EaseInOut()]:
        println r3(e.At(0.25)), r3(e.At(0.5)), r3(e.At(0.75))
    end
    custom := reactive.CubicBezier(0.68, -0.6, 0.32, 1.6)
    println "custom", r3(custom.At(0.25)), r3(custom.At(0.75))
    s := reactive.Steps(3, reactive.StepPosition.JumpStart)
    println "steps", s.At(0.0), r3(s.At(0.5)), s.At(1.0)
    p := reactive.ParseEasing("cubic-bezier(0.1, 0.7, 1.0, 0.1)", reactive.Ease())
    println "parsed", p.kind == reactive.EasingKind.CubicBezier, p.x1, p.y1, p.x2, p.y2
    q := reactive.ParseEasing("steps(5, jump-none)", reactive.Ease())
    println "parsed steps", q.steps, q.position == reactive.StepPosition.JumpNone
    bad := reactive.ParseEasing("bounce", reactive.Linear())
    println "fallback", bad.kind == reactive.EasingKind.Linear
end

Animation

struct Animation
How an animated property moves to a new value: like a CSS transition.
NameTypeDescription
durationint = 0The length of the animation in milliseconds. 0 jumps at once.
delayint = 0Milliseconds to wait after the change before moving; the old value is shown meanwhile.

gui-reactive-Animate.salam
import conv
import gui.reactive

func main:
    reactive.SetTime(0 as i64)
    left := reactive.NewF64(0.0)
    left.Animate(reactive.Animation { duration = 400, easing = reactive.EaseOut() })
    left.Set(200.0)
    println "running", reactive.AnimationsRunning(), reactive.RunningAnimations()
    until reactive.AnimationsRunning():
        println "t", reactive.Now(), "left", left.Get()
        reactive.AdvanceTime(100 as i64)
    end
    println "t", reactive.Now(), "left", left.Get(), "running", reactive.AnimationsRunning()

    reactive.SetTime(1000 as i64)
    tint := reactive.NewInt(0xFF000000 as int)
    tint.Animate(reactive.Animation { duration = 200, delay = 100, color = true })
    tint.Set(0xFFFFFFFF as int)
    repeat 4 in i:
        reactive.SetTime(1000 as i64 + (i * 100) as i64)
        println "tint", conv.FormatHex(((tint.Get() as i64) & 0xFFFFFFFF) as u64)
    end
end

IntProp

struct IntProp
A property holding an int. The handle is a small value you can copy freely; the value itself lives in the runtime. A handle with node -1 is a constant.
NameTypeDescription
nodeint = -1The index of the property in the runtime, or -1 for a constant made with Const.
fixedint = 0The value of a constant property. Unused when node is not -1.
gui-reactive-Basics.salam
import gui.reactive

func main:
    width := reactive.NewInt(3).Name("width")
    height := reactive.NewInt(4).Name("height")
    area := reactive.NewInt(0).Name("area")
    area.Bind((): ret width.Get() * height.Get() end)
    println "area", area.Get(), "bound", area.HasBinding()

    width.Set(10)
    println "dirty after a set", area.IsDirty(), "evaluations so far", reactive.Evaluations()
    println "area", area.Get(), "evaluations so far", reactive.Evaluations()
    println "peek reads without tracking", area.Peek()

    marks := reactive.DirtyMarks()
    height.Set(5)
    height.Set(6)
    println "dirty marks for two sets", reactive.DirtyMarks() - marks

    area.Set(1)
    println "a set replaces the binding", area.HasBinding(), area.Get()
    println "nodes", reactive.LiveNodes(), "edges", reactive.LiveEdges()

    width.Free()
    height.Free()
    area.Free()
    reactive.Reset()
    println "nodes after reset", reactive.LiveNodes()
end

Unlink

func Unlink()
Leaves a two-way link and keeps the current value as a plain value. Does nothing on a property that is not linked.
gui-reactive-Link.salam
import gui.reactive

func main:
    model := reactive.NewStr("hello")
    field := reactive.NewStr("")
    reactive.Link(model, field)
    println model.Get(), field.Get()
    field.Set("typed in the field")
    println model.Get()
    model.Set("set by the program")
    println field.Get()

    volume := reactive.NewInt(5)
    slider := reactive.NewInt(0)
    knob := reactive.NewInt(0)
    reactive.Link(volume, slider)
    reactive.Link(slider, knob)
    knob.Set(9)
    println volume.Get(), slider.Get(), knob.Get()
    slider.Unlink()
    slider.Set(1)
    println volume.Get(), slider.Get(), knob.Get()
end
Outputhello hello typed in the field set by the program 9 9 9 9 1 9

gui-reactive-Types.salam
import gui.reactive
import str

func main:
    bytes := reactive.NewI64(3000000000 as i64)
    scale := reactive.NewF64(1.5)
    visible := reactive.NewBool(true)
    name := reactive.NewStr("Sara")
    greeting := reactive.NewStr("")
    greeting.Bind((): ret "Hello, " + name.Get() + "!" end)
    shown := reactive.NewStr("")
    shown.BindCopy((): ret visible.Get() ? name.Get() : "hidden" end)
    println greeting.Get(), shown.Get()
    name.Set("Reza")
    visible.Set(false)
    println greeting.Get(), shown.Get()
    println bytes.Get(), scale.Get() * 2.0

    gap := reactive.ConstInt(8)
    big := reactive.ConstI64(9000000000 as i64)
    ratio := reactive.ConstF64(0.5)
    flag := reactive.ConstBool(true)
    label := reactive.ConstStr("fixed")
    println gap.Get(), big.Get(), ratio.Get(), flag.Get(), label.Get(), "node", gap.node, "fixed", gap.fixed

    kept := str.Clone(greeting.Get())
    name.Set("Mina")
    println "copy kept", kept, "now", greeting.Get()
end

struct Size:
    pub w: int = 0
    pub h: int = 0
end

func main:

    content.Bind(():

        ret Size { w = s.w - 2 * pad.Get(), h = s.h - 2 * pad.Get() }

    window.Free()
    content.Free()

import gui.reactive

mut redraws := 0

func main:
    x := reactive.NewInt(1)
    y := reactive.NewInt(2)
    t := reactive.Track((): println "  paint at", x.Get(), y.Get() end, (): redraws++ end).Name("paint")
    x.Set(10)
    y.Set(20)
    println "dirty", t.IsDirty(), "redraw requests", redraws
    println "refreshed", t.Refresh()
    println "refresh with nothing new", t.Refresh()

    reactive.BeginBatch()
    x.Set(3)
    println "in batch", reactive.InBatch(), "requests", redraws
    y.Set(4)
    reactive.EndBatch()
    println "after the batch", redraws
    _r := t.Refresh()

    reactive.Batch(():
        x.Set(5)
        y.Set(6)
    end)
    println "after Batch", redraws
    _r2 := t.Refresh()
    t.Free()
end

gui-reactive-ListModel.salam
import gui.reactive

struct Row:
    pub name: str = ""
    pub done: bool = false
end

func main:
    rows := reactive.NewListModel() as reactive.ListModel<Row>
    _id := rows.Listen((ev: reactive.ModelEvent): println "  change", ev.kind as int, "index", ev.index, "count", ev.count end)
    status := reactive.NewStr("")
    status.Bind((): ret "" + rows.Len() + " rows" end)
    rows.Push(Row { name = "milk" })
    rows.Push(Row { name = "bread" })
    rows.Insert(0, Row { name = "eggs" })
    println status.Get(), rows.Get(0).name
    rows.Set(1, Row { name = "milk", done = true })
    rows.Remove(0, 1)
    println status.Get(), rows.Get(0).done
    h := rows.OnChange((): println "  rows changed" end)
    rows.Clear()
    println "handlers", reactive.RunPendingHandlers(), status.Get()
    mut more := Vector {} as Vector<Row>
    more.push(Row { name = "tea" })
    rows.Replace(more)
    more.free()
    rows.Unlisten(_id)
    rows.Push(Row { name = "rice" })
    println status.Get(), reactive.ModelChange.Reset as int
    h.Free()
    rows.Free()
end
