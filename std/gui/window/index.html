gui.window

Native windows that show a pixel buffer, with an event loop for pointer, keyboard, text input, timers and the clipboard.

import gui.window

gui.window opens real windows and shows pixels you draw yourself. It is the bottom layer of Salam's native user interface: the program renders a frame into memory (for example with gui.paint), and this package puts it on the screen and reports what the user does. It talks to the system directly: X11 on Linux, Win32 on Windows and Cocoa on macOS, with no GTK, no web view and no other library. A headless backend keeps windows in memory, so the same program can be tested on any machine.

A frame is a buffer of 32-bit pixels, premultiplied alpha, with the value 0xAARRGGBB (bytes B, G, R, A in memory), described by a pointer, a width, a height and a stride. Sizes and positions in the API are logical pixels; the buffer is in physical pixels, which is the logical size times ScaleFactor.

  • Events are pulled, not pushed: call Wait in a loop. This is what macOS needs, since its windows only work on the main thread.
  • Closing is a request. The close button sends CloseRequested, and the window stays until you call Close.
  • Redraw requests merge. Call RequestRedraw as often as you like; one RedrawRequested arrives.
  • Keys and text are separate. KeyDown names the key by its position, the same on every keyboard language; the characters come in TextInput and, from an input method, in ImePreedit and ImeCommit.

The text of an event belongs to the package and is freed at the next Wait or Poll. Clone it with str.Clone if you keep it.

The event loop

A window program opens a window, then waits for events until it closes. RedrawRequested asks for a frame: allocate a buffer of the physical size, fill it and give it to PresentAll. A repeating timer drives anything that changes over time. This example is headless and uses the test clock, so it runs anywhere and prints the same lines every time; drop headless and SetManualClock and the same loop drives a real window.

gui-window-guide-loop.salam
import gui.window
import mem

func paint(w: window.Window, frame: int):
    size := w.Size()
    width := size.physical_width
    height := size.physical_height
    pixels := mem.AllocateZeroed(((width * height) * 4) as u64) as u32*
    defer mem.Free(pixels as void*)
    shade := (frame * 40) as u32
    repeat width * height in i:
        pixels[i] = (0xFF000000 as u32) | (shade << 8)
    end
    w.PresentAll(pixels as void*, width, height, width * 4)
end

func main:
    window.SetManualClock(true)
    w := window.Open(window.Options { title = "Clock", width = 64.0, height = 48.0, headless = true })
    tick := window.SetTimer(500, true)
    mut frames := 0
    until w.IsOpen():
        e := window.Wait(-1)
        switch e.kind:
            window.EventKind.RedrawRequested:
                paint(w, frames)
                frames++
                println "frame", frames, "at", window.Clock(), "ms"
                break
            end
            window.EventKind.Timer:
                w.RequestRedraw()
                if frames == 3:
                    window.Inject(window.Event { kind = window.EventKind.CloseRequested, window = w.id })
                end
                break
            end
            window.EventKind.CloseRequested:
                println "closing"
                window.ClearTimer(tick)
                w.Close()
                break
            end
            else:
                break
            end
        end
    end
    println "presented", w.PresentCount(), "frames"
end
Outputframe 1 at 0 ms frame 2 at 500 ms frame 3 at 1000 ms closing presented 3 frames

Mouse, keys and text

Pointer events carry logical coordinates and the button. clicks counts a double or triple click: presses of the same button close together in time and place. A key event names the key with Key, by position, and the system's own key code in code. Text never comes from key events: insert the text of TextInput and ImeCommit, and draw ImePreedit as the composition underline. SetImeRect places the input method's candidate window at your caret.

gui-window-guide-input.salam
import gui.window

func main:
    window.SetManualClock(true)
    w := window.Open(window.Options { width = 300.0, height = 200.0, headless = true, visible = false })
    window.Inject(window.Event { kind = window.EventKind.PointerDown, window = w.id, x = 40.0, y = 30.0, button = window.MouseButton.Left })
    window.Advance(120)
    window.Inject(window.Event { kind = window.EventKind.PointerDown, window = w.id, x = 41.0, y = 30.0, button = window.MouseButton.Left })
    window.Inject(window.Event { kind = window.EventKind.KeyDown, window = w.id, key = window.Key.A, code = 38 })
    window.Inject(window.Event { kind = window.EventKind.TextInput, window = w.id, text = "a" })
    window.Inject(window.Event { kind = window.EventKind.ImePreedit, window = w.id, text = "سل", cursor = 4 })
    window.Inject(window.Event { kind = window.EventKind.ImeCommit, window = w.id, text = "سلام" })
    mut e := window.Poll()
    until e.kind != window.EventKind.None:
        switch e.kind:
            window.EventKind.PointerDown:
                println "click", e.clicks, "at", e.x, e.y
                break
            end
            window.EventKind.KeyDown:
                println "key A:", e.key == window.Key.A
                break
            end
            window.EventKind.TextInput, window.EventKind.ImeCommit:
                println "insert", e.text
                break
            end
            window.EventKind.ImePreedit:
                println "composing", e.text, "caret at byte", e.cursor
                break
            end
            else:
                break
            end
        end
        e = window.Poll()
    end
    w.SetImeRect(40.0, 30.0, 2.0, 18.0)
    println "candidate window goes to", w.ImeRect()
end
Outputclick 1 at 40 30 click 2 at 41 30 key A: true insert a composing سل caret at byte 4 insert سلام candidate window goes to Rect {x = 40, y = 30, width = 2, height = 18}

Scale factor and partial updates

On a high density screen one logical pixel is several device pixels. Draw in physical pixels and use ScaleFactor to convert. Moving a window to another screen sends ScaleChanged and then Resized. When only part of the frame changed, pass the changed rectangles to Present; it copies those alone and clips them to the buffer.

gui-window-guide-hidpi.salam
import gui.window
import mem

func main:
    w := window.Open(window.Options { width = 100.0, height = 50.0, headless = true, scale = 1.5, visible = false })
    size := w.Size()
    println "logical", size.width, size.height, "physical", size.physical_width, size.physical_height
    window.Inject(window.Event { kind = window.EventKind.ScaleChanged, window = w.id, scale = 2.0 })
    e := window.Poll()
    println "new scale", e.scale, "logical", e.width, e.height, "physical", e.physical_width, e.physical_height
    window.Inject(window.Event { kind = window.EventKind.Resized, window = w.id, physical_width = 300, physical_height = 100 })
    r := window.Poll()
    println "resized to", r.width, r.height
    width := r.physical_width
    height := r.physical_height
    pixels := mem.AllocateZeroed(((width * height) * 4) as u64) as u32*
    defer mem.Free(pixels as void*)
    w.PresentAll(pixels as void*, width, height, width * 4)
    mut damage := Vector {} as Vector<window.Rect>
    defer damage.free()
    damage.push(window.Rect { x = 10, y = 10, width = 20, height = 8 })
    damage.push(window.Rect { x = 290, y = 90, width = 50, height = 50 })
    w.Present(pixels as void*, width, height, width * 4, damage)
    println "copied", w.LastDamageCount(), "rects inside", w.LastDamage()
end
Outputlogical 100 50 physical 150 75 new scale 2 logical 75 37.5 physical 150 75 resized to 150 50 copied 2 rects inside Rect {x = 10, y = 10, width = 290, height = 90}

Testing without a screen

A headless window behaves like a real one: it gets focus, redraw requests, resize events and timers, and keeps every presented frame. PixelAt reads the frame back, Inject plays any event, and SetManualClock makes time jump straight to the next timer, so a test of a one second animation takes no time. The environment variable SALAM_WINDOW_BACKEND=headless makes every window headless without changing the program.

gui-window-guide-testing.salam
import gui.window
import mem

func main:
    window.SetManualClock(true)
    w := window.Open(window.Options { width = 2.0, height = 2.0, headless = true, visible = false })
    pixels := mem.AllocateZeroed(16 as u64) as u32*
    defer mem.Free(pixels as void*)
    pixels[0] = 0xFFFF0000 as u32
    pixels[3] = 0x80000080 as u32
    w.PresentAll(pixels as void*, 2, 2, 8)
    println "top left red:", w.PixelAt(0, 0) == (0xFFFF0000 as u32)
    println "bottom right half blue:", w.PixelAt(1, 1) == (0x80000080 as u32)
    timer := window.SetTimer(1000, false)
    mut e := window.Wait(-1)
    until e.kind != window.EventKind.Timer:
        println "skipped event", e.kind as int
        e = window.Wait(-1)
    end
    println "timer", e.timer == timer, "after", window.Clock(), "ms of test time"
    w.Close()
end
Outputtop left red: true bottom right half blue: true skipped event 4 timer true after 1000 ms of test time

gui-window-Event.salam
import gui.window

func main:
    window.SetManualClock(true)
    w := window.Open(window.Options { width = 300.0, height = 200.0, headless = true })
    window.Inject(window.Event { kind = window.EventKind.PointerDown, window = w.id, x = 10.0, y = 20.0, button = window.MouseButton.Left })
    window.Inject(window.Event { kind = window.EventKind.PointerDown, window = w.id, x = 10.0, y = 20.0, button = window.MouseButton.Left })
    window.Inject(window.Event { kind = window.EventKind.KeyDown, window = w.id, key = window.Key.Enter, code = 36, modifiers = window.mod_control })
    window.Inject(window.Event { kind = window.EventKind.TextInput, window = w.id, text = "سلام" })
    window.Inject(window.Event { kind = window.EventKind.Wheel, window = w.id, delta_y = 1.0, delta_mode = window.WheelMode.Line })
    mut e := window.Poll()
    until e.kind != window.EventKind.None:
        switch e.kind:
            window.EventKind.PointerDown:
                println "down", e.x, e.y, e.button == window.MouseButton.Left, "clicks", e.clicks
                break
            end
            window.EventKind.KeyDown:
                println "key", e.key == window.Key.Enter, e.code, (e.modifiers & window.mod_control) != 0, e.repeated
                break
            end
            window.EventKind.TextInput:
                println "text", e.text
                break
            end
            window.EventKind.Wheel:
                println "wheel", e.delta_x, e.delta_y, e.delta_mode == window.WheelMode.Line
                break
            end
            else:
                println "other", e.kind as int, e.window
                break
            end
        end
        e = window.Wait(0)
    end
    w.Close()
end

SetTimer

func SetTimer(ms: int, repeating: bool): int
Starts a timer that fires every ms milliseconds, or once; returns its number.
gui-window-Timer.salam
import gui.window

func main:
    window.SetManualClock(true)
    tick := window.SetTimer(100, true)
    once := window.SetTimer(250, false)
    repeat 4:
        e := window.Wait(-1)
        println e.kind == window.EventKind.Timer, e.timer == tick ? "tick" : "once", window.Clock()
    end
    window.ClearTimer(tick)
    window.Advance(1000)
    println window.Clock(), window.Wait(10).kind == window.EventKind.None
    window.Wake()
    println window.Wait(-1).kind == window.EventKind.Wake, once > tick
end
Outputtrue tick 100 true tick 200 true once 250 true tick 300 1300 true true true
