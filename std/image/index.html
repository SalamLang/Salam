image

Reading, editing and saving images, and writing text on them.

import image
On this page
  1. Types
  2. Constants and variables
  3. Functions

This package works with images without any outside library: it reads and writes PNG, JPEG and BMP. It also reads and writes WebP, both lossy and lossless. From a GIF it reads the first frame, and it saves an image in a GIF with at most 256 colors. Open and Save work the format out from the content or the extension. Drawing, resizing, cropping, rotating and filters such as blurring and sharpening are all there.

For text, read a TrueType font with LoadFont and write with DrawTextStyled; Persian and Arabic text is shaped and laid out right to left on its own. For building social media images automatically, DrawTextFitBox finds the size that fits. The font example needs a font file and on the site it is only checked.

import image
import os

func main:
    mut img := image.New(8, 6)
    println img.width, img.height, image.IsEmpty(img), image.InBounds(img, 7, 5), image.InBounds(img, 8, 0)
    image.Fill(img, image.RGB(255, 255, 255))
    image.FillHex(img, "#202040")
    image.SetPixel(img, 0, 0, image.RGB(255, 0, 0))
    image.BlendPixel(img, 1, 0, image.RGBA(255, 255, 255, 128))
    println image.ToHex(image.GetPixel(img, 0, 0)), image.ToHex(image.GetPixel(img, 1, 0)), image.ToHexRGB(image.GetPixel(img, 2, 0))
    image.DrawLine(img, 0, 5, 7, 5, image.Gray(200))
    image.DrawRect(img, 1, 1, 4, 3, image.ParseHex("#ffcc00"))
    image.DrawRectFilled(img, 2, 2, 2, 1, image.ParseHex("#00ff00"))
    image.DrawCircle(img, 6, 2, 1, image.Gray(255))
    image.DrawCircleFilled(img, 6, 2, 0, image.Gray(10))
    println image.ToHexRGB(image.GetPixel(img, 3, 2)), image.ToHexRGB(image.GetPixel(img, 4, 5))

    color := image.ParseHex("#336699")
    println color.r, color.g, color.b, color.a, image.Luminance(color)
    println image.Equals(color, image.RGB(51, 102, 153)), image.ToHex(image.Blend(color, image.RGBA(255, 255, 255, 0)))
    println image.ToHexRGB(image.LerpColor(image.Gray(0), image.Gray(200), 0.5))

    mut labelled := image.NewFilled(40, 10, image.Gray(0))
    image.DrawText(labelled, 1, 1, "SALAM", image.Gray(255), 1)
    println image.TextWidth("SALAM", 1), image.FontWidth, image.FontHeight

    dir := os.TempDir()
    println image.Save(img, dir + "/salam_en.png"), image.SavePNG(img, dir + "/salam2_en.png"), image.SaveBMP(img, dir + "/salam_en.bmp"), image.SaveJPEG(img, dir + "/salam_en.jpg", 90)
    println image.SaveQuality(img, dir + "/salam2_en.jpg", 50), image.SaveWebP(img, dir + "/salam_en.webp")
    mut ok := false
    read := image.Open(dir + "/salam_en.png", ok)
    println ok, read.width, image.Equals(image.GetPixel(read, 0, 0), image.GetPixel(img, 0, 0))
    png := image.OpenPNG(dir + "/salam2_en.png", ok)
    bmp := image.OpenBMP(dir + "/salam_en.bmp", ok)
    jpg := image.OpenJPEG(dir + "/salam_en.jpg", ok)
    println ok, png.width, bmp.height, jpg.width
    println image.FormatFromExt("a.jpeg") == image.FormatJPEG, image.FormatFromExt("a.png") == image.FormatPNG, image.FormatFromExt("a.txt") == image.FormatUnknown

    buf := image.EncodePNG(img)
    println image.DetectFormat(buf.data, buf.len) == image.FormatPNG, image.IsWebP(buf.data, buf.len)
    from_buf := image.DecodePNG(buf.data, buf.len, ok)
    generic := image.Decode(buf.data, buf.len, ok)
    println ok, from_buf.width, generic.height
    b := image.EncodeBMP(img)
    b2 := image.DecodeBMP(b.data, b.len, ok)
    j := image.EncodeJPEG(img, 80)
    j2 := image.DecodeJPEG(j.data, j.len, ok)
    println image.DetectFormat(b.data, b.len) == image.FormatBMP, image.DetectFormat(j.data, j.len) == image.FormatJPEG, b2.width, j2.width
    any := image.Encode(img, image.FormatBMP, 90)
    web := image.EncodeWebP(img)
    lossy := image.EncodeWebPQuality(img, 75)
    println any.len == b.len, web.len > 0, image.FormatWebP
    clear := image.NewFilled(1, 1, image.RGBA(0, 0, 0, 0))
    println image.ToHex(image.GetPixel(image.FlattenAlpha(clear, image.Gray(255)), 0, 0))
    converted := image.Convert(img, image.FormatJPEG)
    println converted.width
    webp := image.DecodeWebP(web.data, web.len, ok)
    println ok, webp.width, image.Equals(image.GetPixel(webp, 1, 0), image.GetPixel(img, 1, 0))
    webp3 := image.DecodeWebP(lossy.data, lossy.len, ok)
    println ok, webp3.width, webp3.height
    webp2 := image.OpenWebP(dir + "/salam_en.webp", ok)
    println image.SaveGIF(img, dir + "/salam_en.gif"), image.FormatFromExt("a.gif") == image.FormatGIF
    gif := image.OpenGIF(dir + "/salam_en.gif", ok)
    g := image.EncodeGIF(img)
    gif2 := image.DecodeGIF(g.data, g.len, ok)
    println ok, gif.width, gif2.height, image.IsGIF(g.data, g.len), image.Equals(image.GetPixel(gif, 0, 0), image.GetPixel(img, 0, 0))
    println ok, webp2.height
    mut copy := image.Clone(img)
    image.Free(copy)
    image.Free(img)
end

image-Edit.salam
import image

func size(name: str, img: image.Image):
    println name, "" + img.width + "x" + img.height
end

func main:
    mut img := image.NewFilled(8, 4, image.Gray(100))
    image.SetPixel(img, 0, 0, image.RGB(255, 0, 0))
    size("nearest", image.ResizeNearestNeighbor(img, 4, 2))
    size("bilinear", image.ResizeBilinearFilter(img, 16, 8))
    size("resize", image.Resize(img, 2, 2, image.ResizeBilinear))
    size("near", image.Resize(img, 2, 2, image.ResizeNearest))
    size("scale", image.Scale(img, 0.5))
    size("crop", image.Crop(img, 2, 1, 3, 2))
    clockwise := image.RotateCW(img)
    size("clockwise", clockwise)
    println image.ToHexRGB(image.GetPixel(clockwise, 3, 0))
    size("counterclockwise", image.RotateCCW(img))
    println image.ToHexRGB(image.GetPixel(image.Rotate180(img), 7, 3))
    println image.ToHexRGB(image.GetPixel(image.FlipHorizontal(img), 7, 0)), image.ToHexRGB(image.GetPixel(image.FlipVertical(img), 0, 3))
    size("fit", image.ResizeFit(img, 4, 4))
    size("fill", image.ResizeFill(img, 4, 4))
    size("thumbnail", image.Thumbnail(img, 2))

    mut c := image.NewFilled(2, 1, image.RGB(200, 100, 50))
    image.Grayscale(c)
    println image.ToHexRGB(image.GetPixel(c, 0, 0))
    image.Invert(c)
    image.Brightness(c, 10)
    image.Contrast(c, 1.5)
    println image.ToHexRGB(image.GetPixel(c, 0, 0))
    image.Threshold(c, 128)
    println image.ToHexRGB(image.GetPixel(c, 0, 0))
    mut dot := image.NewFilled(5, 5, image.Gray(0))
    image.SetPixel(dot, 2, 2, image.Gray(255))
    blurred := image.Blurred(dot, 1)
    println image.GetPixel(blurred, 2, 2).r < 255, image.GetPixel(blurred, 1, 2).r > 0
    println image.GetPixel(image.Sharpened(dot, 1.0), 2, 2).r
    mut base := image.NewFilled(2, 2, image.Gray(0))
    image.Overlay(base, image.NewFilled(1, 1, image.Gray(200)), 1, 1, 0.5)
    println image.ToHexRGB(image.GetPixel(base, 1, 1)), image.ToHexRGB(image.GetPixel(base, 0, 0))

    mut hist := image.ComputeHistogram(dot)
    println image.HistogramGet(hist.luma, 0), image.HistogramGet(hist.luma, 255), image.HistogramGet(hist.r, 255), image.HistogramGet(hist.g, 0), image.HistogramGet(hist.b, 0)
    image.HistogramFree(hist)
    black := image.NewFilled(5, 5, image.Gray(0))
    diff := image.Diff(dot, black)
    println image.GetPixel(diff, 2, 2).r, image.GetPixel(diff, 0, 0).r
    println (image.MeanAbsError(dot, black) * 100.0 + 0.5) as i64, (image.PSNR(dot, black) + 0.5) as i64
end

import image
import io

func main:
    println image.FindSystemFont(), image.FindArabicFont()
    mut ok := false
    mut font := image.LoadSystemFont(ok)
    if not ok:
        font = image.LoadFont("Vazirmatn-Regular.ttf", ok)
    end
    if not ok:
        println "no font was found"
        ret
    end
    println image.FontValid(font), image.UnitsPerEm(font), image.NumGlyphs(font)
    size := 32.0
    println image.ScaleFactor(font, size), image.AscentPixels(font, size), image.DescentPixels(font, size), image.LineHeightPixels(font, size)
    a := image.GlyphIndex(font, 65)
    v := image.GlyphIndex(font, 86)
    println image.GlyphAdvance(font, a), image.KernAdvance(font, a, v), image.GlyphHasOutline(font, a)
    mut x0 := 0
    mut y0 := 0
    mut x1 := 0
    mut y1 := 0
    println image.GlyphBox(font, a, x0, y0, x1, y1), x1 - x0, y1 - y0

    mut img := image.NewFilled(400, 120, image.Gray(255))
    black := image.Gray(0)
    _ := image.DrawTextTTF(img, 10.0, 40.0, "Salam", black, font, size)
    println image.TextWidthTTF(font, "Salam", size)
    style := image.TextStyle { size = 28.0, tracking = 1.0, kern = true, dir = image.TextDirAuto, shape = true }
    _ := image.DrawTextStyled(img, 10.0, 80.0, "سلام دنیا", black, font, style)
    println image.TextWidthStyled(font, "سلام دنیا", style), image.TextInkWidth(font, "سلام", style), image.TextInkHeight(font, "سلام", style)
    mut width := 0.0
    mut height := 0.0
    println image.TextInkSize(font, "سلام", style, width, height), width, height
    mut a0 := 0.0
    mut b0 := 0.0
    mut a1 := 0.0
    mut b1 := 0.0
    println image.TextBounds(font, "سلام", style, a0, b0, a1, b1)
    println image.FitSizeToWidth(font, "سلام", style, 200.0, 96.0), image.FitSizeToBox(font, "سلام", style, 200.0, 50.0, 96.0)
    _ := image.DrawGlyphStyled(img, a, 300.0, 40.0, black, font, style)
    _ := image.DrawCodepointStyled(img, 1600, 340.0, 40.0, black, font, style)
    _ := image.DrawTextCentered(img, 200.0, 60.0, "middle", black, font, style)
    _ := image.DrawTextCenteredIn(img, "middle", black, font, style)
    _ := image.DrawTextInBox(img, 0.0, 0.0, 200.0, 60.0, "box", black, font, style, image.TextAlignRight, image.TextVAlignMiddle)
    _ := image.DrawTextFitBox(img, 0.0, 60.0, 200.0, 60.0, "fit", black, font, style, 0.8, 64.0)
    _ := image.DrawTextFitImage(img, "سلام", black, font, style, 0.9, 120.0)
    println image.TextAlignLeft, image.TextAlignCenter, image.TextVAlignTop, image.TextVAlignBottom, image.TextVAlignBaseline
    _ := image.Save(img, "salam-text-en.png")

    data := io.ReadFile("Vazirmatn-Regular.ttf")
    mut f2 := image.ParseFont(data as void*, data.len(), ok)
    mut f3 := image.ParseFontIndex(data as void*, data.len(), 0, ok)
    mut f4 := image.LoadFontIndex("NotoSansCJK.ttc", 0, ok)
    println image.IsCFFFont(data as void*, data.len())
    image.FreeFont(f2)
    image.FreeFont(f3)
    image.FreeFont(f4)
    image.FreeFont(font)
    image.Free(img)
end
