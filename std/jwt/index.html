jwt

Making and checking JWT tokens, and guarding API routes.

import jwt
On this page
  1. Types
  2. Constants and variables
  3. Functions

A JWT is a token the server signs after a user logs in and the client sends with every request; by checking the signature the server knows who the user is without going to the database. This package supports the HMAC algorithms, HS256, HS384 and HS512, and never accepts a token with the none algorithm.

For an API, make a Guard and call Require or one of its siblings in every handler; when the token is wrong the error response is sent for you. The secret has to be long and random, for instance from RandomToken in crypto.

jwt-Sign.salam
import jwt

func main:
    mut c := jwt.NewClaims()
    c.iss = "salamlang.ir"
    c.sub = "42"
    c.aud = "api"
    c.iat = 1700000000
    c.exp = 1700003600
    jwt.SetStr(c, "name", "Maryam")
    jwt.SetInt(c, "level", 3)
    jwt.SetBool(c, "admin", false)
    jwt.SetRaw(c, "scope", `"read write"`)
    jwt.SetRaw(c, "roles", `["editor","viewer"]`)
    println jwt.ClaimsJSON(c)

    mut err := 0
    token := jwt.SignHS256(c, "a-long-random-secret", err)
    println err == jwt.JWT_OK
    println token

    mut o := jwt.DefaultOptions()
    o.now = 1700001000
    o.issuer = "salamlang.ir"
    o.audience = "api"
    t := jwt.Verify(token, "a-long-random-secret", o, err)
    println t.valid, t.alg, t.typ, t.claims.sub
    println jwt.Str(t, "name"), jwt.Int(t, "level"), jwt.Bool(t, "admin"), jwt.Has(t, "email")
    println jwt.Raw(t, "level"), jwt.HasScope(t, "write"), jwt.HasAudience(t, "api"), jwt.ListHas(t, "roles", "editor")

    _ := jwt.Verify(token, "another-secret", o, err)
    println err == jwt.JWT_ERR_SIGNATURE, jwt.ErrorText(err)
    o.now = 1800000000
    _ := jwt.Verify(token, "a-long-random-secret", o, err)
    println err == jwt.JWT_ERR_EXPIRED, jwt.ErrorText(err)
    o.now = 1700001000
    o.issuer = "example.com"
    println jwt.IsValid(token, "a-long-random-secret", o)
    _ := jwt.Decode("abc", err)
    println err == jwt.JWT_ERR_MALFORMED
end

jwt-Algorithms.salam
import jwt

func main:
    println jwt.JWT_HS256, jwt.JWT_HS384, jwt.JWT_HS512
    println jwt.IsSupportedAlg("HS512"), jwt.IsSupportedAlg("none"), jwt.DigestSize("HS384")

    mut c := jwt.NewClaims()
    c.sub = "7"
    c.jti = "t-1"
    jwt.ExpiresIn(c, 3600)
    jwt.NotBeforeIn(c, 0)
    mut err := 0
    one := jwt.SignHS384(c, "k", err)
    two := jwt.SignHS512(c, "k", err)
    three := jwt.Sign(c, jwt.JWT_HS256, "k", err)
    four := jwt.SignWithKid(c, "HS256", "key-2026", "k", err)
    println jwt.Decode(one, err).alg, jwt.Decode(two, err).alg, jwt.Decode(four, err).kid

    t := jwt.VerifyHS256(three, "k", err)
    println t.valid, jwt.SecondsRemaining(t) > 3500
    b := jwt.Verify(one, "k", jwt.OptionsFor("HS384"), err)
    println b.valid
    _ := jwt.Verify(one, "k", jwt.DefaultOptions(), err)
    println jwt.ErrorText(err)

    raw := jwt.SignJSON(`{"sub":"1"}`, "HS256", "", "k", err)
    println raw
    d := jwt.Decode(raw, err)
    println d.header_json, d.payload_json
    println jwt.SignatureOf(d.signing_input, "HS256", "k") == d.signature

    encoded := jwt.B64UrlEncode("hello?")
    println encoded.len()
    mut ok := false
    println jwt.B64UrlDecode(encoded, ok), ok
    println jwt.B64UrlEncodeBytes("hi?" as void*, 3)
    _ := jwt.SignHS256(c, "", err)
    println err == jwt.JWT_ERR_EMPTY_SECRET, jwt.ErrorText(err)
    println jwt.JWT_ERR_BAD_BASE64, jwt.JWT_ERR_BAD_JSON, jwt.JWT_ERR_ALG_UNSUPPORTED, jwt.JWT_ERR_ALG_MISMATCH
    println jwt.JWT_ERR_NOT_YET_VALID, jwt.JWT_ERR_ISSUED_IN_FUTURE, jwt.JWT_ERR_ISSUER, jwt.JWT_ERR_AUDIENCE
    println jwt.ErrorText(jwt.JWT_ERR_SUBJECT), "/", jwt.ErrorText(jwt.JWT_ERR_MISSING_CLAIM), "/", jwt.ErrorText(jwt.JWT_ERR_TYPE)
end

Guard

struct Guard
An HTTP route guard: the secret, the check options and the cookie name.
NameTypeDescription
secretstr = ""The signing secret.
optsOptionsThe check options.
cookiestr = ""The name of a cookie the token is also read from; empty means the Authorization header only.
realmstr = "api"The name that goes in the WWW-Authenticate header.
jwt-Guard.salam
import jwt
import net.http
import net.router

mut guard := jwt.NewGuard("a-server-secret-only-the-server-knows")

func login(ctx: i64):
    mut err := 0
    mut c := jwt.NewClaims()
    c.sub = "42"
    jwt.SetStr(c, "scope", "read")
    jwt.SetRaw(c, "roles", `["editor"]`)
    token := jwt.Issue(guard, c, 3600, err)
    jwt.SetSessionCookieInsecure(ctx, guard, token, 3600)
    http.Ctx_text(ctx, token)
end

func me(ctx: i64):
    mut t := jwt.Token {}
    if jwt.Require(ctx, guard, t):
        http.Ctx_text(ctx, "hello user " + t.claims.sub)
    end
end

func admin(ctx: i64):
    mut t := jwt.Token {}
    if jwt.RequireScope(ctx, guard, "admin", t):
        http.Ctx_text(ctx, "admin panel")
    end
end

func edit(ctx: i64):
    mut t := jwt.Token {}
    if jwt.RequireClaim(ctx, guard, "roles", "editor", t):
        http.Ctx_text(ctx, "editor")
    end
end

func profile(ctx: i64):
    mut t := jwt.Token {}
    if jwt.RequireSubject(ctx, guard, http.Ctx_param(ctx, "id"), t):
        http.Ctx_text(ctx, "your own profile")
    end
end

func manual(ctx: i64):
    mut err := 0
    t := jwt.Check(ctx, guard, err)
    if jwt.BearerToken(ctx).len() == 0 and jwt.TokenFrom(ctx, guard).len() == 0:
        jwt.MissingCredential(ctx, guard)
    else err != jwt.JWT_OK:
        jwt.Unauthorized(ctx, guard, jwt.ErrorText(err))
    else t.claims.sub == "0":
        jwt.Forbidden(ctx, "the user is blocked")
    else:
        http.Ctx_text(ctx, "valid")
    end
end

func logout(ctx: i64):
    jwt.ClearSessionCookie(ctx, guard)
    http.Ctx_text(ctx, "you are logged out")
end

func main:
    guard.cookie = "session"
    mut r := router.NewRouter()
    router.Post(r, "/login", login)
    router.Get(r, "/me", me)
    router.Get(r, "/admin", admin)
    router.Get(r, "/edit", edit)
    router.Get(r, "/users/:id", profile)
    router.Get(r, "/check", manual)
    router.Post(r, "/logout", logout)
    mut s := http.NewServer(19630)
    http.Use(s, r)
    _ := http.RunBackground(s)
    base := "http://127.0.0.1:19630"

    login_response := http.Post(base + "/login", "")
    token := http.Body(login_response)
    println http.CookieMap(login_response).get("session") == token
    mut headers := http.NewHeaders()
    headers.insert("Authorization", "Bearer " + token)
    println jwt.ParseBearer("Bearer " + token) == token
    ask(base, "/me", headers)
    ask(base, "/admin", headers)
    ask(base, "/edit", headers)
    ask(base, "/users/42", headers)
    ask(base, "/users/7", headers)
    ask(base, "/check", headers)
    ask(base, "/me", http.NewHeaders())
    mut bad := http.NewHeaders()
    bad.insert("Authorization", "Bearer x.y.z")
    ask(base, "/check", bad)
    mut cookie := http.NewHeaders()
    cookie.insert("Cookie", "session=" + token)
    ask(base, "/me", cookie)

    mut access := ""
    mut refresh := ""
    mut err := 0
    println jwt.IssuePair(guard, "42", 900, 86400, access, refresh, err), access != refresh
    one := jwt.IssueFor(guard, "42", 60, err)
    println jwt.Verify(one, guard.secret, guard.opts, err).claims.sub, guard.realm
end

func ask(base: str, path: str, headers: HashMap<str, str>):
    response := http.GetWith(base + path, headers)
    println path, http.Status(response), http.Body(response)
end
