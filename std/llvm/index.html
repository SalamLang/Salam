LLVM is the compiler infrastructure behind Salam's own native backend. This package binds its C API so that a Salam program can be a compiler: build or read IR in a Module, check it with Verify, optimize it, then run it at once with the JIT or write an object file and link it into an executable. Use it when you are writing a language, a query compiler or a code generator. To run Salam code itself you do not need it; salam build already goes through LLVM.

Core ideas

  • Context, module, builder. A Context owns everything LLVM allocates. A Module is one translation unit of functions and globals inside it. A Builder is a cursor that appends instructions to the end of a block.
  • Values are not numbers. Add, Load and the other instruction functions return a Value: the result the generated code will compute when it runs, not a result now.
  • Two ways in. Write IR as text and read it with ParseIR, or build it call by call. Both end in the same Module.
  • Two ways out. RunJIT compiles in memory and calls main. Run with an Options value, or NewTargetMachine, RunOpt and EmitObject step by step, write files for any target LLVM knows.

Things that trip people up

  • ParseIR, RunJIT and Run take the path of a .ll file, not the IR text. Write the text to a file first, as the examples do.
  • RunJIT parses that file again by itself; it does not run a module you already have. Its rc is the i32 that main returned.
  • Invalid IR can crash LLVM later instead of producing an error. Call Verify on everything you build before you optimize or emit it.
  • The linker inside the program (LinkAuto) still needs the target's C library files: a musl sysroot for Linux, MinGW for Windows, a WASI sysroot for WebAssembly. Without one the link fails with a message naming the directory it looked in; point $SALAM_SYSROOTS or Options.sysroot at one.
  • Free in reverse order: builder, then module, then context.

Programs that use this package link against libsalam_llvm, so the site only compiles them. Build the library as the repository's LLVM backend instructions describe, then run them with salam run. The output shown is what they printed with LLVM 22 on x86-64 Linux.

Running IR you wrote as text

The shortest way in is to write LLVM IR as text, the same .ll format clang -S -emit-llvm prints, and save it to a file. ParseIR reads that file into a module in a fresh context and reports syntax errors in err; Verify then checks the rules the parser does not, such as every block ending in a branch or a return. RunJIT compiles the same file in memory and calls its main.

llvm-guide-text.salam
import llvm
import io
import os

func main:
    path := os.TempDir() + "/salam_guide_sum.ll"
    _ := io.WriteFile(path, `define i32 @main() {
entry:
  br label %loop
loop:
  %i = phi i32 [ 1, %entry ], [ %next, %loop ]
  %acc = phi i32 [ 0, %entry ], [ %sum, %loop ]
  %sum = add i32 %acc, %i
  %next = add i32 %i, 1
  %more = icmp sle i32 %next, 10
  br i1 %more, label %loop, label %done
done:
  ret i32 %sum
}
`)
    parsed := llvm.ParseIR(path)
    if not parsed.ok:
        println "parse error:", parsed.err
        ret 1
    end
    checked := llvm.Verify(parsed.mod)
    println "valid:", checked.ok
    llvm.FreeModule(parsed.mod)
    llvm.FreeContext(parsed.ctx)
    r := llvm.RunJIT(path)
    println "sum of 1..10 =", r.rc
end

The loop adds 1 to 10, so the program prints valid: true and then sum of 1..10 = 55. The value of main comes back as rc, just like an exit code, which keeps the round trip simple while you are experimenting.

A tiny compiler with the builder

A real compiler does not glue strings together; it walks its own syntax tree and calls the builder. This one reads arithmetic in reverse Polish notation: a number pushes a constant on a stack, and an operator pops two values and pushes the instruction that combines them. A Value stands for a result that will exist at run time, so the stack holds instructions, not numbers.

llvm-guide-rpn.salam
import llvm
import io
import os
import str

func main:
    program := "2 3 + 4 * 6 -"
    ctx := llvm.NewContext()
    mod := llvm.NewModule(ctx, "rpn")
    b := llvm.NewBuilder(ctx)
    i32 := llvm.Int32Type(ctx)
    none := Vector {} as Vector<llvm.Type>
    fn := llvm.AddFunction(mod, "main", llvm.FunctionType(i32, none, false))
    llvm.PositionAtEnd(b, llvm.AppendBlock(ctx, fn, "entry"))
    mut stack := Vector {} as Vector<llvm.Value>
    each word in str.Split(program, " "):
        if word == "+" or word == "-" or word == "*":
            r := stack.pop()
            l := stack.pop()
            if word == "+": stack.push(llvm.Add(b, l, r, "t")) end
            if word == "-": stack.push(llvm.Sub(b, l, r, "t")) end
            if word == "*": stack.push(llvm.Mul(b, l, r, "t")) end
        else:
            stack.push(llvm.ConstInt(i32, str.ToInt(word) as i64))
        end
    end
    _ := llvm.Ret(b, stack.pop())
    println llvm.Verify(mod).ok
    path := os.TempDir() + "/salam_guide_rpn.ll"
    _ := io.WriteFile(path, llvm.ModuleToString(mod))
    println program, "=", llvm.RunJIT(path).rc
    llvm.FreeBuilder(b)
    llvm.FreeModule(mod)
    llvm.FreeContext(ctx)
end

(2 + 3) * 4 - 6 is 14, and that is what the JIT returns. With constants only, LLVM's builder folds the whole expression while you build it, so the IR is just ret i32 14; with parameters or loaded values you would see the add, mul and sub instructions.

Let Verify catch your mistakes

Bugs in a code generator usually produce IR that is not valid, and LLVM may crash on it later instead of reporting it. Call Verify after building a module and before optimizing or running it. Here the block computes a value but never returns, so the module is rejected with a message that names the problem.

llvm-guide-verify.salam
import llvm

func main:
    ctx := llvm.NewContext()
    mod := llvm.NewModule(ctx, "broken")
    b := llvm.NewBuilder(ctx)
    i32 := llvm.Int32Type(ctx)
    none := Vector {} as Vector<llvm.Type>
    fn := llvm.AddFunction(mod, "answer", llvm.FunctionType(i32, none, false))
    llvm.PositionAtEnd(b, llvm.AppendBlock(ctx, fn, "entry"))
    _ := llvm.Add(b, llvm.ConstInt(i32, 40), llvm.ConstInt(i32, 2), "x")
    checked := llvm.Verify(mod)
    if not checked.ok:
        println "rejected:", checked.err
    end
    llvm.FreeBuilder(b)
    llvm.FreeModule(mod)
    llvm.FreeContext(ctx)
end

An object file and a static program

Run takes the whole pipeline from one Options value: the optimization level, the target triple and what to produce. OUT_OBJ writes an object file for the chosen triple, even one that is not the machine you are on. For x86_64-unknown-linux-musl, LLD inside the program links it into a static executable with no system linker involved; CanLinkTriple says whether that is possible for a triple.

llvm-guide-static.salam
import llvm
import io
import os

func main:
    dir := os.TempDir()
    ll := dir + "/salam_guide_static.ll"
    _ := io.WriteFile(ll, `define i32 @main() {
entry:
  ret i32 7
}
`)
    mut opts := llvm.OptionsDefault()
    opts.opt_level = llvm.OPT_O2
    opts.target_triple = "x86_64-unknown-linux-musl"
    opts.output_mode = llvm.OUT_OBJ
    opts.output_file = dir + "/salam_guide_static.o"
    built := llvm.Run(ll, opts)
    if not built.ok:
        println "compile failed:", built.error
        ret 1
    end
    println "can link here:", llvm.CanLinkTriple(opts.target_triple)
    linked := llvm.LinkAuto(opts.output_file, dir + "/salam_guide_static", opts)
    if not linked.ok:
        println "link failed:", linked.err
        ret 1
    end
    println "built", dir + "/salam_guide_static"
end

On a machine without musl's files in /usr/lib/x86_64-linux-musl (the musl-dev package) or under $SALAM_SYSROOTS, the object is still written but the link step reports no musl sysroot. CanLinkTriple only tells you that LLD is built in and knows the triple; it does not look for the sysroot. The program it builds just exits with 7. For a Windows target use x86_64-w64-windows-gnu and an output ending in .exe; to work out sizes and file names from a triple without LLVM at all, see llvm/triple.

Optimize, then read the result

Looking at the optimized IR and the machine code is how you check that your generator produces what you meant. NewTargetMachine describes the target CPU; give the module the same triple and its data layout with SetTargetTriple and ApplyDataLayout, then RunOpt runs LLVM's standard O2 pipeline on it in place. ModuleToString prints the module as text and EmitAsm writes assembly to a file.

llvm-guide-opt.salam
import llvm
import io
import os
import str

func main:
    path := os.TempDir() + "/salam_guide_square.ll"
    _ := io.WriteFile(path, `define i32 @square_sum(i32 %n) {
entry:
  %a = alloca i32
  store i32 0, ptr %a
  %x = mul i32 %n, %n
  %old = load i32, ptr %a
  %new = add i32 %old, %x
  store i32 %new, ptr %a
  %r = load i32, ptr %a
  ret i32 %r
}
`)
    parsed := llvm.ParseIR(path)
    if not parsed.ok:
        println parsed.err
        ret 1
    end
    made := llvm.NewTargetMachine("x86_64-unknown-linux-gnu", llvm.OPT_O2, false)
    if not made.ok:
        println made.err
        ret 1
    end
    llvm.SetTargetTriple(parsed.mod, "x86_64-unknown-linux-gnu")
    llvm.ApplyDataLayout(parsed.mod, made.tm)
    opt := llvm.RunOpt(parsed.mod, made.tm, llvm.OPT_O2)
    if not opt.ok:
        println opt.err
        ret 1
    end
    text := llvm.ModuleToString(parsed.mod)
    each line in str.Split(text, "\n"):
        if str.StartsWith(line, "define") or str.StartsWith(line, "  "):
            println line
        end
    end
    asm := os.TempDir() + "/salam_guide_square.s"
    emitted := llvm.EmitAsm(parsed.mod, made.tm, asm)
    if not emitted.ok:
        println emitted.err
        ret 1
    end
    each line in str.Split(io.ReadFile(asm), "\n"):
        if str.StartsWith(line, "\t") and not str.StartsWith(line, "\t."):
            println str.Trim(str.Replace(line, "\t", " "))
        end
    end
    llvm.FreeTargetMachine(made.tm)
    llvm.FreeModule(parsed.mod)
    llvm.FreeContext(parsed.ctx)
end
define i32 @square_sum(i32 %n) local_unnamed_addr #0 {
  %x = mul i32 %n, %n
  ret i32 %x
movl %edi, %eax
imull %edi, %eax
retq

The input stores through a stack slot the way a naive code generator would. The optimizer turns the slot into registers, drops the add of zero and leaves one multiply, which becomes two x86 instructions. A front end can emit simple, slow-looking code like this and leave the cleanup to LLVM; that is the usual design. OPT_OS optimizes for size instead, and EmitObject writes a .o instead of assembly.

llvm-Target.salam
import llvm
import os

func build(ctx: llvm.Context): llvm.Module:
    mod := llvm.NewModule(ctx, "calc")
    b := llvm.NewBuilder(ctx)
    i32 := llvm.Int32Type(ctx)
    mut two := Vector {} as Vector<llvm.Type>
    two.push(i32)
    two.push(i32)
    fn := llvm.AddFunction(mod, "salam_add", llvm.FunctionType(i32, two, false))
    llvm.PositionAtEnd(b, llvm.AppendBlock(ctx, fn, "entry"))
    _ := llvm.Ret(b, llvm.Add(b, llvm.Param(fn, 0), llvm.Param(fn, 1), "s"))
    llvm.FreeBuilder(b)
    ret mod
end

func main:
    llvm.InitTargets()
    println llvm.DefaultTriple().len() > 0
    println llvm.CodeGenLevelOf(llvm.OPT_O2) == llvm.CODEGEN_LEVEL_DEFAULT, llvm.PassPipelineOf(llvm.OPT_O3)
    println llvm.OPT_O0, llvm.OPT_O1, llvm.OPT_OS, llvm.OPT_OZ
    println llvm.CODEGEN_LEVEL_NONE, llvm.CODEGEN_LEVEL_LESS, llvm.CODEGEN_LEVEL_AGGRESSIVE
    println llvm.CODEGEN_FILE_ASSEMBLY, llvm.CODEGEN_FILE_OBJECT, llvm.RELOC_DEFAULT, llvm.RELOC_PIC, llvm.CODE_MODEL_DEFAULT

    machine := llvm.NewTargetMachine("", llvm.OPT_O2, false)
    if not machine.ok:
        println "the target machine was not built:", machine.err
        ret
    end
    ctx := llvm.NewContext()
    mod := build(ctx)
    llvm.ApplyDataLayout(mod, machine.tm)
    optimized := llvm.RunOpt(mod, machine.tm, llvm.OPT_O2)
    println optimized.ok
    dir := os.TempDir()
    println llvm.EmitAsm(mod, machine.tm, dir + "/salam_add_en.s").ok
    println llvm.EmitObject(mod, machine.tm, dir + "/salam_add_en.o").ok
    println llvm.EmitBitcode(mod, dir + "/salam_add_en.bc").ok

    println llvm.ParallelJobs() > 0
    parts := llvm.ParallelPartitions(mod, 4)
    println parts >= 1
    mod2 := build(ctx)
    parallel := llvm.OptimizeEmitParallel(mod2, machine.tm, llvm.OPT_O2, 2, dir + "/salam_part_en")
    println parallel.ok, parallel.objects.len() >= 1
    llvm.FreeModule(mod)
    llvm.FreeModule(mod2)
    llvm.FreeTargetMachine(machine.tm)
    llvm.FreeContext(ctx)
end

llvm-Build.salam
import llvm
import io
import os

func main:
    ctx := llvm.NewContext()
    mod := llvm.NewModule(ctx, "fib")
    b := llvm.NewBuilder(ctx)
    i32 := llvm.Int32Type(ctx)

    none := Vector {} as Vector<llvm.Type>
    main_type := llvm.FunctionType(i32, none, false)
    entry_fn := llvm.AddFunction(mod, "main", main_type)
    start := llvm.AppendBlock(ctx, entry_fn, "entry")
    body := llvm.AppendBlock(ctx, entry_fn, "loop")
    tail := llvm.AppendBlock(ctx, entry_fn, "done")

    llvm.PositionAtEnd(b, start)
    _ := llvm.Br(b, body)
    llvm.PositionAtEnd(b, body)
    counter := llvm.Phi(b, i32, "i")
    a := llvm.Phi(b, i32, "a")
    c := llvm.Phi(b, i32, "b")
    total := llvm.Add(b, a, c, "sum")
    next := llvm.Add(b, counter, llvm.ConstInt(i32, 1), "inext")
    more := llvm.ICmp(b, llvm.INT_SLT, next, llvm.ConstInt(i32, 10), "more")
    _ := llvm.CondBr(b, more, body, tail)

    mut sources := Vector {} as Vector<llvm.BasicBlock>
    sources.push(start)
    sources.push(body)
    mut counter_in := Vector {} as Vector<llvm.Value>
    counter_in.push(llvm.ConstInt(i32, 0))
    counter_in.push(next)
    llvm.AddIncoming(counter, counter_in, sources)
    mut a_in := Vector {} as Vector<llvm.Value>
    a_in.push(llvm.ConstInt(i32, 0))
    a_in.push(c)
    llvm.AddIncoming(a, a_in, sources)
    mut c_in := Vector {} as Vector<llvm.Value>
    c_in.push(llvm.ConstInt(i32, 1))
    c_in.push(total)
    llvm.AddIncoming(c, c_in, sources)
    llvm.PositionAtEnd(b, tail)
    _ := llvm.Ret(b, c)

    checked := llvm.Verify(mod)
    println checked.ok
    print llvm.ModuleToString(mod)
    path := os.TempDir() + "/salam_fib_en.ll"
    _ := io.WriteFile(path, llvm.ModuleToString(mod))
    r := llvm.RunJIT(path)
    println r.ok, r.rc
    llvm.FreeBuilder(b)
    llvm.FreeModule(mod)
    llvm.FreeContext(ctx)
end

llvm-Run.salam
import llvm
import io
import os
import mem

func main:
    println llvm.Available()
    dir := os.TempDir()
    ll := dir + "/salam_args_en.ll"
    _ := io.WriteFile(ll, `define i32 @main(i32 %argc, ptr %argv) {
entry:
  %n = mul i32 %argc, 10
  ret i32 %n
}
`)

    parsed := llvm.ParseIR(ll)
    println parsed.ok
    checked := llvm.Verify(parsed.mod)
    println checked.ok, llvm.VERIFY_RETURN_STATUS
    llvm.SetTargetTriple(parsed.mod, llvm.DefaultTriple())
    println llvm.TargetTripleOf(parsed.mod) == llvm.DefaultTriple()
    llvm.FreeModule(parsed.mod)
    llvm.FreeContext(parsed.ctx)

    mut args := Vector {} as Vector<str>
    args.push("a")
    args.push("b")
    println llvm.RunJITArgs(ll, args).rc

    mut opts := llvm.OptionsDefault()
    println opts.opt_level, opts.output_mode
    opts.output_mode = llvm.OUT_JIT
    opts.run_args = args
    println llvm.Run(ll, opts).rc
    opts.output_mode = llvm.OUT_OBJ
    opts.output_file = dir + "/salam_args_en.o"
    println llvm.Run(ll, opts).ok
    opts.output_mode = llvm.OUT_ASM
    opts.output_file = dir + "/salam_args_en.s"
    println llvm.Run(ll, opts).ok
    println llvm.OUT_IR, llvm.OUT_BITCODE, llvm.OUT_EXEC

    println llvm.HostLinkTriple(), llvm.LldAvailable()
    println llvm.CanLinkTriple("x86_64-unknown-linux-musl"), llvm.CanLinkTriple("x86_64-w64-windows-gnu"), llvm.CanLinkTriple("aarch64-apple-darwin")
    println llvm.LLD_ELF, llvm.LLD_MINGW, llvm.LLD_COFF, llvm.LLD_MACHO, llvm.LLD_WASM
    linked := llvm.LinkExternal(dir + "/salam_args_en.o", dir + "/salam_args_en", llvm.OptionsDefault())
    println linked.ok, linked.err
    if linked.ok:
        println os.Run(dir + "/salam_args_en x y")
    end

    mut musl := llvm.OptionsDefault()
    musl.target_triple = "x86_64-unknown-linux-musl"
    println llvm.LinkAuto(dir + "/salam_args_en.o", dir + "/salam_args_en_musl", musl).ok
    _ := llvm.LinkElf(dir + "/a.o", dir + "/a", musl)
    _ := llvm.LinkMingw(dir + "/a.o", dir + "/a.exe", musl)
    _ := llvm.LinkWasi(dir + "/a.o", dir + "/a.wasm", musl)
    mut argv := mem.Allocate(64 as u64) as str*
    argv[0] = "ld.lld"
    argv[1] = "--version"
    println llvm.Link(llvm.LLD_ELF, argv as void*, 2)
    mem.Free(argv as void*)
end

import llvm
import io
import os

func main:
    ctx := llvm.NewContext()
    mod := llvm.NewModule(ctx, "hello")
    b := llvm.NewBuilder(ctx)
    i32 := llvm.Int32Type(ctx)
    i64 := llvm.Int64Type(ctx)
    f64 := llvm.DoubleType(ctx)
    ptr := llvm.PointerType(ctx)
    _ := llvm.Int1Type(ctx)
    _ := llvm.Int8Type(ctx)
    _ := llvm.Int16Type(ctx)
    _ := llvm.FloatType(ctx)
    _ := llvm.ArrayType(i32, 4)

    mut fmt_params := Vector {} as Vector<llvm.Type>
    fmt_params.push(ptr)
    fmt_type := llvm.FunctionType(i32, fmt_params, true)
    _ := llvm.AddFunction(mod, "printf", fmt_type)
    printer := llvm.NamedFunction(mod, "printf")

    mut two_numbers := Vector {} as Vector<llvm.Type>
    two_numbers.push(i64)
    two_numbers.push(i64)
    calc_type := llvm.FunctionType(i64, two_numbers, false)
    calc := llvm.AddFunction(mod, "calc", calc_type)
    llvm.SetLinkage(calc, llvm.LINKAGE_INTERNAL)
    llvm.PositionAtEnd(b, llvm.AppendBlock(ctx, calc, "entry"))
    l := llvm.Param(calc, 0)
    r := llvm.Param(calc, 1)
    t := llvm.Sub(b, llvm.Mul(b, l, r, "m"), llvm.SDiv(b, l, r, "d"), "t")
    _ := llvm.Ret(b, llvm.Add(b, t, llvm.SRem(b, l, r, "r"), "x"))

    count := llvm.AddGlobal(mod, i64, "count")
    llvm.SetInitializer(count, llvm.ConstInt(i64, 40))
    pi := llvm.AddGlobal(mod, f64, "pi")
    llvm.SetInitializer(pi, llvm.ConstReal(f64, 3.14))
    llvm.SetGlobalConstant(pi, true)

    none := Vector {} as Vector<llvm.Type>
    entry_fn := llvm.AddFunction(mod, "main", llvm.FunctionType(i32, none, false))
    llvm.PositionAtEnd(b, llvm.AppendBlock(ctx, entry_fn, "entry"))
    println llvm.InsertBlock(b).handle != null
    slot := llvm.Alloca(b, i64, "n")
    _ := llvm.Store(b, llvm.Load(b, i64, count, "c"), slot)
    mut args := Vector {} as Vector<llvm.Value>
    args.push(llvm.Load(b, i64, slot, "n1"))
    args.push(llvm.ConstInt(i64, 8))
    result := llvm.Call(b, calc_type, calc, args, "v")
    mut fmt_args := Vector {} as Vector<llvm.Value>
    fmt_args.push(llvm.GlobalString(b, "calc(40, 8) = %lld\n", "fmt"))
    fmt_args.push(result)
    _ := llvm.Call(b, fmt_type, printer, fmt_args, "")
    _ := llvm.Ret(b, llvm.ConstInt(i32, 0))

    empty := Vector {} as Vector<llvm.Type>
    noop := llvm.AddFunction(mod, "noop", llvm.FunctionType(llvm.VoidType(ctx), empty, false))
    llvm.SetLinkage(noop, llvm.LINKAGE_PRIVATE)
    llvm.PositionAtEnd(b, llvm.AppendBlock(ctx, noop, "entry"))
    _ := llvm.RetVoid(b)
    println llvm.LINKAGE_EXTERNAL, llvm.Verify(mod).ok
    println llvm.INT_EQ, llvm.INT_NE, llvm.INT_UGT, llvm.INT_UGE, llvm.INT_ULT, llvm.INT_ULE
    println llvm.INT_SGT, llvm.INT_SGE, llvm.INT_SLT, llvm.INT_SLE

    path := os.TempDir() + "/salam_hello_en.ll"
    _ := io.WriteFile(path, llvm.ModuleToString(mod))
    println llvm.RunJIT(path).rc
    llvm.FreeBuilder(b)
    llvm.FreeModule(mod)
    llvm.FreeContext(ctx)
end
