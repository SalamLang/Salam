llvm

Building, optimizing, running and linking code with LLVM.

import llvm
On this page
  1. Types
  2. Constants and variables
  3. Functions

LLVM is the compiler infrastructure that also backs Salam. With this package you can build your own compiler or interpreter: make a Context and a Module, write the instructions with a Builder, test the module with Verify and then run it with the JIT or turn it into an object file.

If you have the IR as text, Run does the whole pipeline with one Options. For linking, LLD works inside the program itself for Linux with musl, Windows with MinGW and WASI; for the rest LinkExternal calls the system linker.

This package needs the libsalam_llvm library. The examples on the site are only checked.

llvm-Target.salam
import llvm
import os

func build(ctx: llvm.Context): llvm.Module:
    mod := llvm.NewModule(ctx, "calc")
    b := llvm.NewBuilder(ctx)
    i32 := llvm.Int32Type(ctx)
    mut two := Vector {} as Vector<llvm.Type>
    two.push(i32)
    two.push(i32)
    fn := llvm.AddFunction(mod, "salam_add", llvm.FunctionType(i32, two, false))
    llvm.PositionAtEnd(b, llvm.AppendBlock(ctx, fn, "entry"))
    _ := llvm.Ret(b, llvm.Add(b, llvm.Param(fn, 0), llvm.Param(fn, 1), "s"))
    llvm.FreeBuilder(b)
    ret mod
end

func main:
    llvm.InitTargets()
    println llvm.DefaultTriple().len() > 0
    println llvm.CodeGenLevelOf(llvm.OPT_O2) == llvm.CODEGEN_LEVEL_DEFAULT, llvm.PassPipelineOf(llvm.OPT_O3)
    println llvm.OPT_O0, llvm.OPT_O1, llvm.OPT_OS, llvm.OPT_OZ
    println llvm.CODEGEN_LEVEL_NONE, llvm.CODEGEN_LEVEL_LESS, llvm.CODEGEN_LEVEL_AGGRESSIVE
    println llvm.CODEGEN_FILE_ASSEMBLY, llvm.CODEGEN_FILE_OBJECT, llvm.RELOC_DEFAULT, llvm.RELOC_PIC, llvm.CODE_MODEL_DEFAULT

    machine := llvm.NewTargetMachine("", llvm.OPT_O2, false)
    if not machine.ok:
        println "the target machine was not built:", machine.err
        ret
    end
    ctx := llvm.NewContext()
    mod := build(ctx)
    llvm.ApplyDataLayout(mod, machine.tm)
    optimized := llvm.RunOpt(mod, machine.tm, llvm.OPT_O2)
    println optimized.ok
    dir := os.TempDir()
    println llvm.EmitAsm(mod, machine.tm, dir + "/salam_add_en.s").ok
    println llvm.EmitObject(mod, machine.tm, dir + "/salam_add_en.o").ok
    println llvm.EmitBitcode(mod, dir + "/salam_add_en.bc").ok

    println llvm.ParallelJobs() > 0
    parts := llvm.ParallelPartitions(mod, 4)
    println parts >= 1
    mod2 := build(ctx)
    parallel := llvm.OptimizeEmitParallel(mod2, machine.tm, llvm.OPT_O2, 2, dir + "/salam_part_en")
    println parallel.ok, parallel.objects.len() >= 1
    llvm.FreeModule(mod)
    llvm.FreeModule(mod2)
    llvm.FreeTargetMachine(machine.tm)
    llvm.FreeContext(ctx)
end

llvm-Build.salam
import llvm
import io
import os

func main:
    ctx := llvm.NewContext()
    mod := llvm.NewModule(ctx, "fib")
    b := llvm.NewBuilder(ctx)
    i32 := llvm.Int32Type(ctx)

    none := Vector {} as Vector<llvm.Type>
    main_type := llvm.FunctionType(i32, none, false)
    entry_fn := llvm.AddFunction(mod, "main", main_type)
    start := llvm.AppendBlock(ctx, entry_fn, "entry")
    body := llvm.AppendBlock(ctx, entry_fn, "loop")
    tail := llvm.AppendBlock(ctx, entry_fn, "done")

    llvm.PositionAtEnd(b, start)
    _ := llvm.Br(b, body)
    llvm.PositionAtEnd(b, body)
    counter := llvm.Phi(b, i32, "i")
    a := llvm.Phi(b, i32, "a")
    c := llvm.Phi(b, i32, "b")
    total := llvm.Add(b, a, c, "sum")
    next := llvm.Add(b, counter, llvm.ConstInt(i32, 1), "inext")
    more := llvm.ICmp(b, llvm.INT_SLT, next, llvm.ConstInt(i32, 10), "more")
    _ := llvm.CondBr(b, more, body, tail)

    mut sources := Vector {} as Vector<llvm.BasicBlock>
    sources.push(start)
    sources.push(body)
    mut counter_in := Vector {} as Vector<llvm.Value>
    counter_in.push(llvm.ConstInt(i32, 0))
    counter_in.push(next)
    llvm.AddIncoming(counter, counter_in, sources)
    mut a_in := Vector {} as Vector<llvm.Value>
    a_in.push(llvm.ConstInt(i32, 0))
    a_in.push(c)
    llvm.AddIncoming(a, a_in, sources)
    mut c_in := Vector {} as Vector<llvm.Value>
    c_in.push(llvm.ConstInt(i32, 1))
    c_in.push(total)
    llvm.AddIncoming(c, c_in, sources)
    llvm.PositionAtEnd(b, tail)
    _ := llvm.Ret(b, c)

    checked := llvm.Verify(mod)
    println checked.ok
    print llvm.ModuleToString(mod)
    path := os.TempDir() + "/salam_fib_en.ll"
    _ := io.WriteFile(path, llvm.ModuleToString(mod))
    r := llvm.RunJIT(path)
    println r.ok, r.rc
    llvm.FreeBuilder(b)
    llvm.FreeModule(mod)
    llvm.FreeContext(ctx)
end

llvm-Run.salam
import llvm
import io
import os
import mem

func main:
    println llvm.Available()
    dir := os.TempDir()
    ll := dir + "/salam_args_en.ll"
    _ := io.WriteFile(ll, `define i32 @main(i32 %argc, ptr %argv) {
entry:
  %n = mul i32 %argc, 10
  ret i32 %n
}
`)

    parsed := llvm.ParseIR(ll)
    println parsed.ok
    checked := llvm.Verify(parsed.mod)
    println checked.ok, llvm.VERIFY_RETURN_STATUS
    llvm.SetTargetTriple(parsed.mod, llvm.DefaultTriple())
    println llvm.TargetTripleOf(parsed.mod) == llvm.DefaultTriple()
    llvm.FreeModule(parsed.mod)
    llvm.FreeContext(parsed.ctx)

    mut args := Vector {} as Vector<str>
    args.push("a")
    args.push("b")
    println llvm.RunJITArgs(ll, args).rc

    mut opts := llvm.OptionsDefault()
    println opts.opt_level, opts.output_mode
    opts.output_mode = llvm.OUT_JIT
    opts.run_args = args
    println llvm.Run(ll, opts).rc
    opts.output_mode = llvm.OUT_OBJ
    opts.output_file = dir + "/salam_args_en.o"
    println llvm.Run(ll, opts).ok
    opts.output_mode = llvm.OUT_ASM
    opts.output_file = dir + "/salam_args_en.s"
    println llvm.Run(ll, opts).ok
    println llvm.OUT_IR, llvm.OUT_BITCODE, llvm.OUT_EXEC

    println llvm.HostLinkTriple(), llvm.LldAvailable()
    println llvm.CanLinkTriple("x86_64-unknown-linux-musl"), llvm.CanLinkTriple("x86_64-w64-windows-gnu"), llvm.CanLinkTriple("aarch64-apple-darwin")
    println llvm.LLD_ELF, llvm.LLD_MINGW, llvm.LLD_COFF, llvm.LLD_MACHO, llvm.LLD_WASM
    linked := llvm.LinkExternal(dir + "/salam_args_en.o", dir + "/salam_args_en", llvm.OptionsDefault())
    println linked.ok, linked.err
    if linked.ok:
        println os.Run(dir + "/salam_args_en x y")
    end

    mut musl := llvm.OptionsDefault()
    musl.target_triple = "x86_64-unknown-linux-musl"
    println llvm.LinkAuto(dir + "/salam_args_en.o", dir + "/salam_args_en_musl", musl).ok
    _ := llvm.LinkElf(dir + "/a.o", dir + "/a", musl)
    _ := llvm.LinkMingw(dir + "/a.o", dir + "/a.exe", musl)
    _ := llvm.LinkWasi(dir + "/a.o", dir + "/a.wasm", musl)
    mut argv := mem.Allocate(64 as u64) as str*
    argv[0] = "ld.lld"
    argv[1] = "--version"
    println llvm.Link(llvm.LLD_ELF, argv as void*, 2)
    mem.Free(argv as void*)
end

INT_EQ

const INT_EQ: i32 = 32
Integer comparison predicate: equal.
llvm-Program.salam
import llvm
import io
import os

func main:
    ctx := llvm.NewContext()
    mod := llvm.NewModule(ctx, "hello")
    b := llvm.NewBuilder(ctx)
    i32 := llvm.Int32Type(ctx)
    i64 := llvm.Int64Type(ctx)
    f64 := llvm.DoubleType(ctx)
    ptr := llvm.PointerType(ctx)
    _ := llvm.Int1Type(ctx)
    _ := llvm.Int8Type(ctx)
    _ := llvm.Int16Type(ctx)
    _ := llvm.FloatType(ctx)
    _ := llvm.ArrayType(i32, 4)

    mut fmt_params := Vector {} as Vector<llvm.Type>
    fmt_params.push(ptr)
    fmt_type := llvm.FunctionType(i32, fmt_params, true)
    _ := llvm.AddFunction(mod, "printf", fmt_type)
    printer := llvm.NamedFunction(mod, "printf")

    mut two_numbers := Vector {} as Vector<llvm.Type>
    two_numbers.push(i64)
    two_numbers.push(i64)
    calc_type := llvm.FunctionType(i64, two_numbers, false)
    calc := llvm.AddFunction(mod, "calc", calc_type)
    llvm.SetLinkage(calc, llvm.LINKAGE_INTERNAL)
    llvm.PositionAtEnd(b, llvm.AppendBlock(ctx, calc, "entry"))
    l := llvm.Param(calc, 0)
    r := llvm.Param(calc, 1)
    t := llvm.Sub(b, llvm.Mul(b, l, r, "m"), llvm.SDiv(b, l, r, "d"), "t")
    _ := llvm.Ret(b, llvm.Add(b, t, llvm.SRem(b, l, r, "r"), "x"))

    count := llvm.AddGlobal(mod, i64, "count")
    llvm.SetInitializer(count, llvm.ConstInt(i64, 40))
    pi := llvm.AddGlobal(mod, f64, "pi")
    llvm.SetInitializer(pi, llvm.ConstReal(f64, 3.14))
    llvm.SetGlobalConstant(pi, true)

    none := Vector {} as Vector<llvm.Type>
    entry_fn := llvm.AddFunction(mod, "main", llvm.FunctionType(i32, none, false))
    llvm.PositionAtEnd(b, llvm.AppendBlock(ctx, entry_fn, "entry"))
    println llvm.InsertBlock(b).handle != null
    slot := llvm.Alloca(b, i64, "n")
    _ := llvm.Store(b, llvm.Load(b, i64, count, "c"), slot)
    mut args := Vector {} as Vector<llvm.Value>
    args.push(llvm.Load(b, i64, slot, "n1"))
    args.push(llvm.ConstInt(i64, 8))
    result := llvm.Call(b, calc_type, calc, args, "v")
    mut fmt_args := Vector {} as Vector<llvm.Value>
    fmt_args.push(llvm.GlobalString(b, "calc(40, 8) = %lld\n", "fmt"))
    fmt_args.push(result)
    _ := llvm.Call(b, fmt_type, printer, fmt_args, "")
    _ := llvm.Ret(b, llvm.ConstInt(i32, 0))

    empty := Vector {} as Vector<llvm.Type>
    noop := llvm.AddFunction(mod, "noop", llvm.FunctionType(llvm.VoidType(ctx), empty, false))
    llvm.SetLinkage(noop, llvm.LINKAGE_PRIVATE)
    llvm.PositionAtEnd(b, llvm.AppendBlock(ctx, noop, "entry"))
    _ := llvm.RetVoid(b)
    println llvm.LINKAGE_EXTERNAL, llvm.Verify(mod).ok
    println llvm.INT_EQ, llvm.INT_NE, llvm.INT_UGT, llvm.INT_UGE, llvm.INT_ULT, llvm.INT_ULE
    println llvm.INT_SGT, llvm.INT_SGE, llvm.INT_SLT, llvm.INT_SLE

    path := os.TempDir() + "/salam_hello_en.ll"
    _ := io.WriteFile(path, llvm.ModuleToString(mod))
    println llvm.RunJIT(path).rc
    llvm.FreeBuilder(b)
    llvm.FreeModule(mod)
    llvm.FreeContext(ctx)
end
