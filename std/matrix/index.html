matrix

Linear algebra: matrices, systems of equations, decompositions, eigenvalues and statistics.

import matrix
On this page
  1. Types
  2. Constants and variables
  3. Functions

This package is complete linear algebra over 64 bit floats: from making a matrix and multiplying through to solving systems, decompositions, eigenvalues and singular values, and statistical tools such as principal component analysis. To solve one system, call Solve; for several systems with one matrix, Decompose once and LUSolve many times.

Matrices hold floats and the results carry a little rounding error; compare with AllClose and display with Chop. For neural networks see tensor.

import matrix

func build(rows: int, cols: int, numbers: str): matrix.Matrix:
    mut data := Vector {} as Vector<f64>
    each number in numbers.split(" "):
        data.push(number.to_float())
    end
    ret matrix.FromArray(rows, cols, data)
end

func show(name: str, m &: matrix.Matrix):
    println name, matrix.ToRepr(m, 1)
end

func main:
    mut a := build(2, 3, "1 2 3 4 5 6")
    mut square := build(3, 3, "1 2 3 4 5 6 7 8 9")
    show("Reshape", matrix.Reshape(a, 3, 2))
    show("Flatten", matrix.Flatten(a))
    show("FlattenCol", matrix.FlattenCol(a))
    mut row := matrix.Row(a, 1)
    mut col := matrix.Col(a, 2)
    show("Row", row)
    show("Col", col)
    mut b := matrix.Clone(a)
    matrix.SetRow(b, 0, row)
    matrix.SetCol(b, 0, col)
    show("SetRow and SetCol", b)
    show("SubMatrix", matrix.SubMatrix(square, 1, 1, 2, 2))
    show("Minor", matrix.Minor(square, 0, 0))
    show("DeleteRow", matrix.DeleteRow(square, 1))
    show("DeleteCol", matrix.DeleteCol(square, 1))
    show("InsertRow", matrix.InsertRow(a, 1, matrix.Row(a, 0)))
    show("InsertCol", matrix.InsertCol(a, 0, matrix.Col(a, 2)))
    matrix.SwapRows(b, 0, 1)
    matrix.SwapCols(b, 0, 2)
    show("swapped", b)
    show("Diagonal", matrix.Diagonal(square))
    show("DiagonalOffset", matrix.DiagonalOffset(square, 1))
    mut c := matrix.Clone(square)
    matrix.SetDiagonal(c, matrix.Ones(3, 1))
    matrix.ShiftDiagonal(c, 10.0)
    show("new diagonal", matrix.Diagonal(c))
    show("HStack", matrix.HStack(a, a))
    show("VStack", matrix.VStack(a, a))
    show("Augment", matrix.Augment(square, matrix.Eye(3)))
