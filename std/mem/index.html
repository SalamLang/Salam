mem

Working with memory directly: allocating and freeing, copying, arenas and pools, and leak checking.

import mem
On this page
  1. Types
  2. Functions

Most programs never need this package: vectors, maps and strings look after their own memory. The mem package is for low-level work, such as talking to a C library or building a data structure of your own. Everything you take with Allocate has to be given back with Free.

An arena is one large block you take pieces off the front of and then release all at once; it is very fast for work with many small objects that share a lifetime. A pool hands out and takes back blocks of one fixed size.

Types

Arena

struct Arena
A memory arena.
NameTypeDescription
bufvoid*The address of the arena's memory.
capu64 = 0The capacity in bytes.
usedu64 = 0How much has been taken.
mem-Arena.salam
import mem

func main:
    mut arena := mem.NewArena(1024 as u64)
    println mem.ArenaCapacity(arena), mem.ArenaUsed(arena)
    a := mem.ArenaAlloc(arena, 100 as u64)
    b := mem.ArenaAllocZeroed(arena, 200 as u64)
    println a != null, b != null, mem.ArenaUsed(arena) >= 300, mem.ArenaRemaining(arena) <= 724
    println mem.ArenaAlloc(arena, 4000 as u64) == null
    mem.ArenaReset(arena)
    println mem.ArenaUsed(arena), arena.cap
    mem.ArenaFree(arena)
end
Output1024 0 true true true true true 0 1024

Pool

struct Pool
A pool of equally sized blocks.
NameTypeDescription
block_sizeu64 = 0The size of each block.
block_countu64 = 0The number of blocks.
blocksvoid*The address of the blocks' memory.
free_listvoid*The list of free blocks.
mem-Pool.salam
import mem

func main:
    mut pool := mem.NewPool(32 as u64, 4 as u64)
    println mem.PoolBlockSize(pool), mem.PoolCapacity(pool)
    b1 := mem.PoolAlloc(pool)
    b2 := mem.PoolAlloc(pool)
    println b1 != null, b2 != null, b1 != b2
    mem.PoolFree(pool, b1)
    b3 := mem.PoolAlloc(pool)
    println b3 == b1
    mem.PoolDestroy(pool)
end
Output32 4 true true true true

Allocate

func Allocate(size: u64): void*
Takes a piece of memory of the given size; its contents are undefined.
mem-Allocate.salam
import mem

func main:
    b := mem.Allocate(8 as u64)
    mem.Set(b, 0, 8 as u64)
    bytes := b as i8*
    bytes[0] = 65 as i8
    bytes[1] = 66 as i8
    b2 := mem.AllocateZeroed(8 as u64)
    mem.Copy(b2, b, 2 as u64)
    bytes2 := b2 as i8*
    println bytes2[0], bytes2[1], bytes2[2]
    mem.MemMove(b, (b as i64 + 1) as void*, 1 as u64)
    println bytes[0]
    b3 := mem.Reallocate(b, 64 as u64)
    arr := mem.AllocateArray(4 as u64, 8 as u64)
    arr64 := arr as i64*
    println arr64[3]
    mem.Free(b3)
    mem.Free(b2)
    mem.Free(arr)
end
Output65 66 0 66 0
