ml

Classic machine learning: preprocessing, regression, classification, trees, clustering and evaluation.

import ml
On this page
  1. Types
  2. Constants and variables
  3. Functions

This package gives you the classic machine learning algorithms behind an interface much like scikit-learn. The data is a matrix where each row is a sample and each column a feature, and the labels are a one-column matrix. Every model is a struct: build it with its parameters, train it with fit and use it with predict.

Before training, prepare the data with StandardScaler and SimpleImputer and split it with TrainTestSplit; after that, measure the model with metrics such as Accuracy, F1 and R2Score, and pick the parameters with GridSearch. For neural networks see nn.

import ml
import matrix

func build(rows: int, cols: int, numbers: str): matrix.Matrix:
    mut data := Vector {} as Vector<f64>
    each number in numbers.split(" "):
        data.push(number.to_float())
    end
    ret matrix.FromArray(rows, cols, data)
end

func f3(number: f64): str:
    ret matrix.ToStringPrec(matrix.Full(1, 1, number), 3)
end

func neighbors(k: int, train: matrix.Matrix, labels: matrix.Matrix, test: matrix.Matrix): matrix.Matrix:
    mut model := ml.KNNClassifier { k = k }
    _ := model.fit(train, labels)
    pred := model.predict(test)
    model.free()
    ret pred
end

func main:
    mut x := build(8, 2, "1 1 1 2 2 1 2 2 6 6 6 7 7 6 7 7")
    mut y := build(8, 1, "0 0 0 0 1 1 1 1")
    mut folds := ml.StratifiedKFold(y, 4, 1)
    mut scores := ml.CrossValScore((a: matrix.Matrix, b: matrix.Matrix, c: matrix.Matrix): ret neighbors(1, a, b, c) end, (a: matrix.Matrix, b: matrix.Matrix): ret ml.Accuracy(a, b) end, x, y, folds)
    println scores.len(), f3(ml.MeanScore(scores))
    mut parallel := ml.CrossValScoreParallel((a: matrix.Matrix, b: matrix.Matrix, c: matrix.Matrix): ret neighbors(1, a, b, c) end, (a: matrix.Matrix, b: matrix.Matrix): ret ml.Accuracy(a, b) end, x, y, folds, 2)
    println f3(ml.MeanScore(parallel))

    mut candidates := Vector {} as Vector<ml.Candidate>
    candidates.push(ml.Candidate { name = "k=1", fit = (a: matrix.Matrix, b: matrix.Matrix, c: matrix.Matrix): ret neighbors(1, a, b, c) end })
    candidates.push(ml.Candidate { name = "k=3", fit = (a: matrix.Matrix, b: matrix.Matrix, c: matrix.Matrix): ret neighbors(3, a, b, c) end })
    mut result := ml.GridSearch(candidates, (a: matrix.Matrix, b: matrix.Matrix): ret ml.Accuracy(a, b) end, x, y, folds, 1)
    println result.scores.len(), result.best >= 0, f3(result.best_score), candidates.get(result.best).name.len() > 0
end
