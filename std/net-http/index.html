net/http

HTTP servers and clients: handlers, text and JSON responses, cookies and sending requests.

import net.http

This package has both halves of HTTP/1.1: a server for web pages and JSON APIs, and a client for calling other services. The server hands each request to a net/router router, which picks the handler by method and path. Every handler takes one i64, the context id of the request; you read the request and write the response with the context functions below, or turn the id into a Context struct with Ctx and call the same names on it.

The packages around it:

  • net/http/middleware for access logs, CORS and cache headers, net/http/session for logins with a cookie, net/http/validate for checking input, and net/http/swagger for API documentation;
  • net/httpx when a client needs hand-written headers, binary bodies or a time limit, net/http2 for HTTP/2, and net/websocket for connections that stay open.

Things people get wrong:

  • Register every route before Use. The server keeps a copy of the router.
  • Post, Put and Patch label a body as a form (application/x-www-form-urlencoded) unless you pass a Content-Type; send JSON with PostWith and that header.
  • The client does not follow redirects and has no time limit. A 302 comes back as a 302 with its Location header; for a request that must not hang, use net/httpx with a timeout.
  • Ok means a 2xx status. When the server could not be reached at all, Status is 0 and Error says why.
  • A value read from a request is freed when the handler returns; see the end of the page.

The examples run the server with RunBackground on 127.0.0.1 and send requests to it from the same program.

A JSON API in one file

The most common job is a small JSON API: list some records, show one by its id, and create new ones. Register each route on the router with its handler, hand the router to the server with Use, and start it. Here the records are Salam structs: json.Marshal turns them into the response body, and json.UnmarshalLenient reads the posted body back into a struct, ignoring fields it does not know and leaving the missing ones at their defaults.

net-http-guide-api.salam
import net.http
import net.router
import encoding.json
import encoding.encerr
import conv

mut books := Vector {} as Vector<Book>

struct Book:
    @json "id"
    pub id: int = 0
    @json "title"
    pub title: str = ""
end

func list(ctx: i64):
    http.Json(ctx, json.Marshal(books))
end

func show(ctx: i64):
    id := conv.Atoi(http.Param(ctx, "id"))
    each b in books:
        if b.id == id:
            http.Json(ctx, json.Marshal(b))
            ret
        end
    end
    http.SetStatus(ctx, 404)
    http.Json(ctx, `{"error":"no such book"}`)
end

func create(ctx: i64):
    mut b := Book {}
    mut e := encerr.NoError()
    if not json.UnmarshalLenient(http.Body(ctx), b, e):
        http.SetStatus(ctx, 400)
        http.Json(ctx, "{" + json.Member("error", e.message) + "}")
        ret
    end
    b.id = books.len() + 1
    books.push(b)
    http.SetStatus(ctx, 201)
    http.SetHeader(ctx, "Location", "/books/" + b.id)
    http.Json(ctx, json.Marshal(b))
end

func main:
    mut r := router.NewRouter()
    router.Get(r, "/books", list)
    router.Get(r, "/books/:id", show)
    router.Post(r, "/books", create)
    mut s := http.NewServer(19761)
    http.Use(s, r)
    _ := http.RunBackground(s)
    base := "http://127.0.0.1:19761"
    a := http.Post(base + "/books", `{"title": "The Little Prince"}`)
    println http.Status(a), http.GetHeader(a, "location"), http.Body(a)
    _ := http.Post(base + "/books", `{"title": "Rumi"}`)
    println http.Body(http.Get(base + "/books"))
    println http.Body(http.Get(base + "/books/2"))
    c := http.Get(base + "/books/9")
    println http.Status(c), http.Body(c)
    d := http.Post(base + "/books", `{"title": 7}`)
    println http.Status(d), http.Body(d)
end
Output201 /books/1 {"id":1,"title":"The Little Prince"} [{"id":1,"title":"The Little Prince"},{"id":2,"title":"Rumi"}] {"id":2,"title":"Rumi"} 404 {"error":"no such book"} 400 {"error": "value does not match the expected type"}

Nothing is sent until the handler returns: the status, the headers and the body are collected and go out together, and a second body call replaces the first. That is why a handler returns right after it writes an error. A 201 with a Location header is the usual answer to a create, and a 404 with a short JSON error is easier for a client to handle than an empty page. Path parameters such as :id always arrive as text, so convert them with conv.Atoi before comparing.

HTML forms, query strings and redirects

A browser form arrives as application/x-www-form-urlencoded. Form reads one field and decodes it, so %40 becomes @ and + becomes a space. Query does the same for the query string. After a successful POST, answer with Redirect so that pressing reload in the browser does not send the form again.

net-http-guide-forms.salam
import net.http
import net.router
import template

mut messages := Vector {} as Vector<str>

func board(ctx: i64):
    mut html := "<ul>"
    each m in messages:
        html = html + "<li>" + template.EscapeHTML(m) + "</li>"
    end
    http.Html(ctx, html + "</ul>")
end

func post(ctx: i64):
    name := http.Form(ctx, "name")
    text := http.Form(ctx, "text")
    if name == "" or text == "":
        http.SetStatus(ctx, 400)
        http.Text(ctx, "name and text are required")
        ret
    end
    messages.push(name + ": " + text)
    http.Redirect(ctx, "/board")
end

func search(ctx: i64):
    word := http.Query(ctx, "q")
    mut found := 0
    each m in messages:
        if m.find(word) >= 0:
            found += 1
        end
    end
    http.Text(ctx, "'" + word + "' found in " + found + " message(s)")
end

func main:
    mut r := router.NewRouter()
    router.Get(r, "/board", board)
    router.Post(r, "/board", post)
    router.Get(r, "/search", search)
    mut s := http.NewServer(19762)
    http.Use(s, r)
    _ := http.RunBackground(s)
    base := "http://127.0.0.1:19762"
    mut form := http.NewHeaders()
    form.insert("name", "Sara")
    form.insert("text", "see you at 5 & bring tea")
    mut headers := http.NewHeaders()
    headers.insert("Content-Type", "application/x-www-form-urlencoded")
    a := http.PostWith(base + "/board", http.EncodeForm(form), headers)
    println http.Status(a), http.GetHeader(a, "location")
    b := http.PostWith(base + "/board", "name=Reza", headers)
    println http.Status(b), http.Body(b)
    println http.Body(http.Get(base + "/board"))
    mut q := http.NewHeaders()
    q.insert("q", "bring tea")
    println http.Body(http.Get(http.WithQuery(base + "/search", q)))
end
Output302 /board 400 name and text are required <ul><li>Sara: see you at 5 &amp; bring tea</li></ul> 'bring tea' found in 1 message(s)

Anything a user typed must be escaped before it goes into HTML; otherwise a message containing <script> runs in every visitor's browser. template.EscapeHTML does that, as the &amp; in the output shows. On the client side, EncodeForm builds a form body from a map and WithQuery adds a query string to a URL, both with correct escaping.

Route groups and a login check

router.Group puts a route under a common prefix, which is how an API is versioned under /api/v1. There is no separate middleware chain: a check that several handlers share is a plain function that returns true when the request may go on, or writes the error response itself and returns false. Each handler calls it on its first line and returns when it fails.

net-http-guide-groups.salam
import net.http
import net.router

func signed_in(ctx: i64): bool:
    if http.Header(ctx, "Authorization") == "Bearer s3cret":
        ret true
    end
    http.SetStatus(ctx, 401)
    http.SetHeader(ctx, "WWW-Authenticate", "Bearer")
    http.Json(ctx, `{"error":"sign in first"}`)
    ret false
end

func health(ctx: i64):
    http.Text(ctx, "ok")
end

func profile(ctx: i64):
    if not signed_in(ctx): ret end
    http.Json(ctx, `{"user":"maryam"}`)
end

func remove(ctx: i64):
    if not signed_in(ctx): ret end
    http.SetStatus(ctx, 204)
    http.Text(ctx, "")
end

func main:
    mut r := router.NewRouter()
    router.Get(r, "/health", health)
    router.Group(r, "/api/v1", "GET", "/profile", profile)
    router.Group(r, "/api/v1", "DELETE", "/posts/:id", remove)
    mut s := http.NewServer(19763)
    http.Use(s, r)
    _ := http.RunBackground(s)
    base := "http://127.0.0.1:19763"
    println http.Body(http.Get(base + "/health"))
    a := http.Get(base + "/api/v1/profile")
    println http.Status(a), http.GetHeader(a, "www-authenticate"), http.Body(a)
    mut auth := http.NewHeaders()
    auth.insert("Authorization", "Bearer s3cret")
    println http.Body(http.GetWith(base + "/api/v1/profile", auth))
    println http.Status(http.DeleteWith(base + "/api/v1/posts/3", auth))
    println http.Status(http.Get(base + "/profile"))
end
Outputok 401 Bearer {"error":"sign in first"} {"user":"maryam"} 204 404

A route that is not registered answers 404 on its own, so /profile without the prefix is not found. For a cookie based login instead of a token, see net/http/session.

Static files and server settings

router.Static serves a whole directory under a prefix, with the content type taken from the file extension. A path that tries to climb out of the directory with .. is refused, and a missing file is a 404. HideTokens drops the version number from the Server header, so the answer says only Salam.

net-http-guide-static.salam
import net.http
import net.router
import os

func main:
    _ := os.MkdirAll("public/css")
    _ := os.WriteFile("public/index.html", "<h1>Welcome</h1>")
    _ := os.WriteFile("public/css/site.css", "body { margin: 0 }")
    mut r := router.NewRouter()
    router.Static(r, "/site", "public")
    mut s := http.NewServer(19764)
    http.HideTokens(s)
    http.Use(s, r)
    _ := http.RunBackground(s)
    base := "http://127.0.0.1:19764"
    a := http.Get(base + "/site/index.html")
    println http.Status(a), http.GetHeader(a, "content-type"), http.Body(a)
    b := http.Get(base + "/site/css/site.css")
    println http.GetHeader(b, "content-type"), http.Body(b)
    println http.Status(http.Get(base + "/site/missing.png"))
    println http.Status(http.Get(base + "/site/../secret.txt"))
    println http.GetHeader(a, "server")
end
Output200 text/html; charset=utf-8 <h1>Welcome</h1> text/css body { margin: 0 } 404 404 Salam

Running a real server

The examples above use RunBackground so that one program can be both the server and its own client. A real service calls Run, which never returns. Read the port from the environment, listen on 0.0.0.0 when the server sits in a container or behind a proxy, and turn on the access log while you are getting it running.

net-http-guide-server.salam
import net.http
import net.router
import os
import conv

func home(ctx: i64):
    http.Html(ctx, "<h1>Salam</h1>")
end

func health(ctx: i64):
    http.Json(ctx, `{"status":"ok"}`)
end

func main:
    mut port := 8080
    if os.Env("PORT") != "":
        port = conv.Atoi(os.Env("PORT"))
    end
    http.SetMaxBodyBytes(1048576)
    http.SetKeepAliveIdle(5000)
    mut r := router.NewRouter()
    router.Get(r, "/", home)
    router.Get(r, "/health", health)
    router.Static(r, "/assets", "public")
    mut s := http.NewServer(port)
    http.SetHost(s, "0.0.0.0")
    http.HideTokens(s)
    http.EnableAccessLog(s)
    http.Use(s, r)
    println "listening on port " + port
    http.Run(s)
end

Run it and try it from a second terminal:

PORT=8080 salam run server.salam
curl http://127.0.0.1:8080/health

The limits are for the whole process: SetMaxBodyBytes and SetMaxHeaderBytes reject requests that are too large, SetKeepAliveIdle closes idle connections, and SetKeepAliveMaxRequests caps how many requests one connection may send. For HTTPS, call RunTLS with a certificate and key file instead of Run, or put the server behind a proxy that handles TLS.

Calling an API as a client

Get, Post and the other one-line calls send a request and return a Response; the ...With forms take a map of extra headers. The program below shows the four answers a client has to handle. A Post without a content type reaches the server as a form, so a JSON API refuses it with 415; the same body through PostWith with Content-Type: application/json creates the ticket. A redirect is not followed, so Get returns the 302 and you read Location yourself. A server that is not running gives status 0 and the reason in Error.

net-http-guide-client.salam
import net.http
import net.router
import encoding.json

func create_ticket(ctx: i64):
    kind := http.Header(ctx, "Content-Type")
    if kind != "application/json":
        http.SetStatus(ctx, 415)
        http.Json(ctx, "{\"error\":\"expected JSON, got " + kind + "\"}")
        ret
    end
    title := json.Get(http.Body(ctx), "title")
    http.SetStatus(ctx, 201)
    http.Json(ctx, "{\"id\":31,\"title\":" + json.Str(title) + "}")
end

func old_tickets(ctx: i64): http.Redirect(ctx, "/v2/tickets") end

func main:
    mut r := router.NewRouter()
    router.Post(r, "/tickets", create_ticket)
    router.Get(r, "/tickets", old_tickets)
    mut s := http.NewServer(18631)
    http.SetHost(s, "127.0.0.1")
    http.Use(s, r)
    if not http.RunBackground(s):
        println "port 18631 is busy"
        ret 1
    end
    base := "http://127.0.0.1:18631"
    body := `{"title": "Printer is out of paper"}`

    plain := http.Post(base + "/tickets", body)
    println "Post:", http.Status(plain), http.Body(plain)

    mut headers := http.NewHeaders()
    headers.insert("Content-Type", "application/json")
    made := http.PostWith(base + "/tickets", body, headers)
    println "PostWith:", http.Status(made), http.Ok(made), "id", json.GetInt(http.Body(made), "id")

    moved := http.Get(base + "/tickets")
    println "Get:", http.Status(moved), "Location:", http.GetHeader(moved, "location")

    down := http.Get("http://127.0.0.1:1/tickets")
    println "dead server:", http.Status(down), http.Ok(down), http.Error(down)
    ret 0
end
OutputPost: 415 {"error":"expected JSON, got application/x-www-form-urlencoded"} PostWith: 201 true id 31 Get: 302 Location: /v2/tickets dead server: 0 false could not connect to 127.0.0.1:1

Gzip and chunked answers are decoded for you, and every request is sent with Connection: close. For a series of calls that share a login, NewClient keeps the cookies from each answer and sends them with the next request, plus any headers set with SetHeader. Its cookie jar is one map for every host, so use one Client per service you talk to. The one-line calls share some state inside the package, so do not run them from several threads at the same time.

Which function for which job

JobFunction
Read a path parameter, query value or form fieldParam, Query, Form
Read the raw body or a request headerBody, Header
Answer with text, JSON or HTMLText, Json, Html
Choose the status or add a headerSetStatus, SetHeader
Send the browser somewhere elseRedirect
Hold the request in a struct instead of the bare idCtx, then the same Param, Json and the rest
One request from a programGet, Post, GetWith, PostWith, Request
Several requests that share cookiesNewClient
Check the answerOk for a 2xx status, Status for the code (0 when nothing answered), Error for the reason

A value you read from the request, such as Param or Form, belongs to that request and is freed when the handler returns. To keep it longer, in a global list or a session, store a copy made with str.Clone or a new string built with +.

import net.http

func main:
    c := http.NewCookie("theme", "dark")
    println c.name, c.value, c.path, c.http_only, c.same_site, c.has_max_age

net-http-Client.salam
import net.http
import net.router

func echo(ctx: i64):
    http.SetHeader(ctx, "X-Tag", "a")
    http.SetCookie(ctx, http.NewCookie("sid", "abc"))
    http.Text(ctx, http.Method(ctx) + " " + http.Header(ctx, "X-Token") + " " + http.Body(ctx))
end

func cookies(ctx: i64):
    all := http.Cookies(ctx)
    http.Text(ctx, "" + all.size() + " " + http.CookieValue(ctx, "lang") + " " + http.HasCookie(ctx, "sid"))
    http.ClearCookie(ctx, "old")
    http.ClearCookieAt(ctx, "scoped", "/admin", "")
end

func main:
    mut r := router.NewRouter()
    each method in ["GET", "POST", "PUT", "DELETE", "PATCH", "OPTIONS", "HEAD"]:
        router.Method(r, method, "/echo", echo)
    end
    router.Get(r, "/cookies", cookies)
    mut s := http.NewServer(19623)
    http.Use(s, r)
    _ := http.RunBackground(s)
    url := "http://127.0.0.1:19623/echo"
    mut headers := http.NewHeaders()
    headers.insert("X-Token", "t1")
    a := http.GetWith(url, headers)
    println http.Ok(a), http.Status(a), http.Body(a), "[" + http.Error(a) + "]"
    println http.Headers(a).has("X-Tag"), http.HeaderValues(a, "set-cookie")
    println http.CookieMap(a).get("sid"), http.RawHeaders(a).len() > 0, http.Raw(a).len() > 0
    println http.Body(http.Post(url, "a")), http.Body(http.PutWith(url, "b", headers))
    println http.Body(http.Delete(url)), http.Body(http.DeleteWith(url, headers))
    println http.Body(http.Patch(url, "c")), http.Body(http.PatchWith(url, "d", headers))
    println http.Body(http.Options(url)), http.Body(http.OptionsWith(url, headers))
    println http.Status(http.Head(url)), http.Status(http.HeadWith(url, headers))
    println http.Body(http.Put(url, "e")), http.Body(http.Get(url))
    println http.Body(http.RequestRaw("POST", url, "X-Token: raw\r\n", "f"))
    mut params := http.NewHeaders()
    params.insert("q", "hello world")
    println http.EncodeForm(params), http.WithQuery("http://x.ir/s", params)
    mut jar := http.NewHeaders()
    jar.insert("lang", "en")
    jar.insert("sid", "abc")
    headers.insert("Cookie", http.CookieHeader(jar))
    b := http.GetWith("http://127.0.0.1:19623/cookies", headers)
    println http.Body(b), http.SetCookieLines(b).len()
    mut c := http.NewClient()
    c.SetCookie("lang", "en")
    println http.Body(c.Get("http://127.0.0.1:19623/cookies"))
    c.ClearCookies()
    mut extra := http.NewHeaders()
    extra.insert("X-Token", "c1")
    println http.Body(c.RequestWith("PUT", url, "g", extra)), http.Body(c.Request("PATCH", url, "h"))
    println http.Body(c.Put(url, "i")), http.Body(c.Delete(url)), http.Body(c.Patch(url, "j")), http.Body(c.Options(url)), http.Status(c.Head(url))
    bad := http.Get("http://127.0.0.1:1/")
    println http.Ok(bad), http.Error(bad) != ""
end

import net.http
import net.router

func hello(h: i64):
    ctx := http.Ctx(h)
    name := http.Query(ctx, "name")
    if http.HasCookie(ctx, "seen"):
        http.Text(ctx, "hello again " + name + " (" + http.CookieValue(ctx, "seen") + ")")
    else:
        http.SetCookie(ctx, http.ExpiringCookie("seen", "1", 3600))
        http.Text(ctx, "hello " + name + " (" + http.RawQuery(ctx) + ")")
    end
end

func book(h: i64):
    ctx := http.Ctx(h)
    http.SetHeader(ctx, "X-Method", http.Method(ctx))
    http.SetStatus(ctx, 202)
    http.Json(ctx, `{"id": "` + http.Param(ctx, "id") + `", "path": "` + http.Path(ctx) + `"}`)
end

func form(h: i64):
    ctx := http.Ctx(h)
    http.Text(ctx, http.Form(ctx, "a") + "|" + http.Body(ctx) + "|" + http.Header(ctx, "Content-Type"))
end

func logout(h: i64):
    ctx := http.Ctx(h)
    http.ClearCookie(ctx, "seen")
    http.Redirect(ctx, "/hello")
end

func page(h: i64):
    ctx := http.Ctx(h)
    cookies := http.Cookies(ctx)
    http.Html(ctx, "<p>" + cookies.size() + "</p>")
end

func main:
    mut r := router.NewRouter()
    router.Get(r, "/hello", hello)
    router.Get(r, "/books/:id", book)
    router.Post(r, "/form", form)
    router.Get(r, "/logout", logout)
    router.Get(r, "/page", page)
    mut s := http.NewServer(19622)
    http.Use(s, r)
    _ := http.RunBackground(s)
    base := "http://127.0.0.1:19622"
    mut c := http.NewClient()
    println http.Body(c.Get(base + "/hello?name=Maryam"))
    println http.Body(c.Get(base + "/hello?name=Maryam"))
    println http.Body(c.Get(base + "/page"))
    a := c.Get(base + "/books/7")
    println http.Status(a), http.Body(a), http.GetHeader(a, "x-method")
    c.SetHeader("Content-Type", "application/x-www-form-urlencoded")
    println http.Body(c.Post(base + "/form", "a=1&b=2"))
    b := c.Get(base + "/logout")
    println http.Status(b), http.SetCookieLines(b).len() > 0

    http.SetKeepAliveMaxRequests(100)

    http.SetMaxBodyBytes(1048576)

    println http.SERVER_VERSION, http.IsEdge()
    if not http.IsEdge():
        http.RunTLS(s, "cert.pem", "key.pem")

    http.Run(s)

import net.http
import net.router

func home(ctx: i64):
    http.Text(ctx, "hello from the server")
end

func book(ctx: i64):
    id := http.Param(ctx, "id")
    format := http.Query(ctx, "format")
    http.SetHeader(ctx, "X-Book", id)
    http.Json(ctx, `{"id": "` + id + `", "format": "` + format + `", "raw": "` + http.RawQuery(ctx) + `"}`)
end

func create(ctx: i64):
    name := http.Form(ctx, "name")
    http.SetStatus(ctx, 201)
    http.Text(ctx, http.Method(ctx) + " " + http.Path(ctx) + " " + name + " " + http.Header(ctx, "X-Client"))
end

func raw(ctx: i64):
    answer := http.Own(ctx, "body: " + http.Body(ctx))
    http.Text(ctx, answer)
end

func page(ctx: i64):
    http.Html(ctx, "<h1>hello</h1>")
end

func old(ctx: i64):
    http.Redirect(ctx, "/")
end

func main:
    mut r := router.NewRouter()
    router.Get(r, "/", home)
    router.Get(r, "/books/:id", book)
    router.Post(r, "/books", create)
    router.Put(r, "/raw", raw)
    router.Get(r, "/page", page)
    router.Get(r, "/old", old)
    mut s := http.NewServer(19621)
    http.Use(s, r)
    println http.RunBackground(s)
    base := "http://127.0.0.1:19621"
    a := http.Get(base + "/")
    println http.Status(a), http.Body(a)
    b := http.Get(base + "/books/42?format=pdf")
    println http.Body(b), http.GetHeader(b, "x-book")
    mut headers := http.NewHeaders()
    headers.insert("Content-Type", "application/x-www-form-urlencoded")
    headers.insert("X-Client", "salam")
    c := http.PostWith(base + "/books", "name=Maryam", headers)
    println http.Status(c), http.Body(c)
    println http.Body(http.Put(base + "/raw", "data"))
    d := http.Get(base + "/page")
    println http.GetHeader(d, "content-type"), http.Body(d)
    e := http.Request("GET", base + "/old", http.NewHeaders(), "")
    println http.Status(e), http.GetHeader(e, "location")
    println http.Status(http.Get(base + "/nope"))
end
