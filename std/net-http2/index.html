net/http2

The HTTP/2 protocol from the ground up: frames, HPACK header compression and streams.

import net.http2
On this page
  1. Types
  2. Constants and variables
  3. Functions

This package implements HTTP/2 at the protocol level: it builds and reads frames, compresses the headers with HPACK and carries several requests at once over one connection as "streams". For building an ordinary web server net/http is simpler; this package is for when you need the protocol itself.

How it goes: build a Transport over a TCP or TLS socket, take a connection with NewConn and do the handshake. Then the client sends a request with SendHeaders and reads the answer in StreamAt after AwaitStreamEnd. Over TLS, the client and the server must have agreed on h2 through ALPN.

net-http2-Conn.salam
import net.http2
import net.tcp
import compress

mut server := tcp.Server {}

func answer():
    conn := tcp.Accept(server)
    mut c := http2.NewConn(http2.PlainTransport(conn.fd), true)
    if not http2.ServerHandshake(c):
        ret
    end
    id := http2.AwaitAnyHeaders(c, 20)
    stream := http2.StreamAt(c, id)
    mut headers := http2.HeaderList {}
    http2.HeadersAdd(headers, ":status", "200")
    http2.HeadersAdd(headers, "content-type", "text/plain")
    _ := http2.SendHeaders(c, id, headers, false)
    _ := http2.SendData(c, id, compress.BufFromStr("hello " + http2.HeadersGet(stream.headers, ":path")), false)
    mut trailers := http2.HeaderList {}
    http2.HeadersAdd(trailers, "x-done", "1")
    _ := http2.SendTrailers(c, id, trailers)
    _ := http2.AwaitStreamEnd(c, 0, 5)
    http2.CloseConn(c)
end

func main:
    server = tcp.Bind(0)
    t := spawn(answer)
    conn := tcp.Dial("127.0.0.1", tcp.Port(server))
    tx := http2.PlainTransport(conn.fd)
    println tx.ok, tx.kind == http2.H2_TRANSPORT_PLAIN
    mut c := http2.NewConn(tx, false)
    println http2.ClientHandshake(c), http2.ConnOk(c)
    id := http2.NextStreamId(c)
    mut headers := http2.HeaderList {}
    http2.HeadersAdd(headers, ":method", "GET")
    http2.HeadersAdd(headers, ":scheme", "http")
    http2.HeadersAdd(headers, ":path", "/books")
    http2.HeadersAdd(headers, ":authority", "127.0.0.1")
    println id, http2.SendHeaders(c, id, headers, true)
    println http2.AwaitStreamEnd(c, id, 20)
    stream := http2.StreamAt(c, id)
    println http2.HeadersGet(stream.headers, ":status"), compress.BufToStrUnsafe(stream.data), stream.got_trailers, http2.HeadersGet(stream.trailers, "x-done"), stream.end_stream
    println http2.StreamCount(c) >= 1, c.remote.max_frame_size
    _ := http2.SendPing(c, compress.BufFromStr("abcdefgh"))
    println http2.Poll(c) >= http2.H2_POLL_CONNECTION
    _ := http2.SendRstStream(c, http2.NextStreamId(c), http2.H2_CANCEL)
    println http2.SendGoAway(c, http2.H2_NO_ERROR, "")
    http2.CloseConn(c)
    join(t)
    tcp.CloseServer(server)
end

import net.http2
import compress

func main:
    mut settings := http2.DefaultSettings()
    println settings.header_table_size, settings.initial_window_size, settings.max_frame_size, settings.enable_push
    frame := http2.MakeSettingsFrame(settings)
    println http2.FrameTypeName(frame.kind), frame.stream_id, frame.payload.len
    settings.max_frame_size = 32768
    println http2.ApplySettings(settings, http2.MakeSettingsFrame(http2.DefaultSettings())), settings.max_frame_size
    ack := http2.MakeSettingsAck()
    println http2.FrameHasFlag(ack, http2.H2_FLAG_ACK)
    win := http2.MakeWindowUpdate(3, 1024)
    println http2.FrameTypeName(win.kind), http2.WindowIncrementOf(win)
    rst := http2.MakeRstStream(5, http2.H2_CANCEL)
    println http2.ErrorCodeOf(rst), http2.ErrorName(http2.ErrorCodeOf(rst))
    away := http2.MakeGoAway(7, http2.H2_NO_ERROR, "bye")
    println http2.FrameTypeName(away.kind), http2.ErrorCodeOf(away)
    ping := http2.MakePing(compress.BufFromStr("12345678"), false)
    println http2.FrameTypeName(ping.kind), ping.payload.len
    mut data := http2.NewFrame(http2.H2_DATA, http2.H2_FLAG_END_STREAM, 1)
    compress.BufPushBytes(data.payload, "hello" as void*, 5)
    mut ok := true
    println compress.BufToStrUnsafe(http2.DataOf(data, ok)), ok
    bytes := http2.FrameEncode(data)
    println bytes.len, compress.BufGet(bytes, 3)
    mut head := http2.NewFrame(http2.H2_HEADERS, http2.H2_FLAG_END_HEADERS, 1)
    compress.BufPush(head.payload, 130)
    println http2.HeaderBlockOf(head, ok).len, ok
    mut one := compress.NewByteBuf()
    http2.PutSetting(one, http2.H2_SETTINGS_MAX_FRAME_SIZE, 16384)
    println one.len
    http2.FreeFrame(data)
    println http2.H2_DATA, http2.H2_PING, http2.H2_PROTOCOL_ERROR, http2.H2_DEFAULT_FRAME_SIZE, http2.H2_DEFAULT_WINDOW, http2.H2_MAX_FRAME_LIMIT
    println http2.FrameTypeName(http2.H2_PRIORITY), http2.FrameTypeName(http2.H2_RST_STREAM), http2.FrameTypeName(http2.H2_PUSH_PROMISE)
    println http2.FrameTypeName(http2.H2_GOAWAY), http2.FrameTypeName(http2.H2_WINDOW_UPDATE), http2.FrameTypeName(http2.H2_CONTINUATION)
    println http2.H2_SETTINGS_HEADER_TABLE_SIZE, http2.H2_SETTINGS_ENABLE_PUSH, http2.H2_SETTINGS_MAX_CONCURRENT_STREAMS, http2.H2_SETTINGS_INITIAL_WINDOW_SIZE, http2.H2_SETTINGS_MAX_HEADER_LIST_SIZE
    println http2.ErrorName(http2.H2_INTERNAL_ERROR), http2.ErrorName(http2.H2_FLOW_CONTROL_ERROR), http2.ErrorName(http2.H2_SETTINGS_TIMEOUT), http2.ErrorName(http2.H2_STREAM_CLOSED)
    println http2.ErrorName(http2.H2_FRAME_SIZE_ERROR), http2.ErrorName(http2.H2_REFUSED_STREAM), http2.ErrorName(http2.H2_COMPRESSION_ERROR), http2.ErrorName(http2.H2_CONNECT_ERROR)
    println http2.ErrorName(http2.H2_ENHANCE_YOUR_CALM), http2.ErrorName(http2.H2_INADEQUATE_SECURITY), http2.ErrorName(http2.H2_HTTP_1_1_REQUIRED)
    println http2.H2_STREAM_IDLE, http2.H2_STREAM_OPEN, http2.H2_STREAM_HALF_CLOSED_LOCAL, http2.H2_STREAM_HALF_CLOSED_REMOTE, http2.H2_STREAM_CLOSED_STATE
    println http2.H2_DEFAULT_TABLE_SIZE, http2.H2_POLL_ERROR
end

net-http2-Hpack.salam
import net.http2
import compress

func main:
    mut list := http2.HeaderList {}
    http2.HeadersAdd(list, ":method", "GET")
    http2.HeadersAdd(list, ":path", "/")
    http2.HeadersAdd(list, "user-agent", "salam")
    http2.HeadersAddSensitive(list, "authorization", "Bearer x")
    println http2.HeadersCount(list), http2.HeadersGet(list, ":path"), http2.HeadersHas(list, "cookie")
    mut sender := http2.NewDynTable(4096)
    mut receiver := http2.NewDynTable(4096)
    block := http2.EncodeHeaderList(sender, list, true)
    println block.len, http2.DynCount(sender), http2.DynSize(sender)
    mut ok := true
    read := http2.DecodeHeaderList(receiver, block, ok)
    println ok, read.items.len(), read.items[2].name, read.items[2].value, read.items[3].sensitive
    block2 := http2.EncodeHeaderList(sender, list, true)
    println block2.len < block.len
    fixed := http2.StaticGet(2)
    println fixed.name, fixed.value, http2.HPACK_STATIC_COUNT, http2.HPACK_ENTRY_OVERHEAD
    dyn := http2.DynGet(receiver, 1)
    println dyn.name
    mut exact := false
    println http2.TableFind(receiver, ":method", "GET", exact), exact
    println http2.TableGet(receiver, 62, ok).name, ok
    mut b := compress.NewByteBuf()
    http2.EncodeTableSizeUpdate(b, sender, 256)
    http2.DynSetMaxSize(receiver, 256)
    println b.len, sender.max_size, receiver.max_size
    mut b2 := compress.NewByteBuf()
    http2.EncodeHeader(b2, sender, http2.NewHeader("x-id", "7"), false)
    http2.DynInsert(receiver, http2.NewSensitiveHeader("k", "v"))
    println b2.len, http2.DynCount(receiver)
end

import net.http2
import encoding
import compress

func main:
    b := http2.HuffmanEncodeStr("www.example.com")
    println encoding.HexEncode(compress.BufToStrUnsafe(b))
    println http2.HuffmanEncodedLenStr("www.example.com"), "www.example.com".len()
    mut ok := true
    println compress.BufToStrUnsafe(http2.HuffmanDecodeBuf(b, ok)), ok
    text := compress.BufFromStr("no-cache")
    println http2.HuffmanEncodedLen(text.data, text.len)

    compress.BufPush(out, 0)

    println compress.BufToStrUnsafe(http2.HuffmanDecode(out, 1, out.len - 1, ok)), ok

net-http2-Transport.salam
import net.http2
import net.tcp
import compress

mut server := tcp.Server {}

func answer():
    conn := tcp.Accept(server)
    mut tx := http2.PlainTransport(conn.fd)
    println "server preface:", http2.ReadPreface(tx)
    frame := http2.ReadFrame(tx, http2.H2_DEFAULT_FRAME_SIZE)
    println "server frame:", frame.ok, http2.FrameTypeName(frame.kind), frame.stream_id, compress.BufToStrUnsafe(frame.payload)
    http2.CloseTransport(tx)
end

func main:
    server = tcp.Bind(0)
    t := spawn(answer)
    conn := tcp.Dial("127.0.0.1", tcp.Port(server))
    mut tx := http2.PlainTransport(conn.fd)
    _ := http2.WritePreface(tx)
    mut data := http2.NewFrame(http2.H2_DATA, 0, 3)
    compress.BufPushBuf(data.payload, compress.BufFromStr("hi"))
    _ := http2.WriteFrame(tx, data)
    join(t)
    http2.CloseTransport(tx)
    tcp.CloseServer(server)
    println http2.H2_PREFACE.len()
end
