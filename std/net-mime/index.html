net/mime

The media type of a file, the Content-Type header, and multipart form bodies.

import net.mime
On this page
  1. Types
  2. Functions

A media type (MIME) tells the browser and other programs what the content is, such as image/png. This package finds the type from a file extension and reads the Content-Type header.

Forms that upload a file send a multipart/form-data body: several parts with a boundary between them. ParseMultipart reads one and MultipartWriter builds one.

net-mime-Types.salam
import net.mime

func main:
    println mime.MimeExt("png"), mime.MimeExt("json"), mime.MimeExt(".png")
    println mime.MimeType("/site/index.html"), mime.MimeType("style.css")
    println mime.StatusText(200), mime.StatusText(404), mime.StatusText(503)
    ct := mime.ParseContentType("text/html; charset=UTF-8; q=0.9")
    println ct.media_type, ct.params.items.len(), mime.ParamGet(ct, "charset"), ct.params.items[1].key, ct.params.items[1].value
end

net-mime-Multipart.salam
import net.mime
import compress
import str

func main:
    mut w := mime.NewMultipartWriter()
    println str.StartsWith(mime.ContentTypeHeader(w), "multipart/form-data; boundary=")
    mime.WriteField(w, "name", "Maryam")
    mime.WriteFileStr(w, "note", "note.txt", "text/plain", "hello")
    data := compress.BufFromStr("PNG")
    mime.WriteFileBytes(w, "pic", "a.png", "image/png", data.data, data.len)
    body := mime.CloseWriter(w)
    println w.closed, body.len > 0
    parts := mime.ParseMultipart(body.data, body.len, w.boundary)
    each p in parts.items:
        println p.name, "[" + p.filename + "]", "[" + p.content_type + "]", compress.BufToStrUnsafe(p.body), p.headers.items.len()
    end
    println parts.items[1].headers.items[0].key
end
Outputtrue true true name [] [] Maryam 1 note [note.txt] [text/plain] hello 2 pic [a.png] [image/png] PNG 2 Content-Disposition
