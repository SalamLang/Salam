net/socketio

A Socket.IO client: events, acknowledgements, namespaces and automatic reconnection.

import net.socketio
On this page
  1. Types
  2. Constants and variables
  3. Functions

Socket.IO is the usual live messaging library in the Node.js world. This package is a client that connects to any Socket.IO version 4 server; it has no server of its own. Dial connects, On sets a listener for an event and the Emit... functions send events. Listeners are called while the program processes messages with Run, RunFor, Pump or Next.

The arguments of every event are a JSON array, read with ArgStr, ArgInt and the like. If the sender wants an answer, the event's ack_id is not negative and you answer it with Reply. For several namespaces on one connection, create a Manager.

net-socketio-Client.salam
import net.socketio
import net.websocket
import str

mut server := websocket.Server {}
mut report := ""

func test_server():
    mut conn := websocket.Accept(server)
    _ := websocket.SendText(conn, `0{"sid":"s1","upgrades":[],"pingInterval":20000,"pingTimeout":5000}`)
    mut going := true
    until going:
        m := websocket.Poll(conn, 3000)
        if m.kind != websocket.TEXT:
            break
        end
        packet := m.text.substr(1, m.text.len() - 1)
        if packet == "0":
            _ := websocket.SendText(conn, `40{"sid":"a1"}`)
        else str.StartsWith(packet, `2["chat"`):
            _ := websocket.SendText(conn, `42["chat","welcome"]`)
            _ := websocket.SendText(conn, `429["question","2+2"]`)
        else str.Contains(packet, `"add"`):
            id := packet.substr(1, str.Find(packet, "[") - 1)
            _ := websocket.SendText(conn, "43" + id + "[5]")
        else str.StartsWith(packet, "39"):
            report += "reply: " + packet.substr(2, packet.len() - 2)
        else packet == "1":
            going = false
        end
    end
    websocket.Close(conn)
end

func main:
    server = websocket.Listen(0)
    t := spawn(test_server)
    mut o := socketio.NewOptions()
    o.transports = socketio.TRANSPORT_WEBSOCKET
    o.reconnection = false
    c := socketio.DialWith("http://127.0.0.1:" + websocket.Port(server), o)
    println socketio.IsConnected(c), socketio.Sid(c), socketio.Id(c), socketio.Namespace(c), socketio.TransportName(c)

    socketio.On(c, "chat", (e: socketio.Event):
        println "message:", socketio.ArgStr(e, 0)
    end)
    socketio.On(c, "question", (e: socketio.Event):
        println "question:", socketio.ArgStr(e, 0), e.ack_id >= 0
        _ := socketio.ReplyStr(e, "four")
    end)
    println socketio.ListenerCount(c, "chat"), socketio.HasListeners(c, "news")
    _ := socketio.EmitStr(c, "chat", "hello")
    socketio.RunFor(c, 500)

    mut a := socketio.NewArgs()
    socketio.AddInt(a, 2)
    socketio.AddInt(a, 3)
    mut k := socketio.EmitAck(c, "add", a, 2000)
    println k.ok, socketio.AckInt(k, 0)
    socketio.AckFree(k)
    socketio.ArgsFree(a)

    socketio.Close(c)
    join(t)
    websocket.CloseServer(server)
    println report
end
Outputtrue s1 a1 / websocket 1 false message: welcome question: 2+2 true true 5 reply: ["four"]

net-socketio-Manager.salam
import net.socketio

func main:
    mut o := socketio.NewOptions()
    o.auth = `{"token":"abc"}`
    o.auto_connect = false
    m := socketio.NewManager("https://example.com", o)
    socketio.SetReconnection(m, true)
    socketio.SetReconnectionAttempts(m, 5)
    socketio.SetReconnectionDelay(m, 500)
    socketio.SetReconnectionDelayMax(m, 10000)
    socketio.SetRandomizationFactor(m, 20)
    socketio.SetTimeout(m, 5000)
    println socketio.OptionsOf(m).reconnection_attempts

    public := socketio.Of(m, "/")
    admin := socketio.Of(m, "/admin")
    socketio.SetAuth(admin, `{"role":"admin"}`)
    println socketio.AuthOf(admin), socketio.Sockets(m).len()
    if not socketio.Open(m):
        println "error:", socketio.ManagerError(m)
        ret
    end
    println socketio.IsOpen(m)
    _ := socketio.Connect(public)
    _ := socketio.Connect(admin)
    if socketio.Refused(admin):
        println "refused:", socketio.RefusalReason(admin)
    end
    println socketio.Active(public), socketio.Disconnected(public), socketio.Recovered(public)
    println socketio.Transport(public), socketio.PingInterval(public), socketio.PingTimeout(public), socketio.MaxPayload(public)
    println socketio.Buffered(public), socketio.Pending(public), socketio.Queued(public), socketio.Attempts(public)
    println socketio.EndpointOf(public).host, socketio.ManagerOf(public).p == m.p
    _ := socketio.PumpManager(m, 1000)
    socketio.Disconnect(admin)
    socketio.CloseManager(m)
end

import net.socketio

func main:

    println text
    d := socketio.DecodePacket(text)
    println d.ok, d.kind, d.nsp, d.id, d.data
    bad := socketio.DecodePacket("x")
    println bad.ok

    payload := socketio.EncodeEngine(socketio.EIO_MESSAGE, text)

    println socketio.EngineKind(payload) == socketio.EIO_MESSAGE
    println socketio.EngineBody(payload)

    mut items := Vector {} as Vector<str>
    items.push("2")

    body := socketio.JoinPayloads(items)
    parts := socketio.SplitPayloads(body)
    println parts.len(), parts.get(0), parts.get(1)

    arr := socketio.SplitJSONArray(`["a", {"b":[1,2]}, 3]`)

A message event is one of the server's messages; the rest report on the connection itself: connecting and disconnecting, errors, reconnection attempts, pings, timeouts and the upgrade to WebSocket. net-socketio-Ack.salam
import net.socketio
import encoding.value

func main:
    c := socketio.Dial("http://localhost:3000")
    if not socketio.IsConnected(c):
        println socketio.LastError(c)
        ret
    end
    socketio.OnAny(c, (e: socketio.Event):
        println "received:", e.name, socketio.ArgCount(e)
    end)
    socketio.Once(c, "welcome", (e: socketio.Event):
        println socketio.Arg(e, 0)
    end)
    socketio.OnAnyOutgoing(c, (e: socketio.Event):
        println "sent:", e.name
    end)

    socketio.PrependAny(c, (e: socketio.Event):
        println "first of all:", e.name
    end)
    socketio.PrependAnyOutgoing(c, (e: socketio.Event):
        println "sending:", e.name
    end)
    println socketio.AnyListenerCount(c)

    _ := socketio.Emit(c, "ping")
    _ := socketio.EmitInt(c, "score", 12)
    _ := socketio.EmitFloat(c, "temp", 21.5)
    _ := socketio.EmitBool(c, "ready", true)
    _ := socketio.EmitJSON(c, "user", `{"name":"Ali"}`)
    _ := socketio.EmitBytes(c, "file", "abc" as void*, 3)
    _ := socketio.Send(c, "a plain message")

    mut a := socketio.NewArgs()
    socketio.AddStr(a, "room-1")
    _ := socketio.EmitArgs(c, "join", a)
    _ := socketio.EmitVolatile(c, "cursor", a)
    _ := socketio.EmitWithAck(c, "join", a, 2000, (k: socketio.Ack):
        println k.ok, socketio.AckCount(k), socketio.AckStr(k, 0)
        println socketio.AckArg(k, 1), socketio.AckFloat(k, 1), socketio.AckBool(k, 2)
        if socketio.AckIsBinary(k, 3):
            println socketio.AckBytes(k, 3).len
        end
        mut v := socketio.AckValue(k, 0)
        value.ValueFree(v)
    end)
    socketio.ArgsFree(a)

    mut e := socketio.Next(c, 5000)
    if e.kind == socketio.EV_EVENT:
        println socketio.ArgInt(e, 0), socketio.ArgFloat(e, 1), socketio.ArgBool(e, 2)
        if socketio.ArgIsBinary(e, 3):
            b := socketio.ArgBytes(e, 3)
            println b.len
        end
        if e.ack_id >= 0:
            _ := socketio.ReplyJSON(e, `{"ok":true}`)
        end
    else e.kind == socketio.EV_DISCONNECT:
        println "disconnected:", e.message
    end
    socketio.EventFree(e)

    mut e2 := socketio.NextEvent(c, 5000)
    mut v2 := socketio.ArgValue(e2, 0)
    println value.IsNumber(v2)
    value.ValueFree(v2)
    socketio.EventFree(e2)

    _ := socketio.Pump(c, 100)
    socketio.Off(c, "welcome")
    socketio.OffAny(c)
    socketio.OffAnyOutgoing(c)
    socketio.OffAll(c)
    socketio.Run(c)
end
