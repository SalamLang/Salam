net/websocket

A two way connection that stays open between browser and server, with rooms for broadcasting.

import net.websocket
On this page
  1. Types
  2. Constants and variables
  3. Functions

A WebSocket is a connection that stays open after one HTTP request, and either side sends a message whenever it likes; for live chat, notifications and games. The server takes connections with Listen and Accept, and a client connects with Dial.

To serve several clients, read each connection in its own thread and use a Hub to send a message to every member of a room.

Server

A WebSocket server.
NameTypeDescription

portint = 0The port.
net-websocket-Echo.salam
import net.websocket

mut server := websocket.Server {}

func responder():
    mut conn := websocket.Accept(server)
    until websocket.IsOpen(conn):
        m := websocket.Receive(conn)
        if websocket.IsText(m):
            _ := websocket.SendText(conn, "echo: " + m.text)
        else websocket.IsBinary(m):
            _ := websocket.SendBinaryBytes(conn, m.data, m.len)
        else websocket.IsClose(m):
            break
        end
    end
    websocket.Close(conn)
end

func main:
    server = websocket.Listen(0)
    println websocket.Ok(server), websocket.Port(server) > 0
    t := spawn(responder)
    mut c := websocket.Dial("127.0.0.1", websocket.Port(server), "/chat")
    println websocket.IsOpen(c), c.path, c.client
    _ := websocket.SendText(c, "hello")
    println websocket.ReceiveText(c)
    _ := websocket.SendBinary(c, "abc")
    m := websocket.Poll(c, 2000)
    println m.kind == websocket.BINARY, m.len
    _ := websocket.Ping(c, "p")
    m2 := websocket.Poll(c, 2000)
    println websocket.IsPong(m2), m2.kind == websocket.PONG
    m3 := websocket.Poll(c, 50)
    println m3.kind == websocket.TIMEOUT
    websocket.Close(c)
    join(t)
    websocket.CloseServer(server)
    println websocket.NONE, websocket.TEXT, websocket.CLOSE, websocket.PING
end
Outputtrue true true /chat true echo: hello true 3 true true true 0 1 8 9

net-websocket-Hub.salam
import net.websocket

mut server := websocket.Server {}
mut sides := Vector {} as Vector<websocket.Conn>

func acceptor():
    repeat 2:
        sides.push(websocket.Accept(server))
    end
end

func main:
    server = websocket.Listen(0)
    port := websocket.Port(server)
    t := spawn(acceptor)
    mut a := websocket.Dial("127.0.0.1", port, "/room")
    mut b := websocket.DialWith("127.0.0.1", port, "/room", "X-Name: b\r\n")
    join(t)
    mut hub := websocket.NewHub()
    each side in sides:
        websocket.Register(hub, "lobby", side)
    end
    println websocket.RoomSize(hub, "lobby"), hub.rooms.size()
    println websocket.Broadcast(hub, "lobby", "welcome")
    println websocket.ReceiveText(a), websocket.ReceiveText(b)
    println websocket.BroadcastBinary(hub, "lobby", "xy")
    println websocket.Receive(a).len
    websocket.Unregister(hub, "lobby", sides[0])
    println websocket.RoomSize(hub, "lobby")
    websocket.Stop(a)
    websocket.Close(b)
    websocket.CloseServer(server)
end
Output2 1 2 welcome welcome 2 2 1

Functions

AcceptKey

func AcceptKey(key: str): str
Builds the handshake accept key from the client key (RFC 6455).
net-websocket-Handshake.salam
import net.websocket
import str

func main:
    println websocket.AcceptKey("dGhlIHNhbXBsZSBub25jZQ==")
    request := websocket.HandshakeRequest("example.com", "/chat", "dGhlIHNhbXBsZSBub25jZQ==")
    println str.Split(request, "\r\n")[0]
    request2 := websocket.HandshakeRequestWith("example.com", "/chat", "dGhlIHNhbXBsZSBub25jZQ==", "Authorization: Bearer x\r\n")
    println str.Contains(request2, "Authorization: Bearer x")
    response := websocket.HandshakeResponse(request)
    println str.Split(response, "\r\n")[0]
    println str.Contains(response, "Sec-WebSocket-Accept: s3pPLMBiTxaQ9kYGzzhZRbK+xOo=")
end
Outputs3pPLMBiTxaQ9kYGzzhZRbK+xOo= GET /chat HTTP/1.1 true HTTP/1.1 101 Switching Protocols true

LoadWSSConfig

func LoadWSSConfig(cert_file: str, key_file: str)
Loads the TLS certificate and key for accepting secure connections.
net-websocket-WSS.salam
import net.websocket

func main:
    mut a := websocket.DialWSS("echo.example.com", 443, "/")
    mut b := websocket.DialWSSWith("echo.example.com", 443, "/", "Origin: https://salamlang.ir\r\n")
    mut c := websocket.DialWSSUnverified("localhost", 8443, "/", "")
    println websocket.IsOpen(a), websocket.IsOpen(b), websocket.IsOpen(c)
    websocket.LoadWSSConfig("cert.pem", "key.pem")
    s := websocket.Listen(8443)
    mut conn := websocket.AcceptWSS(s)
    _ := websocket.SendText(conn, "hello")
    websocket.Close(conn)
end
