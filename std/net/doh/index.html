net.doh

Encrypted DNS: a resolver over DNS over HTTPS and DNS over TLS with server failover and a TTL cache, plain UDP DNS as a fallback, presets, and a local stub server for other programs.

import net.doh

  5. Constants and variables
  6. Functions

Ordinary DNS asks "what is the address of this name?" in plain text over UDP, so the network between you and the resolver can read every name you look up, and can change or block the answer. DNS over HTTPS (DoH) and DNS over TLS (DoT) send the same question inside an encrypted connection. This package speaks both, and plain UDP DNS too, and adds what a program needs on top: a Resolver that tries a list of servers in order, remembers answers for their TTL and answers LookupHost with a list of addresses. A Stub turns the resolver into a small local DNS server, so other programs can use it by pointing their DNS at it. To see how the pieces fit together in an app, read Building a v2ray style or VPN app.

Core ideas:

  • A Server says how to reach one DNS server: KindDoH (HTTPS POST), KindDoHGet (HTTPS GET), KindDoT (TLS on port 853) or KindUDP (plain). Cloudflare, Google, Quad9 and their DoT forms are ready made, and NewServer, NewDoT and NewPlain make your own.
  • Give a server its IP addresses in addrs. Then the program does not need to resolve the server's own name first, which matters when DNS is exactly what is blocked.
  • Query asks one server once. NewResolver and Resolve ask a list: a server that fails, times out or answers SERVFAIL or REFUSED is skipped for the next, and answers are cached (an answer for its TTL, kept between the limits of SetTTLLimits; a "no such name" answer for a short fixed time).
  • Certificates are checked. insecure turns that off for tests only.

Things that catch people out:

  • Encrypted DNS hides the question from the network, not from the resolver. The resolver you pick sees every name you look up. Choose one you trust, and remember that connecting to a well known resolver address is itself visible.
  • Blocked resolvers. A network can block the addresses of public DoH servers. Use several servers, keep a plain or private one as the last in the list, and see IranPlainServers for the plain servers used in Iran. Plain DNS is readable and can be changed on the way.
  • No DNSSEC. Answers are trusted as the server gave them; nothing here checks DNSSEC signatures.
  • The stub answers on UDP and TCP (the same port, TCP with the two byte length prefix) and has no access control. It listens on 127.0.0.1 when the host is empty; give 0.0.0.0 only on purpose, because an open DNS server on a public address is abused.
  • Free what you hold. FreeResponse, FreeAnswer and FreeServers release memory; a Resolver and a Stub live until the program ends.
  • The cache and counters are shared across threads and the resolver takes its own lock, so one resolver can serve many threads.
  • IPv6 resolvers work. A server address may be an IPv6 literal, over UDP, TLS or HTTPS. The built-in servers list IPv4 first; WithIPv6, Cloudflare6, Google6 and Quad96 add or use the IPv6 addresses (2606:4700:4700::1111, 2001:4860:4860::8888, 2620:fe::fe). The local stub can listen on ::1 or ::.
  • Connections are reused. DoH over HTTPS keeps its connections alive and reuses them for later queries (up to 8 idle ones, for 30 seconds); a stale one is replaced by a new connection and the query is repeated once. ReusedConnections counts the reuses and CloseIdleConnections closes the idle ones.
  • Iran presets are unverified. The addresses of Shecan, Server403, Electro, Radar and Begzar come from community lists and could not be confirmed against the official sites of those services. Check them before relying on them.

Using a resolver

Make a Resolver from a list of servers and ask it. The second lookup of the same name comes from the cache and does not touch the network. The example uses a tiny DNS server on loopback so that it runs anywhere; with real servers use DefaultServers or your own list.

net-doh-guide-resolver.salam
import compress
import mem
import net.dns
import net.doh
import net.udp
import sync

mut udp_port := 0
mut queries := 0
mut ready := sync.NewWaitGroup()

func record(name: str, rtype: int, ttl: int, data: str): dns.Record:
    ret dns.Record { name = name, rtype = rtype, rclass = 1, ttl = ttl, data = data }
end

func answer_for(p: void*, n: int): compress.ByteBuf:
    queries++
    question := dns.Decode(p, n)
    asked := question.questions.items.Get(0)
    mut records := Vector {} as Vector<dns.Record>
    mut rcode := 0
    if asked.name == "a.test" and asked.qtype == dns.TypeA:
        records.Push(record("a.test", dns.TypeA, 120, "10.0.0.1"))
        records.Push(record("a.test", dns.TypeA, 300, "10.0.0.2"))
    else asked.name == "v6.test" and asked.qtype == dns.TypeAAAA:
        records.Push(record("v6.test", dns.TypeAAAA, 300, "2001:db8::5"))
    else asked.name == "txt.test" and asked.qtype == dns.TypeTXT:
        records.Push(record("txt.test", dns.TypeTXT, 60, "hello world"))
    else asked.name == "alias.test":
        records.Push(record("alias.test", dns.TypeCNAME, 60, "target.test"))
        if asked.qtype == dns.TypeA:
            records.Push(record("target.test", dns.TypeA, 60, "10.9.9.9"))
        end
    else:
        rcode = 3
    end
    ret doh.EncodeResponse(question.id, asked.name, asked.qtype, rcode, records)
end

func udp_server():
    s := udp.Bind(0)
    udp_port = udp.Port(s)
    sync.Done(ready)
    buf := mem.Allocate(2048 as u64)
    mut from_host := ""
    mut from_port := 0
    until true:
        n := udp.ReceiveBytes(s, buf, 2048, from_host, from_port)
        if n > 0:
            answer := answer_for(buf, n)
            udp.SendBytes(s, from_host, from_port, answer.data, answer.len)
        end
    end
end

func start_mock():
    sync.Add(ready, 1 as i64)
    _t := spawn(udp_server)
    sync.Wait(ready)
end

func mock_server(name: str): doh.Server:
    mut server := doh.NewPlain(name, "127.0.0.1")
    server.port = udp_port
    ret server
end

func main:
    start_mock()
    mut servers := Vector {} as Vector<doh.Server>
    servers.Push(mock_server("mock"))
    resolver := doh.NewResolver(servers)

    addresses := doh.LookupHost(resolver, "a.test")
    println "addresses:", addresses.Len(), addresses.Get(0)
    again := doh.LookupHost(resolver, "a.test")
    println again.Len()
    stats := doh.Stats(resolver)
    println "asked the server", queries, "time(s); cache hits:", stats.hits
end
Outputaddresses: 2 10.0.0.1 2 asked the server 1 time(s); cache hits: 1

For real use, put your servers in the order you trust them and give each one its addresses:

mut servers := Vector {} as Vector<doh.Server>
servers.push(doh.Cloudflare())
servers.push(doh.GoogleDoT())
servers.push(doh.NewPlain("home", "192.168.1.1"))
resolver := doh.NewResolver(servers)
println doh.LookupHost(resolver, "example.com")

DNS when networks interfere

Three kinds of trouble are common. The network can read your lookups: use DoH or DoT. It can change answers for some names, or return nothing: encrypted DNS stops changes on the path, but not a block of the resolver's address. It can block the resolvers themselves: list several, keep addrs filled so that no lookup is needed to reach them, and keep a plain server that works in your network as the last choice. IranPlainServers lists plain servers used in Iran; their traffic is not encrypted.

Run a Stub on 127.0.0.1 and point the operating system or a program at it, and every lookup it makes goes through your resolver and its cache. When a program does name lookups itself through a proxy, as SOCKS5 and Trojan do, let the proxy resolve the name, so that the lookup is not made locally at all.

What to know

  • Encrypted DNS hides the question from the network, not from the resolver you choose.
  • Fill addrs, or set a bootstrap with SetBootstrap, so that the resolver does not need DNS to find its DNS.
  • Failed servers are skipped; a "no such name" answer is final and cached for the short failure time.
  • No DNSSEC validation. The stub has no access control; bind it to loopback.
  • Query has a timeout in milliseconds, and a Resolver has SetTimeout; keep it short enough that a blocked server does not stall your program.
  • No logging happens by itself.

mut queries := 0
mut ready := sync.NewWaitGroup()

func record(name: str, rtype: int, ttl: int, data: str): dns.Record:
    ret dns.Record { name = name, rtype = rtype, rclass = 1, ttl = ttl, data = data }
end

func answer_for(p: void*, n: int): compress.ByteBuf:
    queries++
    question := dns.Decode(p, n)
    asked := question.questions.items.Get(0)
    mut records := Vector {} as Vector<dns.Record>
    mut rcode := 0
    if asked.name == "a.test" and asked.qtype == dns.TypeA:
        records.Push(record("a.test", dns.TypeA, 120, "10.0.0.1"))
        records.Push(record("a.test", dns.TypeA, 300, "10.0.0.2"))
    else asked.name == "v6.test" and asked.qtype == dns.TypeAAAA:
        records.Push(record("v6.test", dns.TypeAAAA, 300, "2001:db8::5"))
    else asked.name == "txt.test" and asked.qtype == dns.TypeTXT:
        records.Push(record("txt.test", dns.TypeTXT, 60, "hello world"))
    else asked.name == "alias.test":
        records.Push(record("alias.test", dns.TypeCNAME, 60, "target.test"))
        if asked.qtype == dns.TypeA:
            records.Push(record("target.test", dns.TypeA, 60, "10.9.9.9"))
        end
    else:
        rcode = 3
    end
    ret doh.EncodeResponse(question.id, asked.name, asked.qtype, rcode, records)
end

func udp_server():
    s := udp.Bind(0)
    udp_port = udp.Port(s)
    sync.Done(ready)
    buf := mem.Allocate(2048 as u64)
    mut from_host := ""
    mut from_port := 0
    until true:
        n := udp.ReceiveBytes(s, buf, 2048, from_host, from_port)
        if n > 0:
            answer := answer_for(buf, n)
            udp.SendBytes(s, from_host, from_port, answer.data, answer.len)
        end
    end

import compress
import mem
import net.dns
import net.doh
import net.tcp
import str
import sync

mut http_port := 0
mut methods := ""
mut ready := sync.NewWaitGroup()

func find_text(b: compress.ByteBuf, pattern: str): int:
    n := pattern.Len()
    mut i := 0
    until i + n <= b.len:
        mut same := true
        repeat n in j:
            if compress.BufGet(b, i + j) != (pattern as u8*)[j] as int:
                same = false
                break
            end
        end
        if same:
            ret i
        end
        i++
    end
    ret 0 - 1
end

func text_at(b: compress.ByteBuf, off: int, n: int): str:
    buf := mem.Allocate((n + 1) as u64)
    mem.Copy(buf, ((b.data as i64) + (off as i64)) as void*, n as u64)
    (buf as u8*)[n] = 0 as u8
    ret buf as str
end

func from_base64url(s: str): compress.ByteBuf:
    alphabet := "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_"
    mut out := compress.NewByteBuf()
    mut acc := 0
    mut bits := 0
    repeat s.Len() in i:
        v := alphabet.Find(s.Substr(i, 1))
        if v < 0:
            break
        end
        acc = (acc << 6) | v
        bits += 6
        if bits >= 8:
            bits -= 8
            compress.BufPush(out, (acc >> bits) & 255)
            acc = acc & ((1 << bits) - 1)
        end
    end
    ret out
end

func serve_one(c: tcp.Conn):
    mut raw := compress.NewByteBuf()
    buf := mem.Allocate(4096 as u64)
    mut head_end := 0 - 1
    mut need := 0
    until true:
        n := tcp.ReadBytes(c, buf, 4096)
        if n <= 0:
            break
        end
        compress.BufPushBytes(raw, buf, n)
        head_end = find_text(raw, "\r\n\r\n")
        if head_end >= 0:
            head := str.Lower(text_at(raw, 0, head_end))
            at := head.Find("content-length:")
            if at >= 0:
                rest := head.Substr(at + 15, head.Len() - at - 15)
                mut line_end := rest.Find("\r\n")
                if line_end < 0:
                    line_end = rest.Len()
                end
                need = str.Trim(rest.Substr(0, line_end)).ToInt()
            end
            if raw.len >= head_end + 4 + need:
                break
            end
        end
    end
    mut wire := compress.NewByteBuf()
    if text_at(raw, 0, 4) == "POST":
        methods += "POST "
        compress.BufPushBytes(wire, ((raw.data as i64) + (head_end as i64) + 4) as void*, need)
    else:
        methods += "GET "
        line := text_at(raw, 0, find_text(raw, "\r\n"))
        rest := line.Substr(line.Find("dns=") + 4, line.Len())
        wire = from_base64url(rest.Substr(0, rest.Find(" ")))
    end
    question := dns.Decode(wire.data, wire.len)
    asked := question.questions.items.Get(0)
    mut records := Vector {} as Vector<dns.Record>
    records.Push(dns.Record { name = asked.name, rtype = dns.TypeA, rclass = 1, ttl = 300, data = "192.0.2.80" })
    answer := doh.EncodeResponse(question.id, asked.name, asked.qtype, 0, records)
    head := "HTTP/1.1 200 OK\r\nContent-Type: application/dns-message\r\nConnection: close\r\nContent-Length: " + answer.len + "\r\n\r\n"
    tcp.Write(c, head)
    tcp.WriteBytes(c, answer.data, answer.len)
    tcp.Close(c)
end

func http_server():
    srv := tcp.Bind(0)
    http_port = tcp.Port(srv)
    sync.Done(ready)
    until true:
        serve_one(tcp.Accept(srv))
    end
end

func main:
    sync.Add(ready, 1 as i64)
    _t := spawn(http_server)
    sync.Wait(ready)
    base := "http://127.0.0.1:" + http_port + "/dns-query"

    post := doh.NewServer("post", doh.KindDoH, base)
    println post.use_tls, post.port == http_port, post.path
    r := doh.Query(post, "web.test", dns.TypeA, 2000)
    println r.ok, r.msg.answers.items.Get(0).data

    get := doh.NewServer("get", doh.KindDoHGet, base)
    g := doh.Query(get, "web.test", dns.TypeA, 2000)
    println g.ok, g.msg.answers.items.Get(0).data
    println methods
end
