net.trojan

The Trojan protocol: a TLS client and server that look like an ordinary HTTPS site, a fallback website for everyone who is not a client, WebSocket mode, UDP and a connector for proxy chains.

import net.trojan

Trojan is a proxy protocol built to look like nothing. A client opens a normal TLS connection to a server and sends one line that holds a password hash and the target it wants; if the hash is right the server connects there and relays bytes. If the first bytes are not a valid Trojan request, the server does not complain. It hands the connection, bytes and all, to an ordinary website, so a visitor or a probe gets a real page from a real site. This package has both sides: Dial is the client and Serve runs the server. It also plugs into net.tunnel as an Inbound and a Connector for routing and proxy chains. To see how the pieces fit together in an app, read Building a v2ray style or VPN app.

Core ideas:

  • A password is never sent. What goes on the wire is its SHA-224 hash (PasswordHash), as 56 hex characters. The server keeps hashes (AddPassword, AddPasswordHash).
  • TLS does the hiding. The Trojan request is inside the TLS stream, and a server with a valid certificate and a real fallback_port website looks like any HTTPS site.
  • Failure is silent. A wrong password, a browser, a scanner: all get the fallback site. fallbacks counts them.
  • A Link is a connection over TCP, TLS or WebSocket. Dial returns one; read and write it with LinkRead and LinkWriteStr, and close it with LinkClose. use_ws carries Trojan inside a WebSocket so that a CDN or reverse proxy can sit in front.
  • DialUDP opens a UDP session where every packet names its own target (SendUDP, ReadUDPPacket).

Things that catch people out:

  • Use TLS, with a real certificate. Serve is the real server. ServePlain and ServeWS have no TLS: the password hash and all traffic are readable by anyone on the path, and the hash can be replayed. Use them on loopback, in tests, or behind a reverse proxy that ends TLS. insecure switches certificate checking off; never use it outside a test.
  • The fallback site must be real. A fallback that is missing, empty or obviously fake makes the server easy to recognise. Without fallback_port, a failed client is just disconnected, which is itself a sign. Serve real content, on the same machine, with the same certificate name.
  • WebSocket mode has no fallback. A failed client of ServeWS is simply closed. Let the reverse proxy in front decide what the outside world sees.
  • No replay protection. The request holds no nonce, so anyone who can read it (a plain connection, a broken certificate check) can replay it. net.censor has a censor.ReplayCache and a censor.ProbeGuard, but this package does not use them for you.
  • Where a server listens. Serve (ModeTLS) listens on every IPv4 interface because it is the public face of the server, so use long random passwords. ServePlain and ServeWS listen on 127.0.0.1 unless bind_host says otherwise; give 0.0.0.0 only when something you control reaches them from outside.
  • The password and its hash are equivalent. Whoever has the hash can authenticate. Protect the config like a password file.
  • A Link is not a tunnel.Stream. Link is the package's own connection type. For tunnel.Chain and routing use the Connector and Inbound adapters, which work on streams.

A server, a client and a fallback

This runs the whole idea on loopback with no certificate: a Trojan server in front of an echo server, with a web server as the fallback. The real client gets the echo; anyone else sends ordinary web traffic to the same port and gets the web page. In production Serve adds TLS in front of exactly this behaviour.

net-trojan-guide-loopback.salam
import mem
import net.tcp
import net.trojan
import sync

mut echo_port := 0
mut web_port := 0
mut web_seen := ""
mut ready := sync.NewWaitGroup()

func echo_server():
    srv := tcp.Bind(0)
    echo_port = tcp.Port(srv)
    sync.Done(ready)
    until true:
        c := tcp.Accept(srv)
        until true:
            data := tcp.Read(c, 4096)
            if data.Len() == 0:
                break
            end
            tcp.Write(c, "echo:" + data)
        end
        tcp.Close(c)
    end
end

func web_server():
    srv := tcp.Bind(0)
    web_port = tcp.Port(srv)
    sync.Done(ready)
    until true:
        c := tcp.Accept(srv)
        web_seen = tcp.Read(c, 4096)
        tcp.Write(c, "HTTP/1.1 200 OK\r\nContent-Length: 12\r\nConnection: close\r\n\r\nhello probes")
        tcp.Close(c)
    end
end

func start_servers():
    sync.Add(ready, 2 as i64)
    _a := spawn(echo_server)
    _b := spawn(web_server)
    sync.Wait(ready)
end

func text_of(p: void*, n: int): str:
    buf := mem.Allocate((n + 1) as u64)
    mem.Copy(buf, p, n as u64)
    (buf as u8*)[n] = 0 as u8
    ret buf as str
end

func read_text(l &: trojan.Link, want: int): str:
    buf := mem.Allocate(8192 as u64)
    defer mem.Free(buf)
    mut out := ""
    until out.Len() < want:
        n := trojan.LinkRead(l, buf, 8192)
        if n <= 0:
            break
        end
        out += text_of(buf, n)
    end
    ret out
end

func main:
    start_servers()
    mut config := trojan.NewServerConfig("", "")
    trojan.AllowPrivate(config, true)
    trojan.AddPassword(config, "secret")
    trojan.SetFallback(config, "127.0.0.1", web_port)
    server := trojan.ServePlain(config, 0)

    mut client := trojan.NewClientConfig("127.0.0.1", server.port, "secret")
    client.use_tls = false
    mut link := trojan.Dial(client, "127.0.0.1", echo_port)
    trojan.LinkWriteStr(link, "through trojan")
    println read_text(link, 19)
    trojan.LinkClose(link)

    stranger := tcp.Dial("127.0.0.1", server.port)
    tcp.Write(stranger, "GET / HTTP/1.1\r\nHost: example.com\r\n\r\n")
    println tcp.Read(stranger, 4096).Substr(0, 15)
    tcp.Close(stranger)
    trojan.Stop(server)
end
Outputecho:through trojan HTTP/1.1 200 OK

With TLS, give the config the certificate and the key and start it with Serve. Point SetFallback at a real web server on the same machine, and use the domain of the certificate as the name clients connect to.

mut config := trojan.NewServerConfig("fullchain.pem", "privkey.pem")
trojan.AddPassword(config, "a long random password")
trojan.SetFallback(config, "127.0.0.1", 8080)
server := trojan.Serve(config, 443)

Using Trojan in a proxy chain

For routing and chains, a Connector writes the Trojan request over any stream that already reaches the server, and an Inbound reads it on the server side. The stream underneath decides what is private: over a plain TCP stream the hop is readable, so put TLS under it. With a tunnel.Dialer that provides TLS as the upstream, tunnel.Via makes a complete outbound:

hop := trojan.NewConnector("proxy.example.com", 443, "a long random password")
outbound := tunnel.Via(your_tls_dialer, hop)

The guide of net.tunnel shows the whole pipeline with this connector in it. For a client that does not need routing, Dial does the TLS itself and is simpler.

What to know

  • Use Serve, with a real certificate and a real fallback website. ServePlain and ServeWS are for loopback, tests and use behind something that ends TLS.
  • No replay protection and no brute force limit of its own. Use long random passwords.
  • The server listens on every IPv4 interface in TLS mode and on loopback in plain and WebSocket mode; change it with SetBind.
  • IPv6 works: the server can listen on ::1 or :: (SetBind), TCP and UDP targets may be IPv6, and loopback tests run over ::1.
  • UDP runs inside the same connection. ReadUDPPacket gives one packet at a time.
  • Free what you allocate: LinkClose for a Link, FreeSession for a Session.
  • No logging happens by itself.

import net.trojan
import compress

func main:
    println trojan.Sha224Hex("")
    println trojan.Sha224Hex("abc")
    println trojan.PasswordHash("password")
    digest := trojan.Sha224("abc" as void*, 3)
    println digest.len, trojan.HexOf(digest) == trojan.Sha224Hex("abc")

    request := trojan.EncodeRequest("password", trojan.CmdConnect, "example.com", 443)
    println trojan.HexOf(request)
    parsed := trojan.ParseRequest(request.data, request.len)
    println parsed.status == trojan.ParseOK, parsed.cmd == trojan.CmdConnect, parsed.atyp == trojan.AtypDomain, parsed.host, parsed.port, parsed.consumed
    println parsed.hash == trojan.PasswordHash("password")
    same := trojan.EncodeRequestHash(trojan.PasswordHash("password"), trojan.CmdConnect, "example.com", 443)
    println trojan.HexOf(same) == trojan.HexOf(request)

    println "short:", trojan.ParseRequest(request.data, request.len - 3).status == trojan.ParseNeedMore
    junk := trojan.ParseRequest("GET / HTTP/1.1\r\nHost: a\r\n\r\n" as void*, 28)
    println "junk:", junk.status == trojan.ParseInvalid

    udp_request := trojan.EncodeRequest("password", trojan.CmdUDP, "2001:db8::1", 53)
    p6 := trojan.ParseRequest(udp_request.data, udp_request.len)
    println p6.cmd == trojan.CmdUDP, p6.atyp == trojan.AtypIPv6, p6.host, p6.port

    packet := trojan.EncodeUDPPacket("8.8.8.8", 53, "hi" as void*, 2)
    println trojan.HexOf(packet)
    up := trojan.ParseUDPPacket(packet.data, packet.len)
    println up.status == trojan.ParseOK, up.atyp == trojan.AtypIPv4, up.host, up.port, up.payload_off, up.payload_len, up.consumed

    addr := trojan.EncodeAddr("example.com", 443)
    println trojan.HexOf(addr)
    mut atyp := 0
    mut host := ""
    mut port := 0
    println trojan.ParseAddr(addr.data, addr.len, 0, atyp, host, port), atyp, host, port

    mut ip4 := compress.NewByteBuf()
    println trojan.ParseIPv4("192.168.1.20", ip4), trojan.FormatIPv4(ip4.data, 0)
    mut ip6 := compress.NewByteBuf()
    println trojan.ParseIPv6("2001:db8::1", ip6), trojan.FormatIPv6(ip6.data, 0)
    println trojan.CmdConnect, trojan.CmdUDP, trojan.AtypIPv4, trojan.AtypIPv6, trojan.ParseNeedMore, trojan.ParseInvalid
end

import mem
import net.tcp
import net.udp
import net.trojan
import net.tunnel
import compress
import sync

mut echo_port := 0
mut udp_port := 0
mut ready := sync.NewWaitGroup()

func text_of(p: void*, n: int): str:
    buf := mem.Allocate((n + 1) as u64)
    mem.Copy(buf, p, n as u64)
    (buf as u8*)[n] = 0 as u8
    ret buf as str
end

func echo_server():
    srv := tcp.Bind(0)
    echo_port = tcp.Port(srv)
    sync.Done(ready)
    until true:
        c := tcp.Accept(srv)
        until true:
            s := tcp.Read(c, 4096)
            if s.Len() == 0:
                break
            end
            _n := tcp.Write(c, s)
        end
        tcp.Close(c)
    end
end

func udp_server():
    s := udp.Bind(0)
    udp_port = udp.Port(s)
    sync.Done(ready)
    until true:
        p := udp.Receive(s, 2048)
        if p.data.Len() > 0:
            _n := udp.Send(s, p.from_host, p.from_port, "echo:" + p.data)
        end
    end
end

func read_all_text(l &: trojan.Link, want: int): str:
    buf := mem.Allocate(8192 as u64)
    defer mem.Free(buf)
    mut out := ""
    until out.Len() < want:
        n := trojan.LinkRead(l, buf, 8192)
        if n <= 0:
            break
        end
        out = out + text_of(buf, n)
    end
    ret out
end

func main:
    sync.Add(ready, 2 as i64)
    _t1 := spawn(echo_server)
    _t2 := spawn(udp_server)
    sync.Wait(ready)

    mut cfg := trojan.NewServerConfig("", "")
    trojan.AllowPrivate(cfg, true)
    trojan.AddPassword(cfg, "secret")
    trojan.SetMode(cfg, trojan.ModePlain)
    stats := tunnel.NewStats()
    trojan.SetStats(cfg, stats)
    srv := trojan.Listen(cfg, 0)
    println "listen ok", srv.ok, srv.host

    mut ccfg := trojan.NewClientConfig("127.0.0.1", srv.port, "secret")
    ccfg.use_tls = false

    mut ulink := trojan.DialUDP(ccfg)
    println trojan.SendUDP(ulink, "127.0.0.1", udp_port, "one" as void*, 3)
    println trojan.SendUDP(ulink, "127.0.0.1", udp_port, "two" as void*, 3)
    mut pkt := trojan.NewUDPPacket()
    mut payload := compress.NewByteBuf()
    println trojan.ReadUDPPacket(ulink, pkt, payload), text_of(payload.data, payload.len)
    println trojan.ReadUDPPacket(ulink, pkt, payload), text_of(payload.data, payload.len)
    trojan.LinkClose(ulink)

    mut link := trojan.Dial(ccfg, "127.0.0.1", echo_port)
    println trojan.LinkWriteStr(link, "hello stats")
    println read_all_text(link, 11)
    trojan.LinkClose(link)
    sync.SleepMs(300)
    println "up", stats.Up() >= 17, "down", stats.Down() >= 17, "total", stats.Total()
    trojan.Stop(srv)

    mut rt := tunnel.NewRouter("direct")
    mut rule := tunnel.NewRule("block")
    println tunnel.RulePort(rule, "" + echo_port)
    tunnel.RouterAdd(rt, rule)
    mut rd := tunnel.NewRouterDialer(rt)
    tunnel.RouterDialerAdd(rd, "block", tunnel.Block())
    mut cfg2 := trojan.NewServerConfig("", "")
    trojan.AllowPrivate(cfg2, true)
    trojan.AddPassword(cfg2, "secret")
    trojan.SetRouter(cfg2, rd)
    trojan.SetBind(cfg2, "127.0.0.1")
    trojan.SetMode(cfg2, trojan.ModePlain)
    srv2 := trojan.Listen(cfg2, 0)
    mut c2 := trojan.NewClientConfig("127.0.0.1", srv2.port, "secret")
    c2.use_tls = false
    mut blocked := trojan.Dial(c2, "127.0.0.1", echo_port)
    println trojan.LinkWriteStr(blocked, "x")
    println "blocked read", read_all_text(blocked, 1).Len()
    trojan.LinkClose(blocked)

    trojan.Stop(srv2)
end

import net.trojan
import sync

mut echo_port := 0
mut web_port := 0
mut web_seen := ""
mut ready := sync.NewWaitGroup()

func echo_server():
    srv := tcp.Bind(0)
    echo_port = tcp.Port(srv)
    sync.Done(ready)
    until true:
        c := tcp.Accept(srv)
        until true:
            data := tcp.Read(c, 4096)
            if data.Len() == 0:
                break
            end
            tcp.Write(c, "echo:" + data)
        end
        tcp.Close(c)
    end
end

func web_server():
    srv := tcp.Bind(0)
    web_port = tcp.Port(srv)
    sync.Done(ready)
    until true:
        c := tcp.Accept(srv)
        web_seen = tcp.Read(c, 4096)
        tcp.Write(c, "HTTP/1.1 200 OK\r\nContent-Length: 12\r\nConnection: close\r\n\r\nhello probes")
        tcp.Close(c)
    end
end

func start_servers():
    sync.Add(ready, 2 as i64)
    _a := spawn(echo_server)
    _b := spawn(web_server)
    sync.Wait(ready)
end

func text_of(p: void*, n: int): str:
    buf := mem.Allocate((n + 1) as u64)
    mem.Copy(buf, p, n as u64)
    (buf as u8*)[n] = 0 as u8
    ret buf as str
end

func read_text(l &: trojan.Link, want: int): str:
    buf := mem.Allocate(8192 as u64)
    defer mem.Free(buf)
    mut out := ""
    until out.Len() < want:
        n := trojan.LinkRead(l, buf, 8192)
        if n <= 0:
            break
        end
        out = out + text_of(buf, n)
    end
    ret out
end

func main:
    start_servers()

    mut config := trojan.NewServerConfig("", "")
    trojan.AllowPrivate(config, true)
    trojan.AddPassword(config, "secret")
    trojan.AddPasswordHash(config, trojan.PasswordHash("second"))
    println config.hashes.Len(), config.handshake_timeout_ms
    server := trojan.ServePlain(config, 0)
    println "server up:", server.ok, trojan.ServerPort(server) == server.port

    mut client := trojan.NewClientConfig("127.0.0.1", server.port, "secret")
    client.use_tls = false
    println client.server_port != 443, client.use_ws, client.lazy_header, client.timeout_ms
    mut link := trojan.Dial(client, "127.0.0.1", echo_port)
    println "dial ok:", trojan.LinkOk(link), link.kind == trojan.LinkTCP
    println trojan.LinkWriteStr(link, "hello trojan")
    println read_text(link, 17)
    big := mem.Allocate(30000 as u64)
    mem.Set(big, 97, 30000 as u64)
    println trojan.LinkWrite(link, big, 30000)
    mut total := 0
    until total < 30005:
        buf := mem.Allocate(8192 as u64)
        n := trojan.LinkRead(link, buf, 8192)
        mem.Free(buf)
        if n <= 0:
            break
        end
        total += n
    end
    println "echoed", total
    mem.Free(big)
    trojan.LinkSetTimeout(link, 1000)
    trojan.LinkClose(link)

    mut second := trojan.NewClientConfig("127.0.0.1", server.port, "second")
    second.use_tls = false
    second.lazy_header = true
    mut lazy := trojan.Dial(second, "127.0.0.1", echo_port)
    println trojan.LinkWriteStr(lazy, "lazy header")
    println read_text(lazy, 16)
    trojan.LinkClose(lazy)

    sync.SleepMs(300)
    stats := trojan.Stats(server)
    println "stats:", stats.accepted, stats.authed, stats.fallbacks
    trojan.Stop(server)
end

Session

struct Session
The result of Handshake: whether a valid client spoke, what it asked for and the bytes involved.
NameTypeDescription
okbool = falseTrue when the request was valid and its password hash is known.
reqRequestThe parsed request: command and target.
seencompress.ByteBufEvery byte read during the handshake. When ok is false these are what Fallback replays to the website.
extracompress.ByteBufPayload that arrived in the same read as the request and must go to the target first.
net-trojan-Handshake.salam
import net.tcp
import net.trojan
import sync

mut port := 0
mut ready := sync.NewWaitGroup()

mut config := trojan.NewServerConfig("", "")
trojan.AllowPrivate(config, true)

func serve_one():
    listener := tcp.Bind(0)
    port = tcp.Port(listener)
    sync.Done(ready)
    c := tcp.Accept(listener)
    mut link := trojan.NewTCPLink(c)
    mut session := trojan.Handshake(link, config)
    println "server saw a valid client:", session.ok
    println session.req.host, session.req.port, session.req.cmd == trojan.CmdConnect
    println "bytes read:", session.seen.len, "payload after the request:", session.extra.len
    trojan.FreeSession(session)
    trojan.LinkClose(link)
end

func main:
    trojan.AddPassword(config, "secret")
    sync.Add(ready, 1 as i64)
    t := spawn(serve_one)
    sync.Wait(ready)

    conn := tcp.Dial("127.0.0.1", port)
    request := trojan.EncodeRequest("secret", trojan.CmdConnect, "example.com", 443)
    mut link := trojan.NewTCPLink(conn)
    trojan.LinkWriteBuf(link, request)
    println trojan.LinkOk(link), trojan.LinkFd(link) > 0
    join(t)
    trojan.LinkClose(link)
end

net-trojan-Adapter.salam
import mem
import net.trojan
import net.tunnel
import compress

func text_of(p: void*, n: int): str:
    buf := mem.Allocate((n + 1) as u64)
    mem.Copy(buf, p, n as u64)
    (buf as u8*)[n] = 0 as u8
    ret buf as str
end

func main:
    hop := trojan.NewConnector("vpn.example.com", 443, "password")
    println hop.Name(), hop.Server().host, hop.Server().port

    pair := tunnel.NewPipe()
    println hop.Handshake(tunnel.PipeA(pair), tunnel.NewAddr("example.com", 443))
    mut expected := trojan.EncodeRequest("password", trojan.CmdConnect, "example.com", 443)
    buf := mem.Allocate(256 as u64)
    n := tunnel.PipeB(pair).ReadBytes(buf, 256)
    println n, n == expected.len, text_of(buf, n) == text_of(expected.data, expected.len)

    mut inbound := trojan.NewInbound()
    trojan.InboundAddPassword(inbound, "password")
    trojan.InboundAddPassword(inbound, "other")
    println inbound.Name(), inbound.hashes.Len(), inbound.replies

    good := tunnel.NewPipe()
    tunnel.PipeB(good).Write(text_of(expected.data, expected.len) + "payload")
    session := inbound.Handshake(tunnel.PipeA(good))
    println session.ok, session.target.host, session.target.port, session.protocol, session.command, session.initial

    web := tunnel.NewPipe()
    tunnel.PipeB(web).Write("GET / HTTP/1.1\r\nHost: x\r\n\r\n")
    stranger := inbound.Handshake(tunnel.PipeA(web))
    println stranger.ok, stranger.error, stranger.initial.Len()

    wrong := trojan.EncodeRequest("nope", trojan.CmdConnect, "example.com", 443)
    bad := tunnel.NewPipe()
    tunnel.PipeB(bad).Write(text_of(wrong.data, wrong.len))
    rejected := inbound.Handshake(tunnel.PipeA(bad))
    println rejected.ok, rejected.initial.Len() == wrong.len
    inbound.Reply(tunnel.PipeA(bad), rejected, false)

    mut chain := tunnel.NewChain(tunnel.Direct())
    tunnel.ChainAdd(chain, hop)
    println chain.Name()
    compress.BufFree(expected)
end

        c := tcp.Accept(srv)
        until true:
            data := tcp.Read(c, 4096)
            if data.Len() == 0:
                break
            end
            tcp.Write(c, "echo:" + data)
        end
        tcp.Close(c)
    end
end

func web_server():
    srv := tcp.Bind(0)
    web_port = tcp.Port(srv)
    sync.Done(ready)
    until true:
        c := tcp.Accept(srv)
        web_seen = tcp.Read(c, 4096)
        tcp.Write(c, "HTTP/1.1 200 OK\r\nContent-Length: 12\r\nConnection: close\r\n\r\nhello probes")
        tcp.Close(c)
    end
end

func start_servers():
    sync.Add(ready, 2 as i64)
    _a := spawn(echo_server)
    _b := spawn(web_server)
    sync.Wait(ready)
end

func text_of(p: void*, n: int): str:
    buf := mem.Allocate((n + 1) as u64)
    mem.Copy(buf, p, n as u64)
    (buf as u8*)[n] = 0 as u8
    ret buf as str
end

func read_text(l &: trojan.Link, want: int): str:
    buf := mem.Allocate(8192 as u64)
    defer mem.Free(buf)
    mut out := ""
    until out.Len() < want:
        n := trojan.LinkRead(l, buf, 8192)
        if n <= 0:
            break
        end
        out = out + text_of(buf, n)
    end
    ret out
end

func main:
    start_servers()
    mut config := trojan.NewServerConfig("", "")
    trojan.AllowPrivate(config, true)
    trojan.AddPassword(config, "wspass")
    server := trojan.ServeWS(config, 0)
    println "ws server up:", server.ok

    mut client := trojan.NewClientConfig("127.0.0.1", server.port, "wspass")
    client.use_tls = false
    client.use_ws = true
    client.ws_path = "/trojan"
    mut link := trojan.Dial(client, "127.0.0.1", echo_port)
    println "ws dial ok:", trojan.LinkOk(link), link.kind == trojan.LinkWS
    trojan.LinkWriteStr(link, "over websocket")
    println read_text(link, 19)
    trojan.LinkShutdown(link)
    trojan.LinkClose(link)
    trojan.Stop(server)
end

import mem
import net.trojan
import net.udp
import sync
import compress

mut udp_port := 0
mut ready := sync.NewWaitGroup()

func udp_server():
    s := udp.Bind(0)
    udp_port = udp.Port(s)
    sync.Done(ready)
    until true:
        p := udp.Receive(s, 2048)
        if p.data.Len() > 0:
            udp.Send(s, p.from_host, p.from_port, "echo:" + p.data)
        end
    end
end

func main:
    sync.Add(ready, 1 as i64)
    _t := spawn(udp_server)
    sync.Wait(ready)

    mut config := trojan.NewServerConfig("", "")
    trojan.AllowPrivate(config, true)
    trojan.AddPassword(config, "secret")
    server := trojan.ServePlain(config, 0)

    mut client := trojan.NewClientConfig("127.0.0.1", server.port, "secret")
    client.use_tls = false
    mut link := trojan.DialUDP(client)
    println "udp session:", trojan.LinkOk(link)
    println trojan.WriteUDPPacket(link, "127.0.0.1", udp_port, "ping" as void*, 4)

    mut packet := trojan.NewUDPPacket()
    mut payload := compress.NewByteBuf()
    println trojan.ReadUDPPacket(link, packet, payload)
    println packet.host, packet.port == udp_port, packet.status
    buf := mem.Allocate((payload.len + 1) as u64)
    mem.Copy(buf, payload.data, payload.len as u64)
    (buf as u8*)[payload.len] = 0 as u8
    println buf as str
    trojan.LinkClose(link)
    trojan.Stop(server)
end
