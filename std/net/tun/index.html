net.tun

Full device VPN mode: a tun device on Linux, macOS and Windows, split routes, DNS capture and a NAT that hands every TCP connection to a net.tunnel dialer.

import net.tun

net.tun gives a VPN app full device mode: every application on the machine goes through the tunnel, not only the ones that honour the system proxy. It creates a layer 3 tunnel device (/dev/net/tun on Linux, utun on macOS, a Wintun adapter on Windows), points the default routes at it and turns the IP packets that arrive into ordinary TCP connections, which it hands to any net.tunnel dialer: a router with direct, proxy and block rules, a VLESS, VMess, Trojan or Shadowsocks dialer, or a chain.

It has no TCP stack of its own. Like the system stack of sing-tun, it rewrites the addresses of each TCP packet so the connection lands on a local listener, and the kernel does all of TCP. Only IP and TCP headers are touched, with incremental checksum updates, so throughput stays close to the kernel.

app -> 1.1.1.1:443                        kernel TCP
  |  routes 0.0.0.0/1, 128.0.0.0/1             ^
  v                                            |
tun device -> Nat.Rewrite: 172.19.0.2:10000 -> 172.19.0.1:listener
                                               |
listener accepts, Nat.Lookup(10000) = 1.1.1.1:443, sniff SNI, dialer.Dial(host:443)

Things to know:

  • It needs privileges. Root or CAP_NET_ADMIN on Linux, root on macOS, administrator rights and wintun.dll next to the executable on Windows (download it from wintun.net). Call Available first and show its reason.
  • Exclude your servers. Put the VPN server names or addresses in exclude. They are resolved before the routes change and keep a route through the old gateway. For direct rules use Bypass, which binds to the physical interface.
  • TCP and DNS only. UDP queries to port 53 are answered through the dialer over TCP. Other UDP (QUIC, games, calls) and ICMP are dropped, so browsers fall back to TCP. IPv6 is blocked rather than leaked.
  • Routes never replace the default route. Two /1 routes win over it and Stop removes everything it added. If the process is killed, the Linux and Windows device disappears with it; host routes for excluded servers stay until reboot.
  • Linux is tested end to end by tools/interop/tun_e2e.sh in a container. The macOS and Windows device code is type checked for every target.

Full device mode in a VPN app

Build the same router dialer you would give a local SOCKS inbound, but use Bypass() for the direct tag so direct traffic leaves through the physical interface instead of looping back into the tunnel. Pass the VPN server in exclude, start the session when the user flips the switch and stop it when they flip it back or the app quits.

import net.tun
import net.tunnel

func start_full_device(proxy: dyn tunnel.Dialer, server: str): tun.Session:
    av := tun.Available()
    if not av.ok:
        println "full device mode is off:", av.reason
        ret tun.Session {}
    end
    rules := tunnel.ParseRules("domain:ir direct\nprivate direct", "proxy")
    mut router := tunnel.NewRouterDialer(rules.router)
    tunnel.RouterDialerAdd(router, "direct", tun.Bypass())
    tunnel.RouterDialerAdd(router, "block", tunnel.Block())
    tunnel.RouterDialerAdd(router, "proxy", proxy)
    mut servers := Vector {} as Vector<str>
    servers.Push(server)
    s := tun.Start(tun.Options { dialer = router, exclude = servers })
    if not s.ok:
        println "could not start:", s.error
    end
    ret s
end

func main():
    s := start_full_device(tunnel.Direct(), "vpn.example.com")
    if s.ok:
        st := tun.SessionStats(s)
        println st.connections, st.dns_queries
        tun.Stop(s)
    end
end

Show Available().reason next to a disabled switch when the process lacks the rights; a desktop app usually starts a small privileged helper for this mode. Read SessionStats on a timer for the traffic counters.

Handling packets yourself

OpenDevice gives a bare device with no address, routes or stack: each Read returns one IP packet and each Write injects one. Combine it with Parse, Nat and the packet builders to write your own stack, a packet logger or a test harness. Configure the address and routes yourself, or look at what BuildPlan would run.

import mem
import sync
import net.tun

func main():
    dev := tun.OpenDevice("salam0", 1500)
    if not dev.ok:
        println dev.error
        ret
    end
    nat := tun.NewNat("172.19.0.1", "172.19.0.2", 7000, 0)
    buf := mem.Allocate(65536 as u64)
    repeat 1000:
        n := dev.Read(buf, 65536, 200)
        if n > 0:
            info := tun.Parse(buf, n)
            println info.src, info.sport, "->", info.dst, info.dport
            if nat.Rewrite(buf, n, sync.NowMs()) == tun.action_reply:
                _w := dev.Write(buf, n)
            end
        end
    end
    mem.Free(buf)
    tun.FreeNat(nat)
    dev.Close()
end

import net.tun

func main():
    body := "hi"
    pk := tun.BuildTcp4(
        "172.19.0.1", 40000, "93.184.216.34", 443, 1 as i64, 0 as i64, tun.tcp_syn, body as void*,
        2
    )
    info := tun.Parse(pk.data, pk.len)
    println info.valid,
        info.version,
        info.proto == tun.proto_tcp,
        info.src,
        info.sport,
        info.dst,
        info.dport
    println "syn",
        (info.flags & tun.tcp_syn) != 0,
        "ack",
        (info.flags & tun.tcp_ack) != 0,
        "payload",
        info.payload_len,
        "of",
        info.total
    println "checksums ok", tun.ChecksumsOk(pk.data, pk.len)
    p := pk.data as u8*
    p[8] = 32 as u8
    println "after edit", tun.ChecksumsOk(pk.data, pk.len)
    tun.FinishChecksums(pk.data, pk.len)
    println "after finish", tun.ChecksumsOk(pk.data, pk.len)
    println "header sum", tun.Checksum(pk.data, 20)
    println "rfc 1624", tun.ChecksumAdjust(56623, 21845, 12933)
    udp := tun.BuildUdp4("10.0.0.2", 5000, "10.0.0.1", 53, null, 0)
    u := tun.Parse(udp.data, udp.len)
    println "udp", u.proto == tun.proto_udp, u.header_len, u.payload_offset, u.fragment
    println "other flags", tun.tcp_fin, tun.tcp_rst, tun.tcp_psh, tun.proto_icmp, tun.proto_icmp6
    tun.FreePacket(pk)
    tun.FreePacket(udp)
end

import net.tun

func main():
    nat := tun.NewNat("172.19.0.1", "172.19.0.2", 7000, 0)
    out := tun.BuildTcp4(
        "172.19.0.1", 51000, "1.1.1.1", 443, 1 as i64, 0 as i64, tun.tcp_syn, null, 0
    )
    println "forward", nat.Rewrite(out.data, out.len, 0 as i64) == tun.action_reply
    seen := tun.Parse(out.data, out.len)
    println "kernel sees", seen.src, seen.sport, seen.dst, seen.dport
    e := nat.Lookup(seen.sport)
    println "original", e.ok, e.src, e.sport, e.dst, e.dport
    back := tun.BuildTcp4(
        "172.19.0.1",
        nat.Port(),
        "172.19.0.2",
        seen.sport,
        9 as i64,
        2 as i64,
        tun.tcp_syn | tun.tcp_ack,
        null,
        0
    )
    _r := nat.Rewrite(back.data, back.len, 1 as i64)
    app := tun.Parse(back.data, back.len)
    println "app sees", app.src, app.sport, app.dst, app.dport
    dns := tun.BuildUdp4("172.19.0.1", 5353, "172.19.0.2", 53, null, 0)
    println "dns", nat.Rewrite(dns.data, dns.len, 2 as i64) == tun.action_dns
    println "drop", tun.action_drop, "ipv6", tun.action_ipv6
    println "active", nat.Active(), "expired", nat.Sweep(3600000 as i64), "active", nat.Active()
    tun.FreePacket(out)
    tun.FreePacket(back)
    tun.FreePacket(dns)
    tun.FreeNat(nat)
end
