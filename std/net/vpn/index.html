net.vpn

One call that registers every proxy protocol with net.tunnel, so a v2ray style JSON config runs both sides, with a ready to use runner.

import net.vpn
On this page
  1. Types
  2. Constants and variables
  3. Functions

This package joins net.tunnel to every protocol package: socks, http, trojan, vless, vmess and shadowsocks, with the v2ray stream settings of net.transport (tcp, tls, ws, httpupgrade, and TLS hello fragmenting). One call to Register and a v2ray style JSON config runs, both the inbounds and the outbounds.

  • Register or RegisterWith installs the makers. After that tunnel.CreateInbound, tunnel.CreateOutbound and tunnel.BuildDialer understand every protocol above.
  • StartAll starts every inbound of a config and RunConfig is a whole program: load, check, start, wait for Ctrl-C.
  • Local by default. An inbound listens on 127.0.0.1 unless its listen field says otherwise. Write "listen": "0.0.0.0" for a server that other machines connect to.
  • With use_doh the names of servers are resolved over DNS over HTTPS, and routing rules with IP ranges can match domain targets, with answers cached.
  • Trojan inbounds start their own server; their bind address follows the trojan package.
net-vpn-guide-app.salam
import net.socks
import net.tcp
import net.tunnel
import net.vpn

mut echo_srv := tcp.Server {}

func echo_once():
    c := tcp.Accept(echo_srv)
    s := tunnel.WrapTcp(c)
    data := s.Read(100)
    s.Write("echo:" + data)
    s.Close()
end

func main:
    echo_srv = tcp.Bind(0)
    _t := spawn(echo_once)
    vpn.Register()
    cfg := tunnel.ParseConfig(`{"inbounds":[{"tag":"in","port":0,"protocol":"mixed"}],"outbounds":[{"tag":"direct","protocol":"freedom"}]}`)
    stats := tunnel.NewStats()
    listeners := vpn.StartAll(cfg, stats)
    println "listening:", vpn.Ok(listeners[0])
    s := tunnel.DialTcp("127.0.0.1", listeners[0].port, 3000)
    r := socks.ClientHandshake(s, socks.Client { version = 5 }, socks.CmdConnect, "127.0.0.1", tcp.Port(echo_srv))
    println "socks handshake:", r.ok
    s.Write("hello")
    println s.Read(100)
    s.Close()
    vpn.Stop(listeners[0])
    tcp.CloseServer(echo_srv)
end
Outputlistening: true socks handshake: true echo:hello

Options

struct Options
Settings for Register: fragmenting the TLS hello and DNS over HTTPS.
NameTypeDescription
fragmentbool = falseSplit the TLS hello of outgoing connections into small pieces.
fragment_minint = 4Smallest piece in bytes.
fragment_maxint = 12Largest piece in bytes.
fragment_delay_msint = 5Pause between pieces.
fragment_modestr = "tcp"What fragmenting splits: "tcp", "tls" or "both".
fragment_snibool = falseAlso cut inside the server name.
fragment_writesint = 1How many of the first handshake writes are fragmented.
read_timeout_msint = 0Idle read timeout of outgoing connections. 0 waits forever.
write_timeout_msint = 0Idle write timeout of outgoing connections. 0 waits forever.
use_dohbool = falseResolve server names through the resolver below.
resolverdoh.ResolverThe DNS over HTTPS resolver.
net-vpn-Register.salam
import net.tunnel
import net.vpn

func main:
    vpn.RegisterWith(vpn.Options { fragment = true, fragment_min = 2, fragment_max = 8 })
    each proto in ["vless", "vmess", "trojan", "socks", "http", "mixed"]:
        ib := tunnel.CreateInbound(tunnel.InboundConfig { protocol = proto, settings = `{"clients":[{"id":"0b3a8c6e-1f3d-4d7e-9a50-3f8e6a1c2b44","password":"secret"}],"method":"aes-128-gcm","password":"secret"}` })
        println "inbound", proto, ib.Name()
    end
    each proto in ["vless", "vmess", "trojan", "shadowsocks", "ss", "socks", "http", "freedom"]:
        ob := tunnel.CreateOutbound(tunnel.OutboundConfig { protocol = proto, settings = `{"address":"vpn.example.com","port":443}` })
        println "outbound", proto, ob.Name()
    end

    cfg := tunnel.ParseConfig(`{"inbounds":[{"tag":"in","port":0,"protocol":"mixed"}],"outbounds":[{"tag":"direct","protocol":"freedom"},{"tag":"out","protocol":"socks","settings":{"address":"127.0.0.1","port":1080}}]}`)
    dialer := vpn.BuildDialer(cfg)
    println dialer.Name()

    vpn.Register()
    println tunnel.CreateInbound(tunnel.InboundConfig { protocol = "vless" }).Name()
end

net-vpn-Settings.salam
import net.transport
import net.tunnel
import net.vpn

func main:
    raw := `{"network":"ws","security":"tls","tlsSettings":{"serverName":"cdn.example.com"},"wsSettings":{"path":"/chat","headers":{"Host":"front.example.com"}}}`
    s := vpn.StreamSettings(raw)
    println s.network, s.tls, s.sni, s.path, s.host

    node := tunnel.Node { protocol = "vless", host = "vpn.example.com", port = 443, network = "ws", tls = true, sni = "vpn.example.com", path = "/ws" }
    n := vpn.NodeSettings(node)
    println n.network, n.tls, n.sni, n.path

    each proto in ["vless", "vmess", "shadowsocks", "trojan", "socks", "http", "wireguard"]:
        println proto, vpn.DialerFromNode(tunnel.Node { protocol = proto, host = "vpn.example.com", port = 443 }).Name()
    end

    t := vpn.ToTunnelAddr(transport.NewAddr("example.com", 443))
    println t.host, t.port
    back := vpn.ToTransportAddr(tunnel.NewAddr("example.org", 80))
    println back.host, back.port
end
Outputws true cdn.example.com /chat front.example.com ws true vpn.example.com /ws vless vless vmess vmess shadowsocks shadowsocks trojan trojan socks socks5 http http wireguard block example.com 443 example.org 80

Constants and variables

PENDING_PER_SOURCE

const PENDING_PER_SOURCE: i32 = 32
How many connections of one address Start lets sit in a handshake at once on each inbound it opens: 32. A client that opens a connection and says nothing is dropped after the handshake time limit.
