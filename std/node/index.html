node

A Salam program under Node.js: modules, files, processes and JavaScript objects.

import node
On this page
  1. Functions

This package is for programs translated to JavaScript with salam js and run under Node.js, the VS Code editor extension for instance. Load any Node or npm module with Require and call its methods with the Call... functions. That is why the examples here are only checked.

JavaScript objects are held as one i64. The call functions are named after the types of their arguments: CallS1 for one string, CallV1 for one JavaScript value.

Functions

Require

func Require(id: str): i64
Loads a Node module with require.
node-Basics.salam
import node

func main:
    println node.Platform().len() > 0, node.IsWindows() == (node.Platform() == "win32")
    folder := node.MkdtempSync(node.PathJoin2(node.TmpDir(), "salam-"))
    path := node.PathJoin3(folder, "data", "note.txt")
    println node.PathDirname(path) == node.PathJoin2(folder, "data")
    fs := node.Require("fs")
    _ := node.CallS1(fs, "mkdirSync", node.PathDirname(path))
    node.WriteFileSync(path, "hello from node")
    println node.ReadFileSync(path)
    node.RmSyncRecursiveForce(folder)

    obj := node.NewObject()
    node.SetStr(obj, "name", "Maryam")
    node.SetNum(obj, "age", 30)
    node.SetBool(obj, "admin", true)
    arr := node.NewArray()
    node.PushStr(arr, "a")
    node.Push(arr, obj)
    node.SetProp(obj, "tags", arr)
    println node.GetStr(obj, "name"), node.GetNum(obj, "age"), node.GetBool(obj, "admin")
    println node.GetNum(node.GetProp(obj, "tags"), "length")
    json := node.Global("JSON")
    println node.ToStr(node.CallS1(json, "stringify", "x"))
    println node.IsNullish(node.GetProp(obj, "missing")), node.ToBool(node.GetProp(obj, "admin"))

    math := node.Global("Math")
    println node.ToStr(node.CallV2(math, "max", node.GetProp(obj, "age"), node.GetProp(obj, "age")))
    date := node.New1(node.Global("Date"), node.GetProp(obj, "age"))
    println node.ToStr(node.Call0(date, "getTime"))
    path_module := node.Require("path")
    println node.ToStr(node.CallS2(path_module, "join", "a", "b"))
    println node.ToStr(node.CallS3(path_module, "join", "a", "b", "c"))
    node.SetPropFn0(obj, "hello", ():
        println "hello from a function"
    end)
    _ := node.Call0(obj, "hello")
    node.SetPropFn1R(obj, "twice", (x: i64): ret node.CallV2(math, "max", x, x) end)
    println node.ToStr(node.Invoke1(node.GetProp(obj, "twice"), node.GetProp(obj, "age")))
    println node.IsNullish(node.Exports())
end

CallV1

func CallV1(obj: i64, method: str, a: i64): i64
With one JavaScript value.
node-Calls.salam
import node

func main:
    arr := node.New0(node.Global("Array"))
    node.PushStr(arr, "b")
    node.PushStr(arr, "a")
    _ := node.Call0(arr, "sort")
    println node.ToStr(node.CallS1(arr, "join", ", "))
    number := node.CallV1(node.Global("Math"), "abs", node.GetProp(node.Global("Number"), "MIN_SAFE_INTEGER"))
    println node.ToStr(number).len() > 0
    println node.ToStr(node.CallV3(node.Global("Math"), "max", number, number, number)) == node.ToStr(number)
    date := node.New4(node.Global("Date"), node.GetProp(node.Global("Number"), "EPSILON"), number, number, number)
    println node.IsNullish(date)
    buf := node.CallSB1(node.NewArray(), "includes", "x", false)
    println node.ToStr(buf)
    println node.ToStr(node.CallB1(node.Global("JSON"), "stringify", true))
    println node.ToStr(node.CallVS1(arr, "fill", node.GetProp(arr, "length"), "x"))

    emitter := node.New0(node.GetProp(node.Require("events"), "EventEmitter"))
    _ := node.CallSFn0(emitter, "on", "ping", ():
        println "the ping event arrived"
    end)
    _ := node.CallS1(emitter, "emit", "ping")
    node.SetPropFn1(emitter, "show", (x: i64):
        println "show:", node.ToStr(x)
    end)
    _ := node.Invoke2(node.GetProp(emitter, "show"), arr, arr)
    _ := node.Invoke0(node.GetProp(emitter, "show"))

    _ := node.CallFn0(node.Global("process"), "nextTick", ():
        println "after the current job"
    end)
    _ := node.CallFn1(arr, "forEach", (x: i64):
        println "element:", node.ToStr(x)
    end)
    promise := node.NewPromise((resolve: i64, reject: i64):
        _ := node.Invoke1(resolve, reject)
    end)
    println node.IsNullish(promise)

    args := node.NewArray()
    node.PushStr(args, "-e")
    node.PushStr(args, "console.log('hello from the child')")
    node.ExecFile("node", args, node.TmpDir(), (failed: bool, out: str, err: str):
        println "child:", failed, out.len() > 0, err.len()
    end)
    node.ExecFileRaw("node", args, node.TmpDir(), (code: i64, out: str, _err: str):
        println "raw child:", node.IsNullish(code), out.len() > 0
    end)
end
