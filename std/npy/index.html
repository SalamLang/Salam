npy

Reading and writing NumPy .npy and .npz files.

import npy
On this page
  1. Types
  2. Constants and variables
  3. Functions

.npy files are the standard way NumPy arrays are stored in Python, and .npz keeps several named arrays in one zip archive. With this package you can move data and model weights between Python and Salam.

Types

NpyArray

struct NpyArray
One multi dimensional NumPy array. The data is always held in memory as 64 bit floats.
NameTypeDescription
shapeVectorThe size of each dimension, such as [2, 3].
dtypeintThe data type in the file; one of the NPY_ constants.
dataVectorThe elements in row order.
npy-Npy.salam
import npy
import matrix

func main:
    mut m := matrix.Matrix { rows = 2, cols = 3 }
    repeat 6 in i:
        m.a.push(i as f64)
    end
    mut a := npy.FromMatrix(m)
    println a.shape, a.ndim(), a.count(), a.dtype == npy.NPY_F64
    bytes := npy.WriteNpy(a)
    println bytes.len > 64
    mut good := false
    mut back := npy.ReadNpy(bytes.data, bytes.len, good)
    println good, back.shape, back.data
    m2 := npy.ToMatrix(back, good)
    println good, m2.at(1, 2)
    println npy.WriteNpyFile(a, "a.npy")
    mut from_file := npy.ReadNpyFile("a.npy", good)
    println good, from_file.count()
    mut keys := Vector {} as Vector<str>
    keys.push("x")
    keys.push("y")
    mut arrays := Vector {} as Vector<npy.NpyArray>
    arrays.push(a)
    arrays.push(back)
    println npy.NpzWrite("ab.npz", keys, arrays)
    println npy.NpzList("ab.npz", good), good
    mut y := npy.NpzRead("ab.npz", "y", good)
    println good, y.shape
    println npy.NPY_F32, npy.NPY_I8, npy.NPY_I16, npy.NPY_I32, npy.NPY_I64, npy.NPY_U8, npy.NPY_U16, npy.NPY_U32, npy.NPY_BOOL
    y.free()
    from_file.free()
    back.free()
    a.free()
end
Output[2, 3] 2 6 true true true [2, 3] [0, 1, 2, 3, 4, 5] true 5 true true 6 true ["x", "y"] true true [2, 3] 1 2 3 4 5 6 7 8 9
