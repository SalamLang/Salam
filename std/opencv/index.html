opencv

Computer vision with OpenCV: image processing, features, face detection, neural networks and video.

import opencv
On this page
  1. Types
  2. Constants and variables
  3. Functions

This package binds the OpenCV library and gives most of the everyday computer vision work: color conversion, resizing, filters, edge detection and thresholds, contours, ORB and SIFT features, face detection with a cascade classifier, running DNN networks, and reading and writing video and camera frames.

It needs OpenCV and the Salam shim at run time; check with Available that the library was loaded. For simpler work with no dependency, see the image package. The examples on the site are only type-checked.

import opencv
import image

func main:
    mut cascade := opencv.LoadCascade("haarcascade_frontalface_default.xml")
    mut photo := opencv.Read("people_en.jpg")
    if cascade.ok():
        each face in cascade.detect(photo.to_gray(), 1.1, 5, 100):
            photo.rectangle(face.x, face.y, face.width, face.height, opencv.Green(), 2)
        end
    end
    cascade.free()

    mut net := opencv.LoadNet("MobileNetSSD_en.caffemodel", "MobileNetSSD_en.prototxt")
    mut single := opencv.LoadNetSingleFile("model_en.onnx")
    if net.ok():
        _ := net.set_input(photo, 0.007843, opencv.RGB(127, 127, 127), false, opencv.Size { width = 300, height = 300 })
        output := net.forward()
        _ := net.forward_layer("detection_out")
        each obj in opencv.ParseSSDDetections(output, photo.width(), photo.height(), 0.5):
            println obj.class_id, obj.confidence, obj.x, obj.y, obj.width, obj.height
        end
    end
    net.free()
    single.free()

    mut video := opencv.OpenVideo("clip_en.mp4")
    mut camera := opencv.OpenCamera(0)
    mut writer := opencv.NewVideoWriter("out_en.mp4", "mp4v", video.fps(), 640, 480)
    println video.ok(), video.frame_count(), camera.ok(), writer.ok()
    opencv.CreateTrackbar("threshold", "preview", 100, 255)
    until video.ok():
        frame := video.read()
        if frame.empty():
            break
        end
        level := opencv.GetTrackbarPos("threshold", "preview")
        edges := frame.to_gray().canny(level as f64, 200.0)
        _ := writer.write(frame)
        opencv.Imshow("preview", edges)
        if opencv.WaitKey(1) == 27:
            break
        end
    end
    opencv.SetTrackbarPos("threshold", "preview", 50)
    opencv.DestroyWindow("preview")
    opencv.DestroyAllWindows()
    video.close()
    camera.close()
    writer.close()

    picture := opencv.ToImage(photo)
    println picture.width
    _ := opencv.FromImage(image.New(8, 8))
end

opencv-Features.salam
import opencv

func main:
    mut scene := opencv.Read("scene_en.jpg")
    mut target := opencv.Read("object_en.jpg")
    mut contours := scene.to_gray().threshold_otsu(255.0).find_contours()
    println contours.ok(), contours.count()
    repeat contours.count() in i:
        box := contours.bounding_rect(i)
        println contours.area(i), contours.point_count(i), contours.point(i, 0).x, box.x, box.y, box.width, box.height
        scene.draw_contour(contours, i, opencv.Green(), 2)
    end
    contours.free()

    corners := scene.to_gray().good_features(50, 0.01, 10.0)
    println corners.len()
    mut keys1 := scene.detect_orb(500)
    mut keys2 := target.detect_orb(500)
    mut sift := scene.detect_sift(500)
    println keys1.ok(), keys1.count(), sift.count()
    if keys1.count() > 0:
        first := keys1.get(0)
        println first.x, first.y, first.size, first.angle, first.response
    end
    matches := opencv.MatchDescriptors(keys1.descriptors(), keys2.descriptors(), 50, opencv.NORM_HAMMING)
    each m in matches:
        println m.query_idx, m.train_idx, m.distance
    end
    println opencv.NORM_L2
    keys1.free()
    keys2.free()
    sift.free()

    found := scene.find_template(target)
    println found.found, found.x, found.y, found.score
    mut quad := Vector {} as Vector<opencv.Point>
    quad.push(opencv.Point { x = 0, y = 0 })
    quad.push(opencv.Point { x = 400, y = 10 })
    quad.push(opencv.Point { x = 410, y = 580 })
    quad.push(opencv.Point { x = 5, y = 590 })
    flat := scene.warp_perspective(quad, 420, 594)
    _ := opencv.Write("document_en.png", flat)
end

import opencv

func main:
    if not opencv.Available():
        println "OpenCV is not available:", opencv.LastError()
        ret
    end
    println opencv.Version()
    mut photo := opencv.Read("photo_en.jpg")
    if not photo.ok() or photo.empty():
        photo = opencv.NewBlank(320, 240, 3, opencv.White())
    end
    dims := photo.size()
    println photo.width(), photo.height(), photo.channels(), dims.width, dims.height, photo.step(), photo.total(), photo.data() != null
    color := photo.at(10, 10)
    println color.r, color.g, color.b
    photo.set_pixel(0, 0, opencv.RGB(255, 128, 0))
    black := opencv.Zeros(10, 10, 1)
    println black.channels(), opencv.Black().r, opencv.Red().r, opencv.Green().g, opencv.Blue().b

    gray := photo.to_gray()
    _ := photo.to_rgb()
    hsv := photo.to_hsv()
    _ := hsv.hsv_to_rgb()
    _ := photo.convert_color(opencv.COLOR_BGR2GRAY)
    _ := gray.convert_color(opencv.COLOR_GRAY2BGR)
    println opencv.COLOR_RGB2GRAY, opencv.COLOR_GRAY2RGB, opencv.COLOR_BGR2RGB, opencv.COLOR_RGB2BGR, opencv.COLOR_BGR2HSV, opencv.COLOR_HSV2BGR, opencv.COLOR_RGB2HSV, opencv.COLOR_HSV2RGB

    small := photo.resize(160, 120, opencv.INTERP_AREA)
    _ := photo.resize_scale(2.0, 2.0)
    _ := photo.crop(10, 10, 100, 80)
    _ := photo.rotate90(opencv.ROTATE_90_CW)
    _ := photo.rotate(45.0, 1.0)
    _ := photo.flip(opencv.FLIP_BOTH)
    _ := photo.flip_horizontal()
    _ := photo.flip_vertical()
    println opencv.INTERP_NEAREST, opencv.INTERP_LINEAR, opencv.INTERP_CUBIC, opencv.INTERP_LANCZOS4, opencv.ROTATE_180, opencv.ROTATE_90_CCW, opencv.FLIP_VERTICAL, opencv.FLIP_HORIZONTAL
    copied := small.clone()
    println copied.save("small_en.png"), opencv.Write("gray_en.png", gray)
    copied.free()
    photo.free()
end

import opencv

func main:
    mut photo := opencv.Read("photo_en.jpg")
    gray := photo.to_gray()
    _ := photo.blur(5)
    _ := photo.gaussian_blur(5, 1.5)
    _ := photo.median_blur(5)
    edges := gray.canny(50.0, 150.0)
    _ := gray.threshold(128.0, 255.0, opencv.THRESH_BINARY)
    _ := gray.threshold_otsu(255.0)
