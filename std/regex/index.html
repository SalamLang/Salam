regex

Searching and replacing with a pattern: finding a number, an email address or a date in text and pulling it apart.

import regex
On this page
  1. Types
  2. Functions

A regular expression is a small language for describing the shape of text; [0-9]+ means "one or more digits". With one you can ask whether text matches a pattern, find every place it matches, and pull pieces out or replace them.

Every function comes in two forms. The one that takes a Regex uses a pattern compiled in advance and is faster when you use it repeatedly. The one whose name ends in Str takes the pattern as text and compiles it each time; that is more convenient for a one-off.

Parentheses in a pattern make a "group", and (?P<name>...) makes a named one. Group zero is the whole match. In a replacement, $1 and $2 stand for the groups.

Types

MatchResult

struct MatchResult
The result of a search. Rather than reading the fields directly, the Group functions are usually simpler.
NameTypeDescription
okbool = falseTrue when a match was found.
startint = -1Where the match starts, in bytes.
stopint = -1Where the match ends, in bytes; that position is not part of it.
countint = 0The number of groups, not counting group zero.
textstr = ""The whole text that was searched, not only the part that matched. For the match itself use Group(m, 0).
patternstr = ""The pattern the search was made with.
gsint[]Where each group starts.
geint[]Where each group ends.
regex-Group.salam
import regex

func main:
    r := regex.Compile("(?P<year>[0-9]{4})-(?P<month>[0-9]{2})")
    m := regex.FindMatch(r, "date: 2025-09")
    println m.ok, m.start, m.stop, m.text
    println regex.GroupCount(m), regex.GroupOk(m, 1)
    println regex.Group(m, 0), regex.Group(m, 1), regex.Group(m, 2)
    println regex.GroupStart(m, 2), regex.GroupEnd(m, 2)
    println regex.GroupIndex(m, "month"), regex.GroupByName(m, "year")
    m2 := regex.FindMatchFrom(r, "2024-01 and 2025-02", 5)
    println regex.Group(m2, 0)
    println regex.Group(regex.FindMatchStr("b+", "aabbbc"), 0)
    println regex.FindSubmatch(r, "2025-09")
    println regex.FindSubmatchStr("(a)(b)", "xab")
end
Outputtrue 6 13 date: 2025-09 2 true 2025-09 2025 09 11 13 2 2025 2025-02 bbb ["2025-09", "2025", "09"] ["ab", "a", "b"]

regex-Compile.salam
import regex

func main:
    r := regex.Compile("[0-9]+")
    println r.pattern, regex.IsValid("[0-9]+"), regex.IsValid("[0-9")
    println regex.Match(r, "abc123"), regex.Find(r, "abc123")
    println regex.Replace(r, "a1 b22 c333", "#")
    println regex.ReplaceAll(r, "a1 b22 c333", "#")
    println regex.MatchStr("^[a-z]+$", "salam"), regex.FindStr("o", "salam donya")
    println regex.ReplaceStr("a", "banana", "o"), regex.ReplaceAllStr("a", "banana", "o")
end
Output[0-9]+ true false true 3 a# b22 c333 a# b# c# true 7 bonana bonono

regex-FindAll.salam
import regex

func main:
    r := regex.Compile("[0-9]+")
    text := "year 1403, month 7, day 12"
    println regex.FindAll(r, text)
    println regex.FindAllN(r, text, 2)
    println regex.FindAllStr("[a-z]+", "one two three")
    println regex.Split(regex.Compile(" *, *"), "a , b,c ,  d")
    println regex.SplitN(regex.Compile(","), "a,b,c,d", 2)
    println regex.SplitStr("-+", "2025--09---29")
    all := regex.FindAllMatches(r, text, -1)
    each m in all:
        println regex.Group(m, 0), m.start, m.stop
    end
end
Output["1403", "7", "12"] ["1403", "7"] ["one", "two", "three"] ["a", "b", "c", "d"] ["a", "b,c,d"] ["2025", "09", "29"] 1403 5 9 7 17 18 12 24 26

Expand

func Expand(m: MatchResult, template: str): str
Fills a template from the groups of a match; $1, $2 and so on stand for the groups.
regex-Expand.salam
import regex

func main:
    r := regex.Compile("([a-z]+)@([a-z]+)")
    text := "ali@home, sara@work"
    m := regex.FindMatch(r, text)
    println regex.Expand(m, "$2 <- $1")
    println regex.ReplaceExpand(r, text, "$2:$1")
    println regex.ReplaceAllExpand(r, text, "$2:$1")
    println regex.ReplaceExpandStr("(\\w+) (\\w+)", "hello world", "$2 $1")
    println regex.ReplaceAllExpandStr("(a)(b)", "abab", "$2$1")
end
Outputhome <- ali home:ali, sara@work home:ali, work:sara world hello baba
