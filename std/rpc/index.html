rpc

JSON-RPC 2.0 and MessagePack-RPC: a server, a client and registering methods.

import rpc
On this page
  1. Types
  2. Constants and variables
  3. Functions

With a remote procedure call one program calls another program's methods as if they were its own routines. On the server side you tie each method to a routine with Register; that routine takes the call's number, reads the parameters with CallParamInt and the like, and answers with Reply... or Fail.

The messages are exchanged with the JSON codec (JSON-RPC 2.0) or MessagePack, and over TCP they are separated by one of three framings, line, header (like LSP) or length prefixed; the client and the server must have the same framing. HandleText does the same thing without a network, so you can put RPC behind an net/http route.

rpc-Serve.salam
import rpc
import net.tcp
import compress

func echo(c: i64):
    rpc.Reply(c, rpc.CallParams(c))
end

func main:
    rpc.Register("echo", echo)
    if rpc.HandleText("").len() == 0:
        _ := rpc.ServeStdio(rpc.RPC_FRAME_HEADER)
        ret
    end

    mut s := rpc.NewServer(4000, rpc.RPC_CODEC_MSGPACK)
    rpc.SetFraming(s, rpc.RPC_FRAME_LENGTH)
    if rpc.Listen(s):
        println "on port", rpc.Port(s), s.running
        rpc.Serve(s)
    end

    conn := tcp.Dial("127.0.0.1", 4001)
    _ := rpc.ServeConn(conn.fd, rpc.RPC_CODEC_JSON, rpc.RPC_FRAME_LINE)

    c := tcp.Dial("127.0.0.1", 4002)
    mut client := rpc.ClientFromFd(c.fd, rpc.RPC_CODEC_JSON)
    println rpc.ResultStr(rpc.CallNoParams(client, "version"))

    mut stream := rpc.NewStream(c.fd)
    _ := rpc.WriteFrameStr(c.fd, rpc.RPC_FRAME_LINE, `{"jsonrpc":"2.0","method":"echo","params":[1],"id":1}`)
    _ := rpc.WriteFrame(c.fd, rpc.RPC_FRAME_LINE, compress.BufFromStr(`{"jsonrpc":"2.0","method":"echo","id":2}`))
    mut ok := false
    println rpc.ReadFrameStr(stream, rpc.RPC_FRAME_LINE, ok)
    answer := rpc.HandlePayload(rpc.RPC_CODEC_JSON, compress.BufFromStr(`{"jsonrpc":"2.0","method":"echo","params":[3],"id":3}`))
    println compress.BufToStrUnsafe(answer)
    rpc.FreeStream(stream)
    rpc.StopServer(s)
end

rpc-Frames.salam
import rpc
import compress

func main:
    line := rpc.FrameEncodeStr(rpc.RPC_FRAME_LINE, `{"a":1}`)
    headed := rpc.FrameEncodeStr(rpc.RPC_FRAME_HEADER, `{"b":2}`)
    sized := rpc.FrameEncode(rpc.RPC_FRAME_LENGTH, compress.BufFromStr(`{"c":3}`))
    println line.len, headed.len, sized.len
    println compress.BufToStrUnsafe(headed).len() > 0

    mut s := rpc.NewMemStream()
    rpc.FeedStr(s, `{"x":1}` + "\n" + `{"x":2}` + "\n")
    mut ok := false
    println rpc.ReadFrameStr(s, rpc.RPC_FRAME_LINE, ok), ok
    println rpc.StreamPending(s)
    println rpc.ReadFrameStr(s, rpc.RPC_FRAME_LINE, ok)
    _ := rpc.ReadFrameStr(s, rpc.RPC_FRAME_LINE, ok)
    println ok, rpc.StreamClosed(s)

    mut s2 := rpc.NewMemStream()
    rpc.FeedBuf(s2, headed)
    rpc.FeedBuf(s2, sized)
    println rpc.ReadFrameStr(s2, rpc.RPC_FRAME_HEADER, ok)
    b := rpc.ReadFrame(s2, rpc.RPC_FRAME_LENGTH, ok)
    println compress.BufToStrUnsafe(b), ok
    println rpc.RPC_READ_CHUNK, rpc.RPC_MAX_FRAME
    rpc.FreeStream(s)
    rpc.FreeStream(s2)
end

rpc-Client.salam
import rpc
import encoding.value
import net.http
import net.router
import sync

mut server := rpc.NewServer(0, rpc.RPC_CODEC_JSON)
mut ready := sync.NewWaitGroup()

func add(c: i64):
    rpc.ReplyInt(c, rpc.CallParamInt(c, "a", 0) + rpc.CallParamInt(c, "b", 1))
end

func clock(c: i64):
    rpc.ReplyStr(c, "12:00")
end

func note(c: i64):
    rpc.ReplyNull(c)
end

func run_server():
    _ := rpc.Listen(server)
    sync.Done(ready)
    _ := rpc.ServeCount(server, 1)
end

func endpoint(ctx: i64):
    http.Ctx_json(ctx, rpc.HandleText(http.Ctx_body(ctx)))
end

func main:
    rpc.Register("add", add)
    rpc.Register("time", clock)
    rpc.Register("note", note)
    rpc.SetFraming(server, rpc.RPC_FRAME_LINE)
    sync.Add(ready, 1)
    t := spawn(run_server)
    sync.Wait(ready)
    println rpc.Port(server) > 0

    mut client := rpc.DialClient("127.0.0.1", rpc.Port(server), rpc.RPC_CODEC_JSON)
    rpc.ClientSetFraming(client, rpc.RPC_FRAME_LINE)
    println rpc.ClientOk(client)
    mut params := value.NewValueArray()
    value.ArrayPush(params, value.ValueInt(2))
    value.ArrayPush(params, value.ValueInt(40))
    println rpc.ResultInt(rpc.Call(client, "add", params))
    println rpc.ResultStr(rpc.CallNoParams(client, "time"))
    mut args := Vector {} as Vector<value.Value>
    args.push(value.ValueInt(1))
    args.push(value.ValueInt(2))
    println rpc.ResultInt(rpc.CallArgs(client, "add", args))
    println rpc.Notify(client, "note", value.NewValueArray())
    mut batch := Vector {} as Vector<rpc.Request>
    batch.push(rpc.NewRequest("add", params, 10))
    batch.push(rpc.NewRequest("missing", value.NewValueArray(), 11))
    answers := rpc.CallBatch(client, batch)
    println answers.items.len(), rpc.ResultInt(answers.items[0])
    println rpc.IsErrorResponse(answers.items[1]), rpc.ErrorCode(answers.items[1]), rpc.ErrorMessage(answers.items[1])
    rpc.CloseClient(client)
    println rpc.ClientOk(client), rpc.ClientError(client).len() >= 0
    join(t)
    rpc.StopServer(server)

    mut r := router.NewRouter()
    router.Post(r, "/rpc", endpoint)
    mut s := http.NewServer(19631)
    http.Use(s, r)
    _ := http.RunBackground(s)
    answer := rpc.CallHTTP("http://127.0.0.1:19631/rpc", "add", params, 7)
    println rpc.ResultInt(answer)
end

rpc-Methods.salam
import rpc
import encoding.value

func add(c: i64):
    a := rpc.CallParamInt(c, "a", 0)
    b := rpc.CallParamInt(c, "b", 1)
    rpc.ReplyInt(c, a + b)
end

func divide(c: i64):
    divisor := rpc.CallParamFloat(c, "b", 1)
    if divisor == 0.0:
        rpc.FailInvalidParams(c, "division by zero")
        ret
    end
    rpc.ReplyFloat(c, rpc.CallParamFloat(c, "a", 0) / divisor)
end

func hello(c: i64):
    name := rpc.CallParamStr(c, "name", 0)
    if name.len() == 0:
        rpc.Fail(c, -32001, "a name is needed")
        ret
    end
    rpc.ReplyStr(c, "hello " + name)
end

func status(c: i64):
    mut info := value.NewValueArray()
    value.ArrayPush(info, value.ValueString(rpc.CallMethodName(c)))
    value.ArrayPush(info, value.ValueInt(rpc.CallParamCount(c)))
    value.ArrayPush(info, value.ValueBool(rpc.CallParamBool(c, "verbose", 0)))
    rpc.Reply(c, info)
end

func log_event(c: i64):
    println "event:", value.AsString(rpc.CallParam(c, "event", 0)), rpc.CallIsNotification(c)
    _ := rpc.CallParams(c)
    _ := rpc.CallRequest(c)
    rpc.ReplyNull(c)
end

func forbidden(c: i64):
    rpc.FailWithData(c, -32003, "you are not allowed", value.ValueString("admin"))
end

func always(c: i64):
    rpc.ReplyBool(c, true)
end

func main:
    rpc.RegisterDoc("add", add, "adds two numbers")
    rpc.Register("divide", divide)
    rpc.Register("hello", hello)
    rpc.Register("status", status)
    rpc.Register("log", log_event)
    rpc.Register("admin", forbidden)
    rpc.Register("ping", always)
    println rpc.MethodCount(), rpc.HasMethod("add"), rpc.MethodDoc("add")
    println rpc.ListMethods().len()

    println rpc.HandleText(`{"jsonrpc":"2.0","method":"add","params":[2,40],"id":1}`)
    println rpc.HandleText(`{"jsonrpc":"2.0","method":"add","params":{"a":5,"b":6},"id":2}`)
    println rpc.HandleText(`{"jsonrpc":"2.0","method":"divide","params":[1,0],"id":3}`)
    println rpc.HandleText(`{"jsonrpc":"2.0","method":"hello","params":["Maryam"],"id":"x"}`)
    println rpc.HandleText(`{"jsonrpc":"2.0","method":"hello","params":[],"id":4}`)
    println rpc.HandleText(`{"jsonrpc":"2.0","method":"status","params":{"verbose":true},"id":5}`)
    println rpc.HandleText(`{"jsonrpc":"2.0","method":"admin","id":6}`)
    println rpc.HandleText(`{"jsonrpc":"2.0","method":"nope","id":7}`)
    println rpc.HandleText(`{"jsonrpc":"2.0","method":"log","params":{"event":"start"}}`).len()
    println rpc.HandleText(`[{"jsonrpc":"2.0","method":"ping","id":8},{"jsonrpc":"2.0","method":"add","params":[1,1],"id":9}]`)
    println rpc.HandleText(`{"jsonrpc":`)

    mut ok := false
    req := rpc.JsonDecodeRequest(`{"jsonrpc":"2.0","method":"add","params":[20,22],"id":10}`, ok)
    answer := rpc.Invoke(req)
    println rpc.ResultInt(answer)
    batch := rpc.JsonDecodeBatch(`[{"jsonrpc":"2.0","method":"ping","id":1}]`)
    println rpc.InvokeBatch(batch).items.len()

    println rpc.Unregister("ping"), rpc.HasMethod("ping")
    rpc.ResetRegistry()
    println rpc.MethodCount()
end
