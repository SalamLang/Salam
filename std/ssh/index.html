ssh

SSH connections, running a command on a server, and working with files over SFTP.

import ssh
On this page
  1. Types
  2. Constants and variables
  3. Functions

This package is an SSH client written in Salam and needs neither OpenSSH nor any other library. Make a Config with the host and the user, pick the login method, and connect with Connect; then send a command with Exec, or move files with SftpOpen. It suits automating deployments and backups.

The host key is checked against the known_hosts file, the way ssh itself does. The buffer and reader functions build and read the data format of the SSH protocol; an ordinary program does not need them.

ssh-Config.salam
import ssh

func main:
    mut cfg := ssh.NewConfig("example.com", ssh.DefaultPort, "deploy")
    println cfg.host, cfg.port, cfg.user, cfg.hostkey_mode == ssh.HOSTKEY_MODE_ACCEPT_NEW
    ssh.WithPassword(cfg, "s3cret")
    println cfg.use_password, cfg.use_key
    ssh.WithKeyPEM(cfg, "-----BEGIN OPENSSH KEY-----")
    println cfg.use_key, cfg.key_pem.len() > 0
    println ssh.WithKeyFile(cfg, "/nonexistent/id_rsa")
    ssh.WithHostKeyMode(cfg, ssh.HOSTKEY_MODE_STRICT)
    ssh.WithKnownHostsPath(cfg, "/home/deploy/.ssh/known_hosts")
    println cfg.hostkey_mode, cfg.known_hosts_path
    println ssh.HOSTKEY_MODE_INSECURE, ssh.ClientVersion
end

ssh-Exec.salam
import ssh

func main:
    mut cfg := ssh.NewConfig("server.example.com", 22, "deploy")
    _ := ssh.WithKeyFile(cfg, "/home/deploy/.ssh/id_rsa")
    mut client := ssh.Connect(cfg)
    if not ssh.Ok(client):
        println "error:", ssh.Error(client)
        ret
    end
    println "host key:", ssh.HostKeyFingerprint(client)

    r := ssh.Exec(client, "uname -a")
    println r.ok, r.exit_code, r.stdout, r.stderr
    r2 := ssh.RunCommand(client, "df -h /")
    println r2.stdout

    mut tail := ssh.NewSession(client)
    _ := ssh.RequestPty(tail, client, "xterm", 120, 40)
    if ssh.Start(tail, client, "tail -f /var/log/app.log"):
        repeat 10:
            _ := ssh.ReadStep(tail, client)
            print ssh.DrainOutput(tail)
            print ssh.DrainStderrOutput(tail)
        end
    end
    ssh.CloseSession(tail, client)

    mut shell := ssh.NewSession(client)
    if ssh.Shell(shell, client):
        _ := ssh.WriteInput(shell, client, "echo hello\nexit\n")
        ssh.CloseInput(shell, client)
        code := ssh.Wait(shell, client)
        println code, ssh.ExitCode(shell), ssh.SessionClosed(shell)
        println ssh.Output(shell), ssh.StderrOutput(shell)
    end
    ssh.CloseSession(shell, client)

    probe := ssh.Dial(cfg)
    println ssh.Ok(probe)
    println ssh.Reconnect(client)
    println ssh.SelfTest()
    ssh.Close(client)
end

ssh-Sftp.salam
import ssh

func main:
    mut cfg := ssh.NewConfig("files.example.com", 22, "ali")
    ssh.WithPassword(cfg, "s3cret")
    mut client := ssh.Connect(cfg)
    mut sftp := ssh.SftpOpen(client)
    if not sftp.ok:
        println ssh.Error(client)
        ret
    end
    println ssh.MkdirAll(client, sftp, "/backup/2026/09")
    println ssh.Upload(client, sftp, "report.pdf", "/backup/2026/09/report.pdf")
    println ssh.UploadDir(client, sftp, "site", "/var/www/site")
    if ssh.Exists(client, sftp, "/backup/2026/09/report.pdf"):
        info := ssh.Stat(client, sftp, "/backup/2026/09/report.pdf")
        println info.name, info.size, info.is_dir, info.permissions
    end
    each info in ssh.ListDir(client, sftp, "/backup"):
        println info.name, info.is_dir
    end
    println ssh.IsLink(client, sftp, "/current"), ssh.Lstat(client, sftp, "/current").name
    _ := ssh.Download(client, sftp, "/etc/hostname", "hostname.txt")
    _ := ssh.DownloadDir(client, sftp, "/var/log/app", "logs")
    _ := ssh.Rename(client, sftp, "/tmp/a", "/tmp/b")
    _ := ssh.Remove(client, sftp, "/tmp/b")
    _ := ssh.Mkdir(client, sftp, "/tmp/x")
    _ := ssh.Rmdir(client, sftp, "/tmp/x")
    _ := ssh.RemoveAll(client, sftp, "/tmp/old")
    ssh.SftpClose(client, sftp)
    ssh.Close(client)
end

NewBuf

func NewBuf(cap0: int): Buf
An empty buffer for building SSH protocol messages.
ssh-Wire.salam
import ssh

func main:
    mut b := ssh.NewBuf(64)
    ssh.BufPutByte(b, 20)
    ssh.BufPutUint32(b, 7 as u32)
    ssh.BufPutUint64(b, 9 as u64)
    ssh.BufPutBool(b, true)
    ssh.BufPutStringStr(b, "ssh-ed25519,rsa-sha2-256")
    ssh.BufPutString(b, "abc" as void*, 3)
    ssh.BufPutBytes(b, "xy" as void*, 2)
    ssh.BufPutZeros(b, 2)
    ssh.BufPutMpintBytes(b, "\x80" as void*, 1)
    println b.len

    mut r := ssh.ReaderFromBuf(b)
    println ssh.RGetByte(r), ssh.RGetUint32(r), ssh.RGetUint64(r), ssh.RGetBool(r)
    names := ssh.RGetNameList(r)
    println names, ssh.NameListContains(names, "rsa-sha2-256")
    println ssh.RGetStringAsStr(r)
    ssh.RSkip(r, 4)
    println ssh.RRemaining(r)
    n := ssh.RGetMpintView(r)
    println n.len, ssh.RRemaining(r)

    println ssh.NegotiateFirstMatch("aes256-ctr,aes128-ctr", "aes128-ctr,aes256-ctr")
    println ssh.ConstantTimeEqual("abcd" as void*, "abcd" as void*, 4), ssh.ConstantTimeEqual("abcd" as void*, "abce" as void*, 4)

    mut b2 := ssh.NewBuf(16)
    ssh.BufPutStringBuf(b2, b)
    ssh.BufPutMpintBuf(b2, b)
    ssh.BufPutBuf(b2, b)
    mut r2 := ssh.NewReader(b2.p, b2.len)
    view := ssh.RGetStringView(r2)
    println view.len == b.len
    mut copied := ssh.RGetStringCopy(ssh.ReaderFromBuf(b2))
    mut number := ssh.RGetMpintCopy(r2)
    println copied.len, number.len, ssh.RRemaining(r2)
    ssh.WipeBuf(b)
    ssh.FreeBuf(b)
    ssh.FreeBuf(b2)
    ssh.FreeBuf(copied)
    ssh.FreeBuf(number)
end
Output59 20 7 9 true ssh-ed25519,rsa-sha2-256 true abc 6 1 0 aes256-ctr true false true 59 59 59
