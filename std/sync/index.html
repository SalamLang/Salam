Standard library · Time and concurrency

sync

Mutexes, condition variables, wait groups, read-write locks, semaphores and once, for coordinating threads.

import sync
On this page
  1. Types
  2. Functions

When several threads change the same variable they have to take turns, or the result depends on the order the threads happen to run in. A mutex is the simplest answer: each thread takes the lock before touching the data and releases it afterwards.

A wait group is for waiting until several pieces of work are done: call Add before starting them, Done at the end of each, and Wait on the main thread. Every tool in this package is created with a New... function and released with a Destroy... one.

Types

Mutex

struct Mutex
A mutex; only one thread holds it at a time.
NameTypeDescription
handlevoid*The operating system's lock.
sync-Mutex.salam
import sync

mut balance := 0
mut lock := sync.Mutex {}
mut group := sync.WaitGroup {}

func deposit():
    repeat 500:
        sync.Lock(lock)
        balance += 1
        sync.Unlock(lock)
    end
    sync.Done(group)
end

func main:
    lock = sync.NewMutex()
    group = sync.NewWaitGroup()
    sync.Wait(group)
    sync.Add(group, 4)
    a := spawn(deposit)
    b := spawn(deposit)
    c := spawn(deposit)
    d := spawn(deposit)
    sync.Wait(group)
    join(a)
    join(b)
    join(c)
    join(d)
    println "balance:", balance
    sync.DestroyWaitGroup(group)
    sync.Destroy(lock)
end
Outputbalance: 2000

CondVar

struct CondVar
A condition variable: threads sleep on it until another thread wakes them.
NameTypeDescription
handlevoid*The operating system's condition variable.
sync-CondVar.salam
import sync

mut ready := false
mut lock := sync.Mutex {}
mut cond := sync.CondVar {}

func producer():
    sync.SleepMs(20)
    sync.Lock(lock)
    ready = true
    sync.Broadcast(cond)
    sync.Unlock(lock)
end

func main:
    lock = sync.NewMutex()
    cond = sync.NewCondVar()
    start := sync.NowMs()
    sync.Lock(lock)
    println sync.WaitCondTimeout(cond, lock, 10)
    t := spawn(producer)
    until not ready:
        sync.WaitCond(cond, lock)
    end
    sync.Unlock(lock)
    join(t)
    println ready, sync.NowMs() - start >= 20
    sync.Signal(cond)
    sync.DestroyCondVar(cond)
    sync.Destroy(lock)
end
Outputtrue true true

RWMutex

struct RWMutex
A read-write lock: several readers together, or one writer alone.
NameTypeDescription
statei64*The internal state: the readers, the writer and the waiting writers.
muvoid*The internal lock.
cvvoid*The internal condition variable.
sync-RWMutex.salam
import sync

func main:
    rw := sync.NewRWMutex()
    sync.RLock(rw)
    sync.RLock(rw)
    println "two readers together"
    sync.RUnlock(rw)
    sync.RUnlock(rw)
    sync.WLock(rw)
    println "one writer alone"
    sync.WUnlock(rw)
    sync.DestroyRWMutex(rw)

    sem := sync.NewSemaphore(2)
    sync.Acquire(sem)
    println sync.TryAcquire(sem), sync.TryAcquire(sem)
    sync.Release(sem)
    println sync.TryAcquire(sem)
    sync.DestroySemaphore(sem)

    once := sync.NewOnce()
    repeat 3:
        sync.Do(once, (): println "only once" end)
    end
    sync.DestroyOnce(once)
end
Outputtwo readers together one writer alone true false true only once
