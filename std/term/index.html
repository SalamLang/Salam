term

Text colour and style, cursor movement, progress bars, reading keys and the mouse, and the terminal size.

import term
On this page
  1. Types
  2. Constants and variables
  3. Functions

The term package has everything a good looking command line program needs: text colour and style, progress bars, and for full screen programs, cursor movement and reading keys one at a time.

The colour functions, such as Red, return a new string and print nothing. When the output goes to a file or another program, or NO_COLOR is set, nothing is added and the text stays plain. SetColorMode changes that behaviour.

The functions whose names start with Seq only return the escape sequence, and the one with the same name without the prefix sends it right away. To read keys, put the terminal into raw mode with MakeRaw or MakeCbreak, and always put it back with Restore.

func main:

    frames.push("◓")
    mut s2 := term.NewSpinnerFrames("loading", frames)
    defer term.SpinnerFree(s2)
    repeat 3:
        println term.SpinnerFrame(s2)
        s2.idx = (s2.idx + 1) % frames.len()
    end
end

term-Decode.salam
import term

func describe(k: term.Key): str:
    if k.code == term.KeyChar:
        ret "char " + k.ch + (k.ctrl ? " with ctrl" : "") + (k.alt ? " with alt" : "")
    else k.code == term.KeyUp:
        ret "up"
    else k.code == term.KeyEnter:
        ret "enter"
    else k.code == term.KeyEscape:
        ret "escape"
    else k.code >= term.KeyF1 and k.code < term.KeyF1 + 12:
        ret "f" + (k.code - term.KeyF1 + 1)
    end
    ret "code " + k.code
end

func main:
    each s in ["a", "\u001b[A", "\r", "\u001b", "\u0001", "\u001bx", "\u001bOQ", "\u001b[3~"]:
        println describe(term.Decode(s))
    end
    m := term.DecodeMouse("\u001b[<0;12;5M")
    println m.ok, m.button, m.col, m.row, m.pressed, m.motion
    println term.KeyDelete, term.KeyTab, term.KeyBackspace, term.KeySpace, term.KeyEOF, term.KeyUnknown
    println term.KeyDown, term.KeyRight, term.KeyLeft, term.KeyHome, term.KeyEnd, term.KeyPageUp, term.KeyPageDown, term.KeyInsert, term.KeyMouse, term.EscDelayMs
end

term-ReadKey.salam
import term

func main:
    if not term.IsTTY(term.StdinFd):
        printerrln "this program has to run in a terminal"
        ret 1
    end
    mut state := term.MakeRaw(term.StdinFd)
    defer term.Restore(state)
    term.Write("press a key (q to quit)\r\n")
    until true:
        if not term.KeyReady(500):
            continue
        end
        k := term.ReadKey()
        if k.code == term.KeyChar and k.ch == "q":
            break
        end
        term.Write("code: " + k.code + "\r\n")
    end
    k2 := term.GetKey()
    b := term.ReadByte()
    password := term.ReadPassword("password: ")
    println k2.code, b, password.len()
    s2 := term.MakeCbreak(term.StdinFd)
    mut s3 := s2
    term.Restore(s3)
    ret 0
end

import term

func main:
    term.EnableAnsi()
    term.SetTitle("dashboard")
    term.EnterAltScreen()
    term.HideCursor()
    term.ClearScreen()
    term.Home()
    term.MoveTo(2, 4)
    term.Write(term.Bold("hello from the dashboard"))
    term.SaveCursor()
    term.Down(2)
    term.Right(4)
    term.Emit("status: good")
    term.RestoreCursor()
    term.Up(1)
    term.Left(1)
    term.NextLine(1)
    term.PrevLine(1)
    term.Column(1)
    term.ClearLine()
    term.ClearToLineEnd()
    term.ClearBelow()
    term.ResetLine()
    term.SetScrollRegion(3, term.Rows() - 1)
    term.ScrollUp(1)
    term.ScrollDown(1)
    term.InsertLines(1)
    term.DeleteLines(1)
    term.ResetScrollRegion()
    term.EnableMouse()
    term.DisableMouse()
    term.Bell()
    term.Flush()
    term.ClearScrollback()
    term.ShowCursor()
    term.ExitAltScreen()
    term.WriteErr("done\n")
    w := term.Size()
    println w.cols, w.rows, term.Cols()
end

term-Width.salam
import term

func main:
    term.SetColorMode(term.ColorAlways)
    colored := term.Green("hello")
    println colored.len(), term.Width(colored), term.Strip(colored)
    println term.Width("hello"), term.Width("日本"), term.Width("ab")
    term.SetColorMode(term.ColorNever)
    println "[" + term.Truncate("the Salam programming language", 10) + "]"
    println "[" + term.PadRight("name", 8) + "]", "[" + term.PadLeft("42", 6) + "]", "[" + term.Center("ok", 6) + "]"
    println term.FormatDuration(9500), term.FormatDuration(3721000)
    println term.FormatBytes(512), term.FormatBytes(1536), term.FormatBytes(3221225472)
end
Output14 5 hello 5 2 2 [the Salam ] [name ] [ 42] [ ok ] 9.5s 1h 02m 512 B 1.5 KB 3.0 GB

term-Progress.salam
import term
import time

func main:
    mut b := term.NewBar(50, "processing")
    repeat 50:
        term.BarAdd(b, 1)
        term.BarDraw(b)
        time.Sleep(20)
    end
    term.BarFinish(b)
    mut s := term.NewSpinner("connecting to the server")
    repeat 20:
        term.SpinnerTick(s)
        time.Sleep(50)
    end
    term.SpinnerStop(s, "connected")
    term.SpinnerFree(s)
end
