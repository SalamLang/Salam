time

The current time, durations, formatting and parsing dates, the monotonic clock, timers and tickers.

import time
On this page
  1. Types
  2. Constants and variables
  3. Functions

There are two kinds of "time" in this package. An instant is a number of seconds since the start of 1970 in UTC, the Unix epoch, and is what you use for dates and clock times. A duration is a span with nanosecond precision, such as "1 hour 30 minutes", which you can add up and compare.

Functions such as Year and FormatDate work in the machine's local time, and their UTC counterparts work in UTC. To measure how long a piece of code took, use the monotonic clock (MonoNanos) rather than Now; the system clock can jump forwards or backwards.

For the Persian and Hijri calendars and for time zones, see the calendar package.

import time
import str

func main:
    t := time.FromLocalParts(2024, 3, 20, 9, 30, 0)
    println time.Year(t), time.Month(t), time.Day(t), time.Hour(t), time.Minute(t), time.Second(t)
    println time.Weekday(t), time.YearDay(t)
    println time.FormatDate(t), time.FormatTime(t)
    println time.Format(t)
    println time.FormatISO(t)
    r := time.FormatRFC3339(t)
    offset := time.LocalOffsetMinutesAt(t)
    println str.StartsWith(r, "2024-03-20T09:30:00"), (offset == 0) == str.EndsWith(r, "Z")
    dt := time.FromEpoch(t)
    println time.Epoch(dt) == t, dt.epoch == t
    println time.Epoch(time.NowDateTime()) >= t
end

import time

func main:
    d := time.Hours(1).add(time.Minutes(30))
    println d.to_str(), d.minutes(), d.secs()
    println time.FormatDuration(time.Milliseconds(1500)), time.FormatDuration(time.Microseconds(250)), time.FormatDuration(time.Nanoseconds(42))
    println time.Seconds(90).to_str(), time.Days(2).hours(), time.SecondsF(0.25).millis()
    p := time.ParseDuration("1h15m30.5s")
    println p.ok, p.value.to_str(), p.value.seconds()
    println time.ParseDuration("ten minutes").ok, time.ParseDurationOr("?", time.Seconds(5)).to_str()
    k := time.Minutes(7)
    println k.sub(time.Minutes(10)).to_str(), k.neg().abs().to_str(), k.neg().is_negative()
    println k.mul(3).to_str(), k.div(2).to_str(), time.Hours(1).ratio(k)
    println k.cmp(time.Seconds(420)), k.less(time.Hours(1)), k.equals(time.Seconds(420))
    t := time.Milliseconds(1750)
    println t.truncate(time.Seconds(1)).to_str(), t.round(time.Seconds(1)).to_str()
    println t.nanos(), t.micros(), t.millis(), time.ZeroDuration().is_zero(), t.days() < 0.001
end

time-UTC.salam
import time

func main:
    t := time.FromUTCParts(2024, 3, 20, 14, 5, 9)
    println t, time.FormatISOUTC(t)
    println time.YearUTC(t), time.MonthUTC(t), time.DayUTC(t), time.HourUTC(t), time.MinuteUTC(t), time.SecondUTC(t)
    println time.FormatHTTPDate(t)
    println time.ParseHTTPDate("Wed, 20 Mar 2024 14:05:09 GMT").epoch == t
    p := time.ParseISO("2024-03-20T17:35:09+03:30")
    println p.ok, p.epoch == t, p.nanos
    println time.ParseISOIn("2024-03-20 17:35:09.25", 210).epoch == t, time.ParseISOIn("2024-03-20 17:35:09.25", 210).nanos
    f := time.ParseFormat("%d/%m/%Y %H:%M %z", "20/03/2024 14:05 +0000")
    println f.ok, time.FormatISOUTC(f.epoch)
    println time.ParseISO("tomorrow").ok, time.ParseISOOr("tomorrow", 0)
    later := time.AddDuration(t, time.Days(10))
    println time.FormatISOUTC(later), time.DiffSeconds(later, t), time.DurationBetween(later, t).to_str()
    println time.SecondsPerMinute, time.SecondsPerHour, time.SecondsPerDay
end

Timer

time-Timer.salam
import time

mut count := 0

func main:
    mut timer := time.NewTimer(time.Milliseconds(100))
    println timer.expired(), timer.remaining().millis() > 0
    timer.wait()
    println timer.expired(), timer.remaining().is_zero()
    timer.reset(time.Seconds(10))
    timer.stop()
    println timer.expired(), timer.active
    mut ticker := time.NewTicker(time.Milliseconds(50))
    println ticker.fired()
    ticker.wait()
    time.Sleep(120)
    println ticker.fired(), ticker.dropped >= 1
    ticker.advance()
    println ticker.remaining().millis() <= 100
    ticker.reset(time.Milliseconds(5))
    ticker.stop()
    println ticker.fired()
    time.AfterFunc(time.Milliseconds(10), (): println "after 10 milliseconds" end)
    time.RepeatEvery(time.Milliseconds(5), 2, (): println "tick" end)
    println time.Every(time.Milliseconds(5), (): count += 1 ret count < 3 end)
    count = 0
    println time.WaitUntil((): count += 1 ret count >= 3 end, time.Milliseconds(5), time.Seconds(5))
    println time.WaitUntil((): ret false end, time.Milliseconds(5), time.Milliseconds(20))
end
Outputfalse true true true false false false true true true false after 10 milliseconds tick tick 3 true false

Now

func Now(): i64
The current instant in seconds since the Unix epoch.
time-Now.salam
import time

func main:
    start := time.Now()
    a := time.MonoNanos()
    time.Sleep(20)
    time.SleepFor(time.Milliseconds(10))
    b := time.MonoNanos()
    println time.ElapsedMs(a, b) >= 30.0, time.ElapsedUs(a, b) >= 30000.0, time.ElapsedSec(a, b) < 5.0
    println time.SinceNanos(a) >= 30000000, time.SinceMs(a) >= 30.0, time.SinceUs(a) >= 30000.0, time.SinceSec(a) < 5.0
    println time.SinceDuration(a).millis() >= 30
    println time.MonoMillis() > 0, time.MonoMicros() > 0
    println time.Now() >= start, time.Since(start) >= 0, time.Until(start + 60) <= 60
    println time.NowMillis() / 1000 >= start, time.NowMicros() / 1000000 >= start, time.NowNanos() / 1000000000 >= start
end

Reads text with a pattern. %Y is a four digit year, %y a two digit one, %m the month, %d the day, %H the hour, %M the minute, %S the second, %f the fraction, %b the English month name, %a the day name, which is ignored, %z an offset such as +0330 and %Z a zone name such as UTC. One space in the pattern matches any number of spaces in the text.

The example for this item is shown with the ones above: time-UTC.salam
