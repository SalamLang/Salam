Standard library · Security and cryptography

tls

Secure TLS 1.3 and 1.2 connections and public key cryptography tools, with no OpenSSL needed.

import tls
On this page
  1. Types
  2. Constants and variables
  3. Functions

The tls package is a complete implementation of TLS written in Salam and needs no outside library. A client connects with Dial and checks the server certificate against the system roots; a server reads its certificate and key with NewServerConfig and secures each connection with WrapServer.

The cryptography tools here are also useful for building protocols of your own: the two sides make a shared secret with ECDH, derive a key from it with HKDF, and encrypt and seal the message with AES-GCM. Keys and data are passed around as base64.

Types

Config

struct Config
The client configuration.
NameTypeDescription
server_namestr = ""The hostname for SNI and for certificate verification.
ca_filestr = ""The file of root certificates; empty means the system roots.
insecure_skip_verifybool = falseDo not verify the certificate; for testing only, since it leaves the connection open to eavesdropping.
alpnstr = ""The protocol for ALPN, such as h2.
aes_onlybool = falseOffer AES ciphers only.
tls-Client.salam
import tls

func main:
    mut cfg := tls.NewConfig("example.com")
    cfg.alpn = "http/1.1"
    mut conn := tls.Dial("example.com", 443, cfg)
    if not tls.ConnOk(conn):
        println "error:", tls.ConnError(conn)
        ret
    end
    println tls.Version(conn), tls.CipherName(conn), tls.PeerCertCount(conn)
    _ := tls.Write(conn, "GET / HTTP/1.1\r\nHost: example.com\r\nConnection: close\r\n\r\n")
    mut body := ""
    until tls.ConnOk(conn):
        chunk := tls.Read(conn, 4096)
        if chunk.len() == 0:
            break
        end
        body += chunk
    end
    println body.len() > 0
    tls.Close(conn)
end

tls-Server.salam
import tls
import net.tcp
import mem

func main:
    cfg := tls.NewServerConfig("cert.pem", "key.pem")
    if not cfg.ok:
        println "the certificate or key could not be read"
        ret
    end
    server := tcp.Bind(8443)
    until true:
        sock := tcp.Accept(server)
        mut conn := tls.WrapServer(sock.fd, cfg)
        if tls.ConnOk(conn):
            if tls.Readable(conn) or tls.Pending(conn) > 0:
                println tls.Read(conn, 1024)
            end
            buffer := mem.Allocate(256 as u64)
            n := tls.ReadBytes(conn, buffer, 256)
            _ := tls.WriteBytes(conn, buffer, n)
            _ := tls.Write(conn, "HTTP/1.1 200 OK\r\nContent-Length: 5\r\n\r\nhello")
            mem.Free(buffer)
            println tls.Fd(conn) >= 0
        end
        tls.Close(conn)
    end
end

The version number of TLS 1.0; not supported.
tls-Certs.salam
import tls
import net.tcp
import encoding
import io

func main:
    roots := tls.LoadRoots("/etc/ssl/cert.pem")
    println tls.RootCount(roots)

    chain := tls.LoadCertChainFile("cert.pem")
    println tls.RootCount(chain)
    mut key := tls.LoadPrivateKeyFile("key.pem")
    mut key2 := tls.ParsePrivateKeyPEM("-----BEGIN ...")
    tls.FreePrivateKey(key2)

    cert := tls.HexDecodeBuf("3082...")
    println tls.CertIsValid(cert), tls.CertSubjectCN(cert)
    code := tls.VerifyCert(cert, roots, "example.com")
    println code, tls.VerifyError(code)

    digest := tls.HashDigest(false, "message")
    sig := tls.NewBuf(512)
    println tls.SignRSAPKCS1(key, false, digest, sig)
    println tls.SignRSAPSS(key, false, digest, sig)
    println tls.VerifyRSAPSSWithKey(key, false, digest, sig)
    r := tls.NewBuf(32)
    s := tls.NewBuf(32)
    if tls.SignECDSA(key, digest, r, s):
        der := tls.ECDSADerEncode(r, s, 32)
        println tls.VerifyECDSADERWithKey(key, digest, der)
    end

    pem := io.ReadFile("rsa-key.pem")
    mut good := false
    println tls.RsaKeyBits(pem)
    println tls.RsaSignPkcs1Sha256(pem, encoding.Base64Encode("..."), good)

    mut cfg := tls.NewConfig("internal.example")
    cfg.ca_file = "company-ca.pem"
    cfg.aes_only = true
    mut conn := tls.Dial("internal.example", 443, cfg)
    println tls.ConnOk(conn)
    tls.Close(conn)
    sock := tcp.Dial("example.com", 443)
    mut conn2 := tls.WrapClient(sock.fd, tls.NewConfig("example.com"))
    println tls.Version(conn2) == "TLS1.3"
    tls.Close(conn2)
    println tls.TLS10, tls.TLS11, tls.TLS12, tls.TLS13
    println tls.SelfTest()
    tls.FreeRoots(roots)
    tls.FreePrivateKey(key)
end

tls-Crypto.salam
import tls
import encoding

func main:
    mut priv1 := ""
    mut pub1 := ""
    mut priv2 := ""
    mut pub2 := ""
    println tls.EcdhP256Generate(priv1, pub1), tls.EcdhP256Generate(priv2, pub2)
    mut good := false
    secret1 := tls.EcdhP256Shared(priv1, pub2, good)
    secret2 := tls.EcdhP256Shared(priv2, pub1, good)
    println good, secret1 == secret2

    point1 := tls.EcdhP256PublicPoint(priv1)
    point2 := tls.EcdhP256PublicPoint(priv2)
    println tls.EcdhP256SharedPoint(priv1, point2, good) == tls.EcdhP256SharedPoint(priv2, point1, good)

    salt := tls.RandomSalt(16)
    key := tls.HkdfSha256(secret1, salt, "chat v1", 32, good)
    println good, key == tls.HkdfSha256(secret2, salt, "chat v1", 32, good)
    println tls.HkdfSha256Raw(secret1, salt, encoding.Base64Encode("chat v1"), 32, good) == key

    nonce := tls.GcmNonce(0)
    sealed := tls.AesGcmSeal(key, nonce, "hello, this message is encrypted", good)
    println good
    println tls.AesGcmOpen(key, nonce, sealed, good), good
    _ := tls.AesGcmOpen(key, tls.GcmNonce(1), sealed, good)
    println good
    binary := tls.AesGcmSealBytes(key, nonce, encoding.Base64Encode("abc"), good)
    println encoding.Base64Decode(tls.AesGcmOpenBytes(key, nonce, binary, good))

    digest := encoding.Base64Encode("0123456789abcdef0123456789abcdef")
    sig := tls.P256Sign(priv1, digest, good)
    println good, tls.P256Verify(point1, digest, sig), tls.P256Verify(point2, digest, sig)

    b := tls.HashDigest(false, "abc")
    println tls.BufLen(b), tls.HexEncodeBuf(b)
    mut hex := tls.HexDecodeBuf("cafe")
    tls.BufAppendStr(hex, "!")
    println tls.BufLen(hex), tls.HexEncodeBuf(hex)
    mut zeros := tls.NewZeroBuf(4)
    println tls.HexEncodeBuf(zeros)
    mut empty := tls.NewBuf(64)
    println tls.BufLen(empty)
    tls.WipeBuf(hex)
    tls.FreeBuf(hex)
    tls.FreeBuf(zeros)
    tls.FreeBuf(empty)
end

import tls
import net.tcp

func main:
