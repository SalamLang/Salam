ai.api_title = OpenAI-compatible API
ai.api_text = Point any OpenAI client at the Salam AI API and use the model <code>salam</code>. Create a key on the keys page of Salam AI.
ai.api_cmd <<
from openai import OpenAI

client = OpenAI(base_url="https://ai-api.salamlang.ir", api_key="sk-salam-...")
reply = client.chat.completions.create(
    model="salam",
    messages=[{"role": "user", "content": "Write hello world in Salam"}],
)
>>
