install.title = Install the Salam programming language on Windows, Linux and macOS | Free download
install.description = Download and install Salam with a single command on Windows, Linux and macOS, or run it in Docker. No administrator access needed.

install.heading = Install Salam
install.lead = The Salam compiler is a single executable. The installer puts it in your home directory and adds it to your path; it does not need administrator access.

install.unix_title = Linux and macOS
install.unix_text = Run this in a terminal. The installer detects your operating system and architecture and adds the Salam directory to your PATH; open a new terminal once it finishes.
install.unix_cmd = { curl -fsSL https://raw.githubusercontent.com/SalamLang/Salam/refs/heads/main/install.sh || wget -qO- https://raw.githubusercontent.com/SalamLang/Salam/refs/heads/main/install.sh; } | sh
install.unix_note = To pass options to the installer, put them after <code>sh -s --</code>, for example <code>… | sh -s -- --version 0.5.0</code>.

install.win_title = Windows
install.win_text = Open PowerShell and run this command.
install.win_cmd = irm https://raw.githubusercontent.com/SalamLang/Salam/refs/heads/main/install.ps1 | iex
install.win_cmd_alt_text = If PowerShell reports a secure connection error, run the installer from the command prompt instead:
install.win_cmd_alt = curl -fsSLo "%TEMP%\salam-install.bat" https://raw.githubusercontent.com/SalamLang/Salam/refs/heads/main/install.bat && "%TEMP%\salam-install.bat"

install.check_title = Check the install
install.check_text = Open a new terminal and ask Salam for its version. If it prints one, you are ready.
install.check_cmd = salam version

install.docker_title = Docker
install.docker_text = If you would rather not install anything, run Salam in Docker. Run this in a directory containing <code>app.salam</code>; the current directory is mounted inside the container. Images are built for x86-64, ARM64 (including Apple silicon), ARMv7 and i386; run <code>docker pull basemax/salam</code> first to get the latest one.
install.docker_cmd = docker run --rm -v "$PWD":/work basemax/salam run app.salam

install.source_title = Build from source
install.source_text = The Salam compiler is written in Salam. To build it you need an installed Salam and a C compiler.
install.source_cmd = git clone https://github.com/SalamLang/Salam && cd Salam && sh tools/bash/build-selfhost.sh
