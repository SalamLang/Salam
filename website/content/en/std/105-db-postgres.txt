pkg.en = db/postgres
pkg.name = db/postgres
pkg.group = Database
pkg.summary = Connecting to PostgreSQL.
pkg.title = The db/postgres package in Salam | PostgreSQL
pkg.description = The db/postgres package in Salam: connecting to PostgreSQL with a connection string, queries and reading rows, transactions and the shared database interface, with an example.
pkg.import = db.postgres
pkg.intro <<
<p>This package implements the Postgres protocol directly and needs no outside library; the example needs a running Postgres server, so it is checked and not run. Every row of a result is fetched from the server at once, so <code>RowCount</code> is known before you read any of them.</p>
>>

doc.Database = One PostgreSQL connection.
doc.Database.handle = The library pointer.
doc.Database.Kind = The name of the database engine.
doc.Database.Ok = It is healthy.
doc.Database.Ping = The connection is alive.
doc.Database.Exec = Runs one statement; zero means success.
doc.Database.Query = Runs a query and returns the rows.
doc.Database.Begin = Starts a transaction.
doc.Database.LastInsertId = The id of the last inserted row.
doc.Database.Changes = The number of rows the last statement changed.
doc.Database.Error = The last error message.
doc.Database.Quote = Returns the string quoted and escaped, safe for SQL.
doc.Database.Close = Closes and frees it.
doc.Rows = The result of one query.
doc.Rows.handle = The result pointer.
doc.Rows.row = The index of the current row.
doc.Rows.count = The number of rows.
doc.Database.affected = The number of rows changed by the last statement.
doc.Rows.Ok = It is healthy.
doc.Rows.Next = Moves to the next row; false when no row is left.
doc.Rows.ColumnCount = The number of columns.
doc.Rows.ColumnName = The column name.
doc.Rows.ColumnType = The type of the column value in the current row, one of the <code>TYPE_...</code> constants in <a href="../db/">db</a>.
doc.Rows.IsNull = The column value is null.
doc.Rows.Text = The column value as text.
doc.Rows.Int = The column value as an integer.
doc.Rows.Int64 = The column value as a 64 bit integer.
doc.Rows.Float = The column value as a float.
doc.Rows.Close = Closes and frees it.
doc.Rows.RowCount = The number of rows in the result.
doc.Txn = One transaction.
doc.Txn.handle = The connection pointer.
doc.Txn.Ok = It is healthy.
doc.Txn.Exec = Runs one statement; zero means success.
doc.Txn.Query = Runs a query and returns the rows.
doc.Txn.Commit = Commits the transaction.
doc.Txn.Rollback = Rolls the transaction back.
doc.Available = The Postgres client is available; since it is inside Salam itself, this is always true.
doc.ClientVersion = The compatible client version as a number, such as 180000.
doc.ServerVersion = The server version as a number.
doc.Open = Connects with a connection string, such as <code>host=... dbname=...</code>, or a <code>postgresql://</code> address.
doc.OpenFull = Connects with a host, port, user, password and database name.
doc.Ok = The connection is up.
doc.Ping = The connection is alive.
doc.Reset = Brings a dropped connection back up.
doc.Close = Closes it.
doc.Error = The last error message.
doc.Exec = Runs one statement; zero means success.
doc.ExecCount = Runs one statement and returns the number of rows changed.
doc.Query = Runs a query.
doc.Quote = Returns the string quoted and escaped, safely.
doc.DatabaseName = The database name.
doc.UserName = The user name.
doc.HostName = The host.
doc.Port = The port.
doc.QueryInt = The first column of the first row as an integer.
doc.QueryText = The same, as text.
doc.Conn = Returns the connection as a shared <code>Connection</code>.
