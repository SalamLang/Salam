pkg.en = db/redis
pkg.name = db/redis
pkg.group = Database
pkg.summary = A Redis client: keys and values, hashes, lists, sets and publishing.
pkg.title = The db/redis package in Salam | Redis
pkg.description = The db/redis package in Salam: key and value storage with expiry, counters, hashes, lists and queues, sets, publishing messages and any command, with an example.
pkg.import = db.redis
pkg.intro <<
<p>Redis is an in memory database for caches, sessions, counters and queues. This package implements the Redis protocol directly and with no outside library; the example needs a running Redis server, so it is only checked. For a command with no function of its own, call <code>Command</code> with the command name and its arguments.</p>
>>

doc.REPLY_STRING = Reply type: a string.
doc.REPLY_ARRAY = Reply type: an array.
doc.REPLY_INTEGER = Reply type: a number.
doc.REPLY_NIL = Reply type: nil, a key that is not there for instance.
doc.REPLY_STATUS = Reply type: a status, such as <code>OK</code>.
doc.REPLY_ERROR = Reply type: an error.
doc.REPLY_DOUBLE = Reply type: a float.
doc.Client = A connection to Redis.
doc.Client.handle = The connection pointer.
doc.Reply = The reply to one command.
doc.Reply.handle = The reply pointer.
doc.Command = Sends any command with its arguments; free the reply with <code>FreeReply</code> when you are done with it.
doc.ReplyOk = A reply arrived.
doc.ReplyType = The reply type.
doc.ReplyIsError = The reply is an error.
doc.ReplyIsNil = The reply is nil.
doc.ReplyInt = The reply as a number.
doc.ReplyStr = The reply as text.
doc.ReplyArrayLen = The number of elements in an array reply.
doc.ReplyArrayGet = One element of an array reply.
doc.FreeReply = Frees the reply.
doc.Set = Sets the value of a key.
doc.SetEx = The same, with an expiry in seconds.
doc.Get = The value of a key; an empty string when there is none.
doc.Exists = The key is there.
doc.Del = Deletes the key.
doc.Expire = Sets the expiry of the key.
doc.TTL = The seconds left until the expiry; -1 means no expiry and -2 means no such key.
doc.Incr = Raises the number of a key by one and returns the new value.
doc.IncrBy = Raises it by the given amount.
doc.Decr = Lowers it by one.
doc.Append = Adds text to the value and returns the new length.
doc.StrLen = The length of the value.
doc.Type = The type of the value of a key, such as <code>string</code> or <code>hash</code>.
doc.Keys = The keys that match a pattern; slow on a large database.
doc.SAdd = Adds a member to a set.
doc.SRem = Removes a member.
doc.SIsMember = It is a member.
doc.SCard = The number of members.
doc.SMembers = Every member.
doc.Connect = Connects to the Redis server.
doc.Ok = The connection is up.
doc.Error = The error message.
doc.Close = Closes the connection.
doc.Ping = The connection is alive.
doc.Select = Picks the numbered database.
doc.Auth = Authenticates with a password.
doc.FlushDB = Clears every key of the current database.
doc.HSet = Sets a field of a hash.
doc.HGet = One field of a hash.
doc.HDel = Deletes a field.
doc.HExists = The field is there.
doc.HLen = The number of fields.
doc.HGetAll = Every field as a map.
doc.LPush = Adds to the front of a list and returns the new length.
doc.RPush = Adds to the end of a list.
doc.LPop = Takes one off the front of a list.
doc.RPop = Takes one off the end.
doc.LLen = The length of the list.
doc.LRange = A part of the list; -1 means the last element.
doc.Publish = Publishes a message to a channel and returns the number of receivers.
