pkg.name = i18n
pkg.group = Text
pkg.summary = A multilingual program: translating messages, placing values into them and choosing singular or plural.
pkg.title = The i18n package in Salam | Multilingual programs
pkg.description = The i18n package in Salam: a translation catalog, a fallback language, message interpolation, plurals, and loading translations from a file, with examples.
pkg.intro <<
<p>If your program has to speak several languages, do not write the messages into the code. Put them in a translation catalog under a key, and ask for the user's language when you display one.</p>
<p>Every catalog has a fallback language. When a message has no translation in the user's language, the fallback is shown, and when that is missing too, the key itself; so the program never shows an empty message.</p>
>>

doc.Catalog = A catalog of translations. Build one with <code>New</code> and release it with <code>Free</code>.
doc.Catalog.keys = The stored keys, in the form "language:key".
doc.Catalog.vals = The translations, in the same order as the keys.
doc.Catalog.fallback = The fallback language.
doc.New = Creates an empty catalog with the given fallback language.
doc.Set = Records the translation of a key for a language, replacing any earlier one.
doc.Has = True when a translation is recorded for this language and key. It does not look at the fallback language.
doc.RawGet = Returns the recorded translation with no substitution at all, or an empty string when there is none.
doc.T = Returns the translation of the key, or the fallback language's translation, or, failing that, the key itself.
doc.Interpolate = Replaces every <code>{name}</code> in the message with the value of that name. The names and the values are given in two vectors in the same order.
doc.Tf = Translation and interpolation in one step.
doc.Plural = Chooses the right form of a message for the number <code>n</code>: the key <code>.one</code> for one and <code>.other</code> for the rest. It does not place the number in the message itself; use <code>Interpolate</code> for that. Languages that need only one form can define <code>.other</code> alone.
doc.LoadLocale = Loads the translations of one language from a file of <code>key = value</code> lines and returns how many keys were read.
doc.DetectLocale = Guesses the user's language from the environment: <code>SALAM_LOCALE</code> first, then <code>LANG</code> and <code>LC_ALL</code>. When none is set it returns <code>en</code>.
doc.Free = Releases the catalog's memory.
