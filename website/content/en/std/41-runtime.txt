pkg.title = The runtime package in Salam | A program's memory and uptime
pkg.description = The runtime package in Salam: the operating system and architecture, memory in use, the program's uptime and its build options, with an example.
pkg.intro <<
<p>The <code>runtime</code> package is useful for reporting a program's state, on a <code>/health</code> route on a server for instance. The uptime is counted from the moment <code>MarkStart</code> was called.</p>
<p>Free and used memory are available on Linux and Windows only; on other systems those functions return minus one.</p>
>>

doc.OS = The name of the operating system.
doc.Arch = The name of the processor architecture.
doc.CpuCount = The number of processor cores.
doc.TotalMemory = The system's total memory.
doc.AvailableMemory = The system's free memory, or minus one where it is not available.
doc.UsedMemory = The system's used memory, total minus free, or minus one.
doc.MarkStart = Records the start instant for the uptime; call it at the top of <code>main</code>.
doc.UptimeMs = The uptime in milliseconds.
doc.UptimeSeconds = The uptime in seconds.
doc.IsAsanBuild = True when the program was built with the ASan memory checker.
doc.IsMemDebugBuild = True when the program was built with allocation counting.
doc.IsSafeBuild = True when the program was built with array bounds checking.
doc.BuildFlags = The program's build options as text.
doc.Summary = A few lines summarising all of the above.
