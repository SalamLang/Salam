pkg.en = encoding/csv
pkg.name = encoding/csv
pkg.group = Encoding and data
pkg.summary = Reading and writing CSV files, with correct quoting and conversion to a dynamic value.
pkg.title = The encoding/csv package in Salam | CSV
pkg.description = The encoding/csv package in Salam: reading and writing CSV, quoted fields, and turning a table into a list of objects using the header row, with an example.
pkg.import = encoding.csv
pkg.intro <<
<p>CSV is the simplest table format: one row per line, fields separated by commas. A field containing a comma, a quote or a newline is wrapped in quotes, and a quote inside it is doubled. This package follows those rules both ways.</p>
<p>Every field is a string. When the first row is a header, <code>ToValue</code> turns each row into an object with those keys.</p>
>>

doc.CSVRow = One row.
doc.CSVRow.fields = The row's fields.
doc.CSVTable = A table.
doc.CSVTable.rows = The rows, including the header row.
doc.CSVOptions = The options for converting to and from a value.
doc.CSVOptions.header = The first row is a header; true by default. When false, each row becomes an array.
doc.ReadAll = Turns a whole piece of CSV text into a table.
doc.FreeTable = Releases the table's memory.
doc.WriteAll = Turns a table into CSV text.
doc.ToValueWith = Turns CSV into a <code>Value</code> with the given options.
doc.ToValue = Turns CSV with a header into an array of objects.
doc.FromValueWith = Turns an array of objects, or of arrays, into CSV with the given options.
doc.FromValue = Turns an array of objects into CSV with a header.
doc.ReadLine = Splits one CSV line into its fields.
doc.WriteLine = Turns fields into one CSV line, quoting wherever it has to.
