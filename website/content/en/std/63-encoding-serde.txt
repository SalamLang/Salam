pkg.title = The encoding/serde package in Salam | Converting between JSON and YAML
pkg.description = The encoding/serde package in Salam: detecting the format of text automatically and converting between JSON, YAML, TOML, XML and CSV, with an example.
pkg.import = encoding.serde
pkg.intro <<
<p>This package reads text in one format, turns it into a <a href="../encoding-value/">Value</a> and writes it out in another. The format names are the short strings <code>json</code>, <code>yaml</code>, <code>toml</code>, <code>xml</code> and <code>csv</code>.</p>
>>

doc.FormatUnknown = An unknown format.
doc.FormatJSON = JSON.
doc.FormatYAML = YAML.
doc.FormatTOML = TOML.
doc.FormatXML = XML.
doc.FormatCSV = CSV.
doc.DetectFormat = Guesses which format the text is in from how it looks and returns its number.
doc.Detect = The same, but it returns the format's short name.
doc.TranscodeStrict = Converts text from the <code>from</code> format to <code>dest</code>; when reading or writing fails, or a format name is wrong, <code>ok</code> becomes false.
doc.Transcode = Converts text from one format to another.
