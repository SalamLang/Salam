pkg.en = net/proxy
pkg.name = net/proxy
pkg.group = Network
pkg.summary = Connecting through an HTTP or SOCKS5 proxy, alone or in a chain.
pkg.title = The net/proxy package in Salam | HTTP and SOCKS5
pkg.description = The net/proxy package in Salam: a TCP connection through an HTTP or SOCKS5 proxy with authentication, and a chain of proxies, with examples.
pkg.import = net.proxy
pkg.intro <<
<p>A proxy is a middleman that makes the connection on behalf of the program. This package builds the tunnel and gives back an ordinary <a href="../net-tcp/">net/tcp</a> connection, read and written exactly as a direct one is.</p>
>>

doc.TypeHTTP = An HTTP proxy, through the CONNECT method.
doc.TypeSOCKS5 = A SOCKS5 proxy.
doc.Proxy = One proxy.
doc.Proxy.kind = The kind; one of the <code>Type</code> constants.
doc.Proxy.host = The host.
doc.Proxy.port = The port.
doc.Proxy.username = The user name, when one is needed.
doc.Proxy.password = The password.
doc.Chain = A chain of proxies used one after another.
doc.Chain.items = The proxies in order.
doc.NewHTTPProxy = An HTTP proxy from a <code>host:port</code> address.
doc.NewHTTPProxyAuth = An HTTP proxy with a user name and password.
doc.NewSOCKS5Proxy = A SOCKS5 proxy.
doc.NewSOCKS5ProxyAuth = A SOCKS5 proxy with a user name and password.
doc.Dial = Connects to the destination through the proxy and gives back a ready TCP connection.
doc.NewChain = An empty chain.
doc.Add = Adds a proxy to the end of the chain.
doc.DialChain = Connects to the destination through every proxy in the chain, in order.
