pkg.title = The net/http/validate package in Salam
pkg.description = The net/http/validate package in Salam: checking a required field, a length, an email address and a pattern, and answering with a 400 error in an API, with an example.
pkg.import = net.http.validate
pkg.intro <<
<p>The functions here check one value and return true or false. <code>Assert</code> sends a 400 response with a JSON message when the condition does not hold; after that the handler must not write another response.</p>
>>

doc.Required = True when the value is not empty; whitespace counts as a value.
doc.Length = True when the number of characters is between the minimum and the maximum.
doc.IsEmail = A very simple email check: the text has to hold an <code>@</code> and a dot. To be sure, send a confirmation email.
doc.Matches = True when the value matches the regular expression.
doc.Assert = When the condition is false, sends a 400 response with <code>{"error": message}</code>.

ex.Required = net-http-validate-Check

ex.Matches = net-http-validate-Check
