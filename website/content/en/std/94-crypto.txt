pkg.en = crypto
pkg.name = crypto
pkg.group = Security and cryptography
pkg.summary = SHA and MD5 digests, HMAC signatures, secure random numbers and password hashing.
pkg.title = The crypto package in Salam | SHA-256, HMAC and passwords
pkg.description = The crypto package in Salam: SHA-256 and SHA-512 digests, HMAC signatures, secure random tokens and numbers, and hashing and checking a password with PBKDF2, with examples.
pkg.import = crypto
pkg.intro <<
<p>This package gives the basic cryptography tools, all of them written in Salam. For the digest of a piece of text the <code>...Hex</code> functions are the simplest way; the <code>...Bytes</code> functions write the digest into memory you pass in yourself.</p>
<p>Never store a user password raw or under a plain digest; use <code>HashPassword</code> and <code>VerifyPassword</code>. To compare tokens, use <code>SecretEquals</code> rather than <code>==</code>. To encrypt data, see the <a href="../tls/">tls</a> package.</p>
>>

doc.HashBytes = A fast, non cryptographic hash of some bytes; for hash tables and the like, not for security.
doc.HashStr = A fast hash of a string.
doc.HashInt = A fast hash of a number.
doc.Md5Bytes = Writes the MD5 digest, 16 bytes, into the output. MD5 is broken; only for compatibility with old systems.
doc.Md5Hex = The MD5 digest as hex.
doc.Sha1Hex = The SHA-1 digest as hex. SHA-1 is no longer strong enough for security either; use SHA-256 for signatures.
doc.Sha1Base64 = The SHA-1 digest as base64; for the WebSocket handshake, for instance.
doc.Sha1Bytes = Writes the SHA-1 digest, 20 bytes, into the output.
doc.Sha256Hex = The SHA-256 digest as hex, 64 characters.
doc.Sha512Hex = The SHA-512 digest as hex, 128 characters.
doc.Sha256Bytes = Writes the SHA-256 digest, 32 bytes, into the output.
doc.Sha384Bytes = The SHA-384 digest, 48 bytes.
doc.Sha512Bytes = The SHA-512 digest, 64 bytes.
doc.Sha256 = The SHA-256 state for data that arrives in pieces, such as a large file.
doc.Sha256.h = The internal state.
doc.Sha256.blk = The partial block.
doc.Sha256.blen = The length of the partial block.
doc.Sha256.total = The total number of bytes.
doc.Sha512 = The SHA-512 or SHA-384 state for data in pieces.
doc.Sha512.h = The internal state.
doc.Sha512.blk = The partial block.
doc.Sha512.blen = The length of the partial block.
doc.Sha512.total = The total number of bytes.
doc.Sha512.outlen = The digest length: 64 for SHA-512 and 48 for SHA-384.
doc.Sha256New = A fresh SHA-256 state.
doc.Sha256Update = Adds the next piece of data.
doc.Sha256Final = Writes the final digest into the output.
doc.Sha384New = A fresh SHA-384 state; carry on with the Sha512 functions.
doc.Sha512New = A fresh SHA-512 state.
doc.Sha512Update = Adds the next piece of data.
doc.Sha512Final = Writes the final digest at the length in <code>outlen</code>.
doc.HmacSha256Hex = The HMAC-SHA256 signature of a message under a key, as hex; for checking that the message is untouched and comes from whoever holds the key.
doc.HmacSha384Hex = HMAC-SHA384 as hex.
doc.HmacSha512Hex = HMAC-SHA512 as hex.
doc.HmacSha1Hex = HMAC-SHA1 as hex; for old services.
doc.HmacSha1Base64 = HMAC-SHA1 as base64; an OAuth 1 signature, for instance.
doc.HmacSha256Bytes = Writes HMAC-SHA256, 32 bytes, into the output.
doc.HmacSha384Bytes = HMAC-SHA384, 48 bytes.
doc.HmacSha512Bytes = HMAC-SHA512, 64 bytes.
doc.HmacSha1Bytes = HMAC-SHA1, 20 bytes.
doc.HmacSha256Raw = HMAC-SHA256 with the key and message given as a pointer and a length; for binary data.
doc.HmacSha384Raw = Raw HMAC-SHA384.
doc.HmacSha512Raw = Raw HMAC-SHA512.
doc.HmacSha1Raw = Raw HMAC-SHA1.
doc.RandomHex = n secure random bytes from the operating system, as hex with 2n characters; for ids and salts.
doc.RandomToken = A secure random token of n bytes, as url safe base64; for session and password reset tokens.
doc.RandomBytes = Writes n secure random bytes into the given memory. When the operating system cannot, the program stops.
doc.Pbkdf2HmacSha256Hex = Derives a key with PBKDF2-HMAC-SHA256 from a password and a salt, at the given iteration count.
doc.HashPassword = Hashes the password with a random salt and returns a string of the form <code>pbkdf2-sha256$iterations$salt$hash</code> to keep in your database. The output is different every time.
doc.VerifyPassword = Checks a password against a string that <code>HashPassword</code> made.
doc.SecretEquals = Compares two secrets in constant time, so the answer time gives nothing away about them; for tokens and signatures.
