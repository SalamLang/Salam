pkg.description = راهنمای فارسی بسته‌ی قالب بندی در زبان برنامه نویسی سلام: تبدیل عدد به متن، عدد اعشاری با دقت دلخواه، پرکردن و وسط‌چین کردن متن و ساختن رشته از قالب با نمونه‌کد.
pkg.intro <<
<p>وقتی می‌خواهید خروجی مرتب و خوانا بسازید، بسته‌ی <code>قالب بندی</code> به کارتان می‌آید. با آن عددها را به متن تبدیل می‌کنید، متن را در ستون‌هایی با پهنای ثابت می‌چینید و جمله‌ها را از یک قالب می‌سازید.</p>
<p>قالب‌ها ساده‌اند: هر <code>{}</code> در قالب با عضو بعدی وکتور مقدارها پر می‌شود. چون مقدارها رشته‌اند، عددها را پیش از آن با تابع‌هایی مثل <code>عدد صحیح</code> به متن تبدیل کنید.</p>
>>

doc.Int = یک عدد صحیح را به متن تبدیل می‌کند، با علامت منفی اگر لازم باشد.
doc.Float = یک عدد اعشاری را به متن تبدیل می‌کند و رقم‌های بی‌فایده‌ی آخر را نمی‌نویسد.
doc.Bool = مقدار منطقی را به <code>"true"</code> یا <code>"false"</code> تبدیل می‌کند.
doc.PadLeft = اگر متن از <code>width</code> کوتاه‌تر باشد، سمت چپش فاصله می‌گذارد تا به آن پهنا برسد. پهنا بر حسب حرف شمرده می‌شود، پس برای متن فارسی هم درست کار می‌کند. در ستون عددها، این تابع عددها را از راست هم‌تراز می‌کند.
doc.PadRight = مثل <code>پرکردن چپ</code>، ولی فاصله‌ها را سمت راست متن می‌گذارد.
doc.Center = متن را وسط پهنای داده‌شده می‌گذارد. اگر فاصله‌ها زوج نباشند، یک فاصله‌ی اضافه سمت راست می‌رود.
doc.Sprintf = قالب را برمی‌دارد و هر <code>{}</code> را به‌ترتیب با عضوهای وکتور <code>args</code> پر می‌کند. اگر مقدار کم بیاید، جای خالی‌ها تهی می‌مانند و مقدارهای اضافه نادیده گرفته می‌شوند.
doc.Fprintf = مثل <code>قالب رشته</code>، ولی نتیجه را مستقیم در پرونده‌ی باز می‌نویسد و تعداد بایت‌های نوشته‌شده را برمی‌گرداند.
doc.FloatPrec = عدد اعشاری را با تعداد رقم اعشار دلخواه به متن تبدیل می‌کند و عدد را گرد می‌کند.
doc.Uint = یک عدد بدون علامت ۶۴ بیتی را به متن تبدیل می‌کند؛ برای عددهای خیلی بزرگ که در صحیح۶۴ جا نمی‌شوند.
doc.Hex = عدد را در مبنای شانزده با حروف کوچک می‌نویسد، بدون پیشوند <code>0x</code>.

ex.Float = fmt-Int
ex.Bool = fmt-Int
ex.FloatPrec = fmt-Int
ex.Uint = fmt-Int
ex.Hex = fmt-Int
ex.PadRight = fmt-PadLeft
ex.Center = fmt-PadLeft
