pkg.intro <<
<p>یونیکد به هر نویسه‌ی همه‌ی خط‌های دنیا یک عدد داده که به آن «کدنقطه» می‌گویند؛ «ش» عدد ۱۵۸۸ است و «A» عدد ۶۵. بسته‌ی <code>یونیکد</code> برای هر کدنقطه می‌گوید چه نوع نویسه‌ای است: حرف است یا رقم، از کدام خط است، از راست به چپ نوشته می‌شود یا از چپ به راست، و شکل بزرگ یا کوچکش چیست.</p>
<p>بیشتر وقت‌ها روال‌های بسته‌ی <a href="/std/str/">رشته</a> کافی‌اند و خودشان از همین بسته استفاده می‌کنند. این بسته برای وقتی است که خودتان متن را نویسه به نویسه بررسی می‌کنید؛ مثلاً در یک تجزیه‌گر، یک ویرایشگر یا ابزاری برای پاک‌سازی متن فارسی. داده‌ها از پایگاه داده‌ی رسمی یونیکد، نسخه‌ی ۱۶، ساخته شده‌اند.</p>
<p>برای گرفتن کدنقطه‌های یک رشته، <code>رشته.کدنقطه ها</code> را به کار ببرید.</p>
>>

doc.ZWNJ = کدنقطه‌ی نیم‌فاصله (U+200C)، که در فارسی دو حرف را کنار هم نگه می‌دارد بی‌آن‌که به هم بچسبند؛ مثل «می‌روم».
doc.ZWJ = کدنقطه‌ی پیونددهنده‌ی بی‌عرض (U+200D)، که دو نویسه را به هم می‌چسباند؛ شکلک‌های ترکیبی مثل خانواده با آن ساخته می‌شوند.
doc.MaxCodePoint = بزرگ‌ترین کدنقطه‌ی ممکن در یونیکد (0x10FFFF).
doc.Version = نسخه‌ی یونیکدی که داده‌های این بسته از آن ساخته شده‌اند.
doc.MaxFoldLen = بیشترین تعداد کدنقطه‌ای که تاشدن کامل یک نویسه ممکن است بدهد. حافظه‌ای به این اندازه برای تاشدن هر نویسه‌ای کافی است.
group.Hangul.title = ثابت‌های هجای هانگول
group.Hangul = عددهایی که برای ساختن و شکستن هجاهای کره‌ای (هانگول) با حساب لازم‌اند. هر هجا از یک صامت آغازی، یک مصوت و شاید یک صامت پایانی ساخته می‌شود و کدنقطه‌اش را می‌توان مستقیم حساب کرد.
group.Cat.title = دسته‌های کلی نویسه
group.Cat = هر نویسه در یونیکد یک «دسته‌ی کلی» دارد. <code>دسته</code> یکی از این عددها را برمی‌گرداند و <code>نام دسته</code> نام کوتاه دوحرفی‌اش را (مثل <code>Lo</code>). دسته‌هایی که حرف اولشان یکی است پشت‌سرهم شماره‌گذاری شده‌اند.
group.Gcb.title = رده‌های شکست خوشه
group.Gcb = رده‌هایی که <code>شکست خوشه</code> برمی‌گرداند. بسته‌ی رشته با آن‌ها تشخیص می‌دهد کجای متن مرز یک «حرف دیده‌شده» است؛ مثلاً یک حرف و اعرابش، یا یک شکلک ترکیبی، یک خوشه‌اند.
group.Incb.title = رده‌های شکست پیوند
group.Incb = رده‌هایی که <code>شکست پیوند</code> برمی‌گرداند. این قاعده صامت‌هایی را که در خط‌های هندی با نشانه‌ی ویرام به هم پیوسته‌اند در یک خوشه نگه می‌دارد.
group.Bc.title = رده‌های دوسویه
group.Bc = رده‌ای که <code>رده دوسویه</code> برمی‌گرداند و الگوریتم متن دوسویه با آن تصمیم می‌گیرد هر نویسه در متن آمیخته‌ی فارسی و انگلیسی کجا نمایش داده شود. حرف‌های فارسی «عربی» هستند، حرف‌های انگلیسی «چپ به راست» و رقم‌ها رده‌ی خودشان را دارند.
group.Pbt.title = گونه‌های کروشه
group.Pbt = نتیجه‌ی <code>گونه کروشه جفت</code>: آیا نویسه کروشه‌ی باز است، بسته، یا اصلاً کروشه نیست.
group.Script.title = شناسه‌ی خط‌ها
group.Script = هر خطی که یونیکد می‌شناسد یک شناسه دارد و <code>خط</code> یکی از این‌ها را برمی‌گرداند. حرف‌های فارسی و عربی هر دو «خط عربی» هستند. نویسه‌های «مشترک» مثل رقم‌ها و نشانه‌ها در چند خط به کار می‌روند و «موروثی» یعنی نویسه خط نویسه‌ی پیش از خودش را می‌گیرد، مثل اعراب.
doc.Category = شماره‌ی دسته‌ی کلی نویسه را برمی‌گرداند؛ یکی از ثابت‌های «دسته» در بالا.
doc.CategoryName = نام دوحرفی دسته را برمی‌گرداند، مثل <code>Lo</code> برای حرف فارسی و <code>Nd</code> برای رقم.
doc.IsAssigned = اگر این کدنقطه در یونیکد به نویسه‌ای داده شده باشد، درست برمی‌گرداند.
doc.IsLetter = اگر نویسه حرف باشد (دسته‌های L)، درست برمی‌گرداند.
doc.IsAlphabetic = مثل <code>حرف است</code>، ولی گسترده‌تر: اعرابی که بخشی از خواندن حرف‌اند را هم حرف حساب می‌کند. بیشتر وقت‌ها منظور از «حرف است؟» همین است.
doc.IsUpper = اگر نویسه حرف بزرگ باشد، درست برمی‌گرداند.
doc.IsLower = اگر نویسه حرف کوچک باشد، درست برمی‌گرداند.
doc.IsTitle = حرف‌های «عنوانی» معدودی که نیمی بزرگ و نیمی کوچک‌اند، مثل «ǅ».
doc.IsMark = اگر نویسه نشانه‌ی ترکیبی باشد، مثل فتحه و تشدید، درست برمی‌گرداند.
doc.IsDigit = فقط رقم‌های دهدهی، از هر خطی؛ رقم فارسی هم حساب می‌شود.
doc.IsNumber = هر نویسه‌ی عددی، از جمله عددهای رومی و کسرها؛ گسترده‌تر از <code>رقم است</code>.
doc.IsAlnum = حرف یا رقم.
doc.IsPunct = نشانه‌های نگارشی، مثل ویرگول فارسی.
doc.IsSymbol = نمادها، مثل نماد ریال و علامت‌های ریاضی.
doc.IsSpace = نویسه‌های فاصله بر اساس ویژگی رسمی White_Space یونیکد. نیم‌فاصله فاصله نیست.
doc.IsControl = نویسه‌های کنترلی.
doc.IsFormat = نویسه‌های قالبی نامرئی، مثل نیم‌فاصله و نشانه‌های جهت.
doc.IsJoinControl = فقط نیم‌فاصله و پیونددهنده‌ی بی‌عرض؛ دو نویسه‌ای که چسبیدن حرف‌ها را کنترل می‌کنند.
doc.ToUpper = شکل بزرگ یک نویسه را برمی‌گرداند؛ اگر نداشته باشد، خود نویسه را.
doc.ToLower = شکل کوچک یک نویسه را برمی‌گرداند.
doc.ToTitle = شکل عنوانی نویسه را برمی‌گرداند.
doc.ToFold = شکل «تاشده»ی ساده‌ی نویسه را برمی‌گرداند؛ شکلی که برای مقایسه‌ی بدون حساسیت به بزرگی و کوچکی به کار می‌رود.
doc.HasFullFold = اگر تاشدن کامل نویسه بیش از یک کدنقطه باشد، درست برمی‌گرداند؛ مثل «ß» که به «ss» تا می‌شود.
doc.FullFoldLen = تعداد کدنقطه‌های تاشدن کامل نویسه.
doc.FullFoldAt = کدنقطه‌ی شماره‌ی <code>k</code> از تاشدن کامل نویسه.
doc.CombiningClass = رده‌ی ترکیبی نشانه را برمی‌گرداند. هنگام یکسان‌سازی، اعرابی که دنبال یک حرف می‌آیند بر اساس این عدد مرتب می‌شوند تا ترتیب تایپشان مهم نباشد. برای نویسه‌های معمولی صفر است.
doc.DecompLen = تعداد کدنقطه‌های تجزیه‌ی متعارف نویسه؛ اگر تجزیه نشود صفر.
doc.DecompAt = کدنقطه‌ی شماره‌ی <code>k</code> از تجزیه‌ی متعارف نویسه.
doc.CompatDecompLen = تعداد کدنقطه‌های تجزیه‌ی سازگاری نویسه. شکل‌های نمایشی عربی اینجا به حرف‌های معمولی باز می‌شوند.
doc.CompatDecompAt = کدنقطه‌ی شماره‌ی <code>k</code> از تجزیه‌ی سازگاری.
doc.Compose = دو کدنقطه را، اگر ترکیب متعارفی داشته باشند، به یک کدنقطه تبدیل می‌کند؛ وگرنه منفی یک برمی‌گرداند. «ا» و مدّ با هم «آ» می‌شوند.
doc.GraphemeBreak = رده‌ی شکست خوشه‌ی نویسه را برمی‌گرداند.
doc.ConjunctBreak = رده‌ی شکست پیوند نویسه را برمی‌گرداند؛ برای خط‌های هندی.
doc.BidiClass = رده‌ی دوسویه‌ی نویسه را برمی‌گرداند.
doc.PairedBracketType = می‌گوید نویسه کروشه‌ی باز است، بسته، یا هیچ‌کدام.
doc.PairedBracket = جفت یک کروشه را برمی‌گرداند؛ جفت «(» می‌شود «)». برای نویسه‌ای که کروشه نیست، خود نویسه را برمی‌گرداند.
doc.Script = شناسه‌ی خط نویسه را برمی‌گرداند.
doc.ScriptName = نام انگلیسی خط نویسه را برمی‌گرداند، مثل <code>Arabic</code>.
