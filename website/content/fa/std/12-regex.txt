pkg.title = بسته‌ی عبارت باقاعده (regex) در زبان برنامه‌نویسی سلام | جستجو و جایگزینی با الگو
pkg.description = راهنمای فارسی عبارت‌های باقاعده (regex) در زبان برنامه نویسی سلام: کامپایل الگو، تطبیق، پیدا کردن همه‌ی موارد، گروه‌های نام‌دار، جایگزینی و شکستن متن با نمونه‌کد.
pkg.intro <<
<p>عبارت باقاعده زبانی کوچک برای توصیف الگوی متن است؛ <code>[0-9]+</code> یعنی «یک یا چند رقم». با آن می‌توانید بپرسید متنی با الگو جور است یا نه، همه‌ی جاهای جور را پیدا کنید، بخش‌هایی از آن را بیرون بکشید یا جایگزین کنید.</p>
<p>هر روال دو نسخه دارد. نسخه‌ای که <code>عبارت باقاعده</code> می‌گیرد با الگوی از پیش کامپایل‌شده کار می‌کند و برای استفاده‌ی تکراری سریع‌تر است. نسخه‌ای که نامش با «رشته» تمام می‌شود، الگو را به شکل متن می‌گیرد و هر بار کامپایلش می‌کند؛ برای کارهای یک‌باره راحت‌تر است.</p>
<p>پرانتز در الگو یک «گروه» می‌سازد و <code>(?P&lt;نام&gt;...)</code> گروهی نام‌دار. گروه صفر کل بخش جورشده است. در جایگزینی، <code>$1</code> و <code>$2</code> جای گروه‌ها را می‌گیرند.</p>
>>

doc.MatchResult = نتیجه‌ی یک جستجو. به‌جای خواندن مستقیم فیلدها، بیشتر وقت‌ها روال‌های «گروه» ساده‌ترند.
doc.MatchResult.ok = اگر جوری پیدا شده باشد، درست است.
doc.MatchResult.start = جای شروع بخش جورشده، بر حسب بایت.
doc.MatchResult.stop = جای پایان بخش جورشده، بر حسب بایت (خود این جایگاه جزو آن نیست).
doc.MatchResult.count = تعداد گروه‌ها، به‌جز گروه صفر.
doc.MatchResult.text = کل متنی که در آن جستجو شده؛ نه فقط بخش جورشده. بخش جورشده را با <code>گروه(ت، ۰)</code> بگیرید.
doc.MatchResult.pattern = الگویی که جستجو با آن انجام شده.
doc.MatchResult.gs = جای شروع هر گروه.
doc.MatchResult.ge = جای پایان هر گروه.
doc.FindMatch = نخستین جای جور با الگو را پیدا می‌کند و همه‌ی جزئیاتش، از جمله گروه‌ها، را برمی‌گرداند.
doc.FindMatchStr = مثل <code>یافتن تطبیق</code>، با الگوی متنی.
doc.FindMatchFrom = مثل <code>یافتن تطبیق</code>، ولی جستجو را از بایت <code>from</code> شروع می‌کند؛ برای پیدا کردن مورد بعدی.
doc.GroupCount = تعداد گروه‌های الگو را برمی‌گرداند، به‌جز گروه صفر.
doc.GroupOk = اگر گروه شماره‌ی <code>i</code> در این جور شرکت کرده باشد، درست برمی‌گرداند. گروه‌های اختیاری ممکن است خالی بمانند.
doc.Group = متن گروه شماره‌ی <code>i</code> را برمی‌گرداند؛ گروه صفر کل بخش جورشده است.
doc.GroupStart = جای شروع گروه، بر حسب بایت.
doc.GroupEnd = جای پایان گروه، بر حسب بایت.
doc.GroupIndex = شماره‌ی یک گروه نام‌دار را برمی‌گرداند.
doc.GroupByName = متن یک گروه نام‌دار را برمی‌گرداند.
doc.FindSubmatch = متن جورشده و همه‌ی گروه‌هایش را در یک وکتور برمی‌گرداند؛ عضو صفر کل بخش جورشده است.
doc.FindSubmatchStr = مثل <code>یافتن زیرتطبیق</code>، با الگوی متنی.
doc.FindAllMatches = همه‌ی جاهای جور را، با جزئیات، برمی‌گرداند. <code>limit</code> منفی یعنی بدون محدودیت.
doc.FindAllN = متن حداکثر <code>limit</code> مورد جور را برمی‌گرداند.
doc.FindAll = متن همه‌ی موارد جور را در یک وکتور برمی‌گرداند.
doc.FindAllStr = مثل <code>یافتن همه</code>، با الگوی متنی.
doc.SplitN = متن را هر جا با الگو جور شود می‌شکند، ولی حداکثر به <code>limit</code> تکه؛ تکه‌ی آخر بقیه‌ی متن است.
doc.Split = متن را هر جا با الگو جور شود می‌شکند. برای جداکننده‌هایی که فاصله‌ی دورشان متغیر است، از <code>رشته.تفکیک کردن</code> انعطاف‌پذیرتر است.
doc.SplitStr = مثل <code>تفکیک کردن</code>، با الگوی متنی.
doc.Expand = یک قالب را با گروه‌های یک نتیجه پر می‌کند؛ <code>$1</code>، <code>$2</code> و… جای گروه‌ها را می‌گیرند.
doc.ReplaceExpand = نخستین مورد جور را با قالب جایگزین می‌کند؛ در قالب می‌توانید از <code>$1</code> و… استفاده کنید.
doc.ReplaceAllExpand = همه‌ی موارد جور را با قالب جایگزین می‌کند.
doc.ReplaceExpandStr = مثل <code>جایگزینی با بسط</code>، با الگوی متنی.
doc.ReplaceAllExpandStr = مثل <code>جایگزینی همه با بسط</code>، با الگوی متنی.
doc.Regex = یک الگوی کامپایل‌شده. آن را با <code>کامپایل</code> بسازید و بارها به کار ببرید.
doc.Regex.pattern = متن الگویی که کامپایل شده.
doc.Compile = الگو را کامپایل می‌کند. اگر الگو ممکن است نادرست باشد، مثلاً از کاربر آمده، اول با <code>معتبر است</code> بررسی‌اش کنید.
doc.IsValid = اگر الگو درست نوشته شده باشد، درست برمی‌گرداند.
doc.Match = اگر جایی از متن با الگو جور شود، درست برمی‌گرداند. برای جور شدن کل متن، الگو را با <code>^</code> و <code>$</code> محصور کنید.
doc.Find = جای نخستین مورد جور را بر حسب بایت برمی‌گرداند، یا منفی یک.
doc.Replace = فقط نخستین مورد جور را با متن تازه جایگزین می‌کند.
doc.ReplaceAll = همه‌ی موارد جور را با متن تازه جایگزین می‌کند.
doc.MatchStr = مثل <code>تطبیق</code>، با الگوی متنی.
doc.FindStr = مثل <code>یافتن</code>، با الگوی متنی.
doc.ReplaceStr = مثل <code>جایگزینی</code>، با الگوی متنی.
doc.ReplaceAllStr = مثل <code>جایگزینی همه</code>، با الگوی متنی.
