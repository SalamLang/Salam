pkg.summary = خواندن و نوشتن پرونده‌های <code>.env</code> با جایگذاری متغیرها.
pkg.title = بسته‌ی پرونده محیط (encoding/envfile) در زبان برنامه‌نویسی سلام | فایل .env
pkg.description = راهنمای فارسی بسته‌ی پرونده محیط در زبان برنامه نویسی سلام: خواندن فایل .env، مقدارهای نقل‌قول‌دار، export، جایگذاری ${VAR}، ادغام و نوشتن دوباره با نمونه‌کد.
pkg.import = کدگذاری.پرونده محیط
pkg.intro <<
<p>پرونده‌ی <code>.env</code> متغیرهای محیطی برنامه را، هر کدام در یک خط <code>KEY=value</code>، نگه می‌دارد. این بسته خط‌های خالی و توضیح‌های <code>#</code> را رد می‌کند، <code>export</code> پیش از کلید را می‌پذیرد، نقل‌قول‌ها را برمی‌دارد و ترتیب کلیدها را حفظ می‌کند.</p>
<p><code>تجزیه</code> ارجاع‌هایی مثل <code>${HOST}</code> را با مقدار همان پرونده جایگذاری می‌کند؛ <code>تجزیه خام</code> آن‌ها را دست‌نخورده نگه می‌دارد. متغیرهای محیطی سیستم‌عامل خوانده نمی‌شوند؛ برای آن‌ها بسته‌ی <a href="../env/">محیط</a> را ببینید.</p>
>>

doc.Doc = محتوای یک پرونده‌ی محیط.
doc.Doc.m = کلیدها و مقدارها به ترتیب.
doc.Doc.flags = برای هر کلید، آیا مقدارش جایگذاری‌پذیر است (نقل‌قول تکی جایگذاری را غیرفعال می‌کند).
doc.New = سند خالی.
doc.Len = تعداد کلیدها.
doc.Has = اگر کلید وجود داشته باشد، درست برمی‌گرداند.
doc.Get = مقدار یک کلید؛ اگر نباشد، رشته‌ی خالی.
doc.GetOr = مقدار یک کلید یا مقدار پیش‌فرض.
doc.Set = مقدار کلید را تنظیم می‌کند؛ کلید تازه به انتها می‌رود.
doc.Remove = کلید را برمی‌دارد و می‌گوید وجود داشت یا نه.
doc.KeyAt = کلید شماره‌ی <code>i</code>.
doc.ValueAt = مقدار شماره‌ی <code>i</code>.
doc.Keys = کلیدها به ترتیب.
doc.MergeInto = کلیدهای <code>src</code> را در <code>dst</code> می‌ریزد؛ کلیدهای تکراری جایگزین می‌شوند.
doc.Free = حافظه‌ی سند را آزاد می‌کند.
doc.ExpandDoc = سند تازه‌ای با ارجاع‌های جایگذاری‌شده می‌سازد. نام‌ها اول در همان سند و بعد در <code>base</code> جستجو می‌شوند.
doc.Expand = ارجاع‌های یک رشته را با مقدارهای سند و <code>base</code> جایگذاری می‌کند.
doc.RefNames = نام متغیرهایی که مقدارهای سند به آن‌ها ارجاع می‌دهند.
doc.RefNamesIn = نام متغیرهای ارجاع‌شده در یک رشته؛ هر دو شکل <code>$A</code> و <code>${A}</code>.
doc.Quote = مقدار را، اگر لازم باشد، در نقل‌قول دوتایی می‌گذارد.
doc.Format = سند را دوباره به متن پرونده‌ی محیط تبدیل می‌کند.
doc.FormatExport = مثل <code>قالب</code>، با <code>export</code> پیش از هر خط؛ برای پوسته.
doc.ParseRaw = پرونده را بدون جایگذاری ارجاع‌ها می‌خواند.
doc.Parse = پرونده را می‌خواند و ارجاع‌ها را جایگذاری می‌کند.
doc.ParseStrict = مثل <code>تجزیه</code>، ولی خط نادرست خطا دارد و <code>err</code> پر می‌شود.
doc.Valid = اگر همه‌ی خط‌ها درست باشند، درست برمی‌گرداند.
doc.ParseScalar = یک مقدار تنها را مثل سمت راست <code>=</code> می‌خواند: نقل‌قول و اسکیپ را برمی‌دارد.
doc.ParseScalarEx = مثل <code>تجزیه مقدار</code>؛ اگر <code>semi</code> درست باشد، <code>;</code> هم مثل <code>#</code> آغاز توضیح است.
