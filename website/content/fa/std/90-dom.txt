doc.Query = نخستین عنصری که با گزینشگر CSS جور است؛ اگر نباشد، صفر.
doc.QueryAll = همه‌ی عنصرهایی که با گزینشگر جورند، به شکل یک فهرست؛ با <code>تعداد</code> و <code>عنصر شماره</code> بخوانیدش.
doc.ById = عنصر با این شناسه.
doc.Exists = عنصری با این شناسه هست.
doc.Create = عنصر تازه با این تگ؛ هنوز در صفحه نیست تا آن را <code>افزودن</code> کنید.
doc.Body = عنصر <code>body</code> صفحه.
doc.Document = شیء <code>document</code>.
doc.Window = شیء <code>window</code>.
doc.Global = یک شیء سراسری جاوااسکریپت با این نام، مثل <code>Math</code>.
doc.Append = عنصر را به انتهای فرزندان عنصر دیگر می‌افزاید.
doc.Remove = عنصر را از صفحه برمی‌دارد.
doc.SetText = متن عنصر را تنظیم می‌کند؛ HTML آن تفسیر نمی‌شود.
doc.GetText = متن عنصر.
doc.SetHtml = HTML درون عنصر را تنظیم می‌کند. متنی را که کاربر نوشته این‌جا نگذارید.
doc.SetValue = مقدار یک جعبه‌ی ورودی.
doc.GetValue = مقدار فعلی جعبه‌ی ورودی.
doc.SetStyle = یک خاصیت CSS را تنظیم می‌کند، مثل <code>color</code>.
doc.GetStyle = مقدار خاصیت CSS که روی خود عنصر تنظیم شده.
doc.SetAttr = یک ویژگی HTML را تنظیم می‌کند.
doc.GetAttr = مقدار یک ویژگی؛ اگر نباشد، رشته‌ی خالی.
doc.AddClass = کلاسی به عنصر می‌افزاید.
doc.RemoveClass = کلاسی را برمی‌دارد.
doc.ToggleClass = اگر کلاس هست برمی‌دارد، وگرنه می‌افزاید.
doc.OnEvent = تابعی را برای یک رویداد عنصر ثبت می‌کند، مثل <code>click</code>.
doc.OnEventE = مثل تابع بالا، ولی تابع شیء رویداد را هم می‌گیرد تا با <code>کلید رویداد</code> و مانند آن بخوانیدش.
doc.OnWindowE = تابع با شیء رویداد برای رویدادهای پنجره، مثل <code>keydown</code>.
doc.OffEvent = تابعی را که پیش‌تر ثبت شده برمی‌دارد؛ باید همان تابع را بدهید، پس آن را در یک متغیر نگه دارید.
doc.OnWindow = تابع برای یک رویداد پنجره، مثل <code>scroll</code>.
doc.OnReady = تابع را وقتی صفحه آماده شد صدا می‌زند؛ اگر آماده است، بی‌درنگ.
doc.OnResize = تابع را با هر تغییر اندازه‌ی پنجره صدا می‌زند.
doc.RequestFrame = تابع را پیش از رسم فریم بعدی یک بار صدا می‌زند؛ تابع زمان را بر حسب میلی‌ثانیه می‌گیرد.
doc.EveryFrame = تابع را در هر فریم صدا می‌زند؛ برای حلقه‌ی بازی و پویانمایی.
doc.SetTimeout = تابع را پس از این قدر میلی‌ثانیه یک بار صدا می‌زند و شناسه‌ای برمی‌گرداند.
doc.SetInterval = تابع را هر این قدر میلی‌ثانیه صدا می‌زند.
doc.ClearTimeout = مهلتی را که هنوز اجرا نشده لغو می‌کند.
doc.ClearInterval = تکرار را متوقف می‌کند.
doc.WinWidth = عرض پنجره بر حسب پیکسل.
doc.WinHeight = ارتفاع پنجره.
doc.DocWidth = عرض سند.
doc.DocHeight = ارتفاع سند.
doc.ScrollX = مقدار پیمایش افقی.
doc.ScrollY = مقدار پیمایش عمودی.
doc.Now = زمان دقیق از بارگذاری صفحه، بر حسب میلی‌ثانیه؛ برای اندازه‌گیری فاصله‌ها.
doc.GetProp = خاصیتی از یک شیء که خودش شیء است.
doc.GetNum = خاصیت عددی.
doc.GetStr = خاصیت رشته‌ای.
doc.GetBool = خاصیت بولی.
doc.SetProp = خاصیتی را به یک شیء تنظیم می‌کند.
doc.SetNum = خاصیت عددی را تنظیم می‌کند.
doc.SetStr = خاصیت رشته‌ای را تنظیم می‌کند.
doc.SetBool = خاصیت بولی را تنظیم می‌کند.
doc.Call0 = متدی از شیء را بی‌آرگومان صدا می‌زند و نتیجه را برمی‌گرداند.
doc.Call1 = با یک آرگومان عددی.
doc.Call2 = با دو آرگومان عددی.
doc.Call3 = با سه آرگومان عددی.
doc.Call4 = با چهار آرگومان عددی.
doc.CallS1 = با یک آرگومان رشته‌ای.
doc.CallS2 = با دو آرگومان رشته‌ای.
doc.GetContext = زمینه‌ی رسم یک عنصر canvas، مثل <code>2d</code>؛ برای رسم ساده‌تر <a href="../web-canvas/">بوم</a> را ببینید.
doc.ItemAt = عنصری از فهرستی که <code>پرسمان همه</code> برگرداند.
doc.CountOf = شمار عنصرهای فهرست.
doc.EventTarget = عنصری که رویداد روی آن رخ داد.
doc.EventX = جای افقی نشانگر موش در رویداد.
doc.EventY = جای عمودی نشانگر.
doc.EventKey = کلید فشرده‌شده، مثل <code>Enter</code> یا <code>a</code>.
doc.PreventDefault = جلوی کار پیش‌فرض مرورگر را می‌گیرد، مثل فرستادن فرم.
doc.StopPropagation = نمی‌گذارد رویداد به عنصرهای بالاتر برسد.
doc.Alert = پیامی در پنجره‌ی مرورگر نشان می‌دهد.
doc.Confirm = پرسشی با دکمه‌های تایید و لغو؛ اگر کاربر تایید کرد، درست.
doc.Prompt = از کاربر متنی می‌پرسد؛ اگر لغو کرد، رشته‌ی خالی.
doc.SetTitle = عنوان زبانه‌ی مرورگر را تنظیم می‌کند.
doc.Random = عدد تصادفی از ۰ تا کمتر از ۱.
doc.RandomInt = عدد صحیح تصادفی از ۰ تا کمتر از n.
doc.Tone = بوق ساده با بسامد و مدت داده‌شده پخش می‌کند؛ مرورگرها صدا را فقط پس از یک کلیک یا کلید اجازه می‌دهند.
doc.LoadImage = بارگذاری تصویری را آغاز می‌کند و شیء آن را برمی‌گرداند.
doc.ImageReady = تصویر کامل بارگذاری شده.
doc.StoreGetInt = عددی که در حافظه‌ی مرورگر (localStorage) ذخیره شده؛ اگر نباشد، صفر.
doc.StoreSetInt = عددی را در حافظه‌ی مرورگر ذخیره می‌کند؛ پس از بستن صفحه هم می‌ماند.
doc.StoreGetStr = متن ذخیره‌شده.
doc.StoreSetStr = متنی را ذخیره می‌کند.
doc.TouchX = جای افقی نخستین انگشت در رویداد لمسی.
doc.TouchY = جای عمودی نخستین انگشت.
