pkg.title = بسته‌ی اعلان وب (webpush) در زبان برنامه‌نویسی سلام | Web Push و VAPID
pkg.description = راهنمای فارسی بسته‌ی اعلان وب در زبان برنامه نویسی سلام: ساختن کلید VAPID، رمزنگاری پیام با RFC 8291 و فرستادن اعلان به کروم و فایرفاکس بدون سرویس واسط با نمونه‌کد.
pkg.import = اعلان وب
pkg.intro <<
<p>اعلان وب پیامی است که سرور حتی وقتی سایت باز نیست به مرورگر کاربر می‌فرستد. روند کار: یک بار با <code>ساخت کلیدها</code> کلید بسازید، کلید عمومی را در جاوااسکریپت صفحه برای اشتراک به کار ببرید، اشتراکی را که مرورگر می‌سازد در سرور ذخیره کنید و با <code>ارسال</code> پیام بفرستید.</p>
<p>پیام پیش از فرستادن طبق RFC 8291 رمز می‌شود، پس سرویس اعلان مرورگر محتوای آن را نمی‌بیند. اگر نتیجه <code>gone</code> بود، کاربر اشتراک را لغو کرده است.</p>
>>

doc.DEFAULT_TTL = ماندگاری پیش‌فرض پیام در سرویس اعلان: ۲۸ روز، بر حسب ثانیه.
doc.Keys = کلیدهای VAPID سرور.
doc.Keys.public_key = کلید عمومی به شکل بیس۶۴ نشانی‌امن؛ همین را در جاوااسکریپت صفحه به <code>applicationServerKey</code> بدهید.
doc.Keys.private_key = کلید خصوصی؛ آن را پنهان نگه دارید.
doc.Keys.ok = کلیدها معتبرند.
doc.Subscription = اشتراکی که مرورگر با <code>pushManager.subscribe</code> می‌سازد و به سرور شما می‌فرستد.
doc.Subscription.endpoint = نشانی سرویس اعلان مرورگر.
doc.Subscription.p256dh = کلید عمومی مرورگر از بخش <code>keys</code> اشتراک.
doc.Subscription.auth = راز احراز هویت از همان بخش.
doc.Options = گزینه‌های فرستادن.
doc.Options.subject = نشانی تماس شما، به شکل <code>mailto:</code> یا نشانی وب؛ سرویس‌ها آن را لازم دارند.
doc.Options.ttl = چند ثانیه سرویس پیام را برای مرورگر خاموش نگه دارد.
doc.Options.urgency = فوریت: <code>very-low</code>، <code>low</code>، <code>normal</code> یا <code>high</code>.
doc.Options.topic = موضوع؛ پیام تازه با همان موضوع جای پیام تحویل‌نشده‌ی قبلی را می‌گیرد.
doc.Result = نتیجه‌ی فرستادن.
doc.Result.ok = سرویس پیام را پذیرفت.
doc.Result.status = کد وضعیت HTTP سرویس.
doc.Result.gone = اشتراک دیگر معتبر نیست (۴۰۴ یا ۴۱۰)؛ آن را از پایگاه داده پاک کنید.
doc.Result.error = پیام خطا.
doc.GenerateKeys = یک جفت کلید VAPID تازه می‌سازد؛ یک بار بسازید و نگه دارید، چون اشتراک‌ها به کلید عمومی بسته‌اند.
doc.KeysFrom = کلیدهای ذخیره‌شده را برمی‌گرداند؛ اگر یکی خالی باشد، <code>ok</code> نادرست است.
doc.VapidHeaderAt = سرآیند <code>Authorization</code> برای سرویسی که نشانی‌اش داده شده، با زمان داده‌شده بر حسب ثانیه‌ی یونیکس. امضا قطعی است، پس با ورودی یکسان خروجی یکسان است.
doc.VapidHeader = مثل تابع بالا، با زمان فعلی.
doc.EncryptWith = پیام را با کلید و نمک داده‌شده به روش RFC 8291 رمز می‌کند؛ برای آزمایش با بردارهای استاندارد. در کار واقعی <code>رمزنگاری</code> را به کار ببرید.
doc.Encrypt = پیام را برای این اشتراک رمز می‌کند؛ هر بار با کلید و نمک تازه، پس دو خروجی یکسان نیستند.
doc.Send = پیام را رمز می‌کند و با سرآیند VAPID به سرویس اعلان مرورگر می‌فرستد. پیام خالی یک اعلان بی‌بدنه است.
doc.NewOptions = گزینه‌های پیش‌فرض با نشانی تماس.
