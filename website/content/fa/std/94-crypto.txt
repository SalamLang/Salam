doc.Sha1Base64 = چکیده‌ی SHA-1 به شکل بیس۶۴؛ مثلاً برای دست‌دهی وب‌سوکت.
doc.Sha1Bytes = چکیده‌ی SHA-1، ۲۰ بایت، را در خروجی می‌نویسد.
doc.Sha256Hex = چکیده‌ی SHA-256 به شکل شانزدهی، ۶۴ نویسه.

doc.Sha256Bytes = چکیده‌ی SHA-256، ۳۲ بایت، را در خروجی می‌نویسد.

doc.Sha512Bytes = چکیده‌ی SHA-512، ۶۴ بایت.

doc.Sha256.total = شمار کل بایت‌ها.
doc.Sha512 = وضعیت SHA-512 یا SHA-384 برای داده‌ی تکه‌تکه.

doc.Sha384New = وضعیت تازه‌ی SHA-384؛ با روال‌های شا۵۱۲ ادامه دهید.
doc.Sha512New = وضعیت تازه‌ی SHA-512.
doc.Sha512Update = تکه‌ی بعدی داده را می‌افزاید.
doc.Sha512Final = چکیده‌ی نهایی را به طول <code>outlen</code> می‌نویسد.
doc.HmacSha256Hex = امضای HMAC-SHA256 پیام با کلید، به شکل شانزدهی؛ برای بررسی اینکه پیام دست نخورده و از طرف دارنده‌ی کلید است.
doc.HmacSha384Hex = HMAC-SHA384 به شکل شانزدهی.
doc.HmacSha512Hex = HMAC-SHA512 به شکل شانزدهی.
doc.HmacSha1Hex = HMAC-SHA1 به شکل شانزدهی؛ برای سرویس‌های قدیمی.
doc.HmacSha1Base64 = HMAC-SHA1 به شکل بیس۶۴؛ مثلاً امضای OAuth 1.
doc.HmacSha256Bytes = HMAC-SHA256، ۳۲ بایت، را در خروجی می‌نویسد.
doc.HmacSha384Bytes = HMAC-SHA384، ۴۸ بایت.
doc.HmacSha512Bytes = HMAC-SHA512، ۶۴ بایت.
doc.HmacSha1Bytes = HMAC-SHA1، ۲۰ بایت.
doc.HmacSha256Raw = HMAC-SHA256 با کلید و پیامی که به شکل اشاره‌گر و طول می‌دهید؛ برای داده‌ی دودویی.
doc.HmacSha384Raw = HMAC-SHA384 خام.
doc.HmacSha512Raw = HMAC-SHA512 خام.
doc.HmacSha1Raw = HMAC-SHA1 خام.
doc.RandomHex = n بایت تصادفی امن از سیستم‌عامل، به شکل شانزدهی با ۲n نویسه؛ برای شناسه‌ها و نمک.
doc.RandomToken = توکن تصادفی امن از n بایت، به شکل بیس۶۴ نشانی‌امن؛ برای توکن نشست و بازیابی گذرواژه.
doc.RandomBytes = n بایت تصادفی امن را در حافظه‌ی داده‌شده می‌نویسد. اگر سیستم‌عامل نتواند، برنامه متوقف می‌شود.
doc.Pbkdf2HmacSha256Hex = کلید را با PBKDF2-HMAC-SHA256 از گذرواژه و نمک با شمار تکرار داده‌شده می‌سازد.
doc.HashPassword = گذرواژه را با نمک تصادفی درهم می‌کند و رشته‌ای به شکل <code>pbkdf2-sha256$تکرار$نمک$درهم</code> برمی‌گرداند که در پایگاه داده نگه می‌دارید. هر بار خروجی فرق می‌کند.
doc.VerifyPassword = گذرواژه را با رشته‌ای که <code>درهم گذرواژه</code> ساخته بررسی می‌کند.
doc.SecretEquals = دو راز را در زمان ثابت مقایسه می‌کند تا زمان پاسخ چیزی درباره‌ی آن‌ها لو ندهد؛ برای توکن‌ها و امضاها.
